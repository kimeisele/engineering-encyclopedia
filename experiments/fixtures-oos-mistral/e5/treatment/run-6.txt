import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass

# Configure logging to include correlation ID in every log line
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - [%(correlation_id)s] - %(message)s'
)
logger = logging.getLogger(__name__)

@dataclass
class ServiceContext:
    correlation_id: str
    service_name: str
    request_data: Dict[str, Any]

class ErrorHandler:
    """
    Error handler that ensures correlation ID is propagated across service boundaries
    and includes sufficient context for tracing requests across multiple services.
    """

    @staticmethod
    def generate_correlation_id() -> str:
        """Generate a unique correlation ID for a new request."""
        return str(uuid.uuid4())

    @staticmethod
    def extract_correlation_id(headers: Optional[Dict[str, str]]) -> str:
        """Extract correlation ID from incoming headers, generating one if missing."""
        if headers and 'X-Correlation-ID' in headers:
            return headers['X-Correlation-ID']
        return ErrorHandler.generate_correlation_id()

    @staticmethod
    def propagate_correlation_id(headers: Dict[str, str], correlation_id: str) -> Dict[str, str]:
        """Add correlation ID to outgoing headers for downstream calls."""
        headers['X-Correlation-ID'] = correlation_id
        return headers

    @staticmethod
    def create_error_context(
        operation: str,
        failing_step: str,
        correlation_id: str,
        service_name: str,
        request_data: Dict[str, Any],
        original_error: Exception
    ) -> Dict[str, Any]:
        """Create a comprehensive error context for logging and tracing."""
        return {
            'operation': operation,
            'failing_step': failing_step,
            'correlation_id': correlation_id,
            'service_name': service_name,
            'request_data': request_data,
            'error_type': type(original_error).__name__,
            'error_message': str(original_error)
        }

    @staticmethod
    def handle_service_error(
        context: ServiceContext,
        operation: str,
        failing_step: str,
        error: Exception,
        is_critical: bool = False
    ) -> None:
        """
        Handle an error in a service, ensuring it's logged with full context.

        Args:
            context: ServiceContext containing correlation ID and request data
            operation: What the service was trying to do
            failing_step: Specific step where failure occurred
            error: The exception that was raised
            is_critical: Whether this is a critical failure requiring immediate attention
        """
        error_context = ErrorHandler.create_error_context(
            operation=operation,
            failing_step=failing_step,
            correlation_id=context.correlation_id,
            service_name=context.service_name,
            request_data=context.request_data,
            original_error=error
        )

        if is_critical:
            logger.critical("CRITICAL ERROR", extra=error_context)
        else:
            logger.error("Service error occurred", extra=error_context)
