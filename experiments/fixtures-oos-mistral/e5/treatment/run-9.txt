import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass

@dataclass
class RequestContext:
    correlation_id: str
    service_name: str
    operation: str
    request_data: Dict[str, Any]

class ErrorHandler:
    def __init__(self):
        self.logger = logging.getLogger(__name__)

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for the request."""
        return str(uuid.uuid4())

    def extract_or_generate_correlation_id(self, headers: Optional[Dict[str, str]]) -> str:
        """Extract correlation ID from headers or generate a new one if missing."""
        if headers and 'X-Correlation-ID' in headers:
            return headers['X-Correlation-ID']
        return self.generate_correlation_id()

    def log_error_with_context(
        self,
        error: Exception,
        context: RequestContext,
        headers: Optional[Dict[str, str]] = None
    ) -> None:
        """Log an error with full context including correlation ID."""
        error_context = {
            'correlation_id': context.correlation_id,
            'service': context.service_name,
            'operation': context.operation,
            'error_type': type(error).__name__,
            'error_message': str(error),
            'request_data': context.request_data
        }

        if headers:
            error_context['incoming_headers'] = headers

        self.logger.error(
            "Request failed: %s in %s (correlation_id=%s)",
            error_context['error_message'],
            error_context['operation'],
            error_context['correlation_id'],
            extra=error_context
        )

    def handle_service_error(
        self,
        error: Exception,
        service_name: str,
        operation: str,
        request_data: Dict[str, Any],
        headers: Optional[Dict[str, str]] = None,
        retry_count: int = 0
    ) -> None:
        """
        Handle errors from service calls with proper context propagation.
        Includes dead-letter queue consideration for final failures.
        """
        correlation_id = self.extract_or_generate_correlation_id(headers)
        context = RequestContext(
            correlation_id=correlation_id,
            service_name=service_name,
            operation=operation,
            request_data=request_data
        )

        # Log the error with full context
        self.log_error_with_context(error, context, headers)

        # Dead-letter queue consideration for final failures
        if retry_count >= 3:  # Example retry limit
