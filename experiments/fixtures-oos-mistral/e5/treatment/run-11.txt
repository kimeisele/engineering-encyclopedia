import logging
import uuid
from typing import Optional, Dict, Any

class CorrelationId:
    """Handles correlation ID generation and propagation across service calls."""

    @staticmethod
    def generate_correlation_id() -> str:
        """Generate a unique correlation ID for a new request."""
        return str(uuid.uuid4())

    @staticmethod
    def ensure_correlation_id(headers: Optional[Dict[str, str]] = None) -> str:
        """Ensure a correlation ID exists in headers, creating one if needed."""
        if headers is None:
            headers = {}
        return headers.get('X-Correlation-ID') or CorrelationId.generate_correlation_id()

class ErrorHandler:
    """Error handler that maintains context across two services for traceability."""

    def __init__(self):
        self.logger = logging.getLogger(__name__)

    def handle_service_error(
        self,
        service_name: str,
        operation: str,
        error: Exception,
        correlation_id: str,
        context: Dict[str, Any]
    ) -> None:
        """
        Handle an error in a service while preserving correlation and context.

        Args:
            service_name: Name of the service where error occurred
            operation: The operation being performed
            error: The exception that occurred
            correlation_id: The correlation ID for the request
            context: Additional context about the request state
        """
        # Include correlation ID in all log messages
        log_context = {
            'correlation_id': correlation_id,
            'service': service_name,
            'operation': operation,
            **context
        }

        # Create a detailed error message with context
        error_message = (
            f"Error in {service_name} during {operation}. "
            f"Correlation ID: {correlation_id}. "
            f"Context: {context}. "
            f"Original error: {str(error)}"
        )

        # Log the error with full context
        self.logger.error(error_message, extra=log_context)

        # If this is the second service in a chain, include first service's context
        if 'first_service_context' in context:
            self.logger.error(
                f"Request spanned multiple services. "
                f"First service context: {context['first_service_context']}",
                extra={'correlation_id': correlation_id}
            )

# Example usage in a two-service request flow
class ServiceA:
    def __init__(self):
        self.error_handler = ErrorHandler()
        self.correlation_id = CorrelationId()

    def make_request_to_service_b(self, request_data: Dict[str, Any]) -> Dict[str, Any]:
        correlation_id = self.correlation_id.ensure_correlation_id(request_data.get('headers'))

        try:
            # Simulate processing
            result = {"status": "success", "data": "processed"}
            return result
        except Exception as e:
            context = {
                'request_data': request_data,
                'step': 'service_a_processing'
            }
            self.error_handler.handle_service_error(
                service_name="ServiceA",
                operation="process_request",
                error=e,
                correlation_id=correlation_id,
                context=context
            )
            raise

class ServiceB:
    def __init__(self):
        self.error_handler = ErrorHandler()
        self.correlation_id = CorrelationId()

    def process_request(self, request_data: Dict[str, Any]) -> Dict[str, Any]:
        correlation_id = self.correlation_id.ensure_correlation_id(request_data.get('headers'))

        try:
            # Simulate processing that might fail
            if request_data.get('should_fail'):
                raise ValueError("Simulated processing error in ServiceB")

            result = {"status": "success", "data": "processed_by_b"}
            return result
        except Exception as e:
            context = {
                'request_data': request_data,
                'step': 'service_b_processing',
                'first_service_context': request_data.get('first_service_context')
            }
            self.error_handler.handle_service_error(
                service_name="ServiceB",
                operation="process_request",
                error=e,
                correlation_id=correlation_id,
                context=context
            )
            raise

# Example of a complete request flow
def handle_two_service_request(initial_request: Dict[str, Any]) -> Dict[str, Any]:
    service_a = ServiceA()
    service_b = ServiceB()

    try:
        # Service A processes first
        response_a = service_a.make_request_to_service_b(initial_request)

        # Pass correlation ID to Service B
        request_for_b = {
            'headers': {'X-Correlation-ID': response_a.get('correlation_id', '')},
            'data': 'for_service_b',
            'first_service_context': {
                'service_a_response': response_a,
                'processing_time': '2023-01-01T00:00:00Z'
            }
        }

        # Service B processes next
        return service_b.process_request(request_for_b)

    except Exception as e:
        # Top-level error handling would go here
        raise
