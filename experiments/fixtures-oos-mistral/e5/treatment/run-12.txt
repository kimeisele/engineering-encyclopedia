import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass

@dataclass
class ServiceRequest:
    correlation_id: str
    request_data: Dict[str, Any]
    service_name: str

class ErrorHandler:
    def __init__(self):
        self.logger = logging.getLogger(__name__)

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for the request."""
        return str(uuid.uuid4())

    def propagate_correlation_id(self, request: ServiceRequest) -> str:
        """Ensure correlation ID is present in all downstream calls."""
        if not request.correlation_id:
            request.correlation_id = self.generate_correlation_id()
        return request.correlation_id

    def handle_error(
        self,
        request: ServiceRequest,
        error: Exception,
        failing_step: str,
        additional_context: Optional[Dict[str, Any]] = None
    ) -> None:
        """
        Handle errors with full context for traceability.

        Args:
            request: The service request containing correlation_id
            error: The exception that occurred
            failing_step: Description of where failure occurred
            additional_context: Any extra context to include
        """
        # Ensure correlation ID exists
        correlation_id = self.propagate_correlation_id(request)

        # Build error context
        error_context = {
            "correlation_id": correlation_id,
            "service": request.service_name,
            "failing_step": failing_step,
            "request_data": request.request_data,
            **(additional_context or {})
        }

        # Log the error with full context
        self.logger.error(
            "Operation failed: %s in %s. Context: %s",
            str(error),
            failing_step,
            error_context,
            exc_info=True
        )

        # After max retries, send to dead-letter queue
        if hasattr(error, "retry_count") and error.retry_count >= 3:
