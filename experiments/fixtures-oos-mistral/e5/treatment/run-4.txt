import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass
from functools import wraps

# Configure logging to include correlation ID
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - [%(correlation_id)s] - %(message)s'
)
logger = logging.getLogger(__name__)

@dataclass
class RequestContext:
    correlation_id: str
    service_name: str
    operation: str
    request_data: Dict[str, Any]

class ErrorHandler:
    def __init__(self, service_name: str):
        self.service_name = service_name

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for a new request."""
        return str(uuid.uuid4())

    def propagate_correlation_id(self, headers: Optional[Dict[str, str]] = None) -> str:
        """Extract or generate correlation ID from headers."""
        if headers and 'X-Correlation-ID' in headers:
            return headers['X-Correlation-ID']
        return self.generate_correlation_id()

    def enrich_error_context(self, exc: Exception, context: RequestContext) -> Exception:
        """Enrich exception with operational context before propagation."""
        error_msg = (
            f"Operation '{context.operation}' failed in {self.service_name} "
            f"(correlation_id: {context.correlation_id}). "
            f"Request data: {context.request_data}. "
            f"Original error: {str(exc)}"
        )
        return Exception(error_msg) from exc

    def handle_service_error(
        self,
        exc: Exception,
        context: RequestContext,
        retry_attempts: int = 3,
        dead_letter_queue: Optional[str] = None
    ) -> None:
        """
        Handle errors from service calls with correlation tracing.

        Args:
            exc: The original exception
            context: Request context with correlation ID
            retry_attempts: Number of retry attempts before dead-lettering
            dead_letter_queue: Queue name for dead-lettered messages
        """
        if retry_attempts <= 0:
            if dead_letter_queue:
                logger.error(
                    f"Dead-lettering request after {retry_attempts} retries",
                    extra={'correlation_id': context.correlation_id}
                )
                # In real implementation, would send to dead-letter queue
            else:
                logger.error(
                    "Request failed after all retries",
                    extra={'correlation_id': context.correlation_id}
                )
            return

        # Enrich and log the error with full context
        enriched_exc = self.enrich_error_context(exc, context)
        logger.error(
            f"Service error occurred: {str(enriched_exc)}",
            exc_info=True,
            extra={'correlation_id': context.correlation_id}
        )

        # In real implementation, would implement retry logic here
        # For this example, we'll just log the retry attempt
        logger.info(
            f"Retrying operation (attempt {3 - retry_attempts + 1}/3)",
            extra={'correlation_id': context.correlation_id}
        )

    def trace_request(self, operation: str, request_data: Dict[str, Any]) -> RequestContext:
        """Create a new request context for tracing."""
        correlation_id = self.generate_correlation_id()
        return RequestContext(
            correlation_id=correlation_id,
            service_name=self.service_name,
            operation=operation,
            request_data=request_data
        )

# Example usage
def example_usage():
    handler = ErrorHandler(service_name="payment-service")

    # Simulate a request coming in with headers
    incoming_headers = {'X-Correlation-ID': 'abc-123'}

    # Create context for this request
    context = handler.trace_request(
        operation="process_payment",
        request_data={"amount": 100.00, "user_id": "user123"}
    )

    # Propagate correlation ID to downstream calls
    downstream_headers = {'X-Correlation-ID': context.correlation_id}

    try:
        # Simulate a service call that might fail
        raise ValueError("Invalid payment method")
    except Exception as e:
        # Handle the error with full context
        handler.handle_service_error(
            exc=e,
            context=context,
            retry_attempts=3,
            dead_letter_queue="payments-dlq"
        )

if __name__ == "__main__":
    example_usage()
