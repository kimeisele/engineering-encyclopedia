import uuid
import logging
from typing import Optional, Dict, Any
from functools import wraps

# Configure logging to include correlation ID
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class CorrelationIdMiddleware:
    """Middleware to generate and propagate correlation IDs across service calls."""

    @staticmethod
    def generate_correlation_id() -> str:
        """Generate a unique correlation ID for a new request."""
        return str(uuid.uuid4())

    @staticmethod
    def get_correlation_id(headers: Optional[Dict[str, str]] = None) -> str:
        """Extract correlation ID from headers or generate a new one."""
        if headers and 'X-Correlation-ID' in headers:
            return headers['X-Correlation-ID']
        return CorrelationIdMiddleware.generate_correlation_id()

    @staticmethod
    def inject_correlation_id(headers: Dict[str, str], correlation_id: str) -> Dict[str, str]:
        """Inject correlation ID into request headers."""
        headers['X-Correlation-ID'] = correlation_id
        return headers

class ErrorHandler:
    """Error handler that ensures correlation IDs and context are preserved across service calls."""

    @staticmethod
    def handle_error(
        exc: Exception,
        operation: str,
        correlation_id: str,
        context: Optional[Dict[str, Any]] = None
    ) -> None:
        """
        Handle an error with proper context and correlation ID.

        Args:
            exc: The exception that occurred
            operation: The operation being performed (e.g., "calling service B")
            correlation_id: The correlation ID for the request
            context: Additional context about the error
        """
        error_context = {
            'operation': operation,
            'correlation_id': correlation_id,
            **(context or {})
        }

        # Log the error with all relevant context
        logger.error(
            f"Operation failed: {operation}. Correlation ID: {correlation_id}",
            exc_info=True,
            extra={'error_context': error_context}
        )

        # Include context in the exception message
        error_msg = f"{operation} failed with correlation ID {correlation_id}"
        if context:
            error_msg += f". Context: {context}"
        raise RuntimeError(error_msg) from exc

def with_error_handling(operation_name: str):
    """
    Decorator to wrap service calls with error handling and correlation ID propagation.

    Args:
        operation_name: Name of the operation being performed (e.g., "calling service B")
    """
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # Extract correlation ID from headers or generate new one
            headers = kwargs.get('headers', {})
            correlation_id = CorrelationIdMiddleware.get_correlation_id(headers)

            try:
                # Inject correlation ID into outgoing request
                if 'headers' in kwargs:
                    kwargs['headers'] = CorrelationIdMiddleware.inject_correlation_id(
                        kwargs['headers'], correlation_id
                    )

                # Call the original function
                return func(*args, **kwargs)

            except Exception as e:
                # Handle the error with proper context
                ErrorHandler.handle_error(
                    exc=e,
                    operation=operation_name,
                    correlation_id=correlation_id,
                    context={
                        'args': args,
                        'kwargs': kwargs
                    }
                )
        return wrapper
    return decorator

    def call_service_b(self, request_data: Dict[str, Any], headers: Optional[Dict[str, str]] = None) -> Dict[str, Any]:
        """Example method that calls service B."""

        # with the correlation ID in headers
        return {"status": "success"}

    def handle_request(self, request_data: Dict[str, Any]) -> Dict[str, Any]:
        """Handle incoming request and propagate correlation ID."""
        correlation_id = CorrelationIdMiddleware.get_correlation_id(request_data.get('headers'))
        logger.info(f"Processing request with correlation ID: {correlation_id}", extra={
            'correlation_id': correlation_id
        })

        try:
            # Process request and call other services
            result = self.call_service_b(request_data, headers=request_data.get('headers'))
            return result
        except Exception as e:
            ErrorHandler.handle_error(
                exc=e,
                operation="processing request in service A",
                correlation_id=correlation_id,
                context={'request_data': request_data}
            )
