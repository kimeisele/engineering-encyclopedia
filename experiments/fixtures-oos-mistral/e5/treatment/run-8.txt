import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass

# Configure logging to include correlation ID in all log lines
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - [%(correlation_id)s] - %(message)s'
)
logger = logging.getLogger(__name__)

@dataclass
class RequestContext:
    """Holds correlation ID and other request-scoped data"""
    correlation_id: str
    service_name: str
    operation: str
    additional_context: Dict[str, Any] = None

def generate_correlation_id() -> str:
    """Generate a unique correlation ID for each logical request"""
    return str(uuid.uuid4())

def propagate_context(context: RequestContext, downstream_service: str) -> RequestContext:
    """Propagate context to downstream service with updated service name"""
    return RequestContext(
        correlation_id=context.correlation_id,
        service_name=downstream_service,
        operation=context.operation,
        additional_context=context.additional_context
    )

def error_handler(service_context: RequestContext, error: Exception) -> None:
    """
    Handle errors in a two-service request while maintaining traceability.

    Args:
        service_context: RequestContext containing correlation ID and service info
        error: The exception that occurred
    """
    # Include operation, failing step, and correlation ID in error message
    error_msg = (
        f"Operation '{service_context.operation}' failed in {service_context.service_name} "
        f"with correlation ID {service_context.correlation_id}. "
        f"Error: {str(error)}"
    )

    # Log the error with full context
    logger.error(
        error_msg,
        extra={
            'correlation_id': service_context.correlation_id,
            'service': service_context.service_name,
            'operation': service_context.operation,
            'error_type': type(error).__name__,
            'additional_context': service_context.additional_context
        }
    )

    # In a real implementation, you might also:
    # 1. Add to dead-letter queue with original context
    # 2. Trigger alerts with the correlation ID
    # 3. Include in error monitoring systems

def process_request(
    service_name: str,
    operation: str,
    request_data: Dict[str, Any],
    downstream_service: str,
    request_processor: callable
) -> Optional[Dict[str, Any]]:
    """
    Process a request across two services with error handling and correlation.

    Args:
        service_name: Name of current service
        operation: Name of operation being performed
        request_data: Input data for the request
        downstream_service: Name of service to call next
        request_processor: Function that processes the request and may call downstream

    Returns:
        Processed result or None if error occurred
    """
    # Generate or propagate correlation ID
    correlation_id = request_data.get('correlation_id') or generate_correlation_id()

    context = RequestContext(
        correlation_id=correlation_id,
        service_name=service_name,
        operation=operation,
        additional_context={
            'input_data': request_data,
            'downstream_service': downstream_service
        }
    )

    try:
        # Include correlation ID in all downstream calls
        propagated_context = propagate_context(context, downstream_service)

        # Process the request (may call downstream service)
        result = request_processor(propagated_context)

        return result

    except Exception as e:
        error_handler(context, e)
        return None

    if result is None:
