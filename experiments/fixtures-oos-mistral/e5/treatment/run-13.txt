import logging
import uuid
from typing import Optional, Dict, Any

# Configure logging to include correlation_id in all log lines
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - [%(correlation_id)s] - %(message)s'
)
logger = logging.getLogger(__name__)

class CorrelationIdFilter(logging.Filter):
    """Filter to inject correlation_id into log records."""

    def filter(self, record):
        record.correlation_id = getattr(record, 'correlation_id', 'none')
        return True

# Add filter to logger
logger.addFilter(CorrelationIdFilter())

class ServiceA:
    """Example service A that calls Service B."""

    def __init__(self):
        self.logger = logger.getChild("ServiceA")

    def call_service_b(self, payload: Dict[str, Any]) -> Dict[str, Any]:
        """Make a request to Service B with correlation ID propagation."""
        correlation_id = payload.get('correlation_id', str(uuid.uuid4()))

        try:
            self.logger.info("Calling Service B", extra={'correlation_id': correlation_id})
            # Simulate service call
            result = ServiceB().process_request(payload)
            return result
        except Exception as e:
            self.logger.error(
                "Service B call failed: operation=call_service_b step=upstream_request",
                extra={
                    'correlation_id': correlation_id,
                    'error': str(e),
                    'payload': payload
                },
                exc_info=True
            )
            raise

class ServiceB:
    """Example service B that receives requests from Service A."""

    def __init__(self):
        self.logger = logger.getChild("ServiceB")

    def process_request(self, payload: Dict[str, Any]) -> Dict[str, Any]:
        """Process a request with error handling and correlation ID propagation."""
        correlation_id = payload.get('correlation_id')
        if not correlation_id:
            correlation_id = str(uuid.uuid4())
            self.logger.warning(
                "No correlation_id provided, generated new one",
                extra={'correlation_id': correlation_id}
            )

        try:
            # Simulate processing
            if payload.get('fail'):
                raise ValueError("Simulated processing failure")

            self.logger.info(
                "Request processed successfully",
                extra={
                    'correlation_id': correlation_id,
                    'processed_data': payload.get('data')
                }
            )
            return {'status': 'success', 'correlation_id': correlation_id}

        except Exception as e:
            self.logger.error(
                "Request processing failed: operation=process_request step=data_processing",
                extra={
                    'correlation_id': correlation_id,
                    'payload': payload,
                    'error': str(e)
                },
                exc_info=True
            )
            raise

# Example usage
if __name__ == "__main__":
    try:
        # Service A initiates request with correlation ID
        request_payload = {
            'data': {'user_id': 123, 'action': 'update'},
            'correlation_id': str(uuid.uuid4()),
            'fail': False  # Set to True to simulate failure
        }

        result = ServiceA().call_service_b(request_payload)
        print("Success:", result)

    except Exception as e:
        logger.error(
            "Request failed at top level: operation=main step=orchestration",
            extra={'error': str(e)},
            exc_info=True
        )
