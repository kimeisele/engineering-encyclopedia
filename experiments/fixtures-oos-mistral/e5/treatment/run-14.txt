```python
import logging
import uuid
from typing import Optional, Dict, Any
from dataclasses import dataclass

@dataclass
class RequestContext:
    correlation_id: str
    service_name: str
    operation: str
    step: str
    state: Dict[str, Any]

class ErrorHandler:
    def __init__(self, service_name: str):
        self.service_name = service_name
        self.logger = logging.getLogger(service_name)

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for the request."""
        return str(uuid.uuid4())

    def extract_correlation_id(self, headers: Optional[Dict[str, str]] = None) -> str:
        """Extract correlation ID from headers or generate a new one."""
        if headers and 'X-Correlation-ID' in headers:
            return headers['X-Correlation-ID']
        return self.generate_correlation_id()

    def log_error(self, context: RequestContext, error: Exception) -> None:
        """Log error with full context for traceability."""
        error_msg = (
            f"Operation '{context.operation}' failed at step '{context.step}' "
            f"in service '{context.service_name}' with correlation ID '{context.correlation_id}'. "
            f"State: {context.state}. Error: {str(error)}"
        )
        self.logger.error(error_msg, exc_info=True)

    def handle_two_service_request(
        self,
        request_headers: Dict[str, str],
        service_a_call: callable,
        service_b_call: callable
    ) -> Dict[str, Any]:
        """Handle a request that spans two services with full error tracing."""
        correlation_id = self.extract_correlation_id(request_headers)

        try:
            # Service A processing
            context_a = RequestContext(
                correlation_id=correlation_id,
                service_name=f"{self.service_name}-A",
                operation="process_request",
                step="service_a",
                state={"input": request_headers}
            )

            result_a = service_a_call()
            context_a.state["result"] = result_a

            # Service B processing
            context_b = RequestContext(
                correlation_id=correlation_id,
                service_name=f"{self.service_name}-B",
                operation="process_request",
                step="service_b",
                state={"input": result_a}
            )

            result_b = service_b_call()
            context_b.state["result"] = result_b

            return result_b

        except Exception as e:
            # Log with full context
            context = RequestContext(
                correlation_id=correlation_id,
                service_name=self.service_name,
                operation="process_request",
                step="error_handler",
                state={"error_type": type(e).__name__}
            )
            self.log_error(context, e)
            raise
```
