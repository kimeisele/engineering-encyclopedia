import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass

# Configure logging to include correlation ID in all log lines
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - [%(correlation_id)s] - %(message)s'
)
logger = logging.getLogger(__name__)

@dataclass
class RequestContext:
    correlation_id: str
    service_name: str
    operation: str
    request_data: Dict[str, Any]

class ErrorHandler:
    def __init__(self):
        self.logger = logger

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for a new request."""
        return str(uuid.uuid4())

    def extract_correlation_id(self, headers: Optional[Dict[str, str]] = None) -> str:
        """Extract correlation ID from headers or generate a new one if missing."""
        if headers and 'X-Correlation-ID' in headers:
            return headers['X-Correlation-ID']
        return self.generate_correlation_id()

    def attach_error_context(
        self,
        exc: Exception,
        context: RequestContext,
        headers: Optional[Dict[str, str]] = None
    ) -> Exception:
        """
        Enhance an exception with context about the failing operation and request.
        Preserves the original exception while adding operational context.
        """
        error_msg = (
            f"Operation '{context.operation}' failed in {context.service_name} "
            f"with correlation ID {context.correlation_id}. "
            f"Request data: {context.request_data}"
        )

        # Create a new exception that wraps the original
        enhanced_exc = Exception(error_msg) from exc

        # Add correlation ID to the exception for easy extraction
        setattr(enhanced_exc, 'correlation_id', context.correlation_id)
        setattr(enhanced_exc, 'service_name', context.service_name)

        return enhanced_exc

    def handle_service_error(
        self,
        service_name: str,
        operation: str,
        request_data: Dict[str, Any],
        headers: Optional[Dict[str, str]] = None,
        max_retries: int = 3
    ) -> Any:
        """
        Handle errors from a service call with proper tracing and context.
        Returns the successful response or raises an enhanced exception.
        """
        correlation_id = self.extract_correlation_id(headers)
        context = RequestContext(
            correlation_id=correlation_id,
            service_name=service_name,
            operation=operation,
            request_data=request_data
        )

        # Log the start of the operation with correlation ID
        self.logger.info(
            f"Starting operation '{operation}'",
            extra={'correlation_id': correlation_id}
        )

        last_exception = None

        for attempt in range(max_retries):
            try:
                # Simulate service call
                response = self._call_service(service_name, operation, request_data, headers)
                self.logger.info(
                    f"Operation '{operation}' completed successfully",
                    extra={'correlation_id': correlation_id}
                )
                return response
            except Exception as exc:
                last_exception = exc
                self.logger.warning(
                    f"Attempt {attempt + 1} failed for operation '{operation}'",
                    exc_info=True,
                    extra={'correlation_id': correlation_id}
                )

                # If this was the last attempt, enhance the error with context
                if attempt == max_retries - 1:
                    enhanced_exc = self.attach_error_context(exc, context, headers)
                    raise enhanced_exc

        # This line should theoretically never be reached
        raise Exception("Unexpected error in error handler") from last_exception

    def _call_service(
        self,
        service_name: str,
        operation: str,
        request_data: Dict[str, Any],
        headers: Optional[Dict[str, str]] = None
    ) -> Any:
        """Simulate calling another service with proper header propagation."""
        # In a real implementation, this would make an HTTP call or similar
        # For this example, we'll just simulate a potential failure
        if "fail" in request_data.get("mode", ""):
            raise Exception("Simulated service failure")

        # Propagate correlation ID to downstream service
        if headers is None:
            headers = {}
        headers['X-Correlation-ID'] = headers.get('X-Correlation-ID', self.generate_correlation_id())

        return {"status": "success", "data": "response"}
