import logging
import uuid
from typing import Optional, Dict, Any
from dataclasses import dataclass

@dataclass
class ServiceError:
    """Container for error information with correlation ID and context."""
    correlation_id: str
    service_name: str
    operation: str
    error_message: str
    context: Dict[str, Any]
    original_exception: Optional[Exception] = None

class TwoServiceErrorHandler:
    """
    Error handler for two-service requests that maintains correlation IDs
    and provides traceability across both services.
    """

    def __init__(self, service_name: str):
        self.service_name = service_name
        self.logger = logging.getLogger(service_name)

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for a new request."""
        return str(uuid.uuid4())

    def extract_correlation_id(self, headers: Dict[str, str]) -> Optional[str]:
        """Extract correlation ID from incoming request headers."""
        return headers.get('X-Correlation-ID')

    def propagate_correlation_id(self, headers: Dict[str, str], correlation_id: str) -> Dict[str, str]:
        """Add correlation ID to outgoing request headers."""
        headers['X-Correlation-ID'] = correlation_id
        return headers

    def handle_error(
        self,
        operation: str,
        error: Exception,
        correlation_id: Optional[str] = None,
        context: Optional[Dict[str, Any]] = None
    ) -> ServiceError:
        """
        Create a structured error with all necessary context for tracing.

        Args:
            operation: The operation being performed when error occurred
            error: The exception that was raised
            correlation_id: Existing correlation ID if available
            context: Additional context about the request state

        Returns:
            ServiceError with all error information
        """
        # Generate correlation ID if not provided
        if not correlation_id:
            correlation_id = self.generate_correlation_id()

        # Build error context
        error_context = {
            'service': self.service_name,
            'operation': operation,
            'error_type': type(error).__name__,
            'error_message': str(error),
            **(context or {})
        }

        # Log the error with correlation ID
        self.logger.error(
            f"Service error in {operation}: {str(error)}",
            extra={
                'correlation_id': correlation_id,
                **error_context
            }
        )

        return ServiceError(
            correlation_id=correlation_id,
            service_name=self.service_name,
            operation=operation,
            error_message=str(error),
            context=error_context,
            original_exception=error
        )

    def handle_dead_letter(
        self,
        original_error: ServiceError,
        retry_count: int,
        max_retries: int
    ) -> None:
        """
        Handle a message that has exhausted retry attempts.

        Args:
            original_error: The original error that occurred
            retry_count: How many times we retried
            max_retries: Maximum allowed retries
        """
        dlq_context = {
            **original_error.context,
            'retry_count': retry_count,
            'max_retries': max_retries,
            'dead_letter_action': 'sent_to_dlq'
        }

        self.logger.critical(
            f"Message moved to dead letter queue after {retry_count}/{max_retries} retries",
