import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass

@dataclass
class RequestContext:
    correlation_id: str
    service_name: str
    operation: str
    request_data: Dict[str, Any]

class ErrorHandler:
    def __init__(self, service_name: str):
        self.service_name = service_name
        self.logger = logging.getLogger(service_name)

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for the request."""
        return str(uuid.uuid4())

    def extract_correlation_id(self, headers: Optional[Dict[str, str]] = None) -> str:
        """Extract correlation ID from headers or generate a new one."""
        if headers and 'X-Correlation-ID' in headers:
            return headers['X-Correlation-ID']
        return self.generate_correlation_id()

    def log_error_with_context(
        self,
        error: Exception,
        context: RequestContext,
        additional_context: Optional[Dict[str, Any]] = None
    ) -> None:
        """Log an error with full context for traceability."""
        error_context = {
            'service': context.service_name,
            'operation': context.operation,
            'correlation_id': context.correlation_id,
            'request_data': context.request_data,
            **(additional_context or {})
        }

        error_message = (
            f"Operation '{context.operation}' failed in {context.service_name} "
            f"(correlation_id: {context.correlation_id}). "
            f"Error: {str(error)}"
        )

        self.logger.error(error_message, extra=error_context)

    def handle_service_error(
        self,
        error: Exception,
        headers: Optional[Dict[str, str]] = None,
        request_data: Optional[Dict[str, Any]] = None,
        operation: str = "unknown_operation"
    ) -> None:
        """Handle errors in a two-service request with proper correlation tracking."""
        correlation_id = self.extract_correlation_id(headers)
        context = RequestContext(
            correlation_id=correlation_id,
            service_name=self.service_name,
            operation=operation,
            request_data=request_data or {}
        )

        self.log_error_with_context(error, context)

        # Additional handling could include:
        # - Dead letter queue processing
        # - Circuit breaker state updates
        # - Metrics collection
