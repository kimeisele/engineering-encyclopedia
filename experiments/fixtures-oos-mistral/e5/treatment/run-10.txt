import uuid
import logging
from typing import Optional, Dict, Any
from dataclasses import dataclass

# Configure logging to include correlation ID
logging.basicConfig(
    format='%(asctime)s - %(name)s - %(levelname)s - [%(correlation_id)s] - %(message)s'
)
logger = logging.getLogger(__name__)

@dataclass
class ServiceRequest:
    correlation_id: str
    payload: Dict[str, Any]
    service_a_response: Optional[Dict[str, Any]] = None
    service_b_response: Optional[Dict[str, Any]] = None

class CorrelationIdMissingError(Exception):
    """Raised when a request lacks a correlation ID"""
    pass

class ServiceA:
    def call(self, request: ServiceRequest) -> Dict[str, Any]:
        logger.info("Calling Service A", extra={'correlation_id': request.correlation_id})
        # Simulate service call
        return {"status": "success", "data": "from_service_a"}

class ServiceB:
    def call(self, request: ServiceRequest) -> Dict[str, Any]:
        logger.info("Calling Service B", extra={'correlation_id': request.correlation_id})
        # Simulate service call
        return {"status": "success", "data": "from_service_b"}

class ErrorHandler:
    def __init__(self):
        self.service_a = ServiceA()
        self.service_b = ServiceB()

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for the request"""
        return str(uuid.uuid4())

    def ensure_correlation_id(self, request: Optional[Dict[str, Any]]) -> str:
        """Extract or generate correlation ID from request"""
        if not request:
            return self.generate_correlation_id()

        correlation_id = request.get('correlation_id')
        if not correlation_id:
            correlation_id = self.generate_correlation_id()
            request['correlation_id'] = correlation_id

        return correlation_id

    def handle_request(self, request_data: Dict[str, Any]) -> ServiceRequest:
        """Handle a request across both services with error handling"""
        try:
            # Ensure we have a correlation ID
            correlation_id = self.ensure_correlation_id(request_data)

            request = ServiceRequest(
                correlation_id=correlation_id,
                payload=request_data
            )

            # Call Service A
            try:
                request.service_a_response = self.service_a.call(request)
            except Exception as e:
                logger.error(
                    f"Service A failed: {str(e)}",
                    extra={
                        'correlation_id': correlation_id,
                        'failing_step': 'service_a_call',
                        'request_payload': request_data
                    },
                    exc_info=True
                )
                raise

            # Call Service B
            try:
                request.service_b_response = self.service_b.call(request)
            except Exception as e:
                logger.error(
                    f"Service B failed: {str(e)}",
                    extra={
                        'correlation_id': correlation_id,
                        'failing_step': 'service_b_call',
                        'request_payload': request_data,
                        'service_a_response': request.service_a_response
                    },
                    exc_info=True
                )
                raise

            logger.info(
                "Request completed successfully",
                extra={
                    'correlation_id': correlation_id,
                    'service_a_response': request.service_a_response,
                    'service_b_response': request.service_b_response
                }
            )
            return request

        except Exception as e:
            logger.critical(
                f"Request failed: {str(e)}",
                extra={
                    'correlation_id': correlation_id if 'correlation_id' in locals() else 'unknown',
                    'request_payload': request_data if 'request_data' in locals() else None
                },
                exc_info=True
            )
            raise
