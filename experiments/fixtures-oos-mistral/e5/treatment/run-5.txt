import logging
import uuid
from typing import Optional, Dict, Any
from dataclasses import dataclass

@dataclass
class ServiceRequest:
    correlation_id: str
    service_a_data: Dict[str, Any]
    service_b_data: Optional[Dict[str, Any]] = None
    error: Optional[Exception] = None

class ErrorHandler:
    def __init__(self):
        self.logger = logging.getLogger(__name__)

    def generate_correlation_id(self) -> str:
        """Generate a unique correlation ID for tracing requests across services."""
        return str(uuid.uuid4())

    def handle_error(
        self,
        request: ServiceRequest,
        service_name: str,
        error: Exception,
        context: Dict[str, Any]
    ) -> ServiceRequest:
        """
        Handle an error in a two-service request, preserving correlation ID and context.

        Args:
            request: The original service request with correlation ID
            service_name: Name of the service where error occurred ('service_a' or 'service_b')
            error: The exception that occurred
            context: Additional context about the operation and state

        Returns:
            Updated request with error information and context
        """
        # Preserve the original correlation ID for traceability
        request.error = error

        # Add error context before wrapping the exception
        error_context = {
            "correlation_id": request.correlation_id,
            "service": service_name,
            "operation": f"{service_name}_operation",
            "state": context,
            "error_type": type(error).__name__
        }

        # Create a new exception with context
        wrapped_error = Exception(f"Service {service_name} failed: {str(error)}") from error
        wrapped_error.__dict__.update(error_context)

        # Log with correlation ID and all relevant context
        self.logger.error(
            "Request failed",
            exc_info=wrapped_error,
            extra={
                "correlation_id": request.correlation_id,
                "service": service_name,
                **context
            }
        )

        return request

    def handle_dead_letter(self, request: ServiceRequest) -> None:
        """
        Handle a message that should go to dead-letter queue after retries exhausted.

        Args:
            request: The failed request to be dead-lettered
        """
        dlq_context = {
            "correlation_id": request.correlation_id,
            "final_state": "dead_letter",
            "original_error": str(request.error) if request.error else "Unknown error"
        }

        self.logger.critical(
            "Message sent to dead-letter queue",
            extra=dlq_context
        )

        # Additional operator notification would go here
        print(f"DEAD LETTER: Correlation ID {request.correlation_id} - {dlq_context}")

# Example usage
if __name__ == "__main__":
    handler = ErrorHandler()

    # Service A receives a request
    req = ServiceRequest(
        correlation_id=handler.generate_correlation_id(),
        service_a_data={"user_id": "123", "action": "process"}
    )

    try:
        # Simulate service A processing
        req.service_b_data = {"status": "pending"}

        # Simulate service B call
        try:
            raise ValueError("Database connection failed")
        except Exception as e:
            req = handler.handle_error(
                request=req,
                service_name="service_b",
                error=e,
                context={"step": "database_call", "retry_count": 3}
            )

        # After retries exhausted, send to dead letter
        handler.handle_dead_letter(req)

    except Exception as e:
        req = handler.handle_error(
            request=req,
            service_name="service_a",
            error=e,
            context={"step": "initial_validation"}
        )
        handler.handle_dead_letter(req)
