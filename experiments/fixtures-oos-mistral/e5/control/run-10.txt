# Error Handler for Cross-Service Request Tracing

Below is a comprehensive error handling solution that allows operators to trace a single request as it flows through two services. This implementation includes:

1. Correlation IDs for request tracking
2. Structured logging with context
3. Error propagation between services
4. Centralized error handling
5. Metrics collection

## Implementation (Python Example)

```python
import uuid
import logging
from typing import Dict, Any, Optional
from datetime import datetime
import json
import requests
from requests.exceptions import RequestException

# Configure structured logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.StreamHandler(),
        logging.FileHandler('service_errors.log')
    ]
)
logger = logging.getLogger(__name__)

class ServiceError(Exception):
    """Base exception for service errors"""
    def __init__(self, message: str, original_exception: Optional[Exception] = None):
        super().__init__(message)
        self.original_exception = original_exception
        self.timestamp = datetime.utcnow().isoformat()

class ServiceAError(ServiceError):
    """Errors specific to Service A"""
    pass

class ServiceBError(ServiceError):
    """Errors specific to Service B"""
    pass

class RequestTracer:
    """Handles request tracing across services"""

    @staticmethod
    def generate_correlation_id() -> str:
        """Generate a unique correlation ID for request tracing"""
        return str(uuid.uuid4())

    @staticmethod
    def log_request_start(correlation_id: str, service_name: str, request_data: Dict[str, Any]):
        """Log the start of a request with correlation ID"""
        logger.info(
            f"Request started - Service: {service_name} - Correlation ID: {correlation_id}",
            extra={
                'correlation_id': correlation_id,
                'service': service_name,
                'request_data': request_data,
                'status': 'started'
            }
        )

    @staticmethod
    def log_request_end(correlation_id: str, service_name: str, response_data: Dict[str, Any], duration: float):
        """Log the completion of a request"""
        logger.info(
            f"Request completed - Service: {service_name} - Correlation ID: {correlation_id}",
            extra={
                'correlation_id': correlation_id,
                'service': service_name,
                'response_data': response_data,
                'duration_ms': duration * 1000,
                'status': 'completed'
            }
        )

    @staticmethod
    def log_error(correlation_id: str, service_name: str, error: ServiceError):
        """Log an error with full context"""
        error_type = type(error).__name__
        error_message = str(error)

        logger.error(
            f"Error in {service_name} - Correlation ID: {correlation_id} - Error: {error_type}: {error_message}",
            extra={
                'correlation_id': correlation_id,
                'service': service_name,
                'error_type': error_type,
                'error_message': error_message,
                'original_exception': str(error.original_exception) if error.original_exception else None,
                'timestamp': error.timestamp,
                'status': 'error'
            }
        )

class ServiceAClient:
    """Client for Service A with error handling"""

    def __init__(self, base_url: str):
        self.base_url = base_url
        self.session = requests.Session()

    def call_service_a(self, data: Dict[str, Any], correlation_id: str) -> Dict[str, Any]:
        """Make a request to Service A with error handling"""
        start_time = datetime.utcnow()

        try:
            RequestTracer.log_request_start(correlation_id, "ServiceA", data)

            headers = {
                'X-Correlation-ID': correlation_id,
                'Content-Type': 'application/json'
            }

            response = self.session.post(
                f"{self.base_url}/api/v1/process",
                json=data,
                headers=headers,
                timeout=10
            )

            response.raise_for_status()

            result = response.json()
            duration = (datetime.utcnow() - start_time).total_seconds()

            RequestTracer.log_request_end(correlation_id, "ServiceA", result, duration)
            return result

        except RequestException as e:
            duration = (datetime.utcnow() - start_time).total_seconds()
            error = ServiceAError(f"Service A request failed: {str(e)}", e)
            RequestTracer.log_error(correlation_id, "ServiceA", error)
            raise error

class ServiceBClient:
    """Client for Service B with error handling"""

    def __init__(self, base_url: str):
        self.base_url = base_url
        self.session = requests.Session()

    def call_service_b(self, data: Dict[str, Any], correlation_id: str) -> Dict[str, Any]:
        """Make a request to Service B with error handling"""
        start_time = datetime.utcnow()

        try:
            RequestTracer.log_request_start(correlation_id, "ServiceB", data)

            headers = {
                'X-Correlation-ID': correlation_id,
                'Content-Type': 'application/json'
            }

            response = self.session.post(
                f"{self.base_url}/api/v1/transform",
                json=data,
                headers=headers,
                timeout=10
            )

            response.raise_for_status()

            result = response.json()
            duration = (datetime.utcnow() - start_time).total_seconds()

            RequestTracer.log_request_end(correlation_id, "ServiceB", result, duration)
            return result

        except RequestException as e:
            duration = (datetime.utcnow() - start_time).total_seconds()
            error = ServiceBError(f"Service B request failed: {str(e)}", e)
            RequestTracer.log_error(correlation_id, "ServiceB", error)
            raise error

class Orchestrator:
    """Orchestrates calls between Service A and Service B with error handling"""

    def __init__(self, service_a_url: str, service_b_url: str):
        self.service_a = ServiceAClient(service_a_url)
        self.service_b = ServiceBClient(service_b_url)

    def process_request(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """Process a request through both services with full tracing"""
        correlation_id = RequestTracer.generate_correlation_id()

        try:
            # Step 1: Call Service A
            service_a_response = self.service_a.call_service_a(input_data, correlation_id)

            # Step 2: Call Service B with the response from Service A
            service_b_response = self.service_b.call_service_b(service_a_response, correlation_id)

            return {
                'status': 'success',
                'correlation_id': correlation_id,
                'service_a_response': service_a_response,
                'service_b_response': service_b_response
            }

        except ServiceAError as e:
            # If Service A fails, we still want to log the correlation ID
            logger.error(
                f"Orchestration failed at Service A - Correlation ID: {correlation_id}",
                extra={
                    'correlation_id': correlation_id,
                    'error_type': 'ServiceAError',
                    'error_message': str(e),
                    'original_exception': str(e.original_exception) if e.original_exception else None,
                    'status': 'failed'
                }
            )
            return {
                'status': 'failed',
                'correlation_id': correlation_id,
                'error': 'Service A processing failed',
                'original_error': str(e)
            }

        except ServiceBError as e:
            # If Service B fails, we want to know what happened in Service A
            logger.error(
                f"Orchestration failed at Service B - Correlation ID: {correlation_id}",
                extra={
                    'correlation_id': correlation_id,
                    'error_type': 'ServiceBError',
                    'error_message': str(e),
                    'original_exception': str(e.original_exception) if e.original_exception else None,
                    'service_a_response': service_a_response if 'service_a_response' in locals() else None,
                    'status': 'failed'
                }
            )
            return {
                'status': 'failed',
                'correlation_id': correlation_id,
                'error': 'Service B processing failed',
                'service_a_response': service_a_response if 'service_a_response' in locals() else None,
                'original_error': str(e)
            }

        except Exception as e:
            logger.error(
                f"Unexpected error in orchestration - Correlation ID: {correlation_id}",
                extra={
                    'correlation_id': correlation_id,
                    'error_type': type(e).__name__,
                    'error_message': str(e),
                    'status': 'failed'
                }
            )
            return {
                'status': 'failed',
                'correlation_id': correlation_id,
                'error': 'Unexpected processing error',
                'original_error': str(e)
            }
```

## Key Features

1. **Correlation IDs**:
   - Unique ID generated for each request
   - Propagated through both services
   - Included in all logs and error messages

2. **Structured Logging**:
   - Consistent log format across services
   - Additional context in each log entry
   - Separate log files for errors

3. **Error Handling**:
   - Service-specific error classes
   - Original exception preservation
   - Timestamps for all errors
   - Duration tracking for performance monitoring

4. **Request Flow Tracking**:
   - Logs at each service boundary
   - Tracks request start and completion
   - Captures response data at each step

5. **Orchestration Layer**:
   - Centralized request processing
   - Consistent error responses
   - Preserves context across service calls

## Usage Example

```python
# Initialize the orchestrator
orchestrator = Orchestrator(
    service_a_url="http://service-a.internal",
    service_b_url="http://service-b.internal"
)

# Process a request
try:
    result = orchestrator.process_request({
        "user_id": "12345",
        "data": {"action": "process"}
    })

    if result['status'] == 'success':
        print(f"Success! Correlation ID: {result['correlation_id']}")
    else:
        print(f"Failed with error: {result['error']}")

except Exception as e:
    print(f"Unexpected error: {str(e)}")
```

## Operator Queries

Operators can now query logs using the correlation ID to trace a request:

```bash
# Find all logs for a specific request
grep "correlation_id: abc123-def456" service_errors.log

# Count errors by type
grep "status: error" service_errors.log | awk '{print $10}' | sort | uniq -c

# Find slow requests
grep "duration_ms" service_errors.log | awk -F'duration_ms": ' '{print $2}' | sort -n
```

This implementation provides comprehensive tracing while maintaining clean separation between services and keeping the error handling consistent across the entire request flow.
