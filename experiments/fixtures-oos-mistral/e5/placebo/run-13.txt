import logging
import uuid
from typing import Dict, Any, Optional

logger = logging.getLogger(__name__)

class TraceableError(Exception):
    """Base class for errors that can be traced across services."""
    def __init__(self, message: str, trace_id: str, context: Dict[str, Any]):
        super().__init__(message)
        self.trace_id = trace_id
        self.context = context
        self.message = f"{message} [trace_id={trace_id}]"

def handle_two_service_request(
    service_a_request: Dict[str, Any],
    service_a_call: callable,
    service_b_call: callable
) -> Dict[str, Any]:
    """
    Execute a request across two services with traceable error handling.

    Args:
        service_a_request: Initial request data
        service_a_call: Function to call service A
        service_b_call: Function to call service B

    Returns:
        Combined result from both services

    Raises:
        TraceableError: When either service fails with trace context
    """
    trace_id = service_a_request.get('trace_id', str(uuid.uuid4()))

    try:
        # Call Service A
        service_a_result = service_a_call(service_a_request)
        service_a_request['service_a_result'] = service_a_result

        # Call Service B with context from Service A
        service_b_request = {
            **service_a_request,
            'operation': 'service_b_processing',
            'failing_step': 'service_b_call'
        }
        service_b_result = service_b_call(service_b_request)

        return {
            'trace_id': trace_id,
            'service_a_result': service_a_result,
            'service_b_result': service_b_result
        }

    except Exception as e:
        # Preserve original context when wrapping exceptions
        context = {
            'operation': 'two_service_request',
            'failing_step': 'service_a_call' if not service_a_result else 'service_b_call',
            'request_id': trace_id,
            'service_a_input': service_a_request
        }

        if not isinstance(e, TraceableError):
            e = TraceableError(
                str(e),
                trace_id,
                context
            )

        # Log with all available context
        logger.error(
            "Request failed across services",
            extra={
                'trace_id': trace_id,
                'error': str(e),
                'context': context,
                'service_a_result': service_a_result if 'service_a_result' in locals() else None
            }
        )

        raise
