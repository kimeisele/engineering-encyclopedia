import logging
from typing import Dict, Any, Optional
import uuid

class TraceableError(Exception):
    """Base exception class for traceable errors across services."""

    def __init__(self, message: str, context: Dict[str, Any]):
        super().__init__(message)
        self.context = context
        self.trace_id = context.get('trace_id', str(uuid.uuid4()))

    def __str__(self):
        base_msg = super().__str__()
        return f"{base_msg} [trace_id={self.trace_id}]"

class ServiceAError(TraceableError):
    """Error handler for Service A operations."""

    def __init__(self, message: str, context: Dict[str, Any]):
        context['service'] = 'ServiceA'
        context['step'] = context.get('step', 'unknown')
        super().__init__(message, context)

class ServiceBError(TraceableError):
    """Error handler for Service B operations."""

    def __init__(self, message: str, context: Dict[str, Any]):
        context['service'] = 'ServiceB'
        context['step'] = context.get('step', 'unknown')
        super().__init__(message, context)

def handle_cross_service_error(
    service_a_error: Optional[Exception],
    service_b_error: Optional[Exception],
    request_context: Dict[str, Any]
) -> TraceableError:
    """
    Combine errors from two services into a single traceable error.

    Args:
        service_a_error: Exception from Service A (if any)
        service_b_error: Exception from Service B (if any)
        request_context: Dictionary containing request metadata

    Returns:
        A TraceableError with combined context from both services
    """
    if service_a_error and service_b_error:
        error_msg = f"Cross-service failure: ServiceA failed during '{request_context.get('service_a_step')}', " \
                   f"ServiceB failed during '{request_context.get('service_b_step')}'"
    elif service_a_error:
        error_msg = f"ServiceA failed during '{request_context.get('service_a_step')}'"
    else:
        error_msg = f"ServiceB failed during '{request_context.get('service_b_step')}'"

    combined_context = {
        **request_context,
        'service_a_error': str(service_a_error) if service_a_error else None,
        'service_b_error': str(service_b_error) if service_b_error else None
    }

    return TraceableError(error_msg, combined_context)

# Example usage
if __name__ == "__main__":
    try:
        # Simulate Service A operation
        raise ServiceAError("Database connection failed", {'request_id': 'req123', 'step': 'db_connect'})
    except ServiceAError as e:
        service_a_error = e
        try:
            # Simulate Service B operation
            raise ServiceBError("Payment processing failed", {'request_id': 'req123', 'step': 'process_payment'})
        except ServiceBError as e:
            service_b_error = e
            combined_error = handle_cross_service_error(service_a_error, service_b_error, {'request_id': 'req123'})
            logging.error("Cross-service error occurred", exc_info=combined_error)
