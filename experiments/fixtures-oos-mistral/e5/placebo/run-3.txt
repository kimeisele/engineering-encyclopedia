import logging
from typing import Dict, Any, Optional
import uuid

class TraceableError(Exception):
    """Base exception class for traceable errors across services."""

    def __init__(self, message: str, context: Optional[Dict[str, Any]] = None):
        super().__init__(message)
        self.context = context or {}
        self.trace_id = self.context.get('trace_id', str(uuid.uuid4()))
        self.request_id = self.context.get('request_id')
        self.service = self.context.get('service')
        self.step = self.context.get('step')
        self.operation = self.context.get('operation')

    def __str__(self) -> str:
        base = super().__str__()
        context_str = ", ".join(f"{k}={v}" for k, v in self.context.items())
        return f"{base} [trace_id={self.trace_id}, request_id={self.request_id}, service={self.service}, step={self.step}, operation={self.operation}, context={context_str}]"

class ServiceErrorHandler:
    """Handler for errors across two services to maintain traceability."""

    def __init__(self, service_name: str):
        self.service_name = service_name
        self.logger = logging.getLogger(f"{service_name}.error_handler")

    def handle_error(
        self,
        error: Exception,
        operation: str,
        request_id: Optional[str] = None,
        step: Optional[str] = None,
        additional_context: Optional[Dict[str, Any]] = None
    ) -> TraceableError:
        """
        Wrap an error with trace context to maintain cross-service traceability.

        Args:
            error: The original exception
            operation: The operation being performed (e.g., "process_payment")
            request_id: Unique identifier for the request
            step: The specific step where failure occurred
            additional_context: Any additional context to include

        Returns:
            A TraceableError with full context
        """
        context = {
            "operation": operation,
            "request_id": request_id or str(uuid.uuid4()),
            "service": self.service_name,
            "step": step or "unknown",
            **(additional_context or {})
        }

        if isinstance(error, TraceableError):
            # Preserve existing trace context and add new step
            error.context.update(context)
            error.step = step or error.step
            self.logger.error("Error in %s at step %s: %s", operation, step, error)
            return error

        trace_error = TraceableError(str(error), context)
        self.logger.error("Error in %s at step %s: %s", operation, step, trace_error)
        return trace_error

    def propagate_error(
        self,
        error: TraceableError,
        next_service: str,
        next_step: str
    ) -> TraceableError:
        """
        Propagate a traceable error to another service.

        Args:
            error: The traceable error to propagate
            next_service: Name of the next service
            next_step: The step in the next service

        Returns:
            The updated error with new service context
        """
        error.context.update({
            "service": next_service,
            "step": next_step
        })
        error.service = next_service
        error.step = next_step
        return error

# Example usage:
if __name__ == "__main__":
    handler = ServiceErrorHandler("payment_service")

    try:
        # Simulate an error in service A
        raise ValueError("Invalid payment amount")
    except Exception as e:
        trace_error = handler.handle_error(
            e,
            operation="process_payment",
            request_id="req_12345",
            step="validate_payment"
        )

    # Propagate to service B
    propagated_error = handler.propagate_error(
        trace_error,
        next_service="fraud_detection",
        next_step="check_risk"
    )

    # Log the propagated error
    logging.error("Propagated error: %s", propagated_error)
