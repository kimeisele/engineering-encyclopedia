import logging
from typing import Dict, Any, Optional
import uuid

class RequestTracingErrorHandler:
    """
    Error handler for tracing requests across two services.
    Preserves context through wrapping layers and includes operation identifiers.
    """

    def __init__(self):
        self.logger = logging.getLogger("request_tracing")

    def handle_service_error(
        self,
        service_name: str,
        operation: str,
        request_id: str,
        step: str,
        error: Exception,
        context: Optional[Dict[str, Any]] = None
    ) -> Exception:
        """
        Wraps an error with tracing context before propagating it.

        Args:
            service_name: Name of the service where error occurred
            operation: The high-level operation being performed
            request_id: Unique identifier for the request
            step: Specific step where failure occurred
            error: Original exception
            context: Additional context to preserve

        Returns:
            Wrapped exception with tracing context
        """
        if context is None:
            context = {}

        # Preserve original error context before wrapping
        error_context = {
            "service": service_name,
            "operation": operation,
            "request_id": request_id,
            "step": step,
            "original_error": str(error),
            **context
        }

        # Create new exception with context
        wrapped_error = RequestTracingError(
            f"Service {service_name} failed during {operation} at {step} for request {request_id}",
            error_context
        )

        # Preserve the original exception chain
        wrapped_error.__cause__ = error
        wrapped_error.__context__ = error.__context__

        return wrapped_error

    def log_error(self, error: Exception) -> None:
        """
        Logs an error with all available tracing context.
        """
        if isinstance(error, RequestTracingError):
            self.logger.error(
                "%s | Request: %s | Operation: %s | Step: %s",
                error.message,
                error.context.get("request_id"),
                error.context.get("operation"),
                error.context.get("step"),
                exc_info=True,
                extra={"error_context": error.context}
            )
        else:
            self.logger.error(
                "Untraced error occurred: %s",
                str(error),
                exc_info=True
            )

class RequestTracingError(Exception):
    """Custom exception that preserves tracing context through wrapping layers."""

    def __init__(self, message: str, context: Dict[str, Any]):
        super().__init__(message)
        self.message = message
        self.context = context

# Example usage:
if __name__ == "__main__":
    handler = RequestTracingErrorHandler()

    try:
        # Simulate service A operation
        request_id = str(uuid.uuid4())
        try:
            # Some operation that fails
            raise ValueError("Invalid input data")
        except ValueError as e:
            # Wrap with tracing context
            wrapped_error = handler.handle_service_error(
                service_name="service_a",
                operation="process_payment",
                request_id=request_id,
                step="validate_input",
                error=e
            )
            raise wrapped_error

    except Exception as e:
        handler.log_error(e)
