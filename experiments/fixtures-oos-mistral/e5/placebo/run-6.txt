import logging
from typing import Dict, Any, Optional
import uuid

logger = logging.getLogger(__name__)

class TraceableError(Exception):
    """Error with trace context for cross-service tracing"""
    def __init__(self, message: str, context: Dict[str, Any]):
        super().__init__(message)
        self.context = context
        self.trace_id = context.get('trace_id', str(uuid.uuid4()))
        self.request_id = context.get('request_id')
        self.service = context.get('service')
        self.operation = context.get('operation')
        self.step = context.get('step')

    def __str__(self):
        base = f"{self.operation} failed at {self.step}: {super().__str__()}"
        if self.request_id:
            base += f" (request: {self.request_id})"
        return base

def handle_two_service_request(
    service1_func: callable,
    service2_func: callable,
    request_id: str,
    operation: str
) -> Optional[Dict[str, Any]]:
    """
    Execute a request across two services with unified error handling.

    Args:
        service1_func: First service function to call
        service2_func: Second service function to call
        request_id: Unique identifier for the request
        operation: Name of the operation being performed

    Returns:
        Result from service2 if successful, None otherwise
    """
    try:
        # Service 1 execution
        context = {
            'trace_id': str(uuid.uuid4()),
            'request_id': request_id,
            'service': 'service1',
            'operation': operation,
            'step': 'service1_processing'
        }
        result1 = service1_func()

        # Service 2 execution
        context.update({
            'service': 'service2',
            'step': 'service2_processing'
        })
        result2 = service2_func(result1)

        return result2

    except Exception as e:
        # Preserve context through exception wrapping
        if not isinstance(e, TraceableError):
            context = {
                'trace_id': getattr(e, 'trace_id', str(uuid.uuid4())),
                'request_id': request_id,
                'service': getattr(e, 'service', 'unknown'),
                'operation': operation,
                'step': getattr(e, 'step', 'unknown')
            }
            e = TraceableError(str(e), context)

        logger.error(
            "Cross-service operation failed",
            extra={
                'trace_id': e.trace_id,
                'request_id': e.request_id,
                'service': e.service,
                'operation': e.operation,
                'step': e.step,
                'error': str(e)
            }
        )
        raise
