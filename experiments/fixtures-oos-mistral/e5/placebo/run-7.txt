import logging
from typing import Dict, Any, Optional
import uuid

class TraceableError(Exception):
    """Error with trace context for cross-service tracing."""

    def __init__(self, message: str, context: Dict[str, Any], original_exception: Optional[Exception] = None):
        super().__init__(message)
        self.context = context
        self.original_exception = original_exception

    def __str__(self):
        base = super().__str__()
        context_str = "; ".join(f"{k}={v}" for k, v in self.context.items())
        return f"{base} [context: {context_str}]"

class TwoServiceErrorHandler:
    """Handles errors across two services with trace context preservation."""

    def __init__(self):
        self.logger = logging.getLogger("service_tracer")

    def handle_service_error(
        self,
        service_name: str,
        operation: str,
        request_id: str,
        step: str,
        error: Exception
    ) -> TraceableError:
        """
        Wraps an error with trace context for cross-service tracing.

        Args:
            service_name: Name of the service where error occurred
            operation: The operation being performed (e.g., "process_payment")
            request_id: Unique identifier for the request
            step: The specific step that failed (e.g., "validate_card")
            error: The original exception

        Returns:
            TraceableError with preserved context
        """
        context = {
            "service": service_name,
            "operation": operation,
            "request_id": request_id,
            "step": step,
            "trace_id": str(uuid.uuid4())  # Unique trace ID for the request
        }

        if isinstance(error, TraceableError):
            # Preserve existing context and add new step
            error.context.update(context)
            return error

        return TraceableError(
            message=f"Error in {service_name} during {operation} at {step}",
            context=context,
            original_exception=error
        )

    def log_error(self, error: TraceableError):
        """Logs the error with all trace context."""
        log_data = {
            "error_message": str(error),
            "context": error.context,
            "original_exception": str(error.original_exception) if error.original_exception else None
        }

        self.logger.error(
            "Service error: %s",
            log_data["error_message"],
            extra=log_data
        )

# Example usage:
if __name__ == "__main__":
    handler = TwoServiceErrorHandler()

    try:
        # Simulate an error in Service A
        raise ValueError("Invalid card number")
    except Exception as e:
        traceable_error = handler.handle_service_error(
            service_name="payment-service",
            operation="process_payment",
            request_id="req_12345",
            step="validate_card",
            error=e
        )
        handler.log_error(traceable_error)

    try:
        # Simulate an error in Service B that wraps the previous error
        raise RuntimeError("Database timeout")
    except Exception as e:
        wrapped_error = handler.handle_service_error(
            service_name="order-service",
            operation="process_payment",
            request_id="req_12345",  # Same request ID
            step="store_order",
            error=e
        )
        handler.log_error(wrapped_error)
