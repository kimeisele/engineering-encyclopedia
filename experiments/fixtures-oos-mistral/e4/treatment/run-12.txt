import time
import requests
from typing import Optional, Callable
import logging

class CircuitBreaker:
    def __init__(self, max_failures: int = 5, reset_timeout: float = 30.0):
        self.max_failures = max_failures
        self.reset_timeout = reset_timeout
        self.failures = 0
        self.last_failure_time = 0.0
        self.state = "closed"  # closed, open, half-open
        self.lock = False

    def call(self, func: Callable, *args, **kwargs):
        if self.state == "open":
            if time.time() - self.last_failure_time > self.reset_timeout:
                self.state = "half-open"
            else:
                raise Exception("Circuit breaker is open")
        elif self.state == "half-open":
            try:
                result = func(*args, **kwargs)
                self.reset()
                return result
            except Exception as e:
                self.trip()
                raise e

        try:
            result = func(*args, **kwargs)
            self.reset()
            return result
        except Exception as e:
            self.record_failure()
            raise e

    def record_failure(self):
        self.failures += 1
        self.last_failure_time = time.time()
        if self.failures >= self.max_failures:
            self.trip()

    def trip(self):
        self.state = "open"

    def reset(self):
        self.state = "closed"
        self.failures = 0
        self.last_failure_time = 0.0

def call_downstream_api(
    url: str,
    method: str = "GET",
    headers: Optional[dict] = None,
    body: Optional[dict] = None,
    max_retries: int = 3,
    backoff_factor: float = 1.0,
    circuit_breaker: Optional[CircuitBreaker] = None
) -> dict:
    """
    Calls a downstream HTTP API with retry and circuit breaker protection.

    Args:
        url: The API endpoint URL
        method: HTTP method (GET, POST, etc.)
        headers: Request headers
        body: Request body
        max_retries: Maximum retry attempts
        backoff_factor: Multiplier for exponential backoff
        circuit_breaker: Optional CircuitBreaker instance

    Returns:
        Response JSON as dict

    Raises:
        Exception: If all retries fail or circuit breaker is open
    """
    session = requests.Session()
    retry_count = 0
    last_exception = None

    # Generate correlation ID for observability
    correlation_id = f"req_{int(time.time() * 1000)}_{id(session)}"

    def make_request():
        nonlocal retry_count, last_exception
        headers_with_cid = headers.copy() if headers else {}
        headers_with_cid["X-Correlation-ID"] = correlation_id

        try:
            response = session.request(
                method=method,
                url=url,
                headers=headers_with_cid,
                json=body,
                timeout=5.0
            )
            response.raise_for_status()
            return response.json()
        except requests.exceptions.RequestException as e:
            last_exception = e
            retry_count += 1
            if retry_count <= max_retries:
                wait_time = backoff_factor * (2 ** (retry_count - 1))
                time.sleep(wait_time)
                return make_request()
            raise

    if circuit_breaker:
        return circuit_breaker.call(make_request)
    return make_request()
