import time
import requests
from typing import Optional, Callable
import logging

class CircuitBreaker:
    def __init__(self, max_failures: int = 5, reset_timeout: float = 30.0):
        self.max_failures = max_failures
        self.reset_timeout = reset_timeout
        self.failures = 0
        self.last_failure_time = 0.0
        self.state = "closed"  # closed, open, half-open
        self.logger = logging.getLogger(__name__)

    def call(self, func: Callable, *args, **kwargs):
        if self.state == "open":
            if time.time() - self.last_failure_time > self.reset_timeout:
                self.state = "half-open"
            else:
                self.logger.warning("Circuit breaker is open, not calling downstream")
                raise Exception("Circuit breaker is open")

        try:
            result = func(*args, **kwargs)
            if self.state == "half-open":
                self._reset()
            return result
        except Exception as e:
            self._record_failure()
            raise e

    def _record_failure(self):
        self.failures += 1
        self.last_failure_time = time.time()
        if self.failures >= self.max_failures:
            self.state = "open"
            self.logger.error("Circuit breaker opened due to failures")

    def _reset(self):
        self.failures = 0
        self.state = "closed"
        self.logger.info("Circuit breaker reset to closed state")

def call_downstream_api(
    api_url: str,
    payload: dict,
    circuit_breaker: CircuitBreaker,
    max_retries: int = 3,
    backoff_factor: float = 1.0,
    correlation_id: Optional[str] = None
) -> dict:
    """
    Calls a downstream HTTP API with circuit breaker and retry logic.

    Args:
        api_url: URL of the downstream API
        payload: Request payload
        circuit_breaker: Circuit breaker instance
        max_retries: Maximum number of retry attempts
        backoff_factor: Multiplier for exponential backoff
        correlation_id: Optional correlation ID for observability

    Returns:
        Response from the API

    Raises:
        Exception: If all retries fail or circuit is open
    """
    headers = {}
    if correlation_id:
        headers["X-Correlation-ID"] = correlation_id

    def make_request():
        try:
            response = requests.post(api_url, json=payload, headers=headers, timeout=5)
            response.raise_for_status()
            return response.json()
        except requests.RequestException as e:
            raise Exception(f"API request failed: {str(e)}")

    try:
        return circuit_breaker.call(make_request)
    except Exception as e:
        if circuit_breaker.state == "open":
            raise Exception(f"API unavailable (circuit breaker open): {str(e)}")

        # Exponential backoff retry logic
        for attempt in range(max_retries):
            wait_time = backoff_factor * (2 ** attempt)
            time.sleep(wait_time)
            try:
                return circuit_breaker.call(make_request)
            except Exception as e:
                if attempt == max_retries - 1:
                    raise Exception(f"All retries failed: {str(e)}")
                continue
