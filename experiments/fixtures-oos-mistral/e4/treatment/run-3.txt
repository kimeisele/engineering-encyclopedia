import time
import random
from typing import Optional, Callable
import requests
from requests.exceptions import RequestException

class CircuitBreaker:
    def __init__(self, max_failures: int = 5, reset_timeout: float = 30.0, half_open_max_calls: int = 1):
        self.max_failures = max_failures
        self.reset_timeout = reset_timeout
        self.half_open_max_calls = half_open_max_calls
        self.failures = 0
        self.state = "closed"  # closed, open, half_open
        self.last_failure_time = 0.0
        self.half_open_calls = 0

    def call(self, func: Callable, *args, **kwargs):
        if self.state == "open":
            if time.time() - self.last_failure_time > self.reset_timeout:
                self.state = "half_open"
                self.half_open_calls = 0
            else:
                raise Exception("Circuit breaker is open")

        try:
            result = func(*args, **kwargs)
            if self.state == "half_open":
                self._reset()
            return result
        except Exception as e:
            self._record_failure()
            raise e

    def _record_failure(self):
        self.failures += 1
        self.last_failure_time = time.time()
        if self.state == "closed" and self.failures >= self.max_failures:
            self.state = "open"
        elif self.state == "half_open":
            self.half_open_calls += 1
            if self.half_open_calls >= self.half_open_max_calls:
                self.state = "open"

    def _reset(self):
        self.failures = 0
        self.state = "closed"

def call_downstream_api(
    url: str,
    method: str = "GET",
    max_retries: int = 3,
    base_backoff: float = 1.0,
    circuit_breaker: Optional[CircuitBreaker] = None,
    correlation_id: Optional[str] = None
) -> dict:
    """
    Calls a downstream HTTP API with circuit breaking and retry semantics.

    Args:
        url: The URL to call
        method: HTTP method (GET, POST, etc.)
        max_retries: Maximum number of retry attempts
        base_backoff: Base seconds for exponential backoff
        circuit_breaker: Optional CircuitBreaker instance
        correlation_id: Optional correlation ID for observability

    Returns:
        Dictionary containing response data

    Raises:
        Exception: If all retries fail or circuit is open
    """
    if circuit_breaker is None:
        circuit_breaker = CircuitBreaker()

    headers = {}
    if correlation_id:
        headers["X-Correlation-ID"] = correlation_id

    def make_request():
        try:
            response = requests.request(
                method,
                url,
                headers=headers,
                timeout=5.0
            )
            response.raise_for_status()
            return response.json()
        except RequestException as e:
            raise Exception(f"API request failed: {str(e)}")

    # Circuit breaker call
    try:
        return circuit_breaker.call(make_request)
    except Exception as e:
        # Retry logic if circuit breaker allows
        for attempt in range(max_retries):
            wait_time = base_backoff * (2 ** attempt) + random.uniform(0, 0.1 * base_backoff)
            time.sleep(wait_time)
            try:
                return circuit_breaker.call(make_request)
            except Exception:
                if attempt == max_retries - 1:
                    raise Exception(f"All {max_retries} retries failed") from e
                continue
        raise Exception("Unexpected error in retry logic")
