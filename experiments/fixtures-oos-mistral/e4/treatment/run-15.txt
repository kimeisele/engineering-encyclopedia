import time
import requests
from requests.exceptions import RequestException
from typing import Optional, Callable

class CircuitBreaker:
    def __init__(self, max_failures: int = 5, reset_timeout: float = 30.0):
        self.max_failures = max_failures
        self.reset_timeout = reset_timeout
        self.failures = 0
        self.last_failure_time = 0.0
        self.state = "closed"  # closed, open, half-open

    def call(self, func: Callable, *args, **kwargs):
        if self.state == "open":
            if time.time() - self.last_failure_time > self.reset_timeout:
                self.state = "half-open"
            else:
                raise Exception("Circuit breaker is open")
        elif self.state == "half-open":
            try:
                result = func(*args, **kwargs)
                self.reset()
                return result
            except Exception as e:
                self.trip()
                raise e

        try:
            result = func(*args, **kwargs)
            self.success()
            return result
        except Exception as e:
            self.failure()
            raise e

    def success(self):
        if self.state == "half-open":
            self.reset()

    def failure(self):
        self.failures += 1
        self.last_failure_time = time.time()
        if self.failures >= self.max_failures:
            self.trip()

    def trip(self):
        self.state = "open"

    def reset(self):
        self.state = "closed"
        self.failures = 0
        self.last_failure_time = 0.0

def call_downstream_api(
    url: str,
    method: str = "GET",
    max_retries: int = 3,
    backoff_factor: float = 1.0,
    circuit_breaker: Optional[CircuitBreaker] = None
):
    """
    Calls a downstream HTTP API with retry and circuit breaker protection.

    Args:
        url: The URL to call
        method: HTTP method (GET, POST, etc.)
        max_retries: Maximum number of retry attempts
        backoff_factor: Multiplier for exponential backoff between retries
        circuit_breaker: Optional CircuitBreaker instance for fault tolerance

    Returns:
        Response from the API

    Raises:
        Exception: If all retries fail or circuit breaker is open
    """
    if circuit_breaker is None:
        circuit_breaker = CircuitBreaker()

    last_exception = None

    for attempt in range(max_retries + 1):
        try:
            response = requests.request(method, url)
            response.raise_for_status()
            return response
        except RequestException as e:
            last_exception = e
            if attempt < max_retries:
                sleep_time = backoff_factor * (2 ** attempt)
                time.sleep(sleep_time)

    # If we get here, all retries failed
    raise Exception(f"All {max_retries} retries failed") from last_exception
