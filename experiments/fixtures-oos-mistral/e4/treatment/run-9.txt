import time
import requests
from requests.exceptions import RequestException
import logging
from functools import wraps

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class CircuitBreaker:
    def __init__(self, max_failures=5, reset_timeout=60):
        self.max_failures = max_failures
        self.reset_timeout = reset_timeout
        self.failures = 0
        self.last_failure_time = None
        self.state = "closed"  # closed, open, half-open

    def __call__(self, func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # Check circuit state
            if self.state == "open":
                if time.time() - self.last_failure_time > self.reset_timeout:
                    self.state = "half-open"
                else:
                    logger.warning("Circuit is open, not calling downstream API")
                    raise Exception("Circuit breaker is open")
            elif self.state == "half-open":
                logger.info("Circuit is half-open, testing recovery")

            try:
                result = func(*args, **kwargs)
                if self.state == "half-open":
                    self._reset()
                return result
            except RequestException as e:
                self._record_failure()
                raise
        return wrapper

    def _record_failure(self):
        self.failures += 1
        self.last_failure_time = time.time()
        if self.failures >= self.max_failures:
            self.state = "open"
            logger.error("Circuit breaker opened due to failures")

    def _reset(self):
        self.failures = 0
        self.state = "closed"
        logger.info("Circuit breaker reset to closed state")

def with_correlation_id(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        import uuid
        correlation_id = str(uuid.uuid4())
        logger.info(f"Request with correlation ID: {correlation_id}")
        # In a real implementation, you would propagate this ID to downstream calls
        return func(correlation_id=correlation_id, *args, **kwargs)
    return wrapper

@with_correlation_id
@CircuitBreaker(max_failures=3, reset_timeout=30)
def call_downstream_api(correlation_id, url, max_retries=3, backoff_factor=1):
    """
    Calls a downstream HTTP API with circuit breaker and retry logic.

    Args:
        correlation_id: Unique identifier for the request
        url: The API endpoint URL
        max_retries: Maximum number of retry attempts
        backoff_factor: Multiplier for exponential backoff between retries

    Returns:
        The API response if successful

    Raises:
        Exception: If all retries fail or circuit is open
    """
    headers = {'X-Correlation-ID': correlation_id}

    for attempt in range(max_retries + 1):
        try:
            response = requests.get(url, headers=headers, timeout=5)
            response.raise_for_status()
            logger.info(f"API call successful for {correlation_id}")
            return response.json()
        except RequestException as e:
            if attempt == max_retries:
                logger.error(f"API call failed after {max_retries} retries: {str(e)}")
                raise
            wait_time = backoff_factor * (2 ** attempt)
            logger.warning(f"Attempt {attempt + 1} failed. Retrying in {wait_time} seconds...")
            time.sleep(wait_time)

    raise Exception("Unexpected error in API call")

# Example usage
if __name__ == "__main__":
    try:
        result = call_downstream_api(
            url="https://api.example.com/data",
            max_retries=3,
            backoff_factor=0.5
        )
        print("API call successful:", result)
    except Exception as e:
        print("Failed to call API:", str(e))
