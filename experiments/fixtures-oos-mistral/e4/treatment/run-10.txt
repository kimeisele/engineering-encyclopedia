import requests
import time
from enum import Enum
import uuid
import logging

class CircuitState(Enum):
    CLOSED = "closed"
    OPEN = "open"
    HALF_OPEN = "half_open"

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=30, half_open_max_calls=3):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = None
        self.half_open_calls = 0
        self.correlation_id = None

    def _generate_correlation_id(self):
        return str(uuid.uuid4())

    def _log_with_correlation(self, message):
        if not self.correlation_id:
            self.correlation_id = self._generate_correlation_id()
        logging.info(f"[{self.correlation_id}] {message}")

    def call(self, func, *args, **kwargs):
        self.correlation_id = self._generate_correlation_id()
        self._log_with_correlation(f"Attempting call in state {self.state.value}")

        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                self._log_with_correlation("Transitioning from OPEN to HALF_OPEN")
            else:
                self._log_with_correlation("Circuit is OPEN, call rejected")
                raise Exception("Circuit breaker is open")

        try:
            result = func(*args, **kwargs)
            if self.state == CircuitState.HALF_OPEN:
                self._log_with_correlation("Half-open call succeeded, transitioning to CLOSED")
                self.state = CircuitState.CLOSED
                self.failure_count = 0
                self.half_open_calls = 0
            return result
        except Exception as e:
            self._handle_failure(e)
            raise

    def _handle_failure(self, exception):
        self.failure_count += 1
        self.last_failure_time = time.time()

        if self.state == CircuitState.CLOSED:
            if self.failure_count >= self.failure_threshold:
                self.state = CircuitState.OPEN
                self._log_with_correlation("Transitioning from CLOSED to OPEN due to failures")
        elif self.state == CircuitState.HALF_OPEN:
            self.half_open_calls += 1
            if self.half_open_calls >= self.half_open_max_calls:
                self._log_with_correlation("Half-open calls exhausted, transitioning back to OPEN")
                self.state = CircuitState.OPEN
            else:
                self._log_with_correlation("Half-open call failed, remaining attempts")

def call_downstream_api(url, payload=None, max_retries=3, initial_backoff=1):
    breaker = CircuitBreaker()

    def api_call():
        headers = {'Content-Type': 'application/json'}
        response = requests.post(url, json=payload, headers=headers, timeout=5)
        response.raise_for_status()
        return response.json()

    def retry_with_backoff(attempt):
        if attempt >= max_retries:
            raise Exception(f"Max retries ({max_retries}) exceeded")

        try:
            return breaker.call(api_call)
        except Exception as e:
            wait_time = initial_backoff * (2 ** attempt) + (0.1 * attempt)  # Exponential backoff with jitter
            time.sleep(wait_time)
            return retry_with_backoff(attempt + 1)

    return retry_with_backoff(0)

# Example usage:
# result = call_downstream_api("https://api.example.com/endpoint", {"key": "value"})
