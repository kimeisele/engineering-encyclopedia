import requests
import time
import uuid
import logging
from dataclasses import dataclass
from typing import Optional, Callable

@dataclass
class CircuitBreakerConfig:
    failure_threshold: int = 5
    recovery_timeout: float = 30.0
    half_open_max_calls: int = 3
    half_open_success_threshold: float = 0.5

class CircuitBreaker:
    def __init__(self, config: CircuitBreakerConfig):
        self.config = config
        self.failure_count = 0
        self.state = "closed"  # closed, open, half-open
        self.last_failure_time = 0.0
        self.half_open_success_count = 0
        self.half_open_call_count = 0
        self.correlation_id = str(uuid.uuid4())

    def call(self, func: Callable, *args, **kwargs):
        if self.state == "open":
            if time.time() - self.last_failure_time > self.config.recovery_timeout:
                self.state = "half-open"
                self.half_open_call_count = 0
                self.half_open_success_count = 0
            else:
                raise Exception("Circuit breaker is open")

        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise e

    def _on_success(self):
        if self.state == "half-open":
            self.half_open_success_count += 1
            self.half_open_call_count += 1
            if self.half_open_call_count >= self.config.half_open_max_calls:
                if self.half_open_success_count / self.half_open_call_count >= self.config.half_open_success_threshold:
                    self.state = "closed"
                else:
                    self.state = "open"
                    self.last_failure_time = time.time()
                self.half_open_call_count = 0
                self.half_open_success_count = 0
        elif self.state == "closed":
            self.failure_count = 0

    def _on_failure(self):
        if self.state == "closed":
            self.failure_count += 1
            if self.failure_count >= self.config.failure_threshold:
                self.state = "open"
                self.last_failure_time = time.time()
        elif self.state == "half-open":
            self.state = "open"
            self.last_failure_time = time.time()

def call_downstream_api(
    url: str,
    method: str = "GET",
    params: Optional[dict] = None,
    headers: Optional[dict] = None,
    timeout: float = 5.0,
    max_retries: int = 3,
    backoff_factor: float = 1.0,
    circuit_breaker_config: Optional[CircuitBreakerConfig] = None
) -> dict:
    """
    Calls a downstream HTTP API with circuit breaker and retry semantics.

    Args:
        url: The API endpoint URL
        method: HTTP method (GET, POST, etc.)
        params: Query parameters
        headers: Request headers
        timeout: Request timeout in seconds
        max_retries: Maximum retry attempts
        backoff_factor: Multiplier for exponential backoff between retries
        circuit_breaker_config: Configuration for circuit breaker

    Returns:
        Dictionary containing response data

    Raises:
        Exception: If all retries fail or circuit breaker is open
    """
    if circuit_breaker_config is None:
        circuit_breaker_config = CircuitBreakerConfig()

    breaker = CircuitBreaker(circuit_breaker_config)
    correlation_id = str(uuid.uuid4())

    def make_request():
        headers_with_id = headers.copy() if headers else {}
        headers_with_id["X-Correlation-ID"] = correlation_id
        headers_with_id["X-Request-ID"] = correlation_id

        try:
            response = requests.request(
                method,
                url,
                params=params,
                headers=headers_with_id,
                timeout=timeout
            )
            response.raise_for_status()
            return response.json()
        except requests.exceptions.RequestException as e:
            logging.error(f"API call failed: {str(e)} (Correlation ID: {correlation_id})")
            raise e

    # Circuit breaker protects the downstream call
    try:
        return breaker.call(make_request)
    except Exception as e:
        # Implement retry logic with exponential backoff
        last_exception = e
        for attempt in range(max_retries):
            wait_time = backoff_factor * (2 ** attempt)
            time.sleep(wait_time)

            try:
                return breaker.call(make_request)
            except Exception as e:
                last_exception = e
                logging.warning(
                    f"Retry attempt {attempt + 1} failed for {url} "
                    f"(Correlation ID: {correlation_id})"
                )

        raise Exception(f"All {max_retries} retry attempts failed") from last_exception
