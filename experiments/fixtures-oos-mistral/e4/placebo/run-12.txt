Here's the implementation of a function that calls a downstream HTTP API with backoff to avoid hammering it when it's down:

```python
import time
import requests
from requests.exceptions import RequestException
from typing import Optional, Dict, Any

def call_downstream_api(
    url: str,
    payload: Optional[Dict[str, Any]] = None,
    max_retries: int = 5,
    initial_backoff: float = 1.0,
    backoff_factor: float = 2.0,
    timeout: int = 10
) -> Optional[Dict[str, Any]]:
    """
    Calls a downstream HTTP API with exponential backoff retry logic.

    Args:
        url: The URL to call
        payload: Optional dictionary of data to send
        max_retries: Maximum number of retry attempts
        initial_backoff: Initial delay between retries in seconds
        backoff_factor: Multiplier for exponential backoff
        timeout: Request timeout in seconds

    Returns:
        Response JSON if successful, None if all retries fail
    """
    attempt = 0
    backoff = initial_backoff

    while attempt <= max_retries:
        try:
            response = requests.post(
                url,
                json=payload,
                timeout=timeout
            )
            response.raise_for_status()
            return response.json()
        except RequestException as e:
            attempt += 1
            if attempt > max_retries:
                # Final attempt failed - log with context
                error_msg = (
                    f"API call failed after {max_retries} attempts. "
                    f"URL: {url}, Error: {str(e)}"
                )
                raise RuntimeError(error_msg) from e

            # Exponential backoff with jitter
            time.sleep(backoff)
            backoff *= backoff_factor

    return None
```
