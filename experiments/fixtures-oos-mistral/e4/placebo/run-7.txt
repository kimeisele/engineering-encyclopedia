Here's a Python function that calls a downstream HTTP API with backoff retry logic to avoid hammering it when it's down:

```python
import time
import requests
from requests.exceptions import RequestException
from typing import Optional, Dict, Any

def call_downstream_api(
    url: str,
    payload: Optional[Dict[str, Any]] = None,
    max_retries: int = 5,
    initial_backoff: float = 1.0,
    backoff_factor: float = 2.0,
    timeout: float = 10.0
) -> Dict[str, Any]:
    """
    Calls a downstream HTTP API with exponential backoff retry logic.

    Args:
        url: The API endpoint URL
        payload: Optional request payload
        max_retries: Maximum number of retry attempts
        initial_backoff: Initial backoff time in seconds
        backoff_factor: Multiplier for exponential backoff
        timeout: Request timeout in seconds

    Returns:
        Dictionary containing the API response

    Raises:
        RequestException: If all retries are exhausted
    """
    attempt = 0
    backoff = initial_backoff

    while attempt <= max_retries:
        try:
            response = requests.post(
                url,
                json=payload,
                timeout=timeout
            )
            response.raise_for_status()
            return response.json()
        except RequestException as e:
            attempt += 1
            if attempt > max_retries:
                raise RequestException(
                    f"API call failed after {max_retries} retries. Last error: {str(e)}"
                ) from e

            # Exponential backoff with jitter
            sleep_time = backoff * (backoff_factor ** (attempt - 1))
            time.sleep(sleep_time)

    raise RequestException("Unexpected error in API call logic")
```
