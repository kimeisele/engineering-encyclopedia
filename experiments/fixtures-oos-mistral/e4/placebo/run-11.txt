import time
import requests
from requests.exceptions import RequestException

def call_downstream_api(url, max_retries=3, initial_backoff=1, max_backoff=10):
    """
    Calls a downstream HTTP API with exponential backoff retry logic.

    Args:
        url: The URL of the downstream API to call.
        max_retries: Maximum number of retry attempts (default: 3).
        initial_backoff: Initial backoff time in seconds (default: 1).
        max_backoff: Maximum backoff time in seconds (default: 10).

    Returns:
        The response from the API if successful.

    Raises:
        RequestException: If the API call fails after all retries.
    """
    retry_count = 0
    backoff = initial_backoff

    while retry_count <= max_retries:
        try:
            response = requests.get(url)
            response.raise_for_status()
            return response
        except RequestException as e:
            if retry_count == max_retries:
                # Add context to the exception before raising
                raise RequestException(
                    f"API call failed after {max_retries} retries. "
                    f"Last error: {str(e)}. URL: {url}"
                ) from e

            # Exponential backoff with jitter to avoid thundering herd
            sleep_time = min(backoff * (2 ** retry_count), max_backoff)
            time.sleep(sleep_time)
            retry_count += 1

    # This line is theoretically unreachable due to the raise above
    raise RequestException("API call failed after all retries")
