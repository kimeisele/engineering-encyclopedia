Here's the implementation of a function that calls a downstream HTTP API with backoff to avoid hammering it when it's down:

```python
import time
import requests
from requests.exceptions import RequestException
from typing import Optional, Dict, Any

def call_downstream_api(
    url: str,
    payload: Optional[Dict[str, Any]] = None,
    max_retries: int = 5,
    initial_backoff: float = 1.0,
    backoff_factor: float = 2.0,
    timeout: int = 10
) -> Optional[Dict[str, Any]]:
    """
    Calls a downstream HTTP API with exponential backoff retry logic.

    Args:
        url: The API endpoint URL
        payload: Optional request payload
        max_retries: Maximum number of retry attempts
        initial_backoff: Initial backoff time in seconds
        backoff_factor: Multiplier for exponential backoff
        timeout: Request timeout in seconds

    Returns:
        Response JSON if successful, None if all retries fail

    Raises:
        RequestException: If the request fails after all retries
    """
    retry_count = 0
    backoff = initial_backoff

    while retry_count <= max_retries:
        try:
            response = requests.post(
                url,
                json=payload,
                timeout=timeout
            )
            response.raise_for_status()
            return response.json()
        except RequestException as e:
            if retry_count == max_retries:
                raise RequestException(
                    f"API call failed after {max_retries} retries: {str(e)}"
                ) from e

            wait_time = backoff * (backoff_factor ** retry_count)
            time.sleep(wait_time)
            retry_count += 1

    return None
```
