Here's the implementation of the function that calls a downstream HTTP API with backoff to avoid hammering it when it's down:

```python
import time
import requests
from requests.exceptions import RequestException
from typing import Optional, Dict, Any

def call_downstream_api(
    url: str,
    payload: Optional[Dict[str, Any]] = None,
    max_retries: int = 5,
    initial_backoff: float = 1.0,
    backoff_factor: float = 2.0,
    timeout: int = 10
) -> Optional[Dict[str, Any]]:
    """
    Calls a downstream HTTP API with exponential backoff when it's down.

    Args:
        url: The API endpoint URL
        payload: Optional request payload
        max_retries: Maximum number of retry attempts
        initial_backoff: Initial backoff time in seconds
        backoff_factor: Multiplier for exponential backoff
        timeout: Request timeout in seconds

    Returns:
        Parsed JSON response if successful, None otherwise
    """
    headers = {'Content-Type': 'application/json'}
    backoff = initial_backoff

    for attempt in range(max_retries + 1):
        try:
            response = requests.post(
                url,
                json=payload,
                headers=headers,
                timeout=timeout
            )
            response.raise_for_status()
            return response.json()
        except RequestException as e:
            if attempt == max_retries:
                # Final attempt failed - log error with context
                error_context = {
                    'operation': 'downstream_api_call',
                    'url': url,
                    'attempt': attempt + 1,
                    'max_retries': max_retries,
                    'error_type': type(e).__name__,
                    'error_message': str(e)
                }
                raise RuntimeError(f"API call failed after {max_retries} attempts") from e

            # Log warning with context before backoff
            warning_context = {
                'operation': 'downstream_api_call',
                'url': url,
                'attempt': attempt + 1,
                'max_retries': max_retries,
                'error_type': type(e).__name__,
                'backoff_seconds': backoff
            }
            print(f"API call failed (attempt {attempt + 1}/{max_retries}): {e}")

            # Exponential backoff with jitter
            time.sleep(backoff)
            backoff = min(backoff * backoff_factor, 30)  # Cap at 30 seconds

    return None
```
