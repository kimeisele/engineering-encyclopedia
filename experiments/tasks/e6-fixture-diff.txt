+import time
+import random

+def process(job_id):

+    if result is None:
+        raise ValueError(f"job {job_id} failed")
+    return result
+

+    time.sleep(0.5)

+
+def run():
+    for job in range(3):
+        try:
+            print(process(job))
+        except ValueError as e:
+            print(f"error: {e}")
+
+if __name__ == "__main__":
+    run()
