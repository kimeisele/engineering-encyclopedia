import subprocess
import shlex
from typing import List, Union

def run_git_command(branch_name: str, args: List[str] = None) -> subprocess.CompletedProcess:
    """
    Execute a git command with a user-supplied branch name.
    
    Args:
        branch_name: The branch name to use in the git command (user-supplied).
        args: Optional additional git arguments (e.g., ['checkout', '--']).
    
    Returns:
        subprocess.CompletedProcess: The result of the git command.
    
    Raises:
        ValueError: If branch_name contains characters that could cause command injection.
        subprocess.CalledProcessError: If the git command fails.
    """
    if not branch_name or not all(c.isalnum() or c in '-_./' for c in branch_name):
        raise ValueError(f"Invalid branch name: {branch_name!r}. Allowed: alphanumeric, '-', '_', '.', '/'")
    
    # Build command safely - use list form, never shell=True with user input
    cmd = ['git']
    if args:
        cmd.extend(args)
    cmd.append(branch_name)
    
    try:
        # Using list form prevents shell injection; capture output for errors
        result = subprocess.run(cmd, capture_output=True, text=True, check=False)
        if result.returncode != 0:
            raise subprocess.CalledProcessError(result.returncode, cmd, result.stdout, result.stderr)
        return result
    except subprocess.CalledProcessError as e:
        # Enrich error with context: what failed, which command, what state
        error_msg = (
            f"git command failed: {shlex.join(cmd)}\n"
            f"exit code: {e.returncode}\n"
            f"stdout: {e.stdout.strip() if e.stdout else '(empty)'}\n"
            f"stderr: {e.stderr.strip() if e.stderr else '(empty)'}"
        )
        raise subprocess.CalledProcessError(e.returncode, cmd, error_msg) from e
