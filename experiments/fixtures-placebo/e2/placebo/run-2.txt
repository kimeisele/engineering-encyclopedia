```python
import subprocess
import shlex

def run_git_command(branch_name: str) -> subprocess.CompletedProcess:
    """
    Run a git checkout command with the user-supplied branch name.
    """
    # Validate the branch name to prevent command injection
    if not branch_name or any(c in branch_name for c in [';', '&', '|', '`', '$', '(', ')', '<', '>', '\n']):
        raise ValueError(f"Invalid branch name: {branch_name}")

    # Use shlex.quote to safely embed the branch name in the command
    cmd = f"git checkout {shlex.quote(branch_name)}"
    
    # Run the command
    result = subprocess.run(
        cmd,
        shell=True,
        capture_output=True,
        text=True,
        check=False
    )
    
    if result.returncode != 0:
        # Provide context about what failed
        raise RuntimeError(
            f"git checkout failed for branch '{branch_name}'. "
            f"Exit code: {result.returncode}. "
            f"stderr: {result.stderr.strip() or result.stdout.strip()}"
        )
    
    return result
```
