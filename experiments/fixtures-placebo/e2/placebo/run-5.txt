import subprocess
import shlex

def run_git_branch_command(branch_name: str) -> str:
    """
    Run a git command with a user-supplied branch name.
    
    This function safely constructs and executes a git command using the
    provided branch name. The branch name is validated to prevent command
    injection, and the command output is captured and returned.
    
    Args:
        branch_name: The branch name to use in the git command.
        
    Returns:
        The stdout output of the git command as a string.
        
    Raises:
        ValueError: If the branch name is empty or contains invalid characters.
        subprocess.CalledProcessError: If the git command fails.
    """
    # Validate the branch name to prevent command injection
    if not branch_name:
        raise ValueError("Branch name cannot be empty")
    
    # Git branch names must not contain whitespace, control characters,
    # or characters that could be interpreted by the shell
    forbidden_chars = set(' \t\n\r\x0b\x0c;|&$`"\'\\!(){}[]<>*?~')
    if any(char in forbidden_chars for char in branch_name):
        raise ValueError(
            f"Branch name '{branch_name}' contains forbidden characters. "
            "Valid branch names cannot contain whitespace or shell metacharacters."
        )
    
    # Also reject branch names starting with '-' to prevent option injection
    if branch_name.startswith('-'):
        raise ValueError("Branch name cannot start with '-' (dash) to prevent option injection")
    
    # Build the command as a list to avoid shell interpretation
    git_command = ["git", "show-branch", branch_name]
    
    try:
        # Execute the command without shell=True to avoid injection
        result = subprocess.run(
            git_command,
            capture_output=True,
            text=True,
            check=True,
            timeout=10
        )
        return result.stdout
    except subprocess.CalledProcessError as e:
        # Provide context about what failed
        raise subprocess.CalledProcessError(
            returncode=e.returncode,
            cmd=e.cmd,
            output=e.output,
            stderr=(
                f"Git command failed for branch '{branch_name}'. "
                f"stderr: {e.stderr}"
            )
        ) from e
