import subprocess
import shlex
from typing import List, Union

def run_git_command(branch_name: str, *args: str) -> subprocess.CompletedProcess:
    """
    Runs a git command with a user-supplied branch name.

    Args:
        branch_name: The branch name to use in the git command.
        *args: Additional git subcommands/arguments (e.g., 'checkout', 'merge').

    Returns:
        subprocess.CompletedProcess: The result of the git command.

    Raises:
        ValueError: If branch_name is empty or contains invalid characters.
        subprocess.CalledProcessError: If the git command fails.
    """
    # Validate the branch name to prevent command injection
    if not branch_name or not branch_name.strip():
        raise ValueError("Branch name cannot be empty.")
    
    # Reject branch names with shell metacharacters or spaces
    invalid_chars = set(";&|<>$`\\\n\r\t ")
    if any(c in invalid_chars for c in branch_name):
        raise ValueError(f"Invalid branch name: {branch_name!r}")
    
    # Build the command safely using a list (no shell=True)
    command: List[str] = ["git", *args, branch_name] if args else ["git", "branch", branch_name]
    
    try:
        # Run the command without shell, capturing output
        result = subprocess.run(
            command,
            capture_output=True,
            text=True,
            check=True,
            timeout=30,
        )
        return result
    except subprocess.CalledProcessError as e:
        # Re-raise with more context
        raise subprocess.CalledProcessError(
            e.returncode, e.cmd, output=e.output, stderr=e.stderr
        ) from e
    except subprocess.TimeoutExpired as e:
        raise subprocess.TimeoutExpired(e.cmd, e.timeout, output=e.output, stderr=e.stderr) from e
