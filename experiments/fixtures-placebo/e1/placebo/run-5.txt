import json
import logging
import time
import uuid
from dataclasses import dataclass
from typing import Any, Callable, Optional

logger = logging.getLogger("payment_worker")


class QueueError(Exception):
    """Raised when there is a problem consuming from the queue."""


class ChargeError(Exception):
    """Raised when charging a customer fails."""


@dataclass
class Job:
    job_id: str
    customer_id: str
    amount_cents: int
    currency: str
    retry_count: int = 0
    payload: dict[str, Any] = None

    def __post_init__(self):
        if self.payload is None:
            self.payload = {}


class QueueClient:
    """Minimal queue interface for demonstration/decoupling."""

    def __init__(self, queue_name: str, poll_interval_seconds: float = 1.0):
        self.queue_name = queue_name
        self.poll_interval = poll_interval_seconds

    def pop(self) -> Optional[Job]:
        """Pop a job from the queue. Returns None when the queue is empty."""
        raise NotImplementedError("Inject a real queue client in production")

    def acknowledge(self, job: Job) -> None:
        """Acknowledge a job as processed successfully."""
        raise NotImplementedError("Inject a real queue client in production")

    def requeue(self, job: Job, delay_seconds: int = 0) -> None:
        """Requeue a job (e.g., for retry)."""
        raise NotImplementedError("Inject a real queue client in production")

    def dead_letter(self, job: Job, reason: str) -> None:
        """Send a job to the dead-letter queue."""
        raise NotImplementedError("Inject a real queue client in production")


class PaymentGateway:
    """Minimal payment gateway interface."""

    def charge(
        self,
        customer_id: str,
        amount_cents: int,
        currency: str,
        idempotency_key: str,
    ) -> str:
        """Charge a customer. Returns a charge ID."""
        raise NotImplementedError("Inject a real payment gateway in production")


class ChargeWorker:
    """Consumes jobs from a queue and charges customers."""

    def __init__(
        self,
        queue: QueueClient,
        payment_gateway: PaymentGateway,
        max_retries: int = 3,
    ):
        self.queue = queue
        self.payment_gateway = payment_gateway
        self.max_retries = max_retries
        self._shutdown = False

    def run(self) -> None:
        """Main loop: poll for jobs and process them."""
        while not self._shutdown:
            try:
                job = self.queue.pop()
                if job is None:
                    time.sleep(self.queue.poll_interval)
                    continue
                self._process_job(job)
            except QueueError as e:
                logger.error("queue_operation_failed error=%s", e, exc_info=True)
                time.sleep(self.queue.poll_interval * 5)
            except Exception:
                logger.exception("worker_unexpected_error")
                time.sleep(self.queue.poll_interval)

    def _process_job(self, job: Job) -> None:
        """Process a single job."""
        operation = "charge_customer"
        try:
            idempotency_key = self._make_idempotency_key(job)

            charge_id = self.payment_gateway.charge(
                customer_id=job.customer_id,
                amount_cents=job.amount_cents,
                currency=job.currency,
                idempotency_key=idempotency_key,
            )

            logger.info(
                "charge_succeeded job_id=%s customer_id=%s charge_id=%s amount_cents=%d",
                job.job_id,
                job.customer_id,
                charge_id,
                job.amount_cents,
            )
            self.queue.acknowledge(job)

        except ChargeError as e:
            error_context = {
                "operation": operation,
                "job_id": job.job_id,
                "customer_id": job.customer_id,
                "amount_cents": job.amount_cents,
                "currency": job.currency,
                "step": "gateway_charge",
                "retry_count": job.retry_count,
            }
            self._handle_charge_failure(job, e, error_context)

        except Exception as e:
            error_context = {
                "operation": operation,
                "job_id": job.job_id,
                "customer_id": job.customer_id,
                "step": "queue_interaction",
            }
            logger.error("job_processing_failed details=%s", error_context, exc_info=True)
            # We don't know if the queue is healthy; requeue for retry.
            self._handle_unknown_failure(job, error_context)

    def _handle_charge_failure(self, job: Job, error: Exception, context: dict) -> None:
        """Handle a charge failure with retry logic."""
        if job.retry_count < self.max_retries:
            job.retry_count += 1
            delay = 2 ** job.retry_count  # exponential backoff
            logger.warning(
                "charge_retry_scheduled job_id=%s retry_count=%d delay_seconds=%d reason=%s",
                job.job_id,
                job.retry_count,
                delay,
                error,
            )
            self.queue.requeue(job, delay_seconds=delay)
        else:
            logger.error(
                "charge_failed_exhausted_retries job_id=%s customer_id=%s amount_cents=%d error=%s",
                job.job_id,
                context["customer_id"],
                context["amount_cents"],
                error,
            )
            self.queue.dead_letter(job, reason=f"max_retries_exceeded: {error}")

    def _handle_unknown_failure(self, job: Job, context: dict) -> None:
        """Handle failure where we're not sure of the state."""
        if job.retry_count < self.max_retries:
            job.retry_count += 1
            logger.warning(
                "job_requeued_after_unknown_failure job_id=%s retry_count=%d context=%s",
                job.job_id,
                job.retry_count,
                context,
            )
            self.queue.requeue(job, delay_seconds=2 ** job.retry_count)
        else:
            logger.error(
                "job_moved_to_dead_letter job_id=%s context=%s",
                job.job_id,
                context,
            )
            self.queue.dead_letter(job, reason="unknown_failure_max_retries")

    @staticmethod
    def _make_idempotency_key(job: Job) -> str:
        """Create an idempotency key from the job ID and retry attempt."""
        return f"{job.job_id}:{job.retry_count}"

        customer_id=customer_id,
        amount_cents=amount_cents,
        currency="USD",
        payload={"source": "web_checkout"},
    )


def main_demo():
    # These would be real clients in production.
    queue = QueueClient("payment-jobs")
    gateway = PaymentGateway()
    worker = ChargeWorker(queue=queue, payment_gateway=gateway, max_retries=3)

    # For the demo, we just show the worker would be started.
    logger.info("worker_started queue=%s", queue.queue_name)
    # worker.run()  # Uncomment to actually start consuming


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    main_demo()
