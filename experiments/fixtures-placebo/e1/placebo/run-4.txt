class BillingService:

@dataclass
class Job:
    """Job wrapper with parsing and validation."""
    job_id: str
    customer_id: str
    amount_cents: int
    raw: dict

    @classmethod
    def from_message(cls, message: dict) -> "Job":
        """Parse and validate a raw message into a Job."""
        required = ["job_id", "customer_id", "amount_cents"]
        if not all(key in message for key in required):
            raise ValueError(f"Missing required fields: {required}")

        job_id = str(message["job_id"])
        customer_id = str(message["customer_id"])
        amount_cents = int(message["amount_cents"])

        if amount_cents <= 0:
            raise ValueError(f"amount_cents must be positive, got {amount_cents}")

        return cls(job_id=job_id, customer_id=customer_id, amount_cents=amount_cents, raw=message)


# --- Worker implementation ---

class PaymentWorker:
    """Consumes jobs from a queue and charges customers."""

    def __init__(
        self,
        queue: QueueConsumer,
        billing: BillingService,
        max_retries: int = 3,
        sleep_seconds: float = 1.0,
    ):
        self.queue = queue
        self.billing = billing
        self.max_retries = max_retries
        self.sleep_seconds = sleep_seconds
        self.running = False

    def run(self) -> None:
        """Main loop: consume jobs indefinitely until stopped."""
        logger.info("Payment worker starting")
        self.running = True
        while self.running:
            try:
                message = self.queue.get_job()
                if message is None:
                    time.sleep(self.sleep_seconds)
                    continue

                self._process_message(message)
            except KeyboardInterrupt:
                logger.info("Worker stopped by signal")
                self.running = False
            except Exception:
                # Top-level safety net: never crash the loop
                logger.exception("Worker loop crashed, continuing")
                time.sleep(self.sleep_seconds)

    def stop(self) -> None:
        """Stop the worker gracefully."""
        self.running = False

    def _process_message(self, message: dict) -> None:
        """Process a single queue message."""
        job_id_placeholder = message.get("job_id", "<unknown>")
        try:
            job = Job.from_message(message)
        except (ValueError, TypeError) as e:
            # Permanently unprocessable: log and dead-letter (no retry)
            logger.error(
                "queue consume failed:parse job_id=%s step=validate error=%s",
                job_id_placeholder,
                str(e),
            )
            # In production: send to DLQ and ACK the message
            self.queue.ack(job_id_placeholder)
            return

        # Attempt the charge with retries
        try:
            self._charge_with_retries(job)
            self.queue.ack(job.job_id)
            logger.info(
                "queue consume succeeded job_id=%s customer_id=%s amount_cents=%d step=charge",
                job.job_id,
                job.customer_id,
                job.amount_cents,
            )
        except Exception as e:
            # Requeue for retry, or dead-letter after max attempts
            logger.error(
                "queue consume failed job_id=%s customer_id=%s step=charge error=%s",
                job.job_id,
                job.customer_id,
                str(e),
            )
            self.queue.nack(job.job_id, requeue=True)

    def _charge_with_retries(self, job: Job) -> dict:
        """Charge the customer with exponential backoff retries."""
        last_exception: Optional[Exception] = None
        for attempt in range(1, self.max_retries + 1):
            try:
                return self.billing.charge(job.customer_id, job.amount_cents)
            except Exception as e:
                last_exception = e
                if attempt < self.max_retries:
                    wait = self.sleep_seconds * (2 ** (attempt - 1))
                    logger.warning(
                        "charge retry job_id=%s customer_id=%s attempt=%d wait=%.1fs error=%s",
                        job.job_id,
                        job.customer_id,
                        attempt,
                        wait,
                        str(e),
                    )
                    time.sleep(wait)

        if last_exception:
            raise last_exception
        raise RuntimeError("Charge failed without exception")


# --- Bootstrap ---

def main() -> None:
    logging.basicConfig(

    worker.run()


if __name__ == "__main__":
    main()
