"""Worker that consumes jobs from a queue and charges a customer."""

import json
import logging
import os
import time
from dataclasses import dataclass, field
from typing import Any, Callable, Dict, Optional
import uuid

logger = logging.getLogger(__name__)


@dataclass
class QueueMessage:
    """A job message with correlation id for observability."""
    job_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    customer_id: str = ""
    amount_cents: int = 0
    currency: str = "USD"
    payload: Dict[str, Any] = field(default_factory=dict)
    retry_count: int = 0


class BillingQueue:
    """Minimal queue interface - replace with real queue (e.g., SQS, RabbitMQ)."""

    def __init__(self, messages: Optional[list] = None):
        self._messages = list(messages or [])

    def receive(self) -> Optional[QueueMessage]:
        if not self._messages:
            return None
        return self._messages.pop(0)

    def acknowledge(self, msg: QueueMessage) -> None:
        logger.info("acknowledged job_id=%s", msg.job_id)


class CustomerChargingService:
    """Service that actually charges a customer."""

    def charge(self, customer_id: str, amount_cents: int, currency: str) -> Dict[str, Any]:
        if amount_cents <= 0:
            raise ValueError(f"invalid amount {amount_cents} for customer {customer_id}")
        if not customer_id:
            raise ValueError("customer_id is required")
        # Simulate charging
        return {"status": "succeeded", "charge_id": f"ch_{customer_id}_{amount_cents}"}


class Worker:
    def __init__(
        self,
        queue: BillingQueue,
        charging_service: CustomerChargingService,
        max_retries: int = 3,
        sleep_seconds: float = 1.0,
        on_fatal: Optional[Callable[..., None]] = None,
        poison_pill_handler: Optional[Callable[[QueueMessage], None]] = None,
    ):
        self._queue = queue
        self._service = charging_service
        self._max_retries = max_retries
        self._sleep = sleep_seconds
        self._on_fatal = on_fatal or self._default_fatal_handler
        self._poison_pill = poison_pill_handler or self._default_poison_pill

    def run(self, once: bool = False) -> None:
        while True:
            msg = self._queue.receive()
            if msg is None:
                if once:
                    break
                time.sleep(self._sleep)
                continue
            try:
                self._process(msg)
                self._queue.acknowledge(msg)
            except Exception as exc:
                self._handle_failure(msg, exc)
            if once:
                break

    def _process(self, msg: QueueMessage) -> None:
        logger.info(
            "charge_attempt job_id=%s customer_id=%s amount_cents=%d currency=%s",
            msg.job_id, msg.customer_id, msg.amount_cents, msg.currency,
        )
        try:
            result = self._service.charge(
                customer_id=msg.customer_id,
                amount_cents=msg.amount_cents,
                currency=msg.currency,
            )
            logger.info(
                "charge_succeeded job_id=%s charge_id=%s",
                msg.job_id, result.get("charge_id"),
            )
        except ValueError as exc:
            raise RuntimeError(
                f"charge_failed job_id={msg.job_id} customer_id={msg.customer_id} "
                f"amount_cents={msg.amount_cents} step=validation"
            ) from exc
        except Exception as exc:
            raise RuntimeError(
                f"charge_failed job_id={msg.job_id} customer_id={msg.customer_id} "
                f"step=charging_service"
            ) from exc

    def _handle_failure(self, msg: QueueMessage, exc: Exception) -> None:
        msg.retry_count += 1
        if msg.retry_count <= self._max_retries:
            logger.warning(
                "charge_retry job_id=%s customer_id=%s attempt=%d error=%s",
                msg.job_id, msg.customer_id, msg.retry_count, exc,
            )
            time.sleep(self._sleep)
            # Re-queue for retry (simulate by adding back to queue)
            self._queue._messages.append(msg)
        else:
            self._on_fatal(msg, exc)

    def _default_fatal_handler(self, msg: QueueMessage, exc: Exception) -> None:
        logger.error(
            "charge_fatal job_id=%s customer_id=%s error=%s",
            msg.job_id, msg.customer_id, exc,
        )
        self._poison_pill(msg)

    def _default_poison_pill(self, msg: QueueMessage) -> None:
        logger.error(
            "poison_pill job_id=%s customer_id=%s - manual intervention required",
            msg.job_id, msg.customer_id,
        )


def main() -> None:
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s %(levelname)s %(name)s %(message)s",
    )
    queue = BillingQueue([
        QueueMessage(customer_id="cust_123", amount_cents=2500, currency="USD"),
        QueueMessage(customer_id="cust_456", amount_cents=-5, currency="USD"),  # fails
        QueueMessage(customer_id="", amount_cents=100, currency="USD"),  # fails validation
    ])
    service = CustomerChargingService()
    worker = Worker(queue, service, max_retries=1, sleep_seconds=0.1)
    worker.run(once=True)


if __name__ == "__main__":
    main()
