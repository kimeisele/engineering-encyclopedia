import time
import logging
import random
import uuid
from typing import Callable, Optional, Any
from dataclasses import dataclass
from enum import Enum
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen

logger = logging.getLogger(__name__)


class CircuitState(str, Enum):
    CLOSED = "closed"          # Normal operation, calls allowed
    OPEN = "open"              # Dependency considered down, calls blocked
    HALF_OPEN = "half_open"    # Probing with a single call to test recovery


@dataclass
class CircuitBreakerConfig:
    failure_threshold: int = 5          # Failures before opening
    recovery_timeout: float = 30.0      # Seconds circuit stays open
    success_threshold: int = 1          # Successes in half-open to close
    base_backoff: float = 1.0           # Base retry delay in seconds
    max_retries: int = 3                # Max retries per call
    correlation_id_header: str = "X-Correlation-ID"


class CircuitBreaker:
    """Circuit breaker to stop hammering a downstream API when it's down."""

    def __init__(self, config: CircuitBreakerConfig):
        self.config = config
        self.state = CircuitState.CLOSED
        self.failures = 0
        self.successes = 0
        self.last_failure_time = 0.0
        self._lock = threading.Lock()

    def _should_allow(self) -> bool:
        with self._lock:
            if self.state == CircuitState.CLOSED:
                return True
            if self.state == CircuitState.OPEN:
                if time.time() - self.last_failure_time >= self.config.recovery_timeout:
                    self.state = CircuitState.HALF_OPEN
                    self.successes = 0
                    return True  # Allow a single probe call
                return False
            # HALF_OPEN: allow only one probe at a time
            if self.state == CircuitState.HALF_OPEN and self.successes < self.config.success_threshold:
                return True
            return False

    def _on_success(self) -> None:
        with self._lock:
            if self.state == CircuitState.HALF_OPEN:
                self.successes += 1
                if self.successes >= self.config.success_threshold:
                    self.state = CircuitState.CLOSED
                    self.failures = 0
                    self.successes = 0
            elif self.state == CircuitState.CLOSED:
                self.failures = 0  # Reset on success

    def _on_failure(self) -> None:
        with self._lock:
            self.failures += 1
            self.last_failure_time = time.time()
            if self.state == CircuitState.HALF_OPEN:
                self.state = CircuitState.OPEN
            elif self.state == CircuitState.CLOSED and self.failures >= self.config.failure_threshold:
                self.state = CircuitState.OPEN


def call_api_with_protection(
    url: str,
    breaker: CircuitBreaker,
    method: str = "GET",
    data: Optional[bytes] = None,
    headers: Optional[dict] = None,
    timeout: float = 5.0,
    correlation_id: str = None,
) -> Any:
    """
    Call a downstream HTTP API with circuit breaking, retries with jitter,
    and correlation ID propagation.
    """
    correlation_id = correlation_id or str(uuid.uuid4())
    request_headers = dict(headers or {})
    request_headers[breaker.config.correlation_id_header] = correlation_id

    if not breaker._should_allow():
        logger.warning("Circuit open; request %s blocked", correlation_id)
        raise ConnectionError("Circuit breaker open")

    attempt = 0
    while attempt <= breaker.config.max_retries:
        try:
            if attempt > 0:
                backoff = breaker.config.base_backoff * (2 ** (attempt - 1))  # Exponential
                backoff += random.uniform(0, backoff * 0.5)  # Jitter
                logger.info("Retrying in %.2fs (attempt %d)", backoff, attempt)
                time.sleep(backoff)

            req = Request(url, data=data, method=method, headers=request_headers)
            with urlopen(req, timeout=timeout) as resp:
                breaker._on_success()
                return resp.read()  # Or parse JSON as needed

        except (HTTPError, URLError, TimeoutError) as e:
            breaker._on_failure()
            logger.error("API call failed (attempt %d): %s", attempt, e)
            attempt += 1
            if attempt > breaker.config.max_retries:
                # Even after retries, we still report to operators
                logger.error("API call failed after %d retries; corr=%s", breaker.config.max_retries, correlation_id)
                raise

    # Unreachable
    raise RuntimeError("Unexpected retry loop exit")
