```python
import logging
import random
import threading
import time
from datetime import datetime, timedelta
from typing import Callable, Optional
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen

logger = logging.getLogger(__name__)


class CircuitBreaker:
    """Circuit breaker to stop hammering a downstream API while it's down.

    State machine:
      CLOSED -> OPEN (when failure threshold reached)
      OPEN -> HALF_OPEN (after cooldown period, on next call attempt)
      HALF_OPEN -> CLOSED (on success) or OPEN (on failure)
    """

    def __init__(
        self,
        failure_threshold: int = 5,
        cooldown_seconds: float = 30.0,
        half_open_max_calls: int = 1,
    ):
        self.failure_threshold = failure_threshold
        self.cooldown_seconds = cooldown_seconds
        self.half_open_max_calls = half_open_max_calls

        self._lock = threading.Lock()
        self._consecutive_failures = 0
        self._state = "CLOSED"  # CLOSED | OPEN | HALF_OPEN
        self._opened_at: Optional[datetime] = None
        self._half_open_calls_in_flight = 0

    def _should_allow_call(self) -> bool:
        with self._lock:
            if self._state == "CLOSED":
                return True
            if self._state == "OPEN":
                # Check if cooldown has elapsed -> transition to HALF_OPEN
                if self._opened_at and datetime.now() - self._opened_at >= timedelta(seconds=self.cooldown_seconds):
                    self._state = "HALF_OPEN"
                    self._half_open_calls_in_flight = 0
                    return True
                return False
            # HALF_OPEN: limit concurrent probe calls
            if self._half_open_calls_in_flight < self.half_open_max_calls:
                self._half_open_calls_in_flight += 1
                return True
            return False

    def _on_success(self):
        with self._lock:
            self._consecutive_failures = 0
            if self._state == "HALF_OPEN":
                self._state = "CLOSED"
                self._opened_at = None
            self._half_open_calls_in_flight = 0

    def _on_failure(self):
        with self._lock:
            self._consecutive_failures += 1
            self._half_open_calls_in_flight = 0
            if self._state == "HALF_OPEN" or self._consecutive_failures >= self.failure_threshold:
                self._state = "OPEN"
                self._opened_at = datetime.now()
                logger.warning("Circuit opened after %d failures", self._consecutive_failures)


def call_with_circuit_breaker(
    url: str,
    *,
    breaker: Optional[CircuitBreaker] = None,
    timeout_seconds: float = 5.0,
    max_retries: int = 2,
    base_backoff_seconds: float = 0.5,
    max_backoff_seconds: float = 5.0,
    fallback: Optional[Callable[[], object]] = None,
) -> object:
    """Call a downstream HTTP API with circuit breaker + retry with jitter.

    Returns:
        The API response body (decoded bytes).

    Raises:
        RuntimeError: If the call fails after all retries or circuit is open
                      and no fallback is provided.
    """
    cb = breaker or CircuitBreaker()
    correlation_id = f"{threading.get_ident()}-{int(time.time() * 1000)}"

    for attempt in range(max_retries + 1):
        if not cb._should_allow_call():
            logger.warning("Circuit open, not calling API (correlation_id=%s)", correlation_id)
            if fallback:
                return fallback()
            raise RuntimeError(f"Downstream API is unavailable (circuit open), correlation_id={correlation_id}")

        try:
            req = Request(url, headers={"X-Correlation-ID": correlation_id})
            with urlopen(req, timeout=timeout_seconds) as resp:
                body = resp.read()
            cb._on_success()
            return body
        except (HTTPError, URLError, TimeoutError, OSError) as exc:
            cb._on_failure()
            logger.error("API call failed (attempt %d/%d): %s (correlation_id=%s)", attempt + 1, max_retries + 1, exc, correlation_id)
            if attempt < max_retries:
                sleep_seconds = min(base_backoff_seconds * (2 ** attempt), max_backoff_seconds)
                sleep_seconds = sleep_seconds * (0.5 + random.random())
                time.sleep(sleep_seconds)
            else:
                if fallback:
                    return fallback()
                raise RuntimeError(f"API call failed after {max_retries + 1} attempts (correlation_id={correlation_id})") from exc

    # Unreachable, but keep linters happy
    return None
```
