import logging
import random
import threading
import time
from datetime import datetime, timedelta
from typing import Optional

log = logging.getLogger(__name__)


class CircuitBreaker:
    """Stateful circuit breaker to protect a downstream HTTP API."""

    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout_s: float = 30.0,
        half_open_probe_interval_s: float = 3.0,
        correlation_id_factory=None,
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout_s = recovery_timeout_s
        self.half_open_probe_interval_s = half_open_probe_interval_s
        self._correlation_id_factory = correlation_id_factory or (lambda: str(random.randint(0, 10**12)))

        self._lock = threading.Lock()
        self._consecutive_failures = 0
        self._state = "closed"  # closed, open, half_open
        self._open_since: Optional[datetime] = None
        self._last_probe_time: Optional[datetime] = None
        self._correlation_id: Optional[str] = None

    def _transition_to_open(self, correlation_id: str) -> None:
        self._state = "open"
        self._open_since = datetime.utcnow()
        self._last_probe_time = None
        log.error(
            "Circuit opened for correlation_id=%s after %d consecutive failures",
            correlation_id,
            self._consecutive_failures,
        )

    def _try_half_open_probe(self) -> bool:
        """Decide whether we may send a probe (for half-open state)."""
        now = datetime.utcnow()
        if self._last_probe_time is None:
            return True
        return (now - self._last_probe_time) >= timedelta(seconds=self._half_open_probe_interval_s)

    def _record_probe_sent(self) -> None:
        self._last_probe_time = datetime.utcnow()

    def call(self, func, *args, **kwargs):
        """Execute `func` (which performs the HTTP call) guarded by the breaker."""
        correlation_id = self._correlation_id_factory()

        with self._lock:
            if self._state == "open":
                if datetime.utcnow() - self._open_since >= timedelta(seconds=self._recovery_timeout_s):
                    self._state = "half_open"
                    self._last_probe_time = None
                else:
                    log.warning(
                        "Blocked call correlation_id=%s because circuit open, will retry in %.1fs",
                        correlation_id,
                        self._recovery_timeout_s - (datetime.utcnow() - self._open_since).total_seconds(),
                    )
                    raise CircuitOpenError(
                        f"Circuit is open; downstream unavailable. correlation_id={correlation_id}"
                    )

            if self._state == "half_open":
                if not self._try_half_open_probe():
                    wait_s = self._half_open_probe_interval_s - (datetime.utcnow() - self._last_probe_time).total_seconds()
                    log.warning(
                        "Blocked half-open probe correlation_id=%s; wait %.1fs before next probe",
                        correlation_id,
                        wait_s,
                    )
                    raise CircuitOpenError(
                        f"Half-open probe throttled; try again in {wait_s:.1f}s. correlation_id={correlation_id}"
                    )
                self._record_probe_sent()

        try:
            # Use correlation id as a header if the caller supports headers.
            # Here we assume func accepts a keyword argument `correlation_id`.
            if "correlation_id" in kwargs:
                kwargs["correlation_id"] = correlation_id
            else:
                kwargs = dict(kwargs, correlation_id=correlation_id)
            result = func(*args, **kwargs)

            with self._lock:
                # Success (including in half_open) closes the circuit.
                self._consecutive_failures = 0
                if self._state != "closed":
                    log.info("Circuit closed after success correlation_id=%s", correlation_id)
                self._state = "closed"
                self._open_since = None
                self._last_probe_time = None
            return result

        except Exception as exc:
            with self._lock:
                self._consecutive_failures += 1
                if self._state == "half_open":
                    # Probe failed → back to open
                    self._transition_to_open(correlation_id)
                elif self._consecutive_failures >= self.failure_threshold:
                    self._transition_to_open(correlation_id)
            log.error(
                "Call failed correlation_id=%s consecutive_failures=%d error=%s",
                correlation_id,
                self._consecutive_failures,
                exc,
            )
            raise


class CircuitOpenError(Exception):
    """Raised when the circuit is open and the call is blocked."""


def make_http_call_with_breaker(
    breaker: CircuitBreaker,
    http_func,
    *args,
    max_retries: int = 3,
    base_backoff_s: float = 0.5,
    **kwargs,
):
    """
    Call `http_func` with a circuit breaker and bounded retries with exponential backoff + jitter.

    - The circuit breaker prevents hammering when the API is down.
    - Retries are bounded (max_retries) with exponential backoff and jitter to avoid storms.
    - This is not idempotent; only retry if the operation is known to be safe.
    """
    attempt = 0
    while True:
        try:
            return breaker.call(http_func, *args, **kwargs)
        except CircuitOpenError:
            # The breaker already throttles; no active retry while open.
            raise
        except Exception as exc:
            attempt += 1
            if attempt > max_retries:
                raise
            # Exponential backoff with full jitter
            sleep_s = base_backoff_s * (2 ** (attempt - 1))
            sleep_s = random.uniform(0, sleep_s)
            log.warning("Retrying attempt %d in %.2fs after error: %s", attempt, sleep_s, exc)
            time.sleep(sleep_s)
