import logging
import random
import threading
import time
from dataclasses import dataclass
from enum import Enum
from typing import Callable, Optional

logger = logging.getLogger(__name__)


class CircuitState(Enum):
    """States of the circuit breaker."""
    CLOSED = "closed"
    OPEN = "open"
    HALF_OPEN = "half_open"


@dataclass
class CircuitConfig:
    """Configuration for the circuit breaker."""
    failure_threshold: int = 5
    recovery_timeout: float = 30.0  # seconds circuit stays open
    half_open_max_calls: int = 1  # probe with only 1 call at a time
    count_window: float = 60.0  # failure count window in seconds


class CircuitBreaker:
    """A simple circuit breaker protecting a downstream HTTP API.

    Implements CLOSED -> OPEN -> HALF_OPEN -> CLOSED state transitions.
    Counts failures within a sliding time window in CLOSED state.
    HALF_OPEN allows one probe call to test recovery.
    """

    def __init__(self, config: CircuitConfig = None):
        self.config = config or CircuitConfig()
        self.state = CircuitState.CLOSED
        self._lock = threading.Lock()
        self._failures: list[float] = []
        self._open_until = 0.0
        self._half_open_calls = 0

    def _record_failure(self) -> None:
        now = time.monotonic()
        with self._lock:
            self._failures.append(now)
            # Prune old failures outside the window
            self._failures = [t for t in self._failures if now - t <= self.config.count_window]
            if len(self._failures) >= self.config.failure_threshold:
                self._open_until = now + self.config.recovery_timeout
                self.state = CircuitState.OPEN
                logger.warning(f"Circuit OPEN after {len(self._failures)} failures")

    def _reset(self) -> None:
        with self._lock:
            self._failures.clear()
            self.state = CircuitState.CLOSED

    def _try_enter_half_open(self) -> bool:
        """Attempt to transition to HALF_OPEN when recovery_timeout has elapsed."""
        now = time.monotonic()
        with self._lock:
            if self.state == CircuitState.OPEN and now >= self._open_until:
                self.state = CircuitState.HALF_OPEN
                self._half_open_calls = 0
                logger.info("Circuit HALF_OPEN, probing recovery")
                return True
            return False

    def _acquire_half_open_slot(self) -> bool:
        """Acquire a probe slot if HALF_OPEN and capacity available."""
        with self._lock:
            if self.state == CircuitState.HALF_OPEN and self._half_open_calls < self.config.half_open_max_calls:
                self._half_open_calls += 1
                return True
            return False

    def _half_open_success(self) -> None:
        with self._lock:
            self._half_open_calls = 0
            self._reset()
            logger.info("Circuit CLOSED after successful probe")

    def _half_open_failure(self) -> None:
        with self._lock:
            self._half_open_calls = 0
            self._open_until = time.monotonic() + self.config.recovery_timeout
            self.state = CircuitState.OPEN
            logger.warning("Circuit back to OPEN after probe failure")

    def call(self, fn: Callable, *args, **kwargs):
        """Execute fn if circuit allows, otherwise raise CircuitOpenError."""
        if self.state == CircuitState.OPEN:
            if self._try_enter_half_open():
                pass  # proceed to slot acquisition
            else:
                raise CircuitOpenError("Circuit is open, not calling API")

        if self.state == CircuitState.HALF_OPEN:
            if not self._acquire_half_open_slot():
                raise CircuitOpenError("Circuit half-open, probe slot busy")

        try:
            result = fn(*args, **kwargs)
            if self.state == CircuitState.HALF_OPEN:
                self._half_open_success()
            self._reset()  # any success in CLOSED resets the failure count
            return result
        except Exception as e:
            if self.state == CircuitState.HALF_OPEN:
                self._half_open_failure()
            else:
                self._record_failure()
            raise CircuitCallError(f"called API and it failed: {e}") from e


class CircuitOpenError(Exception):
    """Raised when circuit is open and call is not attempted."""


class CircuitCallError(Exception):
    """Raised when the downstream call itself fails."""


# ---------------------------------------------------------------------
# Usage example
# ---------------------------------------------------------------------
if __name__ == "__main__":
    import requests

    def call_downstream() -> str:
        resp = requests.get("https://api.example.com/v1/data", timeout=2)
        resp.raise_for_status()
        return resp.text

    breaker = CircuitBreaker(CircuitConfig(failure_threshold=3, recovery_timeout=10.0))

    # Print state for observability (in real code, expose as metrics/logs)
    def report_state():
        print(f"  breaker state: {breaker.state}, failures: {len(breaker._failures)}")

    for i in range(10):
        try:
            print(f"\nAttempt {i}:")
            result = breaker.call(call_downstream)
            print(f"  success: {result[:40]}...")
        except CircuitOpenError:
            print(f"  NOT CALLED - circuit {breaker.state}")
        except CircuitCallError as e:
            print(f"  call failed and recorded: {e}")
        except Exception as e:
            print(f"  unexpected error: {e}")
        report_state()
        time.sleep(1)
