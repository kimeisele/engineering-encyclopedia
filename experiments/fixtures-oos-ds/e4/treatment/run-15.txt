import time
import random
import logging
from typing import Callable, Any, Optional
from enum import Enum

logger = logging.getLogger(__name__)


class CircuitState(Enum):
    CLOSED = "closed"       # Normal operation, calls allowed
    OPEN = "open"           # Failing, calls blocked
    HALF_OPEN = "half_open" # Testing recovery with limited calls


class CircuitBreaker:
    """
    A circuit breaker that protects a downstream HTTP API from being
    hammered while it is down. Uses a sliding window of failures,
    configurable thresholds, and a half-open probe state.
    """

    def __init__(
        self,
        failure_threshold: int = 5,
        window_seconds: float = 30.0,
        open_timeout_seconds: float = 60.0,
        half_open_max_calls: int = 1,
        success_threshold_half_open: int = 1,
        should_trip: Callable[[Exception], bool] = lambda e: True,
    ):
        self.failure_threshold = failure_threshold
        self.window_seconds = window_seconds
        self.open_timeout_seconds = open_timeout_seconds
        self.half_open_max_calls = half_open_max_calls
        self.success_threshold_half_open = success_threshold_half_open
        self.should_trip = should_trip

        self.state = CircuitState.CLOSED
        self.failure_timestamps: list[float] = []
        self.opened_at: Optional[float] = None
        self.half_open_calls = 0
        self.half_open_successes = 0

    def _is_open(self) -> bool:
        if self.state == CircuitState.OPEN:
            assert self.opened_at is not None
            if time.monotonic() - self.opened_at >= self.open_timeout_seconds:
                self._transition_to_half_open()
                return False
            return True
        return False

    def _transition_to_half_open(self) -> None:
        self.state = CircuitState.HALF_OPEN
        self.half_open_calls = 0
        self.half_open_successes = 0
        logger.info("Circuit breaker transitioned to HALF_OPEN for probing")

    def _transition_to_open(self, error: Exception) -> None:
        self.state = CircuitState.OPEN
        self.opened_at = time.monotonic()
        self.failure_timestamps.clear()
        logger.warning(
            "Circuit breaker opened after %d failures: %s",
            self.failure_threshold,
            error,
        )

    def _transition_to_closed(self) -> None:
        self.state = CircuitState.CLOSED
        self.failure_timestamps.clear()
        self.half_open_calls = 0
        self.half_open_successes = 0
        logger.info("Circuit breaker closed - dependency recovered")

    def _record_failure(self) -> None:
        now = time.monotonic()
        self.failure_timestamps.append(now)
        # Keep only failures within the window
        cutoff = now - self.window_seconds
        self.failure_timestamps = [ts for ts in self.failure_timestamps if ts > cutoff]

        if len(self.failure_timestamps) >= self.failure_threshold:
            # Trip the breaker based on the last error captured
            pass  # The calling function will handle the trip

    def _allow_call(self) -> bool:
        """Determine whether a call should be allowed through."""
        if self.state == CircuitState.CLOSED:
            return True

        if self.state == CircuitState.OPEN:
            if self._is_open():
                return False
            # If not open anymore, we've transitioned to half-open
            return self._allow_call()

        # HALF_OPEN state
        if self.half_open_calls < self.half_open_max_calls:
            return True
        return False

    def call(
        self,
        func: Callable[..., Any],
        *args: Any,
        **kwargs: Any,
    ) -> Any:
        """
        Execute the downstream call with circuit breaking.

        Args:
            func: The function that makes the HTTP request
            *args, **kwargs: Passed through to func

        Returns:
            The result of func if called

        Raises:
            CircuitOpenError: If the circuit is open
            The original exception from func if the call fails
        """
        if not self._allow_call():
            raise CircuitOpenError(
                f"Circuit is {self.state.value}, blocking call to downstream API"
            )

        if self.state == CircuitState.HALF_OPEN:
            self.half_open_calls += 1

        try:
            result = func(*args, **kwargs)
        except Exception as e:
            if self.should_trip(e):
                self._record_failure()
                if len(self.failure_timestamps) >= self.failure_threshold:
                    self._transition_to_open(e)
            # Reset half-open progress on failure
            if self.state == CircuitState.HALF_OPEN:
                self.half_open_successes = 0
                # This failed during probe - go back to open
                self._transition_to_open(e)
            raise
        else:
            if self.state == CircuitState.HALF_OPEN:
                self.half_open_successes += 1
                if self.half_open_successes >= self.success_threshold_half_open:
                    self._transition_to_closed()
            # On success in CLOSED, clear failure history gradually
            # (reset window entirely for simplicity - adjust per requirements)
            if self.state == CircuitState.CLOSED:
                self.failure_timestamps.clear()
            return result


class CircuitOpenError(Exception):
    """Raised when the circuit is open and the call is blocked."""


def make_http_call_with_backoff(
    breaker: CircuitBreaker,
    http_func: Callable[..., Any],
    *args: Any,
    **kwargs: Any,
) -> Any:
    """
    Make an HTTP call with circuit breaker and exponential backoff + jitter.

    The breaker prevents hammering while down; the backoff provides
    graceful retry behavior for transient failures without amplifying load.

    Args:
        breaker: Configured CircuitBreaker
        http_func: The function making the HTTP request
        *args, **kwargs: Passed to http_func

    Returns:
        The HTTP response

    Raises:
        CircuitOpenError: If circuit open and backoff exhausted
    """
    max_retries = 3
    base_delay = 0.5
    max_delay = 8.0

    for attempt in range(max_retries):
        try:
            return breaker.call(http_func, *args, **kwargs)
        except CircuitOpenError:
            delay = min(
                max_delay,
                base_delay * (2**attempt) + random.uniform(0, base_delay)
            )
            logger.info(
                f"Circuit open, retry in {delay:.1f}s (attempt {attempt+1}/{max_retries})"
            )
            time.sleep(delay)
        except Exception as e:
            # Only retry on transient-ish errors; let caller handle others
            if attempt < max_retries - 1:
                delay = min(
                    max_delay,
                    base_delay * (2**attempt) + random.uniform(0, base_delay)
                )
                logger.warning(f"HTTP call failed: {e}, retry in {delay:.1f}s")
                time.sleep(delay)
            else:
                raise

    raise CircuitOpenError("Retries exhausted while circuit open")
