import asyncio
import logging
import random
import time
from dataclasses import dataclass, field
from typing import Any, Optional
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen

logger = logging.getLogger(__name__)

class CircuitBreaker:
    """Simple circuit breaker for protecting a downstream HTTP API."""
    
    def __init__(self, failure_threshold: int = 5, recovery_timeout: float = 30.0,
                 half_open_probe_timeout: float = 5.0):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_probe_timeout = half_open_probe_timeout
        self.failure_count = 0
        self.state = "closed"  # closed | open | half-open
        self.last_failure_time: Optional[float] = None
        self.last_probe_time: Optional[float] = None
    
    def record_failure(self):
        self.failure_count += 1
        self.last_failure_time = time.monotonic()
        if self.failure_count >= self.failure_threshold:
            self._open()
            
    def record_success(self):
        self.failure_count = 0
        self.state = "closed"
        
    def _open(self):
        self.state = "open"
        self.last_failure_time = time.monotonic()
        logger.warning(f"Circuit breaker OPEN after {self.failure_count} failures, entering cooldown")
        
    def allow_request(self) -> bool:
        if self.state == "closed":
            return True
        if self.state == "open":
            if time.monotonic() - self.last_failure_time >= self.recovery_timeout:
                self.state = "half-open"
                logger.info("Circuit breaker HALF-OPEN: allowing test probe")
            else:
                return False
        if self.state == "half-open":
            # Allow only one probe at a time
            if self.last_probe_time is None or time.monotonic() - self.last_probe_time >= self.half_open_probe_timeout:
                self.last_probe_time = time.monotonic()
                return True
            return False
        return False

    def reset(self):
        """Reset after successful probe."""
        self.failure_count = 0
        self.state = "closed"
        self.last_failure_time = None
        self.last_probe_time = None


def retry_with_backoff(operation: Any, retries: int = 3, base_delay: float = 0.05, max_delay: float = 1.0) -> Any:
    """Retry with exponential backoff and jitter."""
    if retries <= 0:
        raise ValueError("retries must be at least 1")
    
    for attempt in range(retries):
        try:
            return operation()
        except Exception as e:
            if attempt < retries - 1:
                delay = min(max_delay, base_delay * (2 ** attempt)) * random.uniform(0.5, 1.5)
                logger.info(f"Retrying in {delay:.2f}s (attempt {attempt+2}/{retries}) - error: {e}")
                time.sleep(delay)
            else:
                raise


async def call_api_with_circuit_breaker(
    url: str,
    breaker: CircuitBreaker,
    max_retries: int = 3,
    base_retry_delay: float = 0.05,
    max_retry_delay: float = 1.0
) -> Optional[dict]:
    """Call downstream HTTP API with circuit breaker + retry backoff."""
    correlation_id = f"req-{uuid4().hex[:8]}"
    logger.info(f"[{correlation_id}] Calling {url}")
    
    if not breaker.allow_request():
        logger.warning(f"[{correlation_id}] Circuit open - skipping call to {url}")
        raise ConnectionError("Circuit breaker is open")
    
    def make_request():
        try:
            req = Request(url, headers={"X-Correlation-Id": correlation_id})
            with urlopen(req, timeout=10) as resp:
                return {"status": resp.status, "data": resp.read()}
        except HTTPError as e:
            # Exit retry loop on 4xx errors (non-retryable)
            if 400 <= e.code < 500:
                raise e
            raise
        except URLError as e:
            raise e
        except Exception as e:
            raise e
    
    try:
        result = retry_with_backoff(
            make_request,
            retries=max_retries,
            base_delay=base_retry_delay,
            max_delay=max_retry_delay
        )
        breaker.record_success()
        return result
    except Exception as e:
        breaker.record_failure()
        raise
