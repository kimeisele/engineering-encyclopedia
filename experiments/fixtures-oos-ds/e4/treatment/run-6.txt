import time
import random
import logging
from dataclasses import dataclass
from typing import Callable, Optional
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen

logger = logging.getLogger(__name__)


@dataclass
class CircuitBreakerConfig:
    failure_threshold: int = 5
    open_duration: float = 30.0
    half_open_max_calls: int = 1
    retry_backoff: float = 1.0
    retry_jitter: float = 0.5
    max_attempts: int = 3


class CircuitBreaker:
    """
    A circuit breaker that stops calling a failing API while it's down,
    probes recovery in half-open state, and never hammers the dependency.
    """

    def __init__(self, config: Optional[CircuitBreakerConfig] = None):
        self.config = config or CircuitBreakerConfig()
        self.failure_count = 0
        self.state = "closed"  # closed, open, half_open
        self.open_until = 0.0
        self.half_open_calls = 0
        self.half_open_results = []

    def _allow_call(self) -> bool:
        """Determines if a call is allowed based on breaker state."""
        if self.state == "closed":
            return True
        if self.state == "open":
            if time.monotonic() >= self.open_until:
                self.state = "half_open"
                self.half_open_calls = 0
                self.half_open_results = []
                return True
            return False
        # half-open - allow up to the limit
        if self.half_open_calls < self.config.half_open_max_calls:
            self.half_open_calls += 1
            return True
        return False

    def _record_success(self):
        """Record a successful call."""
        if self.state == "half_open":
            self.half_open_results.append(True)
            if len(self.half_open_results) >= self.config.half_open_max_calls:
                if all(self.half_open_results):
                    self.state = "closed"
                    self.failure_count = 0
        elif self.state == "closed":
            self.failure_count = 0

    def _record_failure(self):
        """Record a failed call."""
        if self.state == "closed":
            self.failure_count += 1
            if self.failure_count >= self.config.failure_threshold:
                self.state = "open"
                self.open_until = time.monotonic() + self.config.open_duration
                logger.warning(
                    "Circuit opened: %d consecutive failures", self.failure_count
                )
        elif self.state == "half_open":
            self.half_open_results.append(False)
            self.state = "open"  # probe failed, open again
            self.open_until = time.monotonic() + self.config.open_duration
            logger.warning("Half-open probe failed; circuit re-opened")


def _with_retry_and_backoff(
    func: Callable,
    max_attempts: int,
    backoff: float,
    jitter: float,
    correlation_id: str,
):
    """Retries with exponential backoff + jitter, capped at max_attempts."""
    attempt = 0
    while attempt < max_attempts:
        try:
            return func()
        except (HTTPError, URLError, TimeoutError) as e:
            attempt += 1
            if attempt >= max_attempts:
                raise
            wait = backoff * (2 ** (attempt - 1)) + random.uniform(0, jitter)
            logger.warning(
                "API call failed (attempt %d/%d): %s; retrying in %.2fs (corr=%s)",
                attempt,
                max_attempts,
                e,
                wait,
                correlation_id,
            )
            time.sleep(wait)
    # Unreachable, but keeps type-checkers happy
    raise RuntimeError("Retry loop exhausted")


def call_downstream_api(
    url: str,
    breaker: CircuitBreaker,
    correlation_id: str,
    timeout: float = 5.0,
    max_attempts: Optional[int] = None,
) -> bytes:
    """
    Calls a downstream HTTP API with circuit-breaking + bounded retries.

    - Does NOT hammer the API while it's down (breaker opens).
    - Uses bounded retries with exponential backoff + jitter.
    - Half-open state probes recovery with a single call.
    - Propagates a correlation id for observability.
    """
    attempts = max_attempts or breaker.config.max_attempts

    if not breaker._allow_call():
        logger.warning(
            "Circuit open; fast-failing call (corr=%s)", correlation_id
        )
        raise RuntimeError(
            f"Circuit open, not calling {url} (correlation_id={correlation_id})"
        )

    def _do_call() -> bytes:
        req = Request(url, headers={"X-Correlation-ID": correlation_id})
        with urlopen(req, timeout=timeout) as resp:
            return resp.read()

    try:
        result = _with_retry_and_backoff(
            _do_call,
            max_attempts=attempts,
            backoff=breaker.config.retry_backoff,
            jitter=breaker.config.retry_jitter,
            correlation_id=correlation_id,
        )
        breaker._record_success()
        return result
    except Exception as e:
        breaker._record_failure()
        raise
