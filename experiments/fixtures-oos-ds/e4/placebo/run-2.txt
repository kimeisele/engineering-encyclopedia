import time
import requests
from typing import Callable, Any, Optional
from functools import wraps

class CircuitBreaker:
    """
    Circuit breaker pattern to avoid hammering a downstream HTTP API when it's down.
    """
    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: float = 30.0,
        half_open_max_attempts: int = 1
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_attempts = half_open_max_attempts
        self.failure_count = 0
        self.state = "closed"  # closed, open, half-open
        self.last_failure_time = 0.0
        self.half_open_attempts = 0
    
    def __call__(self, func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            # Open state: reject immediately
            if self.state == "open":
                if time.time() - self.last_failure_time >= self.recovery_timeout:
                    self.state = "half-open"
                    self.half_open_attempts = 0
                else:
                    raise CircuitOpenError(f"Circuit open, rejecting call to avoid hammering downstream")
            
            # Half-open: allow a limited number of test calls
            if self.state == "half-open":
                if self.half_open_attempts >= self.half_open_max_attempts:
                    raise CircuitOpenError(f"Circuit half-open, max test attempts reached")
                self.half_open_attempts += 1
            
            try:
                result = func(*args, **kwargs)
                # Success: reset circuit
                self.failure_count = 0
                self.half_open_attempts = 0
                self.state = "closed"
                return result
            except (requests.exceptions.ConnectionError, requests.exceptions.Timeout) as e:
                self.failure_count += 1
                self.last_failure_time = time.time()
                if self.failure_count >= self.failure_threshold:
                    self.state = "open"
                raise
        return wrapper

class CircuitOpenError(Exception):
    """Raised when the circuit breaker is open."""

# Example usage with retry logic (optional but common)
def call_downstream_api_with_circuit_breaker(
    api_function: Callable,
    failure_threshold: int = 5,
    recovery_timeout: float = 30.0,
    max_retries: int = 3,
    retry_delay: float = 1.0
) -> Callable:
    breaker = CircuitBreaker(failure_threshold, recovery_timeout)
    
    @wraps(api_function)
    def wrapped(*args, **kwargs):
        for attempt in range(max_retries):
            try:
                return breaker(api_function)(*args, **kwargs)
            except CircuitOpenError as e:
                # Don't retry when circuit is open, just give up quickly
                raise e
            except (requests.exceptions.ConnectionError, requests.exceptions.Timeout) as e:
                if attempt == max_retries - 1:
                    raise
                time.sleep(retry_delay * (attempt + 1))
    return wrapped
