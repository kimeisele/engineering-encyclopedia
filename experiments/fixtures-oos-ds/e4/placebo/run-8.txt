import time
import threading
import logging
from typing import Callable, Optional, Any
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen

logger = logging.getLogger(__name__)

class CircuitBreaker:
    """
    Circuit breaker for downstream HTTP API calls.
    Opens after consecutive failures, closes after cooldown period.
    Prevents hammering a down API by short-circuiting while open.
    """

    def __init__(
        self,
        failure_threshold: int = 5,
        cooldown_seconds: float = 60.0,
        recovery_attempts: int = 1
    ):
        self.failure_threshold = failure_threshold
        self.cooldown_seconds = cooldown_seconds
        self.recovery_attempts = recovery_attempts
        
        self._failure_count = 0
        self._is_open = False
        self._open_until = 0.0
        self._lock = threading.Lock()

    def _trip(self):
        """Open the circuit and schedule a half-open recovery attempt."""
        with self._lock:
            self._is_open = True
            self._open_until = time.monotonic() + self.cooldown_seconds
            logger.warning(
                "Circuit opened: %d consecutive failures, cooldown %s seconds",
                self.failure_threshold, self.cooldown_seconds
            )

    def _reset(self):
        """Reset on successful call."""
        with self._lock:
            self._failure_count = 0
            self._is_open = False
            self._open_until = 0.0
            logger.info("Circuit reset after successful call")

    def _record_failure(self):
        """Increment failure count and potentially open circuit."""
        with self._lock:
            self._failure_count += 1
            if self._failure_count >= self.failure_threshold and not self._is_open:
                self._trip()

    def _allow_request(self) -> bool:
        """Decide if a request can proceed."""
        with self._lock:
            if not self._is_open:
                return True
            # Half-open: allow a limited number of trial requests
            if time.monotonic() >= self._open_until:
                self._open_until = time.monotonic() + self.cooldown_seconds  # reset window
                return False  # The recovery attempt is done once per window
            return False

    def call(self, fn: Callable, *args, **kwargs) -> Any:
        """
        Execute the downstream call, guarding with the circuit.
        Returns result or raises CircuitOpenError/HTTP errors.
        """
        if not self._allow_request():
            raise CircuitOpenError("Circuit is open, not calling downstream API")

        try:
            result = fn(*args, **kwargs)
            self._reset()
            return result
        except (HTTPError, URLError, TimeoutError) as e:
            self._record_failure()
            logger.error("Downstream call failed: %s", e)
            raise


class CircuitOpenError(Exception):
    """Raised when the circuit is open and the call is not executed."""


def guarded_http_call(
    url: str,
    method: str = "GET",
    timeout: float = 5.0,
    breaker: Optional[CircuitBreaker] = None,
    headers: Optional[dict] = None
) -> Optional[dict]:
    """
    Perform a downstream HTTP call with circuit breaker protection.
    
    Args:
        url: Target URL
        method: HTTP method
        timeout: Request timeout in seconds
        breaker: Optional preconfigured CircuitBreaker; creates a default if None
        headers: Optional request headers
    
    Returns:
        Parsed JSON response if applicable, or None
    
    Raises:
        CircuitOpenError: when circuit is open
        HTTPError, URLError, TimeoutError: when downstream fails
    """
    if breaker is None:
        breaker = CircuitBreaker()

    def _make_request():
        req = Request(url, method=method, headers=headers or {})
        with urlopen(req, timeout=timeout) as resp:
            data = resp.read()
            if resp.headers.get("Content-Type", "").startswith("application/json"):
                import json
                return json.loads(data.decode("utf-8"))
            return data

    return breaker.call(_make_request)


# Example usage with retry pattern (respecting circuit state)
def call_with_retry(
    url: str,
    retries: int = 3,
    backoff_seconds: float = 0.5,
    breaker: Optional[CircuitBreaker] = None,
    **kwargs
):
    """
    High-level caller with finite retries, exponential backoff, and circuit breaker.
    """
    for attempt in range(retries):
        try:
            return guarded_http_call(url, breaker=breaker, **kwargs)
        except CircuitOpenError:
            # Don't retry when circuit is open — return None immediately
            logger.warning("Circuit open, aborting retry loop")
            return None
        except (HTTPError, URLError, TimeoutError) as e:
            if attempt == retries - 1:
                raise
            wait_time = backoff_seconds * (2 ** attempt)
            logger.warning("Call failed on attempt %d: %s. Retrying in %.2fs",
                           attempt + 1, e, wait_time)
            time.sleep(wait_time)
    return None
