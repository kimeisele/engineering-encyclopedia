import time
import threading
import requests
from collections import deque
from typing import Optional, Callable, Any


class CircuitBreaker:
    """
    Circuit breaker pattern to avoid hammering a downstream HTTP API that is down.

    Tracks consecutive failures and opens the circuit after a threshold.
    While open, calls fail fast without hitting the network.
    After a cooldown period, the circuit goes to half-open and allows a single
    probe request to test if the service recovered.
    """

    def __init__(
        self,
        failure_threshold: int = 5,
        cooldown_seconds: float = 30.0,
        max_backoff_seconds: float = 300.0,
    ):
        self.failure_threshold = failure_threshold
        self.cooldown_seconds = cooldown_seconds
        self.max_backoff_seconds = max_backoff_seconds

        self._failure_count = 0
        self._state = "closed"  # closed, open, half_open
        self._last_failure_time: Optional[float] = None
        self._lock = threading.Lock()

    def allow_request(self) -> bool:
        """
        Check if a request is allowed based on current state.
        Thread-safe.
        """
        with self._lock:
            if self._state == "closed":
                return True

            if self._state == "open":
                # Check if cooldown has elapsed
                if time.time() - self._last_failure_time >= self.cooldown_seconds:
                    self._state = "half_open"
                    return True
                return False

            # half_open: only allow a single probe request
            # For simplicity, we allow the next request and immediately
            # mark that we're probing. In a real implementation, you'd
            # need a "probe_in_flight" flag.
            return True

    def record_failure(self):
        """Record a failed request. Thread-safe."""
        with self._lock:
            self._failure_count += 1
            self._last_failure_time = time.time()

            if self._failure_count >= self.failure_threshold:
                self._state = "open"
                self._failure_count = 0  # reset for next cycle

    def record_success(self):
        """Record a successful request. Thread-safe."""
        with self._lock:
            self._state = "closed"
            self._failure_count = 0
            self._last_failure_time = None


def call_with_circuit_breaker(
    api_call: Callable[[], Any],
    breaker: CircuitBreaker,
    on_failure: Optional[Callable[[Exception], None]] = None,
) -> Any:
    """
    Calls the downstream API with circuit breaker protection.
    - If circuit is open, raises a CircuitOpenError immediately.
    - If call fails and threshold reached, opens circuit.
    - Calls on_failure callback (if provided) on each failure for logging.
    """
    if not breaker.allow_request():
        raise ConnectionError(
            "Circuit is open. API is likely down. Not attempting call."
        )

    try:
        result = api_call()
        breaker.record_success()
        return result
    except Exception as e:
        breaker.record_failure()
        if on_failure:
            on_failure(e)
        raise


# Example usage
if __name__ == "__main__":
    breaker = CircuitBreaker(failure_threshold=3, cooldown_seconds=5)

    def mock_api():
        # Simulating a flaky/down API
        import random
        if random.random() < 0.7:
            raise requests.exceptions.ConnectionError("Down")
        return {"status": "ok"}

    for i in range(20):
        try:
            result = call_with_circuit_breaker(
                mock_api,
                breaker,
                on_failure=lambda e: print(f"[{time.time():.0f}] Failed: {e}")
            )
            print(f"[{time.time():.0f}] Success: {result}")
        except ConnectionError as ce:
            print(f"[{time.time():.0f}] Rejected: {ce}")
        time.sleep(1)
