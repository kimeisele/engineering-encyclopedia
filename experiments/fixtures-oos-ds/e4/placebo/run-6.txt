import time
import logging
from typing import Callable, Any
from functools import wraps
import requests

logger = logging.getLogger(__name__)


def with_circuit_breaker(
    max_failures: int = 5,
    cooldown_seconds: float = 60.0,
    failure_status_codes: tuple = (500, 502, 503, 504),
):
    """
    Decorator that wraps an HTTP API call to prevent hammering a downstream
    service that is down. Uses circuit breaker pattern: after max_failures
    consecutive failures, opens the circuit and rejects calls immediately
    for cooldown_seconds, then transitions to half-open to probe recovery.

    Args:
        max_failures: number of consecutive failures before opening circuit
        cooldown_seconds: time to wait in open state before probing
        failure_status_codes: HTTP status codes treated as failures
    """

    def decorator(func: Callable) -> Callable:
        # State per wrapped function
        state = {
            "failures": 0,
            "open_until": 0.0,  # timestamp when circuit can transition to half-open
            "is_open": False,
        }

        @wraps(func)
        def wrapper(*args, **kwargs) -> Any:
            now = time.monotonic()

            # If circuit is open
            if state["is_open"]:
                # Check if cooldown has elapsed → transition to half-open (probe)
                if now >= state["open_until"]:
                    logger.info("Circuit half-open: probing downstream API")
                    # Allow one request through; if it fails, reopen
                    try:
                        response = func(*args, **kwargs)
                        # Success → close circuit
                        state["is_open"] = False
                        state["failures"] = 0
                        logger.info("Circuit closed: downstream API recovered")
                        return response
                    except Exception as e:
                        # Still failing → keep open, reset timer
                        state["failures"] = 0  # start fresh failure count
                        state["is_open"] = True
                        state["open_until"] = now + cooldown_seconds
                        logger.warning(
                            "Circuit reopened after probe failure: %s", e
                        )
                        raise
                else:
                    # Still in cooldown → fail fast
                    logger.warning(
                        "Circuit open: rejecting call, retry in %.0fs",
                        state["open_until"] - now,
                    )
                    raise ConnectionError(
                        "Downstream API is unavailable (circuit open)"
                    )

            # Circuit closed → make actual call
            try:
                response = func(*args, **kwargs)

                # Check for failure based on status codes
                if isinstance(response, requests.Response):
                    if response.status_code in failure_status_codes:
                        state["failures"] += 1
                        if state["failures"] >= max_failures:
                            state["is_open"] = True
                            state["open_until"] = now + cooldown_seconds
                            logger.error(
                                "Circuit opened after %d consecutive failures",
                                max_failures,
                            )
                    else:
                        # Success – reset failure counter
                        state["failures"] = 0
                else:
                    # Assume success if not a Response (custom handling)
                    state["failures"] = 0

                return response

            except Exception as e:
                # Network errors, timeouts, etc.
                state["failures"] += 1
                if state["failures"] >= max_failures:
                    state["is_open"] = True
                    state["open_until"] = now + cooldown_seconds
                    logger.error(
                        "Circuit opened after %d consecutive failures: %s",
                        max_failures,
                        e,
                    )
                raise

        # Expose circuit state for testing
        wrapper.get_circuit_state = lambda: dict(state)
        return wrapper

    return decorator


# ============================================================
# Example usage:
# ============================================================
@with_circuit_breaker(max_failures=3, cooldown_seconds=10)
def call_downstream_api(url: str, timeout: float = 5.0) -> requests.Response:
    """Make HTTP GET request to downstream service."""
    logger.info("Making request to %s", url)
    resp = requests.get(url, timeout=timeout)
    resp.raise_for_status()  # raises for 4xx/5xx
    return resp


# Alternative: a version that handles non-Response returns
@with_circuit_breaker(max_failures=2, cooldown_seconds=30)
def call_custom_api(function: Callable) -> Any:
    """Wrap any function; any exception counts as failure."""
    return function()
