import time
import requests
from threading import Lock
from typing import Callable, Optional

class CircuitBreaker:
    """
    A simple circuit breaker to avoid hammering a downstream HTTP API while it's down.
    Uses three states: CLOSED (normal), OPEN (failing fast), HALF_OPEN (test probe).
    """
    
    def __init__(
        self, 
        failure_threshold: int = 5, 
        recovery_timeout: float = 60.0,
        probe_interval: float = 5.0
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.probe_interval = probe_interval
        self._lock = Lock()
        self._failure_count = 0
        self._state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        self._open_until = 0.0
        self._last_probe_time = 0.0
        
    def call(self, func: Callable, *args, **kwargs):
        """Execute the downstream call with circuit breaker protection."""
        with self._lock:
            now = time.time()
            
            if self._state == "OPEN":
                if now < self._open_until:
                    raise ConnectionError(f"Circuit is OPEN, service down until {self._open_until}")
                self._state = "HALF_OPEN"
                self._last_probe_time = now
                
            if self._state == "HALF_OPEN":
                if now - self._last_probe_time < self.probe_interval:
                    raise ConnectionError("Circuit in HALF_OPEN, wait for probe window")
                self._last_probe_time = now
                
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise
        
    def _on_success(self):
        with self._lock:
            self._failure_count = 0
            self._state = "CLOSED"
            
    def _on_failure(self):
        with self._lock:
            self._failure_count += 1
            if self._failure_count >= self.failure_threshold:
                self._state = "OPEN"
                self._open_until = time.time() + self.recovery_timeout
            elif self._state == "HALF_OPEN":
                self._state = "OPEN"
                self._open_until = time.time() + self.recovery_timeout


class DownstreamAPIClient:
    """Client wrapper that uses circuit breaker to avoid hammering a down API."""
    
    def __init__(self, base_url: str, session: Optional[requests.Session] = None):
        self.base_url = base_url.rstrip("/")
        self.session = session or requests.Session()
        self.circuit_breaker = CircuitBreaker(
            failure_threshold=3,
            recovery_timeout=30.0,
            probe_interval=2.0
        )
    
    def get(self, endpoint: str, params: Optional[dict] = None) -> requests.Response:
        """GET request with circuit breaker protection."""
        url = f"{self.base_url}/{endpoint.lstrip('/')}"
        return self.circuit_breaker.call(
            self._do_get, url, params
        )
    
    def _do_get(self, url: str, params: Optional[dict]) -> requests.Response:
        """Actual HTTP call with reasonable timeout."""
        response = self.session.get(
            url, 
            params=params, 
            timeout=5.0,
            headers={"Accept": "application/json"}
        )
        response.raise_for_status()
        return response
    
    def post(self, endpoint: str, json: Optional[dict] = None) -> requests.Response:
        """POST request with circuit breaker protection."""
        url = f"{self.base_url}/{endpoint.lstrip('/')}"
        return self.circuit_breaker.call(
            self._do_post, url, json
        )
    
    def _do_post(self, url: str, json: Optional[dict]) -> requests.Response:
        response = self.session.post(
            url, 
            json=json, 
            timeout=5.0,
            headers={"Accept": "application/json"}
        )
        response.raise_for_status()
        return response


# Example usage
def example():
    client = DownstreamAPIClient("https://httpbin.org")
    try:
        resp = client.get("/status/200")
        print(f"Success: {resp.status_code}")
    except Exception as e:
        print(f"Failed gracefully: {e}")
    
    # Simulate a down service (httpbin's 500 endpoint)
    try:
        for _ in range(5):
            try:
                client.get("/status/500")
            except Exception:
                pass
    
    except Exception as e:
        print(f"Circuit open after failures: {e}")

if __name__ == "__main__":
    example()
