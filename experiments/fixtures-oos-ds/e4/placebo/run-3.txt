import time
import threading
import requests
from typing import Callable, Optional, Any


def call_downstream_with_backoff(
    api_func: Callable[..., Any],
    *args,
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    **kwargs,
) -> Optional[Any]:
    """
    Calls a downstream HTTP API function with exponential backoff.

    Prevents hammering a down service by:
    - Detecting failures (exceptions or 5xx responses)
    - Waiting with exponential backoff between retries
    - Giving up after max_retries and returning None

    Args:
        api_func: Function that makes the HTTP call (e.g. requests.get)
        max_retries: Maximum number of retry attempts (after initial call)
        base_delay: Initial delay in seconds
        max_delay: Maximum delay cap in seconds

    Returns:
        Response object or None if all attempts fail
    """
    delay = base_delay
    attempt = 0

    while attempt <= max_retries:
        try:
            response = api_func(*args, **kwargs)
            # Success on 2xx/3xx/4xx (client errors won't improve with retry)
            if response.status_code < 500:
                return response
            # 5xx - server is down/unhealthy
            error_context = f"API call failed with {response.status_code}"
        except (requests.ConnectionError, requests.Timeout) as e:
            error_context = f"API call failed: {type(e).__name__}"
        except Exception as e:
            error_context = f"Unexpected API error: {type(e).__name__}"
            # Don't retry on programming errors
            raise

        attempt += 1
        if attempt > max_retries:
            print(f"Giving up after {attempt} attempts: {error_context}")
            return None

        print(f"Attempt {attempt}/{max_retries}: {error_context}. Retrying in {delay:.1f}s")
        time.sleep(delay)
        delay = min(delay * 2, max_delay)  # Exponential backoff

    return None


# Thread-safe wrapper for concurrent callers
class RateLimitedAPICaller:
    """Thread-safe wrapper for the backoff logic to prevent concurrent hammering."""

    def __init__(self, api_func, **backoff_kwargs):
        self.api_func = api_func
        self.backoff_kwargs = backoff_kwargs
        self._lock = threading.Lock()

    def call(self, *args, **kwargs):
        with self._lock:
            return call_downstream_with_backoff(self.api_func, *args, **kwargs, **self.backoff_kwargs)
