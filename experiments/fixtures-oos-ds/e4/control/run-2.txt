Here's a robust implementation that avoids hammering a downstream API when it's down, using a circuit breaker pattern with exponential backoff:

```python
import time
import requests
from datetime import datetime, timedelta
from functools import wraps
import threading

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=30, max_backoff=300):
        """
        Circuit breaker with exponential backoff.
        
        Args:
            failure_threshold: Number of consecutive failures before opening circuit
            recovery_timeout: Initial timeout (seconds) before trying again
            max_backoff: Maximum backoff time (seconds)
        """
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.max_backoff = max_backoff
        
        self.consecutive_failures = 0
        self.circuit_open = False
        self.next_attempt_time = None
        self.current_backoff = recovery_timeout
        self.lock = threading.Lock()
    
    def __call__(self, func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            with self.lock:
                if self.circuit_open:
                    if datetime.now() >= self.next_attempt_time:
                        # Time to try again (half-open state)
                        self.circuit_open = False
                        print(f"[CircuitBreaker] Circuit half-open, attempting recovery")
                    else:
                        # Circuit still open, fail fast
                        raise CircuitOpenError(
                            f"Circuit is open until {self.next_attempt_time}"
                        )
            
            try:
                result = func(*args, **kwargs)
                
                # Success - reset counters
                with self.lock:
                    self.consecutive_failures = 0
                    self.current_backoff = self.recovery_timeout
                    
                return result
                
            except Exception as e:
                with self.lock:
                    self.consecutive_failures += 1
                    
                    if self.consecutive_failures >= self.failure_threshold:
                        self.circuit_open = True
                        self.next_attempt_time = datetime.now() + timedelta(
                            seconds=self.current_backoff
                        )
                        # Exponential backoff for next failure
                        self.current_backoff = min(
                            self.current_backoff * 2, 
                            self.max_backoff
                        )
                        print(f"[CircuitBreaker] Circuit opened after {self.consecutive_failures} failures")
                    
                raise  # Re-raise the original exception
                
        return wrapper

class CircuitOpenError(Exception):
    """Raised when circuit is open and request cannot proceed"""
    pass

# Example usage
def make_api_call_with_circuit_breaker():
    circuit_breaker = CircuitBreaker(
        failure_threshold=3,  # Open after 3 consecutive failures
        recovery_timeout=10,  # Wait 10 seconds before first retry
        max_backoff=60        # Never wait more than 60 seconds
    )
    
    @circuit_breaker
    def call_api():
        response = requests.get(
            "https://httpbin.org/status/500",  # Simulates a failing API
            timeout=5
        )
        response.raise_for_status()  # Will raise for 5XX status codes
        return response.json()
    
    return call_api

class ResilientAPIClient:
    def __init__(self, endpoint, failure_threshold=5, recovery_timeout=30):
        self.endpoint = endpoint
        self.failure_count = 0
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.next_attempt = None
        self.backoff = recovery_timeout
        self.max_backoff = 300  # 5 minutes max
    
    def call_api(self):
        """Make API call with circuit protection"""
        
        # Check if we should even attempt
        if self.next_attempt and datetime.now() < self.next_attempt:
            wait_seconds = (self.next_attempt - datetime.now()).total_seconds()
            print(f"API is down. Waiting {wait_seconds:.1f} seconds before retry")
            return None
        
        try:
            response = requests.get(self.endpoint, timeout=5)
            response.raise_for_status()
            
            # Success - reset counters
            self.failure_count = 0
            self.backoff = self.recovery_timeout
            
            return response.json()
            
        except requests.exceptions.RequestException:
            self.failure_count += 1
            
            if self.failure_count >= self.failure_threshold:
                # Calculate exponential backoff
                self.backoff = min(self.backoff * 2, self.max_backoff)
                self.next_attempt = datetime.now() + timedelta(seconds=self.backoff)
                print(f"Circuit opened. Next retry in {self.backoff} seconds")
            
            return None

# Advanced version with concurrency and metrics
class ProductionCircuitBreaker:
    def __init__(self, name="api", failure_threshold=5, recovery_timeout=30):
        self.name = name
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.max_backoff = 300
        
        # Circuit state
        self.failures = 0
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        self.next_attempt = None
        self.backoff = recovery_timeout
        
        # Metrics
        self.request_count = 0
        self.success_count = 0
        self.failure_count = 0
        self.last_failure_time = None
        
        # Thread safety
        self._lock = threading.Lock()
        
    def call(self, func, *args, **kwargs):
        """Execute function with circuit breaker protection"""
        with self._lock:
            self.request_count += 1
            
            if self.state == "OPEN":
                if datetime.now() >= self.next_attempt:
                    print(f"[{self.name}] Circuit half-open, testing connection")
                    self.state = "HALF_OPEN"
                else:
                    wait_seconds = (self.next_attempt - datetime.now()).total_seconds()
                    raise ConnectionError(
                        f"[{self.name}] Circuit is OPEN. Retry in {wait_seconds:.1f}s"
                    )
            
            try:
                result = func(*args, **kwargs)
                
                with self._lock:
                    self.success_count += 1
                    if self.state == "HALF_OPEN":
                        print(f"[{self.name}] Circuit closed after successful test")
                        self.state = "CLOSED"
                    
                    self.failures = 0
                    self.backoff = self.recovery_timeout
                
                return result
                
            except Exception as e:
                with self._lock:
                    self.failure_count += 1
                    self.last_failure_time = datetime.now()
                    self.failures += 1
                    
                    if self.state == "HALF_OPEN" or self.failures >= self.failure_threshold:
                        self.state = "OPEN"
                        self.backoff = min(self.backoff * 2, self.max_backoff)
                        self.next_attempt = datetime.now() + timedelta(
                            seconds=self.backoff
                        )
                        print(f"[{self.name}] Circuit opened. Backoff: {self.backoff}s")
                
                raise  # Re-throw original exception
    
    def get_stats(self):
        """Get circuit breaker statistics"""
        with self._lock:
            return {
                "name": self.name,
                "state": self.state,
                "failures": self.failures,
                "request_count": self.request_count,
                "success_count": self.success_count,
                "failure_count": self.failure_count,
                "last_failure": self.last_failure_time.isoformat() if self.last_failure_time else None,
                "next_retry": self.next_attempt.isoformat() if self.next_attempt else None,
                "current_backoff": self.backoff
            }
```

## Key Features:

1. **Circuit Breaker Pattern**: Three states - CLOSED (normal), OPEN (fail fast), HALF_OPEN (test recovery)
2. **Exponential Backoff**: Increases wait time between attempts
3. **Thread Safety**: Uses locks for concurrent environments
4. **Failure Threshold**: Opens circuit after N consecutive failures
5. **Recovery Testing**: Periodically tests if API is back

## Performance Characteristics:

- **Normal operation**: Zero overhead beyond a simple lock check
- **During outage**: Fails fast (milliseconds) instead of waiting for timeout
- **After recovery**: Automatically detects and closes circuit

## Best Practices Implemented:

1. **Fail Fast** when circuit is open
2. **Automatic Recovery** with half-open state
3. **Exponential Backoff** to prevent thundering herd
4. **Metric Tracking** for monitoring and alerting
5. **Configurable Thresholds** for different service requirements

This implementation prevents hammering by:
- Immediately failing when circuit is open
- Waiting for backoff period before retry
- Using half-open state to test with single request
- Adapting backoff based on failure duration
