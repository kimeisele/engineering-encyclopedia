Here's a robust solution that implements circuit breaker pattern with exponential backoff and jitter:

```python
import time
import random
import threading
import logging
from datetime import datetime
from typing import Callable, Any, Optional

class CircuitBreaker:
    """
    Circuit breaker implementation to protect downstream HTTP APIs.
    Prevents hammering a failing service by:
    - Opening circuit after consecutive failures
    - Using exponential backoff with jitter
    - Allowing half-open state to test recovery
    """
    
    def __init__(self, 
                 failure_threshold: int = 5,
                 recovery_timeout: float = 30.0,
                 max_backoff: float = 300.0,
                 backoff_multiplier: float = 2.0,
                 jitter_factor: float = 0.2,
                 success_threshold: int = 2,
                 logger: Optional[logging.Logger] = None):
        """
        Args:
            failure_threshold: Number of failures before opening circuit
            recovery_timeout: Seconds to wait before testing recovery
            max_backoff: Maximum backoff time in seconds
            backoff_multiplier: Multiplier for exponential backoff
            jitter_factor: Random jitter factor (0-1) to avoid thundering herd
            success_threshold: Number of successes needed to close circuit
            logger: Optional logger instance
        """
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.max_backoff = max_backoff
        self.backoff_multiplier = backoff_multiplier
        self.jitter_factor = jitter_factor
        self.success_threshold = success_threshold
        self.logger = logger or logging.getLogger(__name__)
        
        # State variables
        self._lock = threading.Lock()
        self._failure_count = 0
        self._success_count = 0
        self._state = 'CLOSED'  # CLOSED, OPEN, HALF_OPEN
        self._last_failure_time = None
        self._current_backoff = 1.0  # Start with 1 second
        
    def call(self, func: Callable, *args, **kwargs) -> Any:
        """
        Execute the downstream call with circuit breaker protection.
        
        Args:
            func: The HTTP call function
            *args: Positional arguments for the function
            **kwargs: Keyword arguments for the function
            
        Returns:
            The result of the function call
            
        Raises:
            CircuitBreakerOpenError: When circuit is open and call is rejected
            OriginalException: When function raises during execution
        """
        with self._lock:
            if self._state == 'OPEN':
                if self._should_attempt_recovery():
                    self._state = 'HALF_OPEN'
                    self._success_count = 0
                    self.logger.info("Circuit breaker transitioning to HALF_OPEN state")
                else:
                    wait_time = self._get_wait_time()
                    raise CircuitBreakerOpenError(
                        f"Circuit is OPEN. Try again in {wait_time:.2f} seconds"
                    )
            
            # Execute the function
            try:
                result = func(*args, **kwargs)
                self._handle_success()
                return result
                
            except Exception as e:
                self._handle_failure()
                raise
    
    def _handle_success(self):
        """Handle successful call."""
        with self._lock:
            self._success_count += 1
            self._failure_count = 0
            
            if self._state == 'HALF_OPEN':
                if self._success_count >= self.success_threshold:
                    self._close_circuit()
            elif self._state == 'CLOSED':
                self._reset_backoff()
    
    def _handle_failure(self):
        """Handle failed call."""
        with self._lock:
            self._failure_count += 1
            self._success_count = 0
            self._last_failure_time = datetime.now()
            
            if self._failure_count >= self.failure_threshold:
                self._open_circuit()
            elif self._state == 'HALF_OPEN':
                self._open_circuit()
    
    def _open_circuit(self):
        """Open the circuit breaker."""
        self._state = 'OPEN'
        self._increase_backoff()
        self._failure_count = 0
        self.logger.warning(f"Circuit breaker OPENED. Next attempt in {self._current_backoff:.2f} seconds")
    
    def _close_circuit(self):
        """Close the circuit breaker."""
        self._state = 'CLOSED'
        self._reset_backoff()
        self.logger.info("Circuit breaker CLOSED - service recovered")
    
    def _should_attempt_recovery(self) -> bool:
        """Check if we should attempt recovery."""
        if self._last_failure_time is None:
            return True
        
        elapsed = (datetime.now() - self._last_failure_time).total_seconds()
        return elapsed >= self.recovery_timeout
    
    def _get_wait_time(self) -> float:
        """Calculate how long to wait before next attempt."""
        return self._current_backoff + random.uniform(0, self.jitter_factor * self._current_backoff)
    
    def _increase_backoff(self):
        """Increase backoff time with jitter."""
        base = min(self._current_backoff * self.backoff_multiplier, self.max_backoff)
        # Add jitter to avoid thundering herd
        jitter = random.uniform(0, self.jitter_factor * base)
        self._current_backoff = base + jitter
    
    def _reset_backoff(self):
        """Reset backoff to initial value."""
        self._current_backoff = 1.0

class CircuitBreakerOpenError(Exception):
    """Raised when circuit breaker is open and call is rejected."""
    pass

# Example usage with actual HTTP client

    # Create circuit breaker instance

# Alternative implementation using a simple decorator
def circuit_breaker(func: Callable = None, *, 
                    failure_threshold: int = 5,
                    recovery_timeout: float = 30.0):
    """
    Decorator version of circuit breaker.
    
    Usage:
        @circuit_breaker(failure_threshold=5, recovery_timeout=30)
        def call_api():
            # HTTP call here
            pass
    """
    def decorator(f):
        breaker = CircuitBreaker(
            failure_threshold=failure_threshold,
            recovery_timeout=recovery_timeout
        )
        
        def wrapper(*args, **kwargs):
            return breaker.call(f, *args, **kwargs)
        
        return wrapper
    
    if func is not None:
        return decorator(func)
    return decorator

# Thread-safe version for concurrent access
class ThreadSafeCircuitBreaker(CircuitBreaker):
    """Extended version with thread-safe implementation."""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._lock = threading.RLock()  # Reentrant lock for nested calls
        self._call_count = 0
        self._error_count = 0
    
    def call(self, func: Callable, *args, **kwargs) -> Any:
        with self._lock:
            self._call_count += 1
            return super().call(func, *args, **kwargs)

# Production-ready version with metrics collection
class ProductionCircuitBreaker(CircuitBreaker):
    """Adds metrics collection for monitoring."""
    
    def __init__(self, *args, metrics_collector=None, **kwargs):
        super().__init__(*args, **kwargs)
        self.metrics = {
            'total_calls': 0,
            'successful_calls': 0,
            'failed_calls': 0,
            'rejected_calls': 0,
            'circuit_state_changes': 0
        }
        self.metrics_collector = metrics_collector or self._default_metrics_collector
    
    def _default_metrics_collector(self, metric_name: str, value: Any):
        """Default metrics collector - could be replaced with Prometheus/StatsD."""
        print(f"METRIC: {metric_name} = {value}")
    
    def call(self, func: Callable, *args, **kwargs) -> Any:
        self.metrics['total_calls'] += 1
        
        try:
            result = super().call(func, *args, **kwargs)
            self.metrics['successful_calls'] += 1
            self.metrics_collector('successful_calls', self.metrics['successful_calls'])
            return result
        except CircuitBreakerOpenError:
            self.metrics['rejected_calls'] += 1
            self.metrics_collector('rejected_calls', self.metrics['rejected_calls'])
            raise
        except Exception:
            self.metrics['failed_calls'] += 1
            self.metrics_collector('failed_calls', self.metrics['failed_calls'])
            raise
```

## Key Features:

1. **Circuit Breaker Pattern**:
   - `CLOSED`: Normal operation, calls pass through
   - `OPEN`: Rejects calls immediately after failure threshold
   - `HALF_OPEN`: Tests recovery with limited calls

2. **Exponential Backoff with Jitter**:
   - Starts at 1 second
   - Multiplies by `backoff_multiplier` each failure
   - Adds random jitter to prevent thundering herd
   - Caps at `max_backoff`

3. **Recovery Testing**:
   - After `recovery_timeout` in OPEN state, attempts recovery
   - Requires `success_threshold` consecutive successes to close

4. **Thread Safety**:
   - Uses locks to handle concurrent access
   - Production version includes metrics collection

5. **Error Handling**:
   - Raises `CircuitBreakerOpenError` when rejecting calls
   - Preserves original exceptions from the API call

## Key Benefits:

- **Prevents hammering**: Rejects calls immediately when circuit is open
- **Graceful recovery**: Systematically tests if service is back
- **Scalable**: Backoff times avoid overwhelming down services
- **Observable**: Logs state transitions and includes metrics options
- **Configurable**: All parameters can be tuned for your use case

This implementation protects both your system (by not wasting resources) and the downstream service (by not overloading it during recovery).
