```python
import uuid
import time
import json
import logging
from datetime import datetime
from typing import Dict, Any, Optional
from contextlib import contextmanager
import hashlib

class RequestTracer:
    """
    Unified request tracing and error handling across two services
    """
    
    def __init__(self, service_name: str, log_level: int = logging.INFO):
        self.service_name = service_name
        self.logger = self._setup_logger(log_level)
        
    def _setup_logger(self, level: int) -> logging.Logger:
        """Setup logger with JSON formatting for structured logging"""
        logger = logging.getLogger(f"tracer_{self.service_name}")
        logger.setLevel(level)
        
        handler = logging.StreamHandler()
        handler.setFormatter(JsonFormatter())
        logger.addHandler(handler)
        return logger
    
    @staticmethod
    def generate_trace_id() -> str:
        """Generate a unique trace ID for the entire request flow"""
        return f"trace_{uuid.uuid4().hex[:16]}"
    
    @staticmethod
    def generate_span_id() -> str:
        """Generate a unique span ID for each service call"""
        return f"span_{uuid.uuid4().hex[:12]}"
    
    def start_trace(self, request_data: Dict) -> "TraceContext":
        """Start a new trace for a request"""
        trace_id = self.generate_trace_id()
        return TraceContext(
            trace_id=trace_id,
            service_name=self.service_name,
            start_time=time.time(),
            request_data=request_data
        )
    
    @contextmanager
    def trace_service(self, trace_id: str, service_call: str):
        """
        Context manager for tracing a single service call
        """
        span_id = self.generate_span_id()
        start_time = time.time()
        
        trace_data = {
            "trace_id": trace_id,
            "span_id": span_id,
            "parent_span_id": self.parent_span_id if hasattr(self, 'parent_span_id') else None,
            "service": self.service_name,
            "operation": service_call,
            "start_time": datetime.utcnow().isoformat(),
            "phase": "start"
        }
        
        self.logger.info(f"TRACE_START", extra={"trace": trace_data})
        
        try:
            yield span_id
            trace_data["phase"] = "end"
            trace_data["duration_ms"] = (time.time() - start_time) * 1000
            trace_data["status"] = "success"
            self.logger.info(f"TRACE_END", extra={"trace": trace_data})
            
        except Exception as e:
            trace_data["phase"] = "error"
            trace_data["duration_ms"] = (time.time() - start_time) * 1000
            trace_data["status"] = "error"
            trace_data["error_type"] = type(e).__name__
            trace_data["error_message"] = str(e)
            trace_data["execution_context"] = {
                "request_id": trace_id,
                "operation": service_call
            }
            
            self.logger.error(f"TRACE_ERROR", extra={"trace": trace_data})
            raise

class TraceContext:
    """Container for trace information"""
    def __init__(self, trace_id: str, service_name: str, start_time: float, request_data: Dict):
        self.trace_id = trace_id
        self.service_name = service_name
        self.start_time = start_time
        self.request_data = request_data
        self.spans = []
        self.is_complete = False
        
    def add_span(self, span_data: Dict):
        """Add a span to the trace"""
        self.spans.append(span_data)
        
    def to_dict(self) -> Dict:
        """Convert trace context to dictionary"""
        return {
            "trace_id": self.trace_id,
            "service_name": self.service_name,
            "start_time": self.start_time,
            "request_data": self.request_data,
            "spans": self.spans,
            "duration_ms": (time.time() - self.start_time) * 1000 if self.spans else 0
        }

class JsonFormatter(logging.Formatter):
    """Custom JSON formatter for structured logging"""
    
    def format(self, record: logging.LogRecord) -> str:
        log_entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "message": record.getMessage(),
            "logger": record.name
        }
        
        # Add trace data if present
        if hasattr(record, 'trace'):
            log_entry["trace"] = record.trace
        
        # Add any additional context
        if hasattr(record, 'execution_context'):
            log_entry["execution_context"] = record.execution_context
            
        return json.dumps(log_entry)

class TwoServiceRequestHandler:
    """
    Main handler that orchestrates requests across two services with complete tracing
    """
    
    def __init__(self, service_a_name: str, service_b_name: str):
        self.tracer_a = RequestTracer(service_a_name)
        self.tracer_b = RequestTracer(service_b_name)
        self.storage = {}  # Simple in-memory storage, replace with actual queue/database
        
    def process_request(self, request_data: Dict) -> Dict:
        """
        Process a request that requires both services with full traceability
        
        Args:
            request_data: The incoming request payload
            
        Returns:
            Response dictionary
        """
        # Start the trace
        trace_context = self.tracer_a.start_trace(request_data)
        trace_id = trace_context.trace_id
        
        self._log_request_start(trace_id, request_data)
        
        try:
            # Service A call
            with self.tracer_a.trace_service(trace_id, "service_a_processing"):
                service_a_result = self._call_service_a(request_data)
                
                # Store result for correlation
                span_data = {
                    "service": "A",
                    "request_id": request_data.get('request_id'),
                    "status": "success"
                }
                trace_context.add_span(span_data)
            
            # Pass trace_id to Service B
            enhanced_request = {
                **request_data,
                "trace_id": trace_id,
                "parent_span_id": trace_context.spans[-1].get('span_id')
            }
            
            # Service B call
            with self.tracer_b.trace_service(trace_id, "service_b_processing"):
                service_b_result = self._call_service_b(enhanced_request)
                
                span_data = {
                    "service": "B",
                    "request_id": request_data.get('request_id'),
                    "trace_id": trace_id,
                    "status": "success"
                }
                trace_context.add_span(span_data)
            
            # Store complete trace for retrieval
            self.storage[trace_id] = trace_context.to_dict()
            
            # Log successful completion
            self._log_request_success(trace_id, request_data)
            
            return {
                "status": "success",
                "trace_id": trace_id,
                "service_a_result": service_a_result,
                "service_b_result": service_b_result
            }
            
        except Exception as e:
            # Comprehensive error logging
            self._handle_error(trace_id, request_data, e)
            
            # Store even failed traces for debugging
            self.storage[trace_id] = trace_context.to_dict()
            
            return {
                "status": "error",
                "trace_id": trace_id,
                "error": {
                    "type": type(e).__name__,
                    "message": str(e),
                    "trace": self._build_error_trace(trace_context)
                }
            }
    
    def _call_service_a(self, request_data: Dict) -> Dict:
        """Simulated call to Service A"""
        # In real implementation, this would be an HTTP call to Service A
        return {"service": "A", "processed": True, "echo": request_data}
    
    def _call_service_b(self, request_data: Dict) -> Dict:
        """Simulated call to Service B"""
        # In real implementation, this would be an HTTP call to Service B
        return {"service": "B", "processed": True, "trace_id": request_data.get('trace_id')}
    
    def _log_request_start(self, trace_id: str, request_data: Dict):
        """Logs when a request starts"""
        self.tracer_a.logger.info(
            "REQUEST_START",
            extra={
                "trace": {
                    "trace_id": trace_id,
                    "operation": "full_request",
                    "status": "start"
                }
            }
        )
    
    def _log_request_success(self, trace_id: str, request_data: Dict):
        """Logs successful request completion"""
        self.tracer_a.logger.info(
            "REQUEST_SUCCESS",
            extra={
                "trace": {
                    "trace_id": trace_id,
                    "operation": "full_request",
                    "status": "success",
                    "service_calls": ["A", "B"]
                }
            }
        )
    
    def _handle_error(self, trace_id: str, request_data: Dict, error: Exception):
        """Comprehensive error handling and logging"""
        error_payload = {
            "trace_id": trace_id,
            "error_type": type(error).__name__,
            "error_message": str(error),
            "request_data": request_data,
            "timestamp": datetime.utcnow().isoformat(),
            "service_chain": "A -> B"
        }
        
        # Log the error with full context
        self.tracer_a.logger.error(
            "REQUEST_FAILED",
            extra={
                "trace": error_payload,
                "execution_context": {
                    "request_id": request_data.get('request_id'),
                    "trace_id": trace_id,
                    "service_chain": "ServiceA -> ServiceB"
                }
            }
        )
        
        # Also log to service B tracer
        self.tracer_b.logger.error(
            "REQUEST_FAILED",
            extra={
                "trace": error_payload,
                "execution_context": {
                    "trace_id": trace_id,
                    "source": "service_a"
                }
            }
        )
    
    def _build_error_trace(self, trace_context: TraceContext) -> Dict:
        """Build a human-readable error trace for operator display"""
        return {
            "trace_id": trace_context.trace_id,
            "request_data": trace_context.request_data,
            "duration_ms": (time.time() - trace_context.start_time) * 1000,
            "spans": trace_context.spans,
            "failed_service": "Unknown"  # Set based on which service failed
        }
    
    def retrieve_trace(self, trace_id: str) -> Optional[Dict]:
        """Retrieve a stored trace for debugging by operator"""
        return self.storage.get(trace_id)
    
    def generate_correlation_summary(self, trace_id: str) -> Dict:
        """
        Generate a correlation summary that operators can use
        to trace the request across both services
        """
        trace = self.storage.get(trace_id)
        if not trace:
            return {"error": "Trace not found"}
        
        return {
            "trace_id": trace["trace_id"],
            "correlation_id": trace["request_data"].get("request_id"),
            "service_a": {
                "status": "completed",
                "span_id": trace["spans"][0].get("span_id") if trace["spans"] else None,
                "timestamp": trace["request_data"].get("timestamp")
            },
            "service_b": {
                "status": "completed" if len(trace["spans"]) > 1 else "not_called",
                "span_id": trace["spans"][1].get("span_id") if len(trace["spans"]) > 1 else None
            },
            "full_flow": trace["spans"]
        }
```

## Key Features for Operator Tracing

1. **Unified Trace ID**: Single trace ID propagates across both services
2. **Span Tracking**: Each service call gets its own span ID with parent-child relationships
3. **Structured Logging**: All logs include trace_id, span_id, and correlation context
4. **In-Memory Trace Storage**: Full request lifecycle stored for retrieval
5. **Correlation Summary**: Quick view of how the request flowed through services
6. **Comprehensive Error Context**: Includes request data, error type, and service chain
