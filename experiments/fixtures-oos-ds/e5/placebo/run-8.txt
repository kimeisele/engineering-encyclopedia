import logging
import uuid
from contextlib import contextmanager
from typing import Optional

logger = logging.getLogger(__name__)

class ServiceTracer:
    """Holds trace context for a two-service request."""

    def __init__(self, request_id: Optional[str] = None):
        self.request_id = request_id or str(uuid.uuid4())
        self.services = []

    @contextmanager
    def service(self, service_name: str):
        """Context manager to track which service is being called."""
        service_id = f"{self.request_id}:{service_name}"
        self.services.append(service_name)
        try:
            yield service_id
        except Exception as exc:
            # Attach trace context before re-raising
            exc.request_id = self.request_id
            exc.service_name = service_name
            exc.service_step = len(self.services)
            raise


def handle_two_service_request(service_a_call, service_b_call):
    """
    Error handler that wraps two service calls with a shared trace ID.

    Args:
        service_a_call: callable for first service
        service_b_call: callable for second service

    Returns:
        Result of the second service call
    """
    tracer = ServiceTracer()
    logger.info(
        "Starting two-service request",
        extra={"request_id": tracer.request_id, "operation": "two_service_request"}
    )

    try:
        with tracer.service("service_a"):
            result_a = service_a_call()
            logger.info(
                "Service A completed",
                extra={"request_id": tracer.request_id, "service": "service_a"}
            )

        with tracer.service("service_b"):
            result_b = service_b_call(result_a)
            logger.info(
                "Service B completed",
                extra={"request_id": tracer.request_id, "service": "service_b"}
            )
            return result_b

    except Exception as exc:
        # Capture trace context at failure point
        request_id = getattr(exc, "request_id", tracer.request_id)
        service_name = getattr(exc, "service_name", "unknown")
        step = getattr(exc, "service_step", "unknown")

        logger.error(
            "Two-service request failed",
            extra={
                "request_id": request_id,
                "operation": "two_service_request",
                "failing_service": service_name,
                "failing_step": step,
                "error_type": type(exc).__name__,
                "error_message": str(exc),
            },
            exc_info=True  # Keep full stack trace
        )
        raise


# Example usage
if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)

    def service_a():
        return {"data": "from_a"}

    def service_b(data_a):
        # Simulate failure
        raise ValueError("Service B processing error")

    try:
        handle_two_service_request(service_a, service_b)
    except ValueError:
        pass
