import logging
import uuid
from contextvars import ContextVar
from typing import Optional

logger = logging.getLogger(__name__)

# Context variable to carry the trace ID across the two services
_trace_id: ContextVar[Optional[str]] = ContextVar("trace_id", default=None)


def set_trace_id(trace_id: Optional[str] = None) -> str:
    """Set or generate a trace ID for the current request context."""
    if trace_id is None:
        trace_id = str(uuid.uuid4())
    _trace_id.set(trace_id)
    return trace_id


def get_trace_id() -> Optional[str]:
    return _trace_id.get()


class ServiceError(Exception):
    """Error with full context for tracing across services."""

    def __init__(self, service: str, operation: str, step: str, message: str, trace_id: str, cause: Optional[Exception] = None):
        self.service = service
        self.operation = operation
        self.step = step
        self.trace_id = trace_id
        self.cause = cause
        super().__init__(f"[trace={trace_id}] {service}.{operation} failed at {step}: {message}")


def handle_service_error(error: Exception, service: str, operation: str, step: str) -> None:
    """
    Error handler that enriches any exception with the trace ID and operation context.
    Call this at the boundary of each service (e.g., in a request wrapper).
    """
    trace_id = get_trace_id() or set_trace_id()
    logger.error(
        "Service error",
        extra={
            "trace_id": trace_id,
            "service": service,
            "operation": operation,
            "step": step,
            "error": str(error),
            "error_type": type(error).__name__,
        },
    )
    # Re-raise as ServiceError with full context, preserving the original exception as cause
    raise ServiceError(service, operation, step, str(error), trace_id, cause=error)


# Example usage across two services:

def service_a_entry(request_data):
    trace_id = set_trace_id()  # generate once at entry point
    try:
        # ... some work, maybe call service B
        result = service_b_call(request_data)
        return result
    except Exception as e:
        handle_service_error(e, service="service_a", operation="process_order", step="call_service_b")


def service_b_call(request_data):
    # In real scenario, this would be an HTTP call; here we just simulate
    try:
        # ... service B work
        raise ValueError("DB connection lost")
    except Exception as e:
        handle_service_error(e, service="service_b", operation="process_order", step="query_db")
