import logging
import uuid
from contextvars import ContextVar
from dataclasses import dataclass, field
from typing import Optional

logger = logging.getLogger(__name__)

# Context variable to carry the trace ID across async/sync boundaries
trace_id_var: ContextVar[Optional[str]] = ContextVar('trace_id', default=None)


@dataclass
class RequestContext:
    """Carries the trace ID and service name for a single request."""
    trace_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    service_name: str = "unknown"
    step: str = "start"


def get_trace_id() -> Optional[str]:
    """Return the current request's trace ID, or None if not in a request."""
    return trace_id_var.get()


def set_trace_id(trace_id: str) -> None:
    """Set the trace ID for the current execution context."""
    trace_id_var.set(trace_id)


class ServiceError(Exception):
    """Base error with full request context attached."""

    def __init__(
        self,
        message: str,
        *,
        service: str,
        step: str,
        trace_id: Optional[str] = None,
        cause: Optional[BaseException] = None,
    ):
        super().__init__(message)
        self.message = message
        self.service = service
        self.step = step
        self.trace_id = trace_id or get_trace_id()
        self.cause = cause

    def __str__(self) -> str:
        cause_info = f" | cause: {self.cause}" if self.cause else ""
        return (
            f"[trace:{self.trace_id}] service={self.service} "
            f"step={self.step} message={self.message}{cause_info}"
        )


def error_handler(service_name: str):
    """
    Decorator that wraps a service call and attaches context to any exception.
    Logs the error with the full trace context so an operator can follow the request.
    """

    def decorator(func):
        def wrapper(*args, **kwargs):
            # Ensure trace ID exists (inherit from caller if set, else create one)
            if get_trace_id() is None:
                set_trace_id(str(uuid.uuid4()))

            trace_id = get_trace_id()
            try:
                logger.info("service=%s trace=%s step=enter", service_name, trace_id)
                result = func(*args, **kwargs)
                logger.info("service=%s trace=%s step=exit", service_name, trace_id)
                return result
            except Exception as e:
                # Wrap with context, preserving original exception as cause
                logger.exception(
                    "service=%s trace=%s step=failed error=%r",
                    service_name,
                    trace_id,
                    e,
                )
                raise ServiceError(
                    str(e),
                    service=service_name,
                    step="handler",
                    trace_id=trace_id,
                    cause=e,
                )

        return wrapper

    return decorator


# ============================================================================
# Example usage for a two-service request flow
# ============================================================================

@error_handler("auth-service")
def authenticate(user_id: str):
    # In reality this would call external auth service
    if user_id == "banned":
        raise ValueError("User is banned")
    return {"token": "abc123"}


@error_handler("payment-service")
def process_payment(token: str, amount: float):
    # In reality this would call external payment service
    if amount <= 0:
        raise ValueError("Amount must be positive")
    return {"status": "charged", "amount": amount}


def handle_checkout(user_id: str, amount: float) -> dict:
    """
    Orchestrates a two-service request: auth, then payment.
    The trace ID is propagated across both services via the context variable.
    """
    # Initialize trace ID at the entry point of the request
    request_context = RequestContext(service_name="checkout-orchestrator")
    set_trace_id(request_context.trace_id)
    trace_id = request_context.trace_id

    logger.info("checkout started trace=%s user=%s amount=%s", trace_id, user_id, amount)

    try:
        # Service 1: Authenticate
        auth_result = authenticate(user_id)
        logger.info("auth success trace=%s", trace_id)

        # Service 2: Process payment
        payment_result = process_payment(auth_result["token"], amount)
        logger.info("payment success trace=%s", trace_id)

        return {
            "trace_id": trace_id,
            "auth": auth_result,
            "payment": payment_result,
        }

    except ServiceError as service_error:
        # The error already carries the trace ID; just re-raise or log
        logger.error(
            "checkout failed trace=%s service=%s step=%s message=%s",
            service_error.trace_id,
            service_error.service,
            service_error.step,
            service_error.message,
        )
        raise
    except Exception as unexpected:
        # Capture unexpected errors with the trace context
        raise ServiceError(
            str(unexpected),
            service="checkout-orchestrator",
            step="orchestration",
            trace_id=trace_id,
            cause=unexpected,
        ) from unexpected


def main():
    """Demo: run the flow and show the error output."""
    try:
        # Successful case
        result = handle_checkout(user_id="valid_user", amount=10.99)
        print(f"Success: {result}")
    except ServiceError as e:
        print(f"Failed (expected in demo): {e}")

    print("\n--- Now with an error ---")
    try:
        # Failing case - auth fails
        result = handle_checkout(user_id="banned", amount=10.99)
        print(f"Unexpected success: {result}")
    except ServiceError as e:
        print(f"Failed with context: {e}")


if __name__ == "__main__":
    main()
