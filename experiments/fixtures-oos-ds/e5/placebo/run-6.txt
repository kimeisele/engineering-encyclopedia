import logging
import uuid
from contextlib import contextmanager
from typing import Optional

logger = logging.getLogger(__name__)

REQUEST_ID_ATTR = "request_id"
SERVICE_NAME_ATTR = "service_name"


@contextmanager
def request_context(service_name: str, request_id: Optional[str] = None):
    """
    Context manager that injects a unique request ID and service name into
    all log records emitted within the block. This allows an operator to
    trace a single request as it moves from service A to service B.
    """
    if request_id is None:
        request_id = str(uuid.uuid4())

    # Store the current context so we can restore it after the block
    old_request_id = getattr(logging, REQUEST_ID_ATTR, None)
    old_service_name = getattr(logging, SERVICE_NAME_ATTR, None)

    try:
        logging.request_id = request_id
        logging.service_name = service_name
        yield request_id
    finally:
        # Restore the previous context
        if old_request_id is None:
            delattr(logging, REQUEST_ID_ATTR)
        else:
            logging.request_id = old_request_id
        if old_service_name is None:
            delattr(logging, SERVICE_NAME_ATTR)
        else:
            logging.service_name = old_service_name


class TwoServiceErrorHandler:
    """
    Error handler for a two-service request flow. It captures the request ID
    and service name from the logging context and formats the error with
    enough detail for an operator to trace the request across both services.
    """

    def handle(self, exc: Exception, service_name: str, operation: str,
               failing_step: str, request_id: Optional[str] = None) -> None:
        """
        Log a structured error message with the request ID, service, operation,
        failing step, and the original exception message/stack trace.

        Args:
            exc: The exception that occurred.
            service_name: Name of the service where the error occurred.
            operation: High-level operation being performed (e.g., "create_order").
            failing_step: Specific step that failed (e.g., "validate_payment").
            request_id: Optional request ID. If not provided, it will be
                        taken from the logging context or a new one generated.
        """
        # Get the request ID from the logging context if not provided
        effective_request_id = request_id or getattr(logging, REQUEST_ID_ATTR, None)
        if effective_request_id is None:
            effective_request_id = str(uuid.uuid4())

        # Build the error context
        error_context = {
            "request_id": effective_request_id,
            "service": service_name,
            "operation": operation,
            "failing_step": failing_step,
            "error_type": type(exc).__name__,
            "error_message": str(exc),
        }

        # Include the service name from logging context if it's set (for cross-service tracing)
        logging_service = getattr(logging, SERVICE_NAME_ATTR, None)
        if logging_service and logging_service != service_name:
            error_context["origin_service"] = logging_service

        # Log with the full stack trace for debugging, but the structured
        # context provides the traceability across services.
        logger.error(
            "Request failed: %(service)s / %(operation)s / %(failing_step)s "
            "request_id=%(request_id)s error=%(error_type)s: %(error_message)s",
            error_context,
            exc_info=True,
        )

        # Optional: propagate to an external monitoring system
        self._send_to_monitoring(error_context)


    def _send_to_monitoring(self, error_context: dict) -> None:
        """
        Hook for sending the error context to a monitoring/alerting system.
        Kept as a no-op placeholder—the actual implementation would ship
        the context to your APM (e.g., Datadog, Sentry) with the request ID.
        """
        # Example: sentry_sdk.capture_message(...) with error_context
