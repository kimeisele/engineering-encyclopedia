import logging
import uuid
from contextlib import contextmanager
from typing import Any, Dict, Optional

logger = logging.getLogger(__name__)

class RequestContext:
    """Holds and propagates tracing context for a two-service request."""
    def __init__(self, service_name: str, request_id: Optional[str] = None):
        self.service_name = service_name
        self.request_id = request_id or str(uuid.uuid4())
        self.step = "init"
        self.attrs: Dict[str, Any] = {}

    def add_attr(self, key: str, value: Any) -> None:
        """Add a non-sensitive attribute to the context."""
        self.attrs[key] = value

    def to_dict(self) -> Dict[str, Any]:
        """Serialize context for logging (excludes sensitive keys)."""
        safe_attrs = {k: v for k, v in self.attrs.items() if not k.lower().startswith("secret")}
        return {
            "request_id": self.request_id,
            "service": self.service_name,
            "step": self.step,
            **safe_attrs,
        }


@contextmanager
def step_context(ctx: RequestContext, step_name: str):
    """Set the current step for the request context."""
    ctx.step = step_name
    try:
        yield
    except Exception as e:
        # Attach step context to the exception before re-raised
        e.request_context = ctx.to_dict()  # type: ignore
        raise


class TwoServiceErrorHandler:
    """Error handler that logs the full context chain for traceability."""
    def __init__(self, ctx: RequestContext):
        self.ctx = ctx

    def handle_exception(self, exc: Exception) -> Dict[str, Any]:
        """Handle an exception, log it with tracing info, and return a structured envelope."""
        # Build the trace envelope with the current context plus exception details
        trace_data = {
            **self.ctx.to_dict(),
            "error_type": type(exc).__name__,
            "error_message": str(exc),
            "trace_id": self.ctx.request_id,
        }

        # Preserve any context that was attached to the exception through wrapping
        attached_context = getattr(exc, "request_context", None)
        if attached_context:
            trace_data["original_context"] = attached_context

        logger.error(
            "Request failed across services",
            extra={"request_trace": trace_data},
            exc_info=exc,  # Keep the stack trace for debugging
        )

        # Return a minimal, safe error response (don't leak secrets)
        return {
            "error": "request_failed",
            "request_id": self.ctx.request_id,
            "step": self.ctx.step,
            "service": self.ctx.service_name,
        }


def service_a_request(payload: Dict[str, Any], request_id: Optional[str] = None) -> Dict[str, Any]:
    """Service A entry point - starts the tracing context."""
    ctx = RequestContext("service_a", request_id)
    handler = TwoServiceErrorHandler(ctx)

    try:
        with step_context(ctx, "validate"):
            # Simulate validation
            if not payload.get("data"):
                raise ValueError("Missing required field: data")

        with step_context(ctx, "process"):
            # Simulate processing and calling Service B
            transformed = {"value": payload["data"] * 2}
            ctx.add_attr("transformed_value", transformed["value"])

            # Call Service B (simulated)
            service_b_result = service_b_request(transformed, request_id=ctx.request_id)
            ctx.add_attr("service_b_result", service_b_result)

        with step_context(ctx, "finalize"):
            return {"ok": True, "result": service_b_result, "request_id": ctx.request_id}

    except Exception as e:
        return handler.handle_exception(e)


def service_b_request(payload: Dict[str, Any], request_id: str) -> Dict[str, Any]:
    """Service B - receives the same request_id from Service A."""
    ctx = RequestContext("service_b", request_id)
    handler = TwoServiceErrorHandler(ctx)

    try:
        with step_context(ctx, "validate"):
            if payload["value"] < 0:
                raise ValueError("Value cannot be negative")

        with step_context(ctx, "store"):
            # Simulate storage operation
            ctx.add_attr("stored_value", payload["value"])
            return {"stored": True, "value": payload["value"]}

    except Exception as e:
        return handler.handle_exception(e)


# Example usage:
if __name__ == "__main__":
    logging.basicConfig(level=logging.DEBUG)
    # Successful request
    result = service_a_request({"data": 5})
    print("Success result:", result)

    # Failed request in Service B
    result = service_a_request({"data": -3})
    print("Failure result:", result)

    # Failed request in Service A
    result = service_a_request({"data": None})
    print("Failure result:", result)
