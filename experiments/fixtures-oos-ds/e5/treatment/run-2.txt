import uuid
import logging
from functools import wraps
from typing import Callable, Any

logger = logging.getLogger(__name__)

class CorrelationIdFilter(logging.Filter):
    """Logging filter that injects the correlation id from context vars."""
    def filter(self, record: logging.LogRecord) -> bool:
        record.correlation_id = getattr(record, 'correlation_id', 'unknown')
        return True


def _generate_correlation_id() -> str:
    """Generate a unique correlation id for a logical request."""
    return str(uuid.uuid4())


def get_or_create_correlation_id() -> str:
    """
    Retrieve the current correlation id from context, or create
    one if this is the entry point. The id is stored in a
    context variable so it survives async and threading boundaries
    within a single request.
    """
    import contextvars
    _correlation_id_ctx = contextvars.ContextVar('correlation_id', default=None)
    cid = _correlation_id_ctx.get()
    if cid is None:
        # Inbound call has no id: generate one at the entry point.
        cid = _generate_correlation_id()
        _correlation_id_ctx.set(cid)
    return cid


def with_correlation_id(func: Callable) -> Callable:
    """
    Decorator for service entrypoints. Ensures a correlation id exists
    and attaches it to the log record so all logging in this service
    (and downstream via propagation headers) carries the id.
    """
    @wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        cid = get_or_create_correlation_id()
        # Attach to logging context so all log lines include it.
        # In a real app you'd also propagate via header to the next service.
        logger.info("Processing request with correlation_id=%s", cid)
        return func(*args, **kwargs)
    return wrapper


def error_handler(func: Callable) -> Callable:
    """
    Error handler decorator:
    - Catches any exception
    - Adds correlation id and operation context to the message
    - Logs with full error context
    - Re-raises (so external handler can decide on retry / DLQ)
    """
    @wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        try:
            return func(*args, **kwargs)
        except Exception as e:
            cid = get_or_create_correlation_id()
            # Identify operation: use the function name as the failing step
            operation = func.__qualname__
            # Include the correlation id, operation, and original error message
            err_msg = f"[correlation_id={cid}] Operation '{operation}' failed: {e}"
            # Attach context to the exception for downstream wrapping
            e.add_note = lambda: None  # py<3.11 compat, real code would use add_note
            setattr(e, 'correlation_id', cid)
            setattr(e, 'operation', operation)
            # Log the enriched error
            logger.error(err_msg, exc_info=True)
            # Re-raise for the caller to handle (retry budget, DLQ, etc.)
            raise
    return wrapper
