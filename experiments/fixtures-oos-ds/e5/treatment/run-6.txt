# error_handler.py
import logging
import uuid
from contextlib import contextmanager
from typing import Any, Callable, Dict, Optional

# --- Correlation ID management ---
REQUEST_ID_HEADER = "X-Request-ID"

class RequestContext:
    """Holds the correlation ID that traces one request across services."""
    _contextvars = None
    # Simple context manager using contextvars for async or threading.local for sync
    try:
        import contextvars
        _contextvars = contextvars.ContextVar("request_id", default=None)
    except ImportError:
        import threading
        _local = threading.local()
        _contextvars = None

    @classmethod
    def set(cls, value: str):
        if cls._contextvars:
            cls._contextvars.set(value)
        else:
            cls._local.value = value

    @classmethod
    def get(cls) -> Optional[str]:
        if cls._contextvars:
            return cls._contextvars.get()
        return getattr(cls._local, "value", None)


def generate_request_id() -> str:
    """Generate a new, unique correlation ID at the edge of the request."""
    return str(uuid.uuid4())


@contextmanager
def request_scope(request_id: Optional[str] = None):
    """
    Establish a request scope with a correlation ID.
    If no id is provided, generate one (inbound edge case).
    The ID is stored in the context so all downstream calls and logs can use it.
    """
    incoming_id = request_id or RequestContext.get()
    effective_id = incoming_id or generate_request_id()
    old_id = RequestContext.get()
    RequestContext.set(effective_id)
    try:
        yield effective_id
    finally:
        RequestContext.set(old_id)


def get_correlation_id() -> Optional[str]:
    """Public accessor for the current correlation ID."""
    return RequestContext.get()


# --- Logging setup with correlation ID injection ---
class CorrelationIdFilter(logging.Filter):
    """Adds the correlation ID to every log record."""
    def filter(self, record: logging.LogRecord) -> bool:
        record.correlation_id = get_correlation_id() or "-"
        return True


# Configure root logger with the filter
handler = logging.StreamHandler()
handler.addFilter(CorrelationIdFilter())
logging.basicConfig(level=logging.INFO, handlers=[handler], format="%(asctime)s [%(levelname)s] cid=%(correlation_id)s %(message)s")

logger = logging.getLogger(__name__)


# --- Error handler for the two-service request ---
class ServiceRequestError(Exception):
    """Base error with operation and step context."""
    def __init__(self, operation: str, step: str, message: str, **context: Any):
        self.operation = operation
        self.step = step
        self.context = context
        # Safe message: no secrets, includes operation, step, and correlation ID
        super().__init__(
            f"{operation} failed at step '{step}': {message} "
            f"[cid={get_correlation_id() or 'unknown'}]"
        )


def wrap_service_error(operation: str, step: str, func: Callable, *args, **kwargs) -> Any:
    """
    Execute a service call, wrapping any exception with operation and step context.
    The correlation ID is already in the context from the request scope.
    """
    try:
        # Simulate downstream call with the correlation ID (e.g., HTTP header)
        function_context = {"correlation_id": get_correlation_id()}
        return func(*args, **kwargs, _request_context=function_context)
    except Exception as exc:
        logger.error("Downstream failure", exc_info=exc)
        raise ServiceRequestError(operation, step, str(exc)) from exc


# --- Example: Two-service request handler ---
def service_a_call(payload: Dict, _request_context: Optional[Dict] = None):
    """First service: makes an outbound call. Correlates with the ID."""
    # In a real system, this would include the correlation ID in the outbound HTTP header
    # e.g., requests.post(url, headers={REQUEST_ID_HEADER: _request_context["correlation_id"]})
    if payload.get("force_error"):
        raise ConnectionError("Service A unreachable")
    logger.info("Service A processed")
    return {"ok": True, "from_a": payload["data"]}


def service_b_call(result_a: Dict, _request_context: Optional[Dict] = None):
    """Second service: called after A succeeds."""
    if result_a.get("force_error_b"):
        raise ValueError("Invalid data in service B")
    logger.info("Service B processed")
    return {"final": result_a["from_a"] * 2}


def handle_two_service_request(inbound_request_id: Optional[str], payload: Dict) -> Dict:
    """
    Entry point for the two-service flow.
    Establishes the request scope, then calls both services with error wrapping.
    """
    with request_scope(inbound_request_id) as cid:
        logger.info("Starting two-service request")
        try:
            result_a = wrap_service_error(
                "two_service_flow", "service_a",
                service_a_call, payload
            )
            final_result = wrap_service_error(
                "two_service_flow", "service_b",
                service_b_call, result_a, force_error_b=payload.get("force_error_b")
            )
            logger.info("Two-service request completed successfully")
            return final_result
        except ServiceRequestError as exc:
            # The error message already includes operation, step, and correlation ID
            logger.error("Request failed: %s", exc)
            # In production, this might go to a dead-letter queue or error response
            raise


# --- Usage examples for the operator ---
if __name__ == "__main__":
    # Normal case
    print(handle_two_service_request("req-123", {"data": 5}))

    # Error in service A
    try:
        handle_two_service_request("req-456", {"data": 5, "force_error": True})
    except ServiceRequestError as e:
        print(f"Caught: {e}")

    # Error in service B
    try:
        handle_two_service_request("req-789", {"data": 5, "force_error_b": True})
    except ServiceRequestError as e:
        print(f"Caught: {e}")

    # Inbound without ID → generates one at the edge
    try:
        handle_two_service_request(None, {"data": 3, "force_error": True})
    except ServiceRequestError as e:
        print(f"Caught with generated CID: {e}")
