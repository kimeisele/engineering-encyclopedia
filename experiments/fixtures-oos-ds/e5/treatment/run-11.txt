# error_handler.py
import logging
import uuid
import sys
from contextlib import contextmanager
from typing import Dict, Optional

logger = logging.getLogger("request_trace")


class MissingCorrelationIdError(Exception):
    """Raised when an inbound request has no correlation id."""


def _extract_correlation_id(headers: Dict[str, str]) -> str:
    """Extract or create a correlation id for the inbound request.

    If the inbound request carries a correlation id, use it so we can
    trace across services. If not, create a new one to identify this
    logical request. We never regenerate; this is the one definition of
    the request.
    """
    # Pull the header name we agreed on with the other service.
    inbound_id = headers.get("X-Correlation-Id")
    if inbound_id:
        return inbound_id
    # No inbound id: this is the entry point of the logical request.
    return str(uuid.uuid4())


@contextmanager
def correlation_context(correlation_id: str):
    """Make the correlation id available to every log line in this scope."""
    old = logger.extra.get("correlation_id") if hasattr(logger, "extra") else None
    # In real code you'd use a contextvar, but for clarity we set a thread-local.
    import threading
    _local = threading.local()
    _local.correlation_id = correlation_id
    try:
        yield
    finally:
        _local.correlation_id = old


class TwoServiceErrorHandler:
    """Wraps a two-service request so every failure is traceable."""

    def __init__(self, service_name: str, failure_step: str):
        self.service_name = service_name
        self.failure_step = failure_step

    def __call__(self, headers: Dict[str, str]):
        correlation_id = _extract_correlation_id(headers)
        with correlation_context(correlation_id):
            def wrapper(func):
                def inner(*args, **kwargs):
                    try:
                        # Attach the id to the outbound call so the next service sees it.
                        kwargs.setdefault("headers", {})
                        kwargs["headers"]["X-Correlation-Id"] = correlation_id
                        logger.info(
                            "correlation_id=%s service=%s step=%s starting",
                            correlation_id, self.service_name, self.failure_step,
                        )
                        return func(*args, **kwargs)
                    except Exception as exc:
                        # We know the operation and the failing step here.
                        logger.error(
                            "correlation_id=%s service=%s step=%s error=%s",
                            correlation_id, self.service_name, self.failure_step,
                            exc,
                        )
                        # Re-raise with the correlation id attached for the caller.
                        raise RequestError(
                            f"{self.service_name} failed at {self.failure_step} "
                            f"(correlation_id={correlation_id}): {exc}"
                        ) from exc
                return inner
            return wrapper


class RequestError(Exception):
    """Error with correlation context so an operator can trace it."""


def handle_dead_letter(exc: Exception, correlation_id: str, dead_letter_topic: str) -> None:
    """After the retry budget is exhausted, move the message to a dead-letter queue.

    We never silently drop; the dead letter is the surfaced end-of-budget state.
    """
    # In production this would publish to a DLQ topic. Here we log it clearly.
    logger.error(
        "DEAD_LETTER correlation_id=%s error=%s dlq=%s — operator attention required",
        correlation_id, exc, dead_letter_topic,
    )


# Example usage for a two-service request:
# service_a calls service_b; the correlation id flows through both.
def service_a_workflow(headers):
    handler_a = TwoServiceErrorHandler("service_a", "fetch_user_data")
    handler_b = TwoServiceErrorHandler("service_b", "persist_order")

    @handler_a(headers)
    def call_a():
        # The decorator attached the correlation id to headers.
        # In a real app, this would be an HTTP call to service B.
        outbound_headers = {"X-Correlation-Id": headers["X-Correlation-Id"]}
        return handler_b(outbound_headers)(lambda: "order_saved")()

    result = call_a()
    return result
