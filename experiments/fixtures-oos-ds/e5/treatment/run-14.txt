# error_handler.py
import logging
import uuid
from typing import Optional, Callable, Any
from contextlib import contextmanager
from dataclasses import dataclass, field
from enum import Enum

logger = logging.getLogger(__name__)

class ServiceName(Enum):
    SERVICE_A = "service-a"
    SERVICE_B = "service-b"

@dataclass
class CorrelationContext:
    """Holds the correlation ID for the current request thread."""
    correlation_id: Optional[str] = None
    service_name: Optional[str] = None

    def generate_if_missing(self) -> str:
        if not self.correlation_id:
            self.correlation_id = str(uuid.uuid4())
        return self.correlation_id

# Thread-local storage to hold the current correlation context
import threading
_thread_local = threading.local()

def get_correlation_context() -> CorrelationContext:
    """Get the current thread's correlation context."""
    if not hasattr(_thread_local, 'correlation_context'):
        _thread_local.correlation_context = CorrelationContext()
    return _thread_local.correlation_context

@contextmanager
def correlation_scope(correlation_id: Optional[str] = None, service_name: str = "unknown"):
    """
    Context manager that sets up a correlation ID scope for a request.
    - If correlation_id is provided (from inbound request), use it.
    - If None, generate a new one.
    - Sets thread-local state so all logging in the scope includes the ID.
    """
    ctx = get_correlation_context()
    previous_cid = ctx.correlation_id
    previous_service = ctx.service_name
    
    # Use provided ID or generate new one — never regenerate mid-request
    ctx.correlation_id = correlation_id or ctx.generate_if_missing()
    ctx.service_name = service_name
    
    # Create a logger adapter that includes the correlation ID
    log_adapter = logging.LoggerAdapter(
        logger,
        {"correlation_id": ctx.correlation_id, "service": service_name}
    )
    
    try:
        yield log_adapter
    finally:
        # Restore previous state (for nested scopes)
        ctx.correlation_id = previous_cid
        ctx.service_name = previous_service

def propagate_correlation_id_to_downstream(
    downstream_call: Callable[[str], Any],
    correlation_id: str
) -> Any:
    """
    Propagate the correlation ID to a downstream service call.
    In reality, this would attach the ID to an HTTP header, message queue
    property, or RPC metadata. Here we demonstrate the pattern with a
    caller that accepts the ID as a parameter.
    """
    return downstream_call(correlation_id)

def build_error_message(
    operation: str,
    failing_step: str,
    correlation_id: Optional[str],
    service: Optional[str],
    instance: Optional[str] = None,
    context: Optional[dict] = None
) -> str:
    """
    Build a structured error message with full context for tracing.
    Includes: operation, failing step, correlation ID, service, instance
    and any additional non-sensitive context.
    """
    parts = [
        f"operation={operation}",
        f"step={failing_step}",
    ]
    if correlation_id:
        parts.append(f"correlation_id={correlation_id}")
    if service:
        parts.append(f"service={service}")
    if instance:
        parts.append(f"instance={instance}")
    if context:
        # Filter out potential sensitive keys — in production this would
        # use a proper allowlist, not just a denylist
        safe_keys = [k for k in context.keys() if 'password' not in k.lower() 
                     and 'token' not in k.lower() and 'secret' not in k.lower()]
        safe_context = {k: context[k] for k in safe_keys}
        parts.append(f"context={safe_context}")
    
    return " | ".join(parts)

class ServiceError(Exception):
    """Base error for service failures that carries correlation context."""
    
    def __init__(
        self,
        operation: str,
        failing_step: str,
        correlation_id: Optional[str],
        service: Optional[str],
        message: str = "",
        context: Optional[dict] = None
    ):
        self.operation = operation
        self.failing_step = failing_step
        self.correlation_id = correlation_id
        self.service = service
        self.context = context or {}
        
        full_message = build_error_message(
            operation=operation,
            failing_step=failing_step,
            correlation_id=correlation_id,
            service=service,
            context=self.context
        )
        if message:
            full_message = f"{message} | {full_message}"
        super().__init__(full_message)

def error_handler_for_operator(
    func: Callable,
    operation_name: str,
    service: ServiceName,
    instance_id: str
) -> Callable:
    """
    Decorator that wraps a request handler to:
    1. Ensure correlation ID exists (from inbound requests or new generation)
    2. Catch all exceptions and wrap them with full operator context
    3. Log with the correlation ID in every line
    4. Re-raise the wrapped exception so upstream error handling continues
    """
    def wrapper(*args, **kwargs):
        ctx = get_correlation_context()
        correlation_id = ctx.correlation_id
        service_name = service.value
        
        # If no correlation ID in thread-local (e.g., started this service),
        # generate one now
        if not correlation_id:
            correlation_id = ctx.generate_if_missing()
            ctx.service_name = service_name
        
        log_adapter = logging.LoggerAdapter(
            logger,
            {"correlation_id": correlation_id, "service": service_name}
        )
        log_adapter.info(f"Starting {operation_name} on instance={instance_id}")
        
        try:
            return func(*args, **kwargs)
        except ServiceError as e:
            # Already has correlation context; just re-raise
            log_adapter.error(
                f"ServiceError in {operation_name}: {str(e)}",
                exc_info=True
            )
            raise
        except Exception as exc:
            # Wrap with correlation context and operator-traceable info
            wrapped = ServiceError(
                operation=operation_name,
                failing_step="unknown",
                correlation_id=correlation_id,
                service=service_name,
                message=f"Unhandled exception: {exc}",
                context={"instance": instance_id}
            )
            log_adapter.error(
                f"Unhandled exception in {operation_name}: {str(wrapped)}",
                exc_info=True
            )
            raise wrapped from exc
    
    return wrapper

# Example usage for a two-service request flow:
# ============================================================

# In Service A (the entry point receiving the user request):
def service_a_request_handler(request_payload: dict):
    """
    Entry point for a request that will call Service B.
    This is where the correlation ID is created (if not in inbound headers)
    and propagated down to Service B.
    """
    # In real code, correlation_id would come from request headers (X-Correlation-ID)
    inbound_correlation_id = request_payload.get("headers", {}).get("X-Correlation-ID")
    
    # Enter correlation scope for this request — generates if missing
    with correlation_scope(correlation_id=inbound_correlation_id, service_name="service-a"):
        ctx = get_correlation_context()
        correlation_id = ctx.correlation_id
        log = logging.LoggerAdapter(logger, {"correlation_id": correlation_id, "service": "service-a"})
        log.info(f"Service A received request for user={request_payload['user_id']}")
        
        try:
            # Process in Service A
            processed_data = process_in_service_a(request_payload["data"])
            
            # Call Service B, propagating the correlation ID
            # In real code: HTTP header X-Correlation-ID, or MQ message property
            service_b_response = propagate_correlation_id_to_downstream(
                lambda cid: call_service_b(cid, processed_data),
                correlation_id
            )
            
            return service_b_response
        except ServiceError:
            # Already has full context
            raise
        except Exception as e:
            raise ServiceError(
                operation="service_a_request_handler",
                failing_step="processing_or_downstream_call",
                correlation_id=correlation_id,
                service="service-a",
                message=f"Failed to process request: {e}",
                context={"user_id": request_payload["user_id"]}
            )

# In Service B (the downstream service):
def call_service_b(correlation_id: str, data: dict):
    """
    Service B receives the correlation ID from the downstream call.
    It enters its own correlation scope using the provided ID.
    """
    with correlation_scope(correlation_id=correlation_id, service_name="service-b"):
        ctx = get_correlation_context()
        log = logging.LoggerAdapter(logger, {"correlation_id": ctx.correlation_id, "service": "service-b"})
        log.info(f"Service B processing data received from Service A")
        
        try:
            # Do work in Service B
            result = process_in_service_b(data)
            log.info(f"Service B completed processing: {result}")
            return {"status": "success", "data": result}
        except Exception as e:
            log.error(f"Service B failed: {e}", exc_info=True)
            raise ServiceError(
                operation="call_service_b",
                failing_step="process_in_service_b",
                correlation_id=ctx.correlation_id,
                service="service-b",
                message=f"Service B processing failed: {e}",
                context={"input_schema": list(data.keys())}
            )

def process_in_service_a(data: dict):
    # Simulated processing
    return {"processed_a": data}

def process_in_service_b(data: dict):
    # Simulated processing — could raise
    return {"processed_b": data}

# ============================================================
# End-to-end demonstration:
# ============================================================
if __name__ == "__main__":
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(name)s | cid=%(correlation_id)s | svc=%(service)s | %(message)s"
    )
    
    # Simulate a user request to Service A
    user_request = {
        "headers": {},  # No correlation ID — Service A will generate one
        "user_id": "user-12345",
        "data": {"action": "purchase", "amount": 99.99}
    }
    
    try:
        response = service_a_request_handler(user_request)
        print(f"Response: {response}")
    except ServiceError as e:
        print(f"Error: {e}")
        print(f"  correlation_id: {e.correlation_id}")
        print(f"  service: {e.service}")
        print(f"  operation: {e.operation}")
        print(f"  step: {e.failing_step}")
