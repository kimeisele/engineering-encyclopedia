import logging
import uuid
from contextvars import ContextVar
from typing import Optional, Callable, Any

# --- Correlation ID propagation ---
# ContextVar so the id survives async boundaries without threading it
# through every function signature.
_correlation_id: ContextVar[Optional[str]] = ContextVar("correlation_id", default=None)

def get_correlation_id() -> Optional[str]:
    """Return the current correlation id, or None if not set."""
    return _correlation_id.get()

def set_correlation_id(cid: str) -> None:
    """Set the correlation id for the current execution context."""
    _correlation_id.set(cid)

def new_correlation_id() -> str:
    """Generate a new correlation id (UUID4)."""
    return str(uuid.uuid4())

def correlation_id_middleware(handler: Callable[..., Any]):
    """ASGI/WSGI-style middleware: create or accept id at the entry point."""
    def wrapper(scope, receive, send):
        # Accept inbound id if present (e.g. from HTTP header X-Correlation-Id)
        inbound = scope.get("headers", {}).get(b"X-correlation-id")
        cid = inbound.decode() if inbound else new_correlation_id()
        set_correlation_id(cid)
        # Ensure the id is sent back in the response header
        original_send = send
        async def send_with_header(message):
            if message["type"] == "http.response.start":
                headers = list(message.get("headers", []))
                headers.append((b"X-Correlation-Id", cid.encode()))
                message["headers"] = headers
            await original_send(message)
        return handler(scope, receive, send_with_header)
    return wrapper


# --- Error handler with traceable context ---
def error_handler(exc: Exception, operation: str, step: str, extra_context: Optional[dict] = None):
    """
    Wrap an exception with operation and step context plus correlation id.
    Returns a new exception with a message that an operator can search on.
    """
    cid = get_correlation_id() or "NO_CORRELATION_ID"
    context = {
        "operation": operation,
        "step": step,
        "correlation_id": cid,
    }
    if extra_context:
        context.update(extra_context)
    # Build message: includes the original message but preserves the chain
    # via __cause__ if the caller does `raise new from exc`.
    msg = f"[correlation_id={cid}] {operation} failed at {step}: {exc}"
    new_exc = type(exc)(msg)
    new_exc.__context__ = exc  # preserve original for stack trace
    return new_exc

# --- Circuit breaker integration (prevent hammering) ---
class CircuitBreaker:
    """
    Minimal circuit breaker: after N failures in a window, open for cooldown.
    When open, callers fail fast with a clear error including the correlation id.
    """
    def __init__(self, failure_threshold: int = 5, cooldown_seconds: float = 60.0):
        self._threshold = failure_threshold
        self._cooldown = cooldown_seconds
        self._failures = 0
        self._opened_at = None

    def record_failure(self):
        self._failures += 1
        if self._failures >= self._threshold:
            self._opened_at = time.time()
            self._failures = 0

    def record_success(self):
        self._failures = 0
        self._opened_at = None

    def allow_call(self) -> bool:
        if self._opened_at is None:
            return True
        if time.time() - self._opened_at > self._cooldown:
            self._opened_at = None
            return True
        return False

    def call(self, func: Callable, *args, **kwargs):
        if not self.allow_call():
            raise CircuitOpenError(get_correlation_id())
        try:
            result = func(*args, **kwargs)
            self.record_success()
            return result
        except Exception:
            self.record_failure()
            raise

class CircuitOpenError(Exception):
    def __init__(self, cid):
        super().__init__(f"Circuit open; skipping call (correlation_id={cid})")
