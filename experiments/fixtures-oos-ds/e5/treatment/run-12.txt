# error_handler.py
import uuid
import logging
from contextvars import ContextVar
from typing import Optional, Dict, Any

logger = logging.getLogger(__name__)
correlation_id_var: ContextVar[Optional[str]] = ContextVar("correlation_id", default=None)


class CorrelationIdMiddleware:
    """Creates or propagates correlation ID at service entry."""
    
    def __init__(self, get_response):
        self.get_response = get_response
    
    def __call__(self, request):
        # Extract from inbound header, or create new one
        cid = request.headers.get("X-Correlation-ID")
        if not cid:
            cid = str(uuid.uuid4())
            # Log that we generated a new one (so operator knows it started here)
            logger.warning("No correlation ID in inbound request; generated %s", cid)
        correlation_id_var.set(cid)
        
        # Attach to response so downstream/upstream can propagate
        response = self.get_response(request)
        response["X-Correlation-ID"] = cid
        return response


class ErrorHandler:
    """Wraps exceptions with correlation context and sends to DLQ if needed."""
    
    def __init__(self, dlq_publisher=None):
        self.dlq_publisher = dlq_publisher  # callable for dead-letter
    
    def __call__(self, exc, context: Dict[str, Any] = None):
        cid = correlation_id_var.get() or "unknown"
        context = context or {}
        
        # Build error with operation, failing step, and correlation ID
        operation = context.get("operation", "unknown")
        step = context.get("step", "unknown")
        details = {
            "correlation_id": cid,
            "operation": operation,
            "step": step,
            "error_type": type(exc).__name__,
            "error_message": str(exc)[:500],  # limit size, no secrets
            "service": context.get("service", "unknown"),
        }
        
        # Log with correlation id in every line
        logger.error(
            "Correlation=%s Operation=%s Step=%s Error=%s: %s",
            cid, operation, step, type(exc).__name__, str(exc),
            extra={"correlation_id": cid}
        )
        
        # Dead-letter if configured — after retry budget exhausted
        if self.dlq_publisher:
            self.dlq_publisher(details)
        else:
            # Without DLQ, we still trace but don't drop silently
            logger.critical("No DLQ configured; message will be retried or lost: %s", details)
        
        # Return structured error for caller
        return {
            "error": type(exc).__name__,
            "correlation_id": cid,
            "operation": operation,
            "step": step,
        }


class ServiceAClient:
    """Client that propagates correlation ID to downstream service."""
    
    def call_service_b(self, payload):
        import requests
        cid = correlation_id_var.get()
        headers = {"X-Correlation-ID": cid} if cid else {}
        # Circuit breaker would protect this call (see reliability.circuit-breaker)
        try:
            resp = requests.post("https://service-b/internal", json=payload, headers=headers, timeout=5)
            resp.raise_for_status()
            return resp.json()
        except requests.RequestException as e:
            handler = ErrorHandler()
            handler(e, {"operation": "call_service_b", "step": "http_post", "service": "service_a"})
            raise


# Usage example in a service:
def handle_request(request):
    cid = correlation_id_var.get()
    handler = ErrorHandler()
    try:
        # ... do work ...
        result = ServiceAClient().call_service_b({"data": "x"})
        return {"status": "ok", "correlation_id": cid}
    except Exception as e:
        # Attach context BEFORE wrapping to preserve original error
        return handler(e, {"operation": "handle_request", "step": "main", "service": "service_a"})
