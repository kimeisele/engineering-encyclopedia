# errors.py
import logging
import uuid
from functools import wraps
from typing import Callable

logger = logging.getLogger(__name__)

class ServiceError(Exception):
    """Base service error with correlation context."""

    def __init__(self, message: str, correlation_id: str, service: str, operation: str, step: str):
        super().__init__(message)
        self.correlation_id = correlation_id
        self.service = service
        self.operation = operation
        self.step = step
        self.message = message

    def to_dict(self) -> dict:
        return {
            "error": self.message,
            "correlation_id": self.correlation_id,
            "service": self.service,
            "operation": self.operation,
            "step": self.step,
        }


def correlation_id_middleware(get_response: Callable):
    """Extract or create correlation id, propagate to downstream."""
    @wraps(get_response)
    def middleware(request):
        correlation_id = request.headers.get("X-Correlation-Id")
        if not correlation_id:
            correlation_id = str(uuid.uuid4())
        request.correlation_id = correlation_id
        response = get_response(request)
        response["X-Correlation-Id"] = correlation_id
        return response
    return middleware


def error_handler(fn: Callable) -> Callable:
    """Wrap a service function to catch exceptions, attach correlation context."""
    @wraps(fn)
    def wrapper(*args, **kwargs):
        try:
            return fn(*args, **kwargs)
        except ServiceError as e:
            logger.error(
                "Service error: %s correlation=%s service=%s operation=%s step=%s",
                e.message, e.correlation_id, e.service, e.operation, e.step
            )
            raise
        except Exception as e:
            correlation_id = getattr(kwargs.get('request', None), 'correlation_id', "unknown")
            logger.exception(
                "Unhandled error: %s correlation=%s",
                str(e), correlation_id
            )
            raise ServiceError(
                message=str(e),
                correlation_id=correlation_id,
                service=fn.__module__.split('.')[-1],
                operation=fn.__name__,
                step="unknown"
            ) from e
    return wrapper


# service_a.py
class ServiceA:
    @correlation_id_middleware
    def handle(self, request):
        correlation_id = request.correlation_id
        try:
            # Step 1: validate request
            self._validate(request)
            # Step 2: call Service B with correlation id propagated
            response = self._call_service_b(correlation_id, request.payload)
            return response
        except Exception:
            raise

    @error_handler
    def _validate(self, request):
        if not request.payload.get("id"):
            raise ServiceError(
                message="Missing required field 'id'",
                correlation_id=request.correlation_id,
                service="service-a",
                operation="validate",
                step="validation"
            )

    @error_handler
    def _call_service_b(self, correlation_id: str, payload: dict):
        # In real code: make HTTP call with X-Correlation-Id header
        import requests
        headers = {"X-Correlation-Id": correlation_id}
        try:
            resp = requests.post(
                "http://service-b/api/process",
                json=payload,
                headers=headers,
                timeout=5
            )
            resp.raise_for_status()
            return resp.json()
        except requests.exceptions.RequestException as e:
            raise ServiceError(
                message=f"Downstream call failed: {str(e)}",
                correlation_id=correlation_id,
                service="service-a",
                operation="call_service_b",
                step="http_call"
            ) from e
