import logging
import uuid
from typing import Optional, Dict, Any

logger = logging.getLogger(__name__)

class CorrelationIdFilter(logging.Filter):
    """Injects correlation_id into every log record."""
    def filter(self, record: logging.LogRecord) -> bool:
        record.correlation_id = getattr(record, 'correlation_id', 'no-correlation-id')
        return True

def get_or_create_correlation_id(headers: Optional[Dict[str, str]] = None) -> str:
    """
    Extract existing correlation id from inbound request, or generate one.
    Propagate it downstream via headers.
    """
    if headers and 'X-Correlation-Id' in headers:
        return headers['X-Correlation-Id']
    return str(uuid.uuid4())

def make_downstream_headers(correlation_id: str) -> Dict[str, str]:
    """Attach correlation id to outbound calls."""
    return {'X-Correlation-Id': correlation_id}

class ServiceError(Exception):
    """Base error with operation context."""
    def __init__(self, message: str, operation: str, step: str, correlation_id: str, **context: Any):
        self.operation = operation
        self.step = step
        self.correlation_id = correlation_id
        self.context = context
        # Include identifying state, exclude secrets (override in subclasses)
        state = ", ".join(f"{k}={v}" for k, v in context.items())
        super().__init__(f"[{correlation_id}] {operation} failed at {step}: {message} ({state})")

def handle_service_request(request_headers: Dict[str, str], service_call, max_retries: int = 3):
    """
    Two-service request handler with correlation id propagation, retry budget,
    and error wrapping. Returns (response) or raises.
    """
    correlation_id = get_or_create_correlation_id(request_headers)
    logger.info("request_started", extra={'correlation_id': correlation_id})

    call_headers = make_downstream_headers(correlation_id)

    for attempt in range(max_retries):
        try:
            response = service_call(call_headers)
            logger.info("request_succeeded", extra={'correlation_id': correlation_id})
            return response
        except Exception as e:
            logger.warning(
                "request_attempt_failed",
                extra={'correlation_id': correlation_id, 'attempt': attempt + 1}
            )
            if attempt == max_retries - 1:
                # Dead-letter: surface to operator, don't silently drop
                error = ServiceError(
                    str(e), 
                    operation="two_service_request",
                    step=f"call_attempt_{attempt+1}",
                    correlation_id=correlation_id
                )
                logger.error("request_dead_lettered", exc_info=error)
                raise error

def error_handler(exc: Exception) -> None:
    """
    Central error handler for operator tracing.
    Assumes correlation_id is available from exception or log context.
    """
    if isinstance(exc, ServiceError):
        # Already enriched - log with full context
        logger.error(
            f"Operation {exc.operation} failed: {exc}",
            extra={'correlation_id': exc.correlation_id}
        )
    else:
        # Unexpected error - try to extract correlation id from current context
        correlation_id = getattr(logging.currentframe().f_locals, 'correlation_id', 'unknown')
        logger.error(
            f"Unhandled error: {exc}",
            extra={'correlation_id': correlation_id}
        )
