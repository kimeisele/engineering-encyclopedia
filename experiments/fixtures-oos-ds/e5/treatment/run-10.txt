# error_handler.py
import logging
import uuid
from contextvars import ContextVar
from typing import Optional

logger = logging.getLogger(__name__)

# Correlation ID propagation using ContextVar (works within a single process/thread)
correlation_id_var: ContextVar[Optional[str]] = ContextVar('correlation_id', default=None)


def get_correlation_id() -> Optional[str]:
    """Return the current correlation ID or None."""
    return correlation_id_var.get()


def set_correlation_id(cid: Optional[str]) -> None:
    """Set the correlation ID for the current context."""
    correlation_id_var.set(cid)


def extract_or_create_correlation_id(headers: dict) -> str:
    """
    Extract the correlation ID from inbound headers, or generate a new one.
    Returns the ID to use for this request.
    """
    cid = headers.get('X-Correlation-ID') or headers.get('x-correlation-id')
    if not cid:
        cid = str(uuid.uuid4())
        logger.warning("No correlation ID in inbound request; generated %s", cid)
    set_correlation_id(cid)
    return cid


def outbound_headers() -> dict:
    """Attach the current correlation ID to outbound headers."""
    cid = get_correlation_id()
    if cid:
        return {'X-Correlation-ID': cid}
    return {}


class CorrelatedError(Exception):
    """Base error carrying correlation ID and operation context."""

    def __init__(self, operation: str, step: str, correlation_id: Optional[str] = None,
                 instance_id: Optional[str] = None):
        self.operation = operation
        self.step = step
        self.correlation_id = correlation_id or get_correlation_id()
        self.instance_id = instance_id
        super().__init__(self._format_message())

    def _format_message(self) -> str:
        parts = [
            f"operation={self.operation}",
            f"step={self.step}",
            f"correlation_id={self.correlation_id}",
        ]
        if self.instance_id:
            parts.append(f"instance_id={self.instance_id}")
        return " ".join(parts)


class ServiceErrorHandler:
    """
    Error handler for a two-service request flow.
    Assumes Service A calls Service B; both use this handler.
    """

    def __init__(self, service_name: str, retry_budget: int = 3):
        self.service_name = service_name
        self.retry_budget = retry_budget

    def handle_downstream_error(self, operation: str, step: str, error: Exception,
                                instance_id: Optional[str] = None) -> None:
        """
        Wrap an error from a downstream call with correlation context.
        Used in Service A when calling Service B.
        """
        cid = get_correlation_id() or str(uuid.uuid4())
        wrapped = CorrelatedError(
            operation=operation,
            step=step,
            correlation_id=cid,
            instance_id=instance_id
        )
        # Attach original error as cause, keep original stack
        wrapped.__cause__ = error
        logger.error("Downstream call failed: %s; cause: %s", wrapped, error)
        raise wrapped

    def handle_request_error(self, operation: str, step: str, error: Exception,
                             correlation_id: Optional[str] = None) -> None:
        """
        Handle an error at the top-level request boundary.
        Ensures correlation ID is logged even if not set earlier.
        Used in both services for their entry points.
        """
        cid = correlation_id or get_correlation_id() or str(uuid.uuid4())
        wrapped = CorrelatedError(operation=operation, step=step, correlation_id=cid)
        wrapped.__cause__ = error
        logger.error("Request failed in %s: %s", self.service_name, wrapped)
        raise wrapped

    def handle_dead_letter(self, operation: str, message_id: str, final_error: Exception,
                           correlation_id: Optional[str] = None) -> None:
        """
        After retry budget exhausted, route to dead-letter handling.
        Logs clearly and does NOT silently drop.
        """
        cid = correlation_id or get_correlation_id() or str(uuid.uuid4())
        wrapped = CorrelatedError(operation=operation, step="dead-letter", correlation_id=cid)
        wrapped.__cause__ = final_error
        logger.error(
            "DEAD LETTER after %d retries: message_id=%s operation=%s correlation_id=%s",
            self.retry_budget, message_id, operation, cid
        )
        # In production: push to DLQ, alert operator, store original payload
        raise wrapped


# ===== Usage example =====
# Service A entry point
def service_a_entry(request_headers: dict, request_body: dict):
    cid = extract_or_create_correlation_id(request_headers)
    handler = ServiceErrorHandler("service-a")
    try:
        # ... process request ...
        # Call service B with correlation ID
        downstream_headers = outbound_headers()
        try:
            # hypothetical http call
            # response = http.post("service-b/process", headers=downstream_headers)
            pass
        except Exception as e:
            handler.handle_downstream_error(operation="process-order", step="call-service-b", error=e)
    except CorrelatedError as ce:
        # Already logged, return error response with correlation ID
        return {"error": str(ce), "correlation_id": cid}, 500
    except Exception as e:
        handler.handle_request_error(operation="process-order", step="entry", error=e, correlation_id=cid)


# Service B entry point
def service_b_entry(request_headers: dict, request_body: dict):
    cid = extract_or_create_correlation_id(request_headers)
    handler = ServiceErrorHandler("service-b")
    try:
        # ... process ...
        pass
    except CorrelatedError as ce:
        return {"error": str(ce), "correlation_id": cid}, 500
    except Exception as e:
        handler.handle_request_error(operation="process-order", step="entry", error=e, correlation_id=cid)
