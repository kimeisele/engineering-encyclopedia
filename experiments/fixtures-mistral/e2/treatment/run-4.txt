```python
import subprocess
import re

def run_git_command(branch_name: str) -> bool:
    """
    Run a git command with a user-supplied branch name safely.

    Args:
        branch_name: The branch name provided by the user

    Returns:
        bool: True if the command succeeded, False otherwise
    """
    # Validate the branch name to ensure it's safe
    if not re.match(r'^[a-zA-Z0-9_\-\./]+$', branch_name):
        raise ValueError("Invalid branch name")

    # Build the command as a list to avoid shell injection
    command = ['git', 'checkout', branch_name]

    try:
        # Run the command without shell=True to prevent injection
        result = subprocess.run(
            command,
            check=True,
            capture_output=True,
            text=True
        )
        return True
    except subprocess.CalledProcessError as e:
        print(f"Git command failed: {e.stderr}")
        return False
