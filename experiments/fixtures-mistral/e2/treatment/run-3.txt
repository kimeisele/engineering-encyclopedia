import subprocess
import re

def run_git_command(branch_name):
    """
    Run a git command with a user-supplied branch name safely.

    Args:
        branch_name (str): The branch name to use in the git command.

    Returns:
        subprocess.CompletedProcess: The result of the git command execution.

    Raises:
        ValueError: If the branch name is invalid.
    """
    # Validate the branch name to ensure it's safe to use
    if not re.match(r'^[a-zA-Z0-9_\-\./]+$', branch_name):
        raise ValueError("Invalid branch name")

    # Construct the command as a list to avoid shell injection
    command = ['git', 'checkout', branch_name]

    # Run the command without shell=True to prevent injection
    result = subprocess.run(
        command,
        check=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True
    )

    return result
