import subprocess
import re

def run_git_command(branch_name):
    """
    Run a git command with a user-supplied branch name safely.

    Args:
        branch_name: The branch name to use in the git command.

    Returns:
        subprocess.CompletedProcess: The result of the git command execution.

    Raises:
        ValueError: If the branch name is invalid.
        subprocess.CalledProcessError: If the git command fails.
    """
    # Validate branch name: alphanumeric, hyphens, underscores, slashes, dots
    # Disallow sequences like '..', '//', '@{', and control characters
    if not re.match(r'^[a-zA-Z0-9_\-\./]+$', branch_name):
        raise ValueError("Invalid branch name")

    # Prevent path traversal or command injection by ensuring it's a single branch name
    if '..' in branch_name or branch_name.startswith('/') or branch_name.endswith('/'):
        raise ValueError("Invalid branch name")

    # Use list form with no shell=True to prevent injection
    command = ['git', 'checkout', branch_name]

    # Run the command safely
    result = subprocess.run(
        command,
        check=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True
    )

    return result
