import subprocess
import re

def run_git_command(branch_name: str) -> subprocess.CompletedProcess:
    """
    Run a git command with a user-supplied branch name safely.

    Args:
        branch_name: The branch name to use in the git command.

    Returns:
        The CompletedProcess result from the git command.

    Raises:
        ValueError: If the branch name is invalid.
        subprocess.CalledProcessError: If the git command fails.
    """
    # Validate branch name: only alphanumeric, hyphens, underscores, and slashes
    if not re.match(r'^[a-zA-Z0-9_\-/]+$', branch_name):
        raise ValueError("Invalid branch name")

    # Construct command as a list to avoid shell injection
    command = ['git', 'checkout', branch_name]

    # Run the command without shell=True to prevent injection
    return subprocess.run(
        command,
        check=True,
        capture_output=True,
        text=True
    )
