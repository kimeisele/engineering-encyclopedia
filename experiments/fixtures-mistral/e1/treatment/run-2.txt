class IdempotentPaymentWorker:

        self.queue = queue
        self.payment_processor = payment_processor

        if not self.completed_jobs_file.exists():
            self.completed_jobs_file.write_text("{}")
        self.completed_jobs = json.loads(self.completed_jobs_file.read_text())

    def _is_job_completed(self, idempotency_key: str) -> bool:
        """Check if a job with this idempotency key was already completed."""
        return idempotency_key in self.completed_jobs

    def _record_completion(self, idempotency_key: str):
        """Record that a job with this idempotency key was completed."""
        self.completed_jobs[idempotency_key] = True
        # Write to a temp file first, then atomically replace the target file
        temp_file = self.completed_jobs_file.with_suffix(".tmp")
        temp_file.write_text(json.dumps(self.completed_jobs))
        # Flush data to disk
        with temp_file.open("rb") as f:
            os.fsync(f.fileno())
        # Atomic rename
        temp_file.replace(self.completed_jobs_file)

    def process_job(self, job: Job) -> bool:
        """Process a single job with idempotency."""
        if self._is_job_completed(job.idempotency_key):
            logger.info(f"Job {job.job_id} already completed (idempotency key: {job.idempotency_key})")
            return True

        # Charge the customer
        success = self.payment_processor.charge_customer(job.customer_id, job.amount)
        if success:
            self._record_completion(job.idempotency_key)
            logger.info(f"Successfully processed job {job.job_id}")
            return True
        else:
            logger.error(f"Failed to process job {job.job_id}")
            return False
