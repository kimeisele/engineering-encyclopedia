import os
import uuid
import json
import time
import logging
from typing import Dict, Any
from pathlib import Path

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class PaymentWorker:
    def __init__(self, queue, payment_service, idempotency_store: Path, config_path: Path):
        self.queue = queue
        self.payment_service = payment_service
        self.idempotency_store = idempotency_store
        self.config_path = config_path
        self.idempotency_store.parent.mkdir(parents=True, exist_ok=True)

    def _get_operation_id(self, job: Dict[str, Any]) -> str:
        """Generate a unique operation ID from the job data."""
        # Use a hash of the job data to create a stable ID
        job_hash = hash(frozenset(job.items()))
        return f"op_{job_hash}_{uuid.uuid4()}"

    def _is_completed(self, operation_id: str) -> bool:
        """Check if an operation was already completed by looking at durable storage."""
        try:
            if self.idempotency_store.exists():
                with open(self.idempotency_store, 'r') as f:
                    completed_ops = json.load(f)
                    return operation_id in completed_ops
        except Exception as e:
            logger.error(f"Error reading idempotency store: {e}")
        return False

    def _record_completion(self, operation_id: str) -> bool:
        """Atomically record that an operation is completed."""
        try:
            # Create temp file in same directory
            temp_path = self.idempotency_store.with_suffix('.tmp')
            with open(temp_path, 'w') as f:
                # Load existing completed ops or start fresh
                if self.idempotency_store.exists():
                    with open(self.idempotency_store, 'r') as existing:
                        completed_ops = json.load(existing)
                else:
                    completed_ops = set()

                completed_ops.add(operation_id)
                json.dump(list(completed_ops), f)
                f.flush()
                os.fsync(f.fileno())  # Ensure durability

            # Atomic rename
            temp_path.replace(self.idempotency_store)
            return True
        except Exception as e:
            logger.error(f"Error recording completion: {e}")
            if temp_path.exists():
                temp_path.unlink()
            return False

    def _load_config(self) -> Dict[str, Any]:
        """Load configuration atomically."""
        try:
            if not self.config_path.exists():
                return {}
            with open(self.config_path, 'r') as f:
                return json.load(f)
        except Exception as e:
            logger.error(f"Error loading config: {e}")
            return {}

    def _save_config(self, config: Dict[str, Any]) -> bool:
        """Save configuration atomically."""
        try:
            temp_path = self.config_path.with_suffix('.tmp')
            with open(temp_path, 'w') as f:
                json.dump(config, f)
                f.flush()
                os.fsync(f.fileno())

            temp_path.replace(self.config_path)
            return True
        except Exception as e:
            logger.error(f"Error saving config: {e}")
            if temp_path.exists():
                temp_path.unlink()
            return False

    def process_job(self, job: Dict[str, Any]) -> bool:
        """Process a single job with idempotency and atomic operations."""
        operation_id = self._get_operation_id(job)

        # Check if already completed (idempotency)
        if self._is_completed(operation_id):
            logger.info(f"Operation {operation_id} already completed")
            return True

        # Load config to get customer balance
        config = self._load_config()
        customer_id = job.get('customer_id')
        amount = job.get('amount', 0)

        if customer_id not in config:
            config[customer_id] = {'balance': 0}

        # Check-then-act with atomic state update
        new_balance = config[customer_id]['balance'] - amount
        if new_balance < 0:
            logger.warning(f"Insufficient funds for customer {customer_id}")
            return False

        # Attempt payment (this is the "effect" that needs idempotency)
        try:
            payment_result = self.payment_service.charge(customer_id, amount)
            if not payment_result:
                logger.warning(f"Payment failed for customer {customer_id}")
                return False
        except Exception as e:
            logger.error(f"Payment service error: {e}")
            return False

        # Update balance atomically
        config[customer_id]['balance'] = new_balance
        if not self._save_config(config):
            logger.error("Failed to save config, payment may be lost")
            return False

        # Record completion (must be durable before we consider the operation complete)
        if not self._record_completion(operation_id):
            logger.error("Failed to record completion, operation not idempotent")
            # Attempt to rollback balance
            config[customer_id]['balance'] += amount
            self._save_config(config)
            return False

        logger.info(f"Successfully processed job {job} with operation {operation_id}")
        return True

    def run(self):
        """Main worker loop."""
        while True:
            try:
                job = self.queue.get(timeout=5)
                self.process_job(job)
            except Exception as e:
                logger.error(f"Worker error: {e}")
                time.sleep(1)  # Backoff

# Mock payment service for demonstration
class PaymentService:
    def charge(self, customer_id: str, amount: float) -> bool:
        """Mock payment service that always succeeds."""
        logger.info(f"Charging customer {customer_id} ${amount}")
        return True
