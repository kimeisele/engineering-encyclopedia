import os
import uuid
import time
import json
import logging
from typing import Dict, Any
from pathlib import Path

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class PaymentWorker:
    def __init__(self, queue, payment_service, idempotency_store: Path):
        self.queue = queue
        self.payment_service = payment_service
        self.idempotency_store = idempotency_store
        # Ensure idempotency store directory exists
        self.idempotency_store.mkdir(parents=True, exist_ok=True)

    def _get_idempotency_key(self, job: Dict[str, Any]) -> str:
        """Generate a stable idempotency key from job data."""
        # Use a hash of relevant job fields to create a stable key
        job_hash = hash(frozenset(job.items()))
        return f"payment:{job['customer_id']}:{job_hash}"

    def _is_completed(self, idempotency_key: str) -> bool:
        """Check if payment was already completed for this idempotency key."""
        # Use atomic file read to check completion status
        file_path = self.idempotency_store / idempotency_key
        try:
            with file_path.open('r') as f:
                data = json.load(f)
                return data.get('completed', False)
        except (FileNotFoundError, json.JSONDecodeError):
            return False

    def _record_completion(self, idempotency_key: str, success: bool, details: Dict[str, Any]):
        """Atomically record completion status using file replacement."""
        temp_path = self.idempotency_store / f"{idempotency_key}.tmp"
        final_path = self.idempotency_store / idempotency_key

        # Write to temp file first
        with temp_path.open('w') as f:
            json.dump({
                'completed': success,
                'timestamp': time.time(),
                'details': details
            }, f)
            f.flush()
            os.fsync(f.fileno())  # Ensure data is written to disk

        # Atomic rename (POSIX guarantees atomicity)
        temp_path.replace(final_path)
        # Clean up temp file if it still exists (shouldn't happen)
        temp_path.unlink(missing_ok=True)

    def process_job(self, job: Dict[str, Any]) -> bool:
        """Process a single payment job with idempotency guarantees."""
        idempotency_key = self._get_idempotency_key(job)

        # Check if already completed (idempotency check)
        if self._is_completed(idempotency_key):
            logger.info(f"Job already completed for key: {idempotency_key}")
            return True

        try:
            # Attempt payment
            payment_result = self.payment_service.charge(
                customer_id=job['customer_id'],
                amount=job['amount'],
                currency=job.get('currency', 'USD')
            )

            # Record completion atomically
            self._record_completion(
                idempotency_key=idempotency_key,
                success=True,
                details={'payment_id': payment_result['payment_id']}
            )
            logger.info(f"Successfully processed payment for {job['customer_id']}")
            return True

        except Exception as e:
            logger.error(f"Payment failed for {job['customer_id']}: {str(e)}")
            # Record failure (so we don't retry indefinitely)
            self._record_completion(
                idempotency_key=idempotency_key,
                success=False,
                details={'error': str(e)}
            )
            return False

    def run(self):
        """Main worker loop."""
        while True:
            try:
                job = self.queue.get(timeout=30)
                self.process_job(job)
                self.queue.ack(job)
            except Exception as e:
                logger.error(f"Worker error: {str(e)}")
                time.sleep(5)  # Backoff before retry
