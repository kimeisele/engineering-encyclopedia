class PaymentWorker:
    def __init__(self, job_queue: Queue, idempotency_store: Dict[str, bool]):
        self.job_queue = job_queue
        self.idempotency_store = idempotency_store
        self.idempotency_lock = Lock()
        self.max_retries = 3
        self.retry_delay = 1.0

    def _is_processed(self, idempotency_key: str) -> bool:
        """Check if operation was already completed."""
        with self.idempotency_lock:
            return self.idempotency_store.get(idempotency_key, False)

    def _mark_processed(self, idempotency_key: str) -> None:
        """Record completion before charging (durable write)."""
        with self.idempotency_lock:
            self.idempotency_store[idempotency_key] = True

    def _charge_customer(self, amount: float, customer_id: str) -> bool:
        """Simulate payment processing (idempotent operation)."""
        # In real system, this would call payment gateway
        logger.info(f"Charging customer {customer_id} for ${amount}")
        time.sleep(0.1)  # Simulate network delay
        return True

    def _generate_idempotency_key(self, job_data: Dict[str, Any]) -> str:
        """Create stable key from job data."""
        # Use stable fields like customer_id and job_id
        return f"charge:{job_data['customer_id']}:{job_data['job_id']}"

    def process_job(self, job_data: Dict[str, Any]) -> None:
        """Process a single job with idempotency and retry logic."""
        idempotency_key = self._generate_idempotency_key(job_data)
        amount = job_data['amount']
        customer_id = job_data['customer_id']

        # Check idempotency first (before any effect)
        if self._is_processed(idempotency_key):
            logger.info(f"Job already processed: {idempotency_key}")
            return

        # Retry loop with backoff
        for attempt in range(self.max_retries):
            try:
                # Effect happens here
                success = self._charge_customer(amount, customer_id)

                if success:
                    # Record completion BEFORE considering this successful
                    self._mark_processed(idempotency_key)
                    logger.info(f"Successfully charged customer {customer_id}")
                    return

            except Exception as e:
                logger.warning(f"Attempt {attempt + 1} failed: {str(e)}")
                if attempt < self.max_retries - 1:
                    time.sleep(self.retry_delay * (2 ** attempt))  # Exponential backoff

        logger.error(f"Failed to charge customer {customer_id} after {self.max_retries} attempts")

    def run(self) -> None:
        """Main worker loop."""
        while True:
            try:
                job_data = self.job_queue.get()
                if job_data is None:  # Sentinel value to stop
                    break
                self.process_job(job_data)
            except Exception as e:
                logger.error(f"Worker error: {str(e)}")
            finally:
                self.job_queue.task_done()
