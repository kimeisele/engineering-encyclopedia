def make_idempotency_key(job: ChargeJob) -> str:
    """Create a durable idempotency key from job data."""
    return f"charge-{job.customer_id}-{job.amount}-{job.idempotency_key}"

def process_job(job: ChargeJob, processor: PaymentProcessor, queue: JobQueue) -> bool:
    """Process a single job with retry logic."""

    # Check if we've already processed this operation
    if key in processor.processed_keys:
        logger.info(f"Job already processed: {job.idempotency_key}")
        return True

    # Attempt the charge with retry logic
    success = False

            success = processor.charge_customer(job.customer_id, job.amount, key)
            if success:
                break
        except Exception as e:
            logger.error(f"Charge failed (attempt {attempt + 1}): {e}")
            time.sleep(1 * (2 ** attempt))  # Exponential backoff with jitter

    if not success:
        logger.error(f"Failed to charge after retries: {job.idempotency_key}")
        # Requeue the job for later processing
        queue.produce(job)
        return False

    return True
