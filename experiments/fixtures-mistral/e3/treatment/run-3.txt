import json
import os
import tempfile
from typing import Any, Dict

def update_json_config(file_path: str, new_config: Dict[str, Any]) -> None:
    """
    Atomically updates a JSON config file on disk.

    Args:
        file_path: Path to the JSON config file to update.
        new_config: Dictionary containing the new configuration to write.

    Raises:
        ValueError: If the new_config is invalid or cannot be serialized.
        OSError: If file operations fail (permissions, disk full, etc.)
    """
    # Validate input at trust boundary
    if not isinstance(new_config, dict):
        raise ValueError("Config must be a dictionary")

    # Create temp file in same directory as target
    dir_name = os.path.dirname(file_path) or "."
    with tempfile.NamedTemporaryFile(
        mode='w',
        dir=dir_name,
        delete=False,
        suffix='.tmp'
    ) as temp_file:
        temp_path = temp_file.name

        try:
            # Write new content to temp file
            json.dump(new_config, temp_file, indent=2)
            # Ensure data is flushed to disk
            temp_file.flush()
            os.fsync(temp_file.fileno())

            # Atomic rename (POSIX) - readers see either old or new content
            os.replace(temp_path, file_path)
        except Exception:
            # Clean up temp file if anything fails
            try:
                os.unlink(temp_path)
            except OSError:
                pass
            raise
