import json
import os
import tempfile
from typing import Any

def update_json_config(file_path: str, new_config: dict[str, Any]) -> None:
    """
    Atomically updates a JSON config file on disk.

    Writes the new config to a temporary file in the same directory, flushes it,
    and renames it over the target file. This ensures readers see either the old
    or the new content, never a corrupted middle state.

    Args:
        file_path: Path to the JSON config file to update.
        new_config: Dictionary representing the new configuration.

    Raises:
        ValueError: If the new_config is invalid or the write fails.
        OSError: If file operations fail (e.g., permissions, disk full).
    """
    # Validate input at the trust boundary
    if not isinstance(new_config, dict):
        raise ValueError("Config must be a dictionary")

    # Create a temporary file in the same directory as the target
    dir_name = os.path.dirname(file_path) or "."
    with tempfile.NamedTemporaryFile(
        mode="w",
        encoding="utf-8",
        dir=dir_name,
        delete=False,
    ) as temp_file:
        temp_path = temp_file.name

        # Write the new config to the temp file
        json.dump(new_config, temp_file, indent=2, ensure_ascii=False)
        temp_file.flush()  # Ensure data is written to disk

        # Sync the file to disk to prevent data loss on power failure
        os.fsync(temp_file.fileno())

    # Atomically replace the target file with the temp file
    os.replace(temp_path, file_path)
