```python
import json
import os
import tempfile
from typing import Any, Dict

def update_json_config(file_path: str, new_config: Dict[str, Any]) -> None:
    """
    Atomically updates a JSON config file on disk.

    Writes to a temporary file in the same directory, flushes it to disk,
    then renames it over the target file. This ensures atomicity on POSIX
    systems and prevents readers from seeing partial writes.

    Args:
        file_path: Path to the JSON config file to update.
        new_config: Dictionary containing the new configuration data.

    Raises:
        OSError: If file operations fail (e.g., permission issues).
        ValueError: If the new_config is not serializable to JSON.
    """
    # Validate input at the trust boundary
    if not isinstance(new_config, dict):
        raise ValueError("Config must be a dictionary")

    # Create temp file in same directory as target
    dir_name = os.path.dirname(file_path) or "."
    with tempfile.NamedTemporaryFile(
        mode='w',
        dir=dir_name,
        delete=False,
        suffix='.tmp'
    ) as temp_file:
        temp_path = temp_file.name

        try:
            # Write JSON data to temp file
            json.dump(new_config, temp_file, indent=2)
            temp_file.flush()  # Ensure data is written to disk

            # Ensure directory metadata is flushed (fsync on file and parent dir)
            os.fsync(temp_file.fileno())
            parent_dir = os.path.dirname(temp_path)
            if parent_dir:
                with open(parent_dir, 'rb') as parent:
                    os.fsync(parent.fileno())

            # Atomic rename (POSIX guarantees this is atomic)
            os.replace(temp_path, file_path)
        except Exception:
            # Clean up temp file if something went wrong
            try:
                os.unlink(temp_path)
            except OSError:
                pass
            raise
```
