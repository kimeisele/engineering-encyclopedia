import json
import os
import tempfile
from pathlib import Path
from typing import Any, Dict

def update_json_config(file_path: str, new_config: Dict[str, Any]) -> None:
    """
    Atomically updates a JSON config file on disk.

    Args:
        file_path: Path to the JSON config file to update.
        new_config: Dictionary containing the new configuration data.

    Raises:
        ValueError: If the new_config is invalid (not a dict).
        OSError: If file operations fail.
    """
    # Validate input at trust boundary (security.input-validation)
    if not isinstance(new_config, dict):
        raise ValueError("Config must be a dictionary")

    path = Path(file_path)
    temp_fd, temp_path = tempfile.mkstemp(dir=path.parent, prefix=path.name + '.tmp')

    try:
        # Write to temp file
        with os.fdopen(temp_fd, 'w') as f:
            json.dump(new_config, f, indent=2)
            # Ensure data is flushed to disk (python.files.atomic-replacement)
            f.flush()
            os.fsync(f.fileno())

        # Atomic rename on POSIX (python.files.atomic-replacement)
        os.replace(temp_path, path)
    except Exception:
        # Clean up temp file if write fails (python.files.atomic-replacement)
        try:
            os.unlink(temp_path)
        except OSError:
            pass
        raise
    finally:
        # Ensure temp file is removed even if exceptions occur
        if os.path.exists(temp_path):
            try:
                os.unlink(temp_path)
            except OSError:
                pass
