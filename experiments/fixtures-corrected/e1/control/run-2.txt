class JobStateStore:
    """Manages job state in Redis for tracking and recovery"""
    
    def __init__(self, redis_url: str):
        self.redis = redis.from_url(redis_url)
        self.state_prefix = "charge_job:"
    
    def set_status(self, job_id: str, status: JobStatus, metadata: Optional[Dict] = None):
        """Store job status in Redis"""
        state = {
            'status': status.value,
            'updated_at': datetime.utcnow().isoformat(),
            **metadata if metadata else {}
        }
        key = f"{self.state_prefix}{job_id}"
        self.redis.hset(key, mapping=state)
        self.redis.expire(key, 86400)  # expire in 24 hours
    
    def get_status(self, job_id: str) -> Dict:
        """Get the current status of a job"""
        key = f"{self.state_prefix}{job_id}"
        return self.redis.hgetall(key)

        try:

                currency=job.currency.lower(),
                customer=job.customer_id,
                description=f"Charge for job {job.id}",
                metadata={
                    'job_id': job.id,
                    'customer_id': job.customer_id,
                    **job.metadata or {}
                }
            )
            return {
                'success': True,
                'charge_id': charge.id,
                'raw_response': charge
            }
        except stripe.error.CardError as e:
            # Card was declined
            return {
                'success': False,
                'error': f"Card declined: {e.error.message}",
                'error_code': e.code,
                'retryable': False
            }
        except stripe.error.RateLimitError as e:
            # Too many requests made to the API too quickly
            return {
                'success': False,
                'error': f"Rate limit: {e}",
                'retryable': True
            }
        except stripe.error.APIConnectionError as e:
            # Network communication with Stripe failed

# ==================== Message Queue Handler ====================

class MessageQueueHandler:
    """Wrapper around RabbitMQ/AMQP for consuming jobs"""
    
    def __init__(self, queue_handler_config):
        self.config = queue_handler_config
        self.connection = None
        self.channel = None
        self.is_running = False
    
    def connect(self):
        """Establish connection and declare queues"""
        self.connection = pika.BlockingConnection(
            pika.URLParameters(self.config.RABBITMQ_URL)
        )
        self.channel = self.connection.channel()
        
        # Declare main queue
        self.channel.queue_declare(
            queue=self.config.QUEUE_NAME,
            durable=True,
            arguments={
                'x-dead-letter-exchange': '',
                'x-dead-letter-routing-key': self.config.DEAD_LETTER_QUEUE
            }
        )
        
        # Declare dead letter queue
        self.channel.queue_declare(
            queue=self.config.DEAD_LETTER_QUEUE,
            durable=True
        )
        
        # Set QoS - prefetch 1 to avoid overwhelm
        self.channel.basic_qos(prefetch_count=1)
    
    def consume(self, callback):
        """Start consuming messages"""
        self.channel.basic_consume(
            queue=self.config.QUEUE_NAME,
            on_message_callback=callback,
            auto_ack=False  # Manual ack for at-least-once processing
        )
        self.channel.start_consuming()
    
    def acknowledge(self, delivery_tag):
        """Acknowledge a successfully processed message"""
        self.channel.basic_ack(delivery_tag)
    
    def reject(self, delivery_tag, requeue=True):
        """Reject a message, optionally requeue for retry"""
        self.channel.basic_reject(
            delivery_tag=delivery_tag,
            requeue=requeue
        )
    
    def stop(self):
        """Gracefully stop consuming"""
        self.is_running = False
        if self.channel:
            self.channel.stop_consuming()
        if self.connection:
            self.connection.close()

# ==================== Main Worker ====================

class ChargeWorker:
    """Main worker that processes charge jobs"""
    
    def __init__(self):
        self.config = Config()
        self.setup_logging()
        
        self.queue_handler = MessageQueueHandler(self.config)
        self.state_store = JobStateStore(self.config.REDIS_URL)
        self.payment_processor = PaymentProcessor(self.config.STRIPE_API_KEY)
        self.running = False
        
        # Signal handlers for graceful shutdown
        signal.signal(signal.SIGINT, self.signal_handler)
        signal.signal(signal.SIGTERM, self.signal_handler)
    
    def setup_logging(self):
        """Configure logging for this worker"""
        logging.basicConfig(
            level=getattr(logging, self.config.LOG_LEVEL),
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
            handlers=[
                logging.FileHandler(self.config.LOG_FILE),
                logging.StreamHandler(sys.stdout)
            ]
        )
        self.logger = logging.getLogger(f"charge_worker.{self.config.WORKER_ID}")
    
    def signal_handler(self, signum, frame):
        """Handle shutdown signals"""
        self.logger.info(f"Received signal {signum}, shutting down")
        self.running = False
        self.queue_handler.stop()
    
    def handle_message(self, ch, method, properties, body):
        """Process a message from the queue"""
        job_id = f"Unknown-{method.delivery_tag}"
        try:
            # Parse job
            job = ChargeJob.from_json(body.decode('utf-8'))
            job_id = job.id
            self.logger.info(f"[{self.config.WORKER_ID}] Processing job {job_id}")
            
            # Update state to processing
            self.state_store.set_status(
                job_id, JobStatus.PROCESSING,
                {'worker_id': self.config.WORKER_ID}
            )
            
            # Get retry count from message headers or database
            retry_count = self.get_retry_count(method, job_id)
            
            # Process the charge
            result = self.payment_processor.process_charge(job)
            
            if result['success']:
                # Log success and acknowledge
                self.logger.info(f"Job {job_id} completed successfully, charge_id={result['charge_id']}")
                self.state_store.set_status(
                    job_id, JobStatus.COMPLETED,
                    {'charge_id': result['charge_id']}
                )
                ch.basic_ack(delivery_tag=method.delivery_tag)
            else:
                # Handle failure
                self.handle_failure(
                    job_id, method, result['error'],
                    retryable=result.get('retryable', False),
                    retry_count=retry_count
                )
            
        except json.JSONDecodeError as e:
            self.logger.error(f"Invalid JSON in message: {e}")
            ch.basic_reject(delivery_tag=method.delivery_tag, requeue=False)
        except Exception as e:
            self.logger.error(f"Unexpected error processing job {job_id}: {e}")
            # Update state
            self.state_store.set_status(
                job_id, JobStatus.FAILED,
                {'error': str(e)}
            )
            # Reject with requeue to try again
            ch.basic_nack(delivery_tag=method.delivery_tag, requeue=True)
    
    def get_retry_count(self, method, job_id):
        """Get the current retry count for this job"""
        # Check if we have retry tracking in headers
        if method.headers and 'x-retry-count' in method.headers:
            return method.headers['x-retry-count']
        
        # Or check state store
        state = self.state_store.get_status(job_id)
        return int(state.get('retry_count', 0)) if state else 0
    
    def handle_failure(self, job_id, method, error, retryable, retry_count):
        """Handle failed job processing with retry logic"""
        self.logger.warning(f"Job {job_id} failed: {error} (retryable={retryable}, attempts={retry_count})")
        
        if retryable and retry_count < self.config.MAX_RETRIES:
            # Increment retry count
            new_retry_count = retry_count + 1
            
            # Update state
            self.state_store.set_status(
                job_id, JobStatus.RETRYING,
                {
                    'retry_count': new_retry_count,
                    'last_error': error
                }
            )
            
            # Set custom header for next attempt
            properties = pika.BasicProperties(
                headers={'x-retry-count': new_retry_count}
            )
            
            # Reject message (will go back to queue)
            self.logger.info(f"Job {job_id} will be retried (attempt {new_retry_count})")
            ch.basic_nack(
                delivery_tag=method.delivery_tag,
                requeue=True,
                # Note: With RabbitMQ, we can't set properties on requeue
                # Alternative: Re-publish with new headers and delete original
            )
        else:
            # Give up - send to dead letter queue
            self.logger.error(f"Job {job_id} failed permanently, routing to dead letter queue")
            self.state_store.set_status(
                job_id, JobStatus.DEAD_LETTERED,
                {'error': error, 'final_retry_count': retry_count}
            )
            ch.basic_reject(
                delivery_tag=method.delivery_tag,
                requeue=False  # This will trigger the DEAD LETTER exchange
            )
            
            # Optional: Send notification to ops
            self.notify_failure(job_id, error)
    
    def notify_failure(self, job_id, error):
        """Send notification about failed job (e.g., PagerDuty, email)"""
        # This would go to a notification service
        self.logger.info(f"Notification would be sent for job {job_id}: {error}")
    
    def run(self):
        """Start the worker"""
        self.logger.info(f"Starting charge worker {self.config.WORKER_ID}")
        self.running = True
        
        try:
            # Connect to queue
            self.queue_handler.connect()
            
            # Start consuming
            self.queue_handler.consume(self.handle_message)
            
        except KeyboardInterrupt:
            self.logger.info("Interrupted")
        except Exception as e:
            self.logger.error(f"Worker crashed: {e}")
            # Attempt restart with backoff
            time.sleep(5)
            self.run()
        finally:
            self.cleanup()
    
    def cleanup(self):
        """Cleanup resources"""
        self.logger.info("Cleaning up worker resources")
        if hasattr(self, 'queue_handler'):
            self.queue_handler.stop()
        self.state_store.redis.close()
        self.logger.info("Worker stopped")

# ==================== Entry Point ====================

def main():
    """Main entry point for the worker"""
    worker = ChargeWorker()
    worker.run()

if __name__ == "__main__":
    main()

**Key Features of this Worker:**

1. **At-least-once delivery**: Manual acknowledgment ensures jobs aren't lost
2. **Retry with backoff**: Configurable retry mechanism with dead-letter queue
3. **State tracking**: Redis maintains job status for monitoring
4. **Graceful shutdown**: Handles SIGINT/SIGTERM signals
5. **Error classification**: Distinguishes retryable vs permanent failures
6. **Monitoring**: JSON logs with worker ID and job ID for tracing
7. **Resource management**: Connection pooling and proper cleanup

**Production Considerations:**

- Use **exponential backoff** for retries with jitter to avoid thundering herd
- Implement **heartbeat monitoring** and health checks
- Add **circuit breaker** for the payment gateway
- Consider **dead-letter monitoring** alerts
- Use **correlation IDs** across your system for tracing
- For high throughput, consider **worker pools** with autoscaling
- Implement **idempotency keys** to prevent double charges
- Add **audit logging** for all financial operations

This worker handles the critical requirements of ordering/scheduling, retry logic, and production readiness for a payment system.
