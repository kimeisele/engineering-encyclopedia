class DatabaseManager:

# ============== Queue Manager ==============

class QueueManager:
    """Manages queue operations using Redis or AWS SQS"""
    
    def __init__(self, config: WorkerConfig):
        self.config = config
        self.queue_name = config.queue_name
        
        if config.use_sqs:
            self._init_sqs()
        else:
            self._init_redis()
    
    def _init_redis(self):
        try:
            self.redis_client = redis.Redis(
                host=self.config.redis_host,
                port=self.config.redis_port,
                db=self.config.redis_db,
                decode_responses=True
            )
            self.redis_client.ping()
            logger.info("Connected to Redis")
        except redis.ConnectionError as e:
            logger.error(f"Failed to connect to Redis: {e}")
            raise
    
    def _init_sqs(self):
        try:
            import boto3
            self.sqs_client = boto3.client('sqs', region_name=self.config.aws_region)
            if not self.config.aws_queue_url:
                response = self.sqs_client.get_queue_url(QueueName=self.queue_name)
                self.config.aws_queue_url = response['QueueUrl']
            logger.info(f"Connected to SQS queue: {self.queue_name}")
        except Exception as e:
            logger.error(f"Failed to connect to SQS: {e}")
            raise
    
    def get_job(self) -> Optional[Dict[str, Any]]:
        """Get a job from the queue"""
        if self.config.use_sqs:
            response = self.sqs_client.receive_message(
                QueueUrl=self.config.aws_queue_url,
                MaxNumberOfMessages=1,
                WaitTimeSeconds=1
            )
            messages = response.get('Messages', [])
            if messages:
                # Store the receipt handle for deletion
                message = messages[0]
                job_data = json.loads(message['Body'])
                job_data['_message_id'] = message['MessageId']
                job_data['_receipt_handle'] = message['ReceiptHandle']
                return job_data
        else:
            # Redis list operation
            job_data = self.redis_client.rpop(self.queue_name)
            if job_data:
                return json.loads(job_data)
        return None
    
    def complete_job(self, job: Dict[str, Any]) -> bool:
        """Mark job as completed and remove from queue"""
        if self.config.use_sqs:
            receipt_handle = job.get('_receipt_handle')
            if receipt_handle:
                try:
                    self.sqs_client.delete_message(
                        QueueUrl=self.config.aws_queue_url,
                        ReceiptHandle=receipt_handle
                    )
                    return True
                except Exception as e:
                    logger.error(f"Failed to delete SQS message: {e}")
                    return False
        # Redis doesn't need explicit completion since we already popped
        return True
    
    def failed_job(self, job: Dict[str, Any]) -> bool:
        """Retry a failed job - push back to queue"""
        # Remove SQS specific fields before pushing back
        job_copy = {k: v for k, v in job.items() if not k.startswith('_')}
        
        if self.config.use_sqs:
            # For SQS, we'll push a new message (or handle visibility timeout differently)
            try:
                self.sqs_client.send_message(
                    QueueUrl=self.config.aws_queue_url,
                    MessageBody=json.dumps(job_copy)
                )
                return True
            except Exception as e:
                logger.error(f"Failed to requeue message: {e}")
                return False
        else:
            # Redis list operation
            try:
                self.redis_client.lpush(self.queue_name, json.dumps(job_copy))
                return True
            except Exception as e:
                logger.error(f"Failed to requeue message: {e}")
                return False

# ============== Charge Worker ==============

class ChargeWorker:
    def __init__(self, config: WorkerConfig):
        self.config = config
        self.db = DatabaseManager(config.db_url)
        self.queue = QueueManager(config)
        self.payment_gateway = PaymentGatewayClient(
            api_key=config.payment_gateway_api_key,
            base_url=config.payment_gateway_url,
            timeout=config.payment_gateway_timeout
        )
        self.running = True
        self.stats = {
            'total_processed': 0,
            'successful': 0,
            'failed': 0,
            'retries': 0,
            'last_processing_time': None
        }
        
    def process_charge(self, job: Dict[str, Any]) -> bool:
        """Process a single charge job"""
        charge_id = job.get('charge_id')
        customer_id = job.get('customer_id')
        amount = job.get('amount')
        
        logger.info(f"Processing charge {charge_id} for customer {customer_id} amount ${amount}")
        
        # Validate input
        if not all([charge_id, customer_id, amount]):
            logger.error(f"Invalid job data: {job}")
            return False
        
        # Fetch customer
        customer = self.db.get_customer(customer_id)
        if not customer:
            logger.error(f"Customer not found: {customer_id}")
            return False
        
        # Update charge status to processing
        self.db.update_charge_status(charge_id, ChargeStatus.PROCESSING)
        
        # Process through payment gateway
        success = self.payment_gateway.process_charge(charge_id, customer, amount)
        
        if success:
            logger.info(f"Charge {charge_id} successful")
            self.stats['successful'] += 1
            return True
        else:
            logger.warning(f"Charge {charge_id} failed")
            self.stats['failed'] += 1
            return False
    
    def process_job(self, job: Dict[str, Any]) -> bool:
        """Process a job with retry logic"""
        max_attempts = self.config.max_retries
        attempts = 0
        
        while attempts < max_attempts:
            attempts += 1
            
            if attempts > 1:
                logger.info(f"Retry attempt {attempts} for job")
                self.stats['retries'] += 1
                # Wait before retry
                time.sleep(self.config.retry_delay)
            
            try:
                success = self.process_charge(job)
                if success:
                    return True
                else:
                    logger.error(f"Processing failed, attempt {attempts}/{max_attempts}")
            except Exception as e:
                logger.error(f"Error processing job: {e}, attempt {attempts}/{max_attempts}")
            
            if attempts >= max_attempts:
                logger.error("Job failed after all retries")
                # Mark as failed in database
                if job.get('charge_id'):
                    self.db.update_charge_status(
                        job['charge_id'], 
                        ChargeStatus.FAILED,
                        f"Failed after {max_attempts} attempts"
                    )
                return False
        
        return False
    
    def run(self):
        """Main worker loop"""
        logger.info(f"Starting charge worker. Polling queue: {self.config.queue_name}")
        
        while self.running:
            try:
                # Get job from queue
                job = self.queue.get_job()
                
                if job:
                    self.stats['total_processed'] += 1
                    self.stats['last_processing_time'] = datetime.utcnow()
                    
                    # Process the job
                    success = self.process_job(job)
                    
                    if success:
                        # Complete the job
                        self.queue.complete_job(job)
                    else:
                        # Job failed - decide whether to retry or give up
                        if job.get('retry_count', 0) < self.config.max_retries:
                            job['retry_count'] = job.get('retry_count', 0) + 1
                            if not self.queue.failed_job(job):
                                logger.error("Failed to requeue job")
                        else:
                            logger.error(f"Job exceeded max retries: {job}")
                            self.queue.complete_job(job)  # Remove dead letter
                    
                    # Log stats periodically
                    if self.stats['total_processed'] % 10 == 0:
                        logger.info(f"Stats: {json.dumps(self.stats)}")
                else:
                    # No jobs available, wait
                    time.sleep(self.config.poll_interval)
                    
            except KeyboardInterrupt:
                logger.info("Worker stopped by user")
                self.running = False
            except Exception as e:
                logger.error(f"Unexpected error in main loop: {e}")
                time.sleep(self.config.poll_interval)
        
        logger.info("Worker stopped")
        logger.info(f"Final stats: {json.dumps(self.stats)}")

    def stop(self):
        """Gracefully stop the worker"""
        self.running = False
