```python
"""
Job worker that charges a customer exactly once per logical operation.

Design notes (see application report for mapping to context pack):

- The queue provides a `job.id` that identifies one logical operation.
- We keep a durable deduplication record (SQLite) and mark completion *before*
  issuing the charge. This turns the "at-least-once" delivery of the queue into
  an exactly-once effect for our charging operation.
- The effect (charge) and completion record are not in the same transaction:
  we use a claim token so only one worker can be mid-flight for a given job.
- We write a small status file atomically for observability/debugging; it is
  never used as the source of truth.

The worker is safe against:
- Two workers seeing the same job as not-yet-done (claim token in DB).
- A crash between charge and completion record (claim token + idempotency key
  sent to the payment provider; a retry reuses the same key).
- A retry loop that is not bounded (explicit max attempts with exponential
  backoff + jitter).
"""

import json
import logging
import os
import random
import sqlite3
import tempfile
import time
import uuid
from dataclasses import dataclass

logger = logging.getLogger(__name__)

# --- Configuration -----------------------------------------------------------

DB_PATH = os.environ.get("WORKER_DB_PATH", "/var/lib/charger/worker.db")
STATUS_DIR = os.environ.get("WORKER_STATUS_DIR", "/var/lib/charger/status")
MAX_ATTEMPTS = int(os.environ.get("WORKER_MAX_ATTEMPTS", "5"))
BASE_BACKOFF_MS = int(os.environ.get("WORKER_BASE_BACKOFF_MS", "100"))


# --- Domain ------------------------------------------------------------------

class PaymentProvider:
    """Mocks an external payment API. In production this would be a real SDK.

    The important contract: `charge` is idempotent for a given
    `idempotency_key` -- if the provider already processed that key, it returns
    the previous result instead of charging again.
    """

    def __init__(self):
        self._processed = {}

    def charge(self, *, customer_id: str, amount_cents: int, idempotency_key: str) -> str:
        # Simulate network latency and a rare transient failure.
        time.sleep(0.01)
        if random.random() < 0.02:
            raise RuntimeError("transient network error")
        if idempotency_key in self._processed:
            return self._processed[idempotency_key]
        receipt = f"receipt-{uuid.uuid4()}"
        self._processed[idempotency_key] = receipt
        logger.info("charged customer=%s amount=%d cents receipt=%s",
                    customer_id, amount_cents, receipt)
        return receipt


@dataclass(frozen=True)
class Job:
    """A job from the queue. Fields are populated by a real queue consumer."""
    id: str                    # unique job id provided by the queue
    customer_id: str
    amount_cents: int
    attempt: int = 1           # current delivery attempt

    @classmethod
    def from_payload(cls, payload: dict) -> "Job":
        if "id" not in payload or "customer_id" not in payload or "amount_cents" not in payload:
            raise ValueError("malformed job payload")
        return cls(
            id=str(payload["id"]),
            customer_id=str(payload["customer_id"]),
            amount_cents=int(payload["amount_cents"]),
            attempt=int(payload.get("attempt", 1)),
        )


# --- Persistence -------------------------------------------------------------

class DedupStore:
    """SQLite-backed store for deduplication and claim tracking.

    Schema:
      job_id      TEXT PRIMARY KEY
      status      TEXT CHECK(status IN ('claimed','done'))
      claim_token TEXT NOT NULL
      receipt     TEXT
      updated_at  INTEGER NOT NULL

    Guarantees:
      - `claim_if_unclaimed` is atomic: only one of two concurrent callers
        gets the claim for a given job.
      - `mark_done` is durable (WAL + synchronous=FULL).
    """

    def __init__(self, path: str):
        self._conn = sqlite3.connect(path, timeout=30)
        self._conn.execute("PRAGMA journal_mode=WAL")
        self._conn.execute("PRAGMA synchronous=FULL")
        self._conn.execute(
            """
            CREATE TABLE IF NOT EXISTS jobs (
                job_id      TEXT PRIMARY KEY,
                status      TEXT NOT NULL CHECK(status IN ('claimed','done')),
                claim_token TEXT NOT NULL,
                receipt     TEXT,
                updated_at  INTEGER NOT NULL
            )
            """
        )
        self._conn.commit()

    def claim_if_unclaimed(self, job_id: str, claim_token: str) -> bool:
        """Atomically claim a job if it is not already done.

        Returns True only if this caller got the claim.
        """
        # This single INSERT ... WHERE NOT EXISTS is atomic under SQLite's
        # single-writer model. Two concurrent workers both try; only one
        # succeeds.
        cur = self._conn.execute(
            """
            INSERT OR IGNORE INTO jobs (job_id, status, claim_token, updated_at)
            VALUES (?, 'claimed', ?, ?)
            """,
            (job_id, claim_token, int(time.time())),
        )
        self._conn.commit()
        if cur.rowcount == 1:
            return True

        # Job existed. Check whether the existing claim is ours (a retry of the
        # same worker after a crash) or whether it's done.
        row = self._conn.execute(
            "SELECT status, claim_token FROM jobs WHERE job_id = ?", (job_id,)
        ).fetchone()
        if row is None:
            return False
        if row[0] == "done":
            return False
        # 'claimed' -- only the holder of the token may proceed.
        return row[1] == claim_token

    def mark_done(self, job_id: str, claim_token: str, receipt: str) -> None:
        """Durably record completion. Must be called before/after the effect."""
        self._conn.execute(
            """
            UPDATE jobs
            SET status = 'done', receipt = ?, updated_at = ?
            WHERE job_id = ? AND claim_token = ?
            """,
            (receipt, int(time.time()), job_id, claim_token),
        )
        self._conn.commit()

    def close(self):
        self._conn.close()


def write_status_file(job_id: str, state: str) -> None:
    """Write a small status file atomically for debugging/observability.

    This is not the source of truth (SQLite is), it's just a convenience for
    operators. Atomic replacement ensures readers never see a truncated file.
    """
    os.makedirs(STATUS_DIR, exist_ok=True)
    target = os.path.join(STATUS_DIR, f"{job_id}.json")
    fd, tmp_path = tempfile.mkstemp(dir=STATUS_DIR, suffix=".tmp")
    try:
        with os.fdopen(fd, "w") as f:
            json.dump({"job_id": job_id, "state": state, "ts": time.time()}, f)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp_path, target)  # atomic on POSIX
    except Exception:
        os.unlink(tmp_path)
        raise


# --- Worker loop -------------------------------------------------------------

class Worker:
    def __init__(self, provider: PaymentProvider, store: DedupStore,
                 max_attempts: int = MAX_ATTEMPTS,
                 base_backoff_ms: int = BASE_BACKOFF_MS):
        self.provider = provider
        self.store = store
        self.max_attempts = max_attempts
        self.base_backoff_ms = base_backoff_ms

    def _make_claim_token(self, job: Job) -> str:
        """A fresh token per delivery attempt. Survives worker crash because
        we reuse the same token for the same (job, attempt)."""
        # Deterministic per (job_id, attempt) so that a crash-and-redeliver of
        # the *same* attempt reuses the token and can proceed. A new attempt
        # (from the queue after a failure) gets a new token.
        return f"{job.id}:{job.attempt}"

    def process(self, job: Job) -> None:
        """Process one job delivery. Raises on unrecoverable error.

        The queue will redeliver on failure; our dedup store makes the
        retry safe.
        """
        claim = self._make_claim_token(job)
        if not self.store.claim_if_unclaimed(job.id, claim):
            logger.info("job %s already done or claimed by another worker", job.id)
            return

        # We hold the claim. Perform the charge with an idempotency key that
        # the provider understands. If we crash right here before marking done,
        # the queue redelivers the same job; our claim token lets us retry and
        # the provider's idempotency key prevents a double charge.
        idempotency_key = f"charge:{job.id}:{job.attempt}"
        receipt = self.provider.charge(
            customer_id=job.customer_id,
            amount_cents=job.amount_cents,
            idempotency_key=idempotency_key,
        )

        # Record completion *after* the effect. If we crash here, a redelivery
        # will call the provider again with the same idempotency key, getting
        # the same receipt back -- no double charge.
        self.store.mark_done(job.id, claim, receipt)
        write_status_file(job.id, "done")

    def run_with_retry(self, job: Job) -> bool:
        """Attempt a job with bounded exponential backoff + jitter.

        Returns True on success, False after exhausting attempts.
        """
        for attempt in range(1, self.max_attempts + 1):
            try:
                self.process(job)
                return True
            except Exception:
                if attempt == self.max_attempts:
                    logger.error("job %s failed after %d attempts", job.id, attempt)
                    write_status_file(job.id, "failed")
                    return False
                wait_ms = self.base_backoff_ms * (2 ** (attempt - 1))
                wait_ms += random.randint(0, wait_ms // 2 or 1)  # jitter
                logger.warning("job %s attempt %d failed, retrying in %d ms",
                               job.id, attempt, wait_ms)
                time.sleep(wait_ms / 1000.0)
        return False


def consume_queue_forever(queue, worker: Worker) -> None:
    """Blocking loop; `queue` is a stand-in for an actual queue consumer."""
    while True:
        raw = queue.next_job()
        if raw is None:
            time.sleep(0.5)
            continue
        try:
            job = Job.from_payload(raw)
        except ValueError as exc:
            logger.error("discarding malformed job: %s", exc)
            continue
        worker.run_with_retry(job)
        queue.ack(job.id)


# --- Main / test --------------------------------------------------------------

def main():
    logging.basicConfig(level=logging.INFO)
    provider = PaymentProvider()
    store = DedupStore(DB_PATH)
    worker = Worker(provider, store)
    try:
        # In a real deployment this comes from the queue SDK.
        queue = FakeQueue()

        # Simulate delivery with multiple jobs; also test concurrent duplicate
        # delivery.
        jobs = [
            Job(id="job-1", customer_id="cust-A", amount_cents=1000),
            Job(id="job-1", customer_id="cust-A", amount_cents=1000),  # duplicate
            Job(id="job-2", customer_id="cust-B", amount_cents=2500),
            Job(id="job-3", customer_id="cust-C", amount_cents=500),
        ]
        for job in jobs:
            ok = worker.run_with_retry(job)
            print(f"job {job.id} -> {'ok' if ok else 'failed'}")
        print(f"provider received {len(provider._processed)} unique charges")
    finally:
        store.close()


class FakeQueue:
    """Minimal stand-in for a queue consumer; not part of the production code."""
    def next_job(self):
        return None

    def ack(self, job_id):
        pass


if __name__ == "__main__":
    main()
```
