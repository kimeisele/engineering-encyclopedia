```python
"""
Worker that consumes jobs from a queue and charges a customer.

Design:
- Each job carries an `operation_id` identifying one logical charge.
- Completion is durably recorded (fsync'd) *before* the charge effect.
- A lock file prevents two workers from racing on the same operation.
- Retries use exponential backoff with jitter and a bounded attempt count.
- State files are written atomically (temp + fsync + rename).
"""

import json
import os
import random
import time
from pathlib import Path

class ChargeWorker:
    def __init__(self, queue, state_dir, max_attempts=5):
        self.queue = queue
        self.state_dir = Path(state_dir)
        self.state_dir.mkdir(parents=True, exist_ok=True)
        self.max_attempts = max_attempts

    def _state_path(self, operation_id):
        return self.state_dir / f"{operation_id}.json"

    def _is_complete(self, operation_id):
        """Durable check: does the completion record exist?"""
        return self._state_path(operation_id).exists()

    def _write_completion(self, operation_id):
        """Durably record completion BEFORE the effect."""
        path = self._state_path(operation_id)
        temp = path.with_suffix(".tmp")
        with open(temp, "w") as f:
            json.dump({"operation_id": operation_id, "completed_at": time.time()}, f)
            f.flush()
            os.fsync(f.fileno())          # flush data
        os.replace(temp, path)            # atomic rename (same dir)
        dir_fd = os.open(self.state_dir, os.O_DIRECTORY)
        try:
            os.fsync(dir_fd)              # flush directory metadata
        finally:
            os.close(dir_fd)

    def _lock_path(self, operation_id):
        return self.state_dir / f"{operation_id}.lock"

    def _acquire_lock(self, operation_id):
        """Try to claim the operation exclusively (O_EXCL). Returns True if claimed."""
        try:
            fd = os.open(self._lock_path(operation_id), os.O_CREAT | os.O_EXCL | os.O_WRONLY)
            os.write(fd, str(os.getpid()).encode())
            os.close(fd)
            return True
        except FileExistsError:
            return False

    def _release_lock(self, operation_id):
        try:
            os.unlink(self._lock_path(operation_id))
        except FileNotFoundError:
            pass

    def _charge(self, customer, amount, operation_id):
        """
        The actual charge. In real code this would call a payment service.
        Must be side-effect free when called twice for same operation_id.
        We rely on the completion record written BEFORE calling this.
        """
        # idempotency guard inside the charge itself, best-effort
        if self._is_complete(operation_id):
            return
        # charge the customer...
        print(f"Charging {customer} ${amount} (op={operation_id})")
        # NOTE: if this throws, the completion record remains, so retry will
        # see it and NOT re-charge.

    def process_one(self):
        job = self.queue.dequeue()
        if not job:
            return False

        operation_id = job["operation_id"]
        customer = job["customer"]
        amount = job["amount"]

        # 1. Idempotency: if already complete, drop.
        if self._is_complete(operation_id):
            return True

        # 2. Race: claim with lock file.
        if not self._acquire_lock(operation_id):
            # Another worker is handling this; drop.
            return True

        try:
            # 3. Re-check after acquiring the lock (check-then-act is now safe).
            if self._is_complete(operation_id):
                return True

            # 4. Durable completion record BEFORE the effect.
            self._write_completion(operation_id)

            # 5. The effect (charge).
            self._charge(customer, amount, operation_id)

            return True
        finally:
            self._release_lock(operation_id)

    def run(self):
        """Main loop with retry semantics."""
        while True:
            attempt = 0
            while attempt < self.max_attempts:
                try:
                    if not self.process_one():
                        return  # no more jobs
                    break
                except Exception as e:
                    attempt += 1
                    if attempt >= self.max_attempts:
                        print(f"Giving up: {e}")
                        return
                    sleep = (2 ** attempt) + random.uniform(0, 0.5)
                    time.sleep(sleep)


if __name__ == "__main__":
    # Example queue with deque
    from collections import deque

    class SimpleQueue:
        def __init__(self, jobs):
            self.jobs = deque(jobs)
        def dequeue(self):
            return self.jobs.popleft() if self.jobs else None

    jobs = [
        {"operation_id": "op-001", "customer": "acme", "amount": 100.0},
        {"operation_id": "op-002", "customer": "globex", "amount": 250.0},
    ]
    q = SimpleQueue(jobs)
    worker = ChargeWorker(q, "/tmp/charge_state")
    worker.run()
```
