import json
import logging
import os
import random
import sqlite3
import tempfile
import time
import uuid
from dataclasses import dataclass
from typing import Optional

logger = logging.getLogger(__name__)
logging.basicConfig(level=logging.INFO)


@dataclass
class ChargeResult:
    job_id: str
    customer_id: str
    amount_cents: int
    status: str  # "completed", "duplicate", "failed"
    charge_id: Optional[str] = None


class ChargeWorker:
    """
    Consumes jobs from a queue and charges customers.
    """

    def __init__(self, queue, db_path: str, max_retries: int = 3):
        self.queue = queue
        self.db_path = db_path
        self.max_retries = max_retries
        self._init_db()

    def _init_db(self):
        """Create tables for deduplication and charge records."""
        with sqlite3.connect(self.db_path) as conn:
            conn.execute(
                """
                CREATE TABLE IF NOT EXISTS processed_jobs (
                    job_id TEXT PRIMARY KEY,
                    charge_id TEXT NOT NULL,
                    processed_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
                )
                """
            )
            conn.execute(
                """
                CREATE TABLE IF NOT EXISTS charges (
                    charge_id TEXT PRIMARY KEY,
                    job_id TEXT NOT NULL,
                    customer_id TEXT NOT NULL,
                    amount_cents INTEGER NOT NULL,
                    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
                )
                """
            )
            conn.commit()

    def _is_processed(self, conn: sqlite3.Connection, job_id: str) -> Optional[str]:
        """Check if job was already processed. Returns charge_id or None."""
        row = conn.execute(
            "SELECT charge_id FROM processed_jobs WHERE job_id = ?", (job_id,)
        ).fetchone()
        return row[0] if row else None

    def _record_charge(self, conn: sqlite3.Connection, job_id: str, customer_id: str,
                       amount_cents: int) -> str:
        """Record a charge and mark job processed atomically."""
        charge_id = str(uuid.uuid4())
        conn.execute(
            "INSERT INTO charges (charge_id, job_id, customer_id, amount_cents) VALUES (?, ?, ?, ?)",
            (charge_id, job_id, customer_id, amount_cents),
        )
        conn.execute(
            "INSERT INTO processed_jobs (job_id, charge_id) VALUES (?, ?)",
            (job_id, charge_id),
        )
        return charge_id

    def _process_job(self, job: dict) -> ChargeResult:
        """
        Process a single job. Idempotent: check-then-act is protected by
        the database transaction, ensuring no double-charging even with
        concurrent workers.
        """
        job_id = job["job_id"]
        customer_id = job["customer_id"]
        amount_cents = job["amount_cents"]

        with sqlite3.connect(self.db_path) as conn:
            # Transaction begins implicitly. Both the check and the insert
            # happen atomically — no race between two workers.
            existing = self._is_processed(conn, job_id)
            if existing:
                return ChargeResult(job_id, customer_id, amount_cents, "duplicate", existing)

            charge_id = self._record_charge(conn, job_id, customer_id, amount_cents)
            conn.commit()

        # From this point, the dedup record is durable. Simulated external
        # charge call would go here — in production, we'd use the charge_id
        # as the idempotency key for the payment provider.
        logger.info("Charged customer %s %d cents (charge_id=%s)", customer_id, amount_cents, charge_id)
        return ChargeResult(job_id, customer_id, amount_cents, "completed", charge_id)

    def run(self, poll_interval: float = 1.0):
        """Main loop: poll queue, process jobs, handle retries."""
        while True:
            job = self.queue.dequeue()
            if job is None:
                time.sleep(poll_interval)
                continue

            attempt = 0
            while attempt <= self.max_retries:
                try:
                    result = self._process_job(job)
                    if result.status == "completed":
                        self.queue.ack(job["job_id"])
                    elif result.status == "duplicate":
                        # Already processed — ack it so it leaves the queue.
                        self.queue.ack(job["job_id"])
                    break
                except Exception as e:
                    attempt += 1
                    if attempt > self.max_retries:
                        logger.error("Job %s failed after %d attempts: %s",
                                     job["job_id"], attempt - 1, e)
                        self.queue.fail(job["job_id"], reason=str(e))
                    else:
                        backoff_ms = (2 ** attempt) * 100 + random.randint(0, 100)
                        time.sleep(backoff_ms / 1000.0)


# ---- Minimal queue stub for demonstration ----
class InMemoryQueue:
    def __init__(self):
        self.jobs = []
        self.acked = set()
        self.failed = {}

    def enqueue(self, job: dict):
        self.jobs.append(job)

    def dequeue(self) -> Optional[dict]:
        return self.jobs.pop(0) if self.jobs else None

    def ack(self, job_id: str):
        self.acked.add(job_id)

    def fail(self, job_id: str, reason: str = ""):
        self.failed[job_id] = reason


# ---- Config file: atomic replacement per python.files.atomic-replacement ----
def write_config_atomic(path: str, config: dict):
    """Write config to temp file in same dir, fsync, rename over target."""
    dirname = os.path.dirname(path) or "."
    fd, tmp_path = tempfile.mkstemp(dir=dirname, prefix=".config.", suffix=".tmp")
    try:
        with os.fdopen(fd, "w") as f:
            json.dump(config, f, indent=2)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp_path, path)  # atomic on POSIX
    except Exception:
        try:
            os.unlink(tmp_path)
        except OSError:
            pass
        raise
