"""
Worker that consumes jobs from a queue and charges a customer.

Design principles (from context pack):
- Idempotency: each job has a stable idempotency key; we record completion
  durably BEFORE charging, and the charge call itself is idempotent.
- Race safety: claim the job atomically in the queue (atomic read-and-delete
  or lease). Never check-then-act without atomicity.
- Atomic config writes: write config via temp file + fsync + rename.
- Retry: bounded, with exponential backoff and jitter; only retry idempotent
  operations.
"""

import json
import os
import random
import tempfile
import time
import uuid
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Optional


# ---------------------------------------------------------------------------
# Configuration & storage (simplified for demonstration)
# ---------------------------------------------------------------------------

@dataclass
class Config:
    queue_dir: Path
    completion_dir: Path
    max_retries: int = 5
    base_backoff_ms: float = 100.0
    max_backoff_ms: float = 5000.0


class DurableStore:
    """Minimal durable key-value store using atomic file replacement."""

    def __init__(self, directory: Path):
        self.directory = directory
        self.directory.mkdir(parents=True, exist_ok=True)

    def _path_for(self, key: str) -> Path:
        # key is a stable idempotency key (uuid) — safe as filename
        return self.directory / f"{key}.json"

    def read(self, key: str) -> Optional[dict]:
        path = self._path_for(key)
        if not path.exists():
            return None
        with path.open("r") as f:
            return json.load(f)

    def write_atomic(self, key: str, data: dict) -> None:
        """Atomic write: temp file in same dir, flush, fsync, rename."""
        path = self._path_for(key)
        fd, tmp_path_str = tempfile.mkstemp(
            dir=self.directory,
            prefix=f".{key}.",
            suffix=".tmp",
        )
        tmp_path = Path(tmp_path_str)
        try:
            with os.fdopen(fd, "w") as f:
                json.dump(data, f)
                f.flush()
                os.fsync(f.fileno())
            os.replace(tmp_path, path)
        except Exception:
            # cleanup on failure
            tmp_path.unlink(missing_ok=True)
            raise

    def claim_job(self, queue: "JobQueue", job_id: str) -> bool:
        """Atomically claim a job — no check-then-act race.

        In a real system this would be a database UPDATE ... WHERE status='pending'
        or a queue claim operation. Here we use the filesystem as a simplified
        stand-in: rename the job file to a 'claimed' state atomically.
        """
        pending = queue.pending_dir / f"{job_id}.json"
        claimed = queue.claimed_dir / f"{job_id}.json"
        try:
            os.rename(pending, claimed)
            return True
        except FileNotFoundError:
            return False


# ---------------------------------------------------------------------------
# Job queue (simplified filesystem-backed)
# ---------------------------------------------------------------------------

class JobQueue:
    def __init__(self, root: Path):
        self.root = root
        self.pending_dir = root / "pending"
        self.claimed_dir = root / "claimed"
        for d in (self.pending_dir, self.claimed_dir):
            d.mkdir(parents=True, exist_ok=True)

    def read_job(self, job_id: str) -> Optional[dict]:
        for d in (self.claimed_dir, self.pending_dir):
            p = d / f"{job_id}.json"
            if p.exists():
                with p.open("r") as f:
                    return json.load(f)
        return None


# ---------------------------------------------------------------------------
# Charging service (external — we only call it; it is idempotent)
# ---------------------------------------------------------------------------

class ChargingService:
    """External service. Its charge() is idempotent when given the same
    idempotency_key — it will not double-charge even if called twice."""

    def charge(self, customer_id: str, amount_cents: int, idempotency_key: str) -> bool:
        # In production this would be an HTTP call with the key in a header.
        # We simulate: it always succeeds and records the key.
        return True


# ---------------------------------------------------------------------------
# Worker
# ---------------------------------------------------------------------------

class ChargeWorker:
    def __init__(self, config: Config, queue: JobQueue, store: DurableStore,
                 charger: ChargingService):
        self.config = config
        self.queue = queue
        self.store = store
        self.charger = charger

    def process_job(self, job_id: str) -> None:
        # 1. Atomically claim the job — prevents two workers charging twice
        if not self.store.claim_job(self.queue, job_id):
            # Another worker claimed it, or it doesn't exist; skip.
            return

        job = self.queue.read_job(job_id)
        if job is None:
            return

        customer_id = job["customer_id"]
        amount_cents = job["amount_cents"]
        # The stable, externally-supplied idempotency key:
        idempotency_key = job["idempotency_key"]

        # 2. Check completion record (racing with another worker is fine because
        #    only one worker can hold the claim from step 1).
        completion = self.store.read(idempotency_key)
        if completion and completion.get("status") == "completed":
            return

        # 3. Record "in-progress" BEFORE the external effect, durably.
        self.store.write_atomic(idempotency_key, {"status": "in_progress"})

        # 4. Charge with retries (bounded, jittered backoff)
        success = self._charge_with_retry(customer_id, amount_cents, idempotency_key)

        # 5. Record the final outcome durably.
        status = "completed" if success else "failed"
        self.store.write_atomic(idempotency_key, {"status": status})

    def _charge_with_retry(self, customer_id: str, amount_cents: int,
                           idempotency_key: str) -> bool:
        attempt = 0
        while attempt <= self.config.max_retries:
            try:
                if self.charger.charge(customer_id, amount_cents, idempotency_key):
                    return True
            except Exception:
                pass  # in production, log with error context
            attempt += 1
            if attempt <= self.config.max_retries:
                backoff_ms = min(
                    self.config.base_backoff_ms * (2 ** attempt),
                    self.config.max_backoff_ms,
                )
                # Add jitter to avoid retry storms
                backoff_ms = backoff_ms * random.uniform(0.5, 1.5)
                time.sleep(backoff_ms / 1000.0)
        return False


# ---------------------------------------------------------------------------
# Example usage
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    work_root = Path(tempfile.mkdtemp()) / "work"
    queue_root = work_root / "queue"
    completion_root = work_root / "completions"

    config = Config(
        queue_dir=queue_root,
        completion_dir=completion_root,
    )
    queue = JobQueue(queue_root)
    store = DurableStore(completion_root)
    charger = ChargingService()

    worker = ChargeWorker(config, queue, store, charger)

    # Simulate a job arriving in the queue
    job_id = str(uuid.uuid4())
    idem_key = str(uuid.uuid4())
    job_data = {
        "customer_id": "cust_123",
        "amount_cents": 500,
        "idempotency_key": idem_key,
    }
    with (queue.pending_dir / f"{job_id}.json").open("w") as f:
        json.dump(job_data, f)

    worker.process_job(job_id)

    # Verify
    final_state = store.read(idem_key)
    assert final_state["status"] == "completed", final_state
