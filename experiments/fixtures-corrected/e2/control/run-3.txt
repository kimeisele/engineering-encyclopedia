Here's a Python function that runs git commands with user-supplied branch names:

```python
import subprocess
import shlex
from typing import List, Optional, Union

def run_git_command(branch_name: str, command: Optional[str] = "checkout", 
                   extra_args: Optional[List[str]] = None) -> subprocess.CompletedProcess:
    """
    Run a git command with a specified branch name.
    
    Args:
        branch_name: The name of the branch to use
        command: The git command to run (default: "checkout")
        extra_args: Additional arguments to pass to the git command
    
    Returns:
        subprocess.CompletedProcess: The result of the subprocess execution
    
    Raises:
        ValueError: If branch_name is empty or contains invalid characters
        subprocess.CalledProcessError: If the git command fails
    """
    
    # Validate branch name
    if not branch_name or not branch_name.strip():
        raise ValueError("Branch name cannot be empty")
    
    # Basic branch name validation (reject problematic characters)
    invalid_chars = [' ', '..', '~', '^', ':', '?', '*', '[', '\\']
    for char in invalid_chars:
        if char in branch_name:
            raise ValueError(f"Invalid character '{char}' in branch name")
    
    # Build the git command
    git_command = ["git", command, branch_name]
    
    # Add extra arguments if provided
    if extra_args:
        git_command.extend(extra_args)
    
    try:
        # Execute the command
        result = subprocess.run(
            git_command,
            capture_output=True,
            text=True,
            check=False  # We'll handle error checking ourselves
        )
        
        # Check if the command was successful
        if result.returncode != 0:
            raise subprocess.CalledProcessError(
                result.returncode, 
                git_command,
                output=result.stdout,
                stderr=result.stderr
            )
        
        return result
    
    except FileNotFoundError:
        raise FileNotFoundError("Git command not found. Make sure Git is installed and in PATH.")

def run_git_safe(branch_name: str, command: str = "checkout", 
                extra_args: Optional[List[str]] = None) -> dict:
    """
    A safer version that returns a dictionary with detailed results.
    This version handles errors gracefully without raising exceptions.
    
    Returns:
        dict: Contains 'success', 'output', 'error', and 'returncode'
    """
    
    if not branch_name or not branch_name.strip():
        return {
            'success': False,
            'output': '',
            'error': 'Branch name cannot be empty',
            'returncode': -1
        }
    
    # Build the command using shlex to properly handle arguments
    cmd_parts = ["git", command, branch_name]
    if extra_args:
        cmd_parts.extend(extra_args)
    
    try:
        result = subprocess.run(
            cmd_parts,
            capture_output=True,
            text=True,
            timeout=30,  # Add a timeout to prevent hanging
            check=False
        )
        
        return {
            'success': result.returncode == 0,
            'output': result.stdout,
            'error': result.stderr if result.returncode != 0 else '',
            'returncode': result.returncode
        }
    
    except subprocess.TimeoutExpired:
        return {
            'success': False,
            'output': '',
            'error': f'Command timed out after 30 seconds',
            'returncode': -2
        }
    
    except subprocess.SubprocessError as e:
        return {
            'success': False,
            'output': '',
            'error': f'Subprocess error: {str(e)}',
            'returncode': -3
        }
    
    except FileNotFoundError:
        return {
            'success': False,
            'output': '',
            'error': 'Git not found. Is it installed and in PATH?',
            'returncode': -4
        }

# Example usage and additional helper functions

def example_usage():
    """
    Example usage of the git command functions
    """
    
    # Basic usage - checkout a branch
    try:
        result = run_git_command("feature/my-feature")
        print(f"Checkout output: {result.stdout}")
    except subprocess.CalledProcessError as e:
        print(f"Error: {e.stderr}")
    except ValueError as e:
        print(f"Validation error: {e}")
    
    # Using the safe version
    result = run_git_safe("main", "pull")
    print(f"Success: {result['success']}")
    print(f"Output: {result['output']}")
    if not result['success']:
        print(f"Error: {result['error']}")
    
    # Custom command with extra arguments
    result = run_git_safe("release/v1.0", "log", ["--oneline", "-5"])
    print(f"Log output (last 5 commits):\n{result['output']}")

def git_checkout(branch_name: str, create: bool = False) -> dict:
    """
    Helper function specifically for checkout operations
    
    Args:
        branch_name: Branch to checkout
        create: If True, creates the branch if it doesn't exist
    
    Returns:
        dict: Result of the operation
    """
    if create:
        return run_git_safe(branch_name, "checkout", ["-b"])
    else:
        return run_git_safe(branch_name, "checkout")

def git_branch_list(branch_name: Optional[str] = None) -> dict:
    """
    List branches, optionally filtered by a pattern
    
    Args:
        branch_name: Optional branch pattern to match
    
    Returns:
        dict: Result of the operation
    """
    if branch_name:
        return run_git_safe(branch_name, "branch", ["-a", "--list"])
    else:
        return run_git_safe("", "branch", ["-a"])

if __name__ == "__main__":
    example_usage()
```

Here's also a simple command-line interface version:

```python
import argparse
import subprocess
import sys

def main():
    parser = argparse.ArgumentParser(description="Run git commands with branch names")
    parser.add_argument("branch", help="Branch name to use")
    parser.add_argument("command", nargs="?", default="checkout", 
                       help="Git command to run (default: checkout)")
    parser.add_argument("args", nargs="*", help="Additional git arguments")
    
    args = parser.parse_args()
    
    # Build and execute the command
    cmd = ["git", args.command, args.branch] + args.args
    
    try:
        result = subprocess.run(cmd, capture_output=True, text=True)
        print(result.stdout, end='')
        if result.stderr:
            print(result.stderr, file=sys.stderr, end='')
        sys.exit(result.returncode)
    except FileNotFoundError:
        print("Error: Git not found. Make sure it's installed and in PATH.", 
              file=sys.stderr)
        sys.exit(1)

if __name__ == "__main__":
    main()
```

5. **Helper functions**: Convenience functions for common operations
6. **CLI support**: Can be used from the command line
