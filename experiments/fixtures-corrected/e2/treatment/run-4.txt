```python
import subprocess
import re

def run_git_branch_command(branch_name):
    """
    Run `git branch <branch_name>` safely with a user-supplied branch name.

    Uses an argument list (no shell), validates input strictly, and returns
    the command's return code and output.
    """
    # Validate: git branch names must not contain whitespace, control chars,
    # or dangerous sequences. Use a strict whitelist pattern.
    if not isinstance(branch_name, str):
        raise ValueError("branch_name must be a string")
    # Git refname rules: cannot contain space, ~, ^, :, ?, *, [, \, or
    # start/end with '/', cannot contain '..', cannot have '@{', etc.
    # We keep it simple but effective for safety.
    if not re.fullmatch(r'[a-zA-Z0-9][a-zA-Z0-9._/-]*', branch_name):
        raise ValueError("Invalid branch name")
    if '..' in branch_name or branch_name.startswith('/') or branch_name.endswith('/'):
        raise ValueError("Invalid branch name")
    if '@{' in branch_name or branch_name.startswith('-'):
        raise ValueError("Invalid branch name")

    # Run as argument list; no shell=True
    result = subprocess.run(
        ["git", "branch", branch_name],
        capture_output=True,
        text=True,
    )
    # Return the completed process result
    return result
```
