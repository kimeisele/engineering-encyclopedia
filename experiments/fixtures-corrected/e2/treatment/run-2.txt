```python
import subprocess
import re
from typing import List

def run_git_command(branch_name: str) -> str:
    """
    Runs a safe git command with a user-supplied branch name.
    
    The branch name is validated against a strict pattern before being
    passed as a list argument to subprocess. This prevents shell injection
    and ensures only a branch name (not arbitrary commands) is accepted.
    """
    # Validate branch name: only alphanumeric, dash, underscore, dot, and slash
    if not re.fullmatch(r'[A-Za-z0-9._/-]+', branch_name):
        raise ValueError(f"Invalid branch name: {branch_name}")
    
    # Reject dangerous patterns (e.g., leading dashes that could be flags)
    if branch_name.startswith('-'):
        raise ValueError("Branch name cannot start with a dash")
    
    # Build the command as a list — never a string with shell=True
    command: List[str] = ["git", "checkout", branch_name]
    
    try:
        # Run without shell=True; subprocess handles argument separation safely
        result = subprocess.run(
            command,
            capture_output=True,
            text=True,
            check=True
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        raise RuntimeError(f"Git command failed: {e.stderr.strip()}") from e
```
