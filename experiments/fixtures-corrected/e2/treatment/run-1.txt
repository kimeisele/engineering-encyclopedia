```python
import subprocess
import re

def run_git_branch_command(branch_name):
    """
    Run a safe git command (e.g., `git branch <branch_name>`) with a
    user-supplied branch name.

    Args:
        branch_name (str): The branch name provided by the user.

    Returns:
        subprocess.CompletedProcess: The result of the git command.

    Raises:
        ValueError: If the branch name is not a valid git branch name.
    """
    if not isinstance(branch_name, str):
        raise ValueError("Branch name must be a string")

    # Validate: allow only safe characters for a git branch name
    # Git branch refs must not contain spaces, ~, ^, :, ?, *, [, \, or
    # start with '-' or contain '..'
    if not re.fullmatch(r"[A-Za-z0-9._/-]+", branch_name):
        raise ValueError("Invalid branch name: contains disallowed characters or patterns")

    if branch_name.startswith("-") or ".." in branch_name:
        raise ValueError("Invalid branch name: must not start with '-' or contain '..'")

    # Build the command as a list (no shell=True), with the validated input
    cmd = ["git", "branch", branch_name]
    try:
        result = subprocess.run(cmd, check=False, capture_output=True, text=True)
        return result
    except FileNotFoundError:
        raise RuntimeError("Git executable not found")
```
