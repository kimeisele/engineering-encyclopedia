```python
import subprocess
import re

def run_git_command(branch_name: str) -> subprocess.CompletedProcess:
    """
    Run a git command (git checkout) with a user-supplied branch name.

    Args:
        branch_name: The branch name provided by the user.

    Returns:
        The CompletedProcess result of the git command.

    Raises:
        ValueError: If the branch name is not a valid git branch name format.
    """
    # Validate that the branch name matches git's safe branch name pattern.
    # Git branch names must not start with '-', must not contain '..',
    # must not contain '~', '^', ':', '?', '*', '[', '\\', space, or control chars,
    # and must not end with '/', '.', or '.lock'.
    if not re.fullmatch(r'[a-zA-Z0-9][a-zA-Z0-9._/-]*', branch_name):
        raise ValueError(f"Invalid git branch name: {branch_name!r}")

    # Pass the command as a list, never as a string with shell=True.
    # The branch name is an argument element, not interpolated into a shell string.
    result = subprocess.run(
        ["git", "checkout", branch_name],
        capture_output=True,
        text=True,
        check=False,
    )
    return result
```
