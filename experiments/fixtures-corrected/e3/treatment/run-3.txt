```python
import json
import os
import tempfile
from typing import Any, Callable


def update_json_config(
    filepath: str,
    updater: Callable[[dict[str, Any]], dict[str, Any]],
    *,
    validator: Callable[[dict[str, Any]], bool] | None = None,
    fsync: bool = True,
) -> dict[str, Any]:
    """
    Atomically update a JSON config file on disk.

    The file is read, the updater is applied to its contents, the result is
    validated (if a validator is provided), and then written to a temporary
    file in the same directory, flushed, and atomically renamed over the
    original.

    Args:
        filepath: Path to the JSON config file.
        updater: Callable that receives the current config (dict) and
            returns the new config (dict).
        validator: Optional callable that receives the new config and
            returns True if it is acceptable, False otherwise.
        fsync: If True (default), fsync both the file and the directory
            to ensure durability across power loss.

    Returns:
        The new config that was written.

    Raises:
        FileNotFoundError: If the target file does not exist.
        ValueError: If the new config fails validation.
        OSError: On I/O errors (including cross-filesystem rename errors).
    """
    # Read the current config
    with open(filepath, "r", encoding="utf-8") as f:
        current_config = json.load(f)

    # Apply the updater
    new_config = updater(current_config)

    # Validate the new config (reject invalid data)
    if validator is not None and not validator(new_config):
        raise ValueError("New config failed validation")

    # Write to a temp file in the same directory (atomic rename requires
    # same filesystem)
    directory = os.path.dirname(os.path.abspath(filepath))
    fd, temp_path = tempfile.mkstemp(dir=directory, suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(new_config, f, indent=2)
            f.flush()
            if fsync:
                os.fsync(f.fileno())

        # Atomic rename over the target
        os.replace(temp_path, filepath)

        # Flush directory metadata so the rename is durable
        if fsync:
            dir_fd = os.open(directory, os.O_RDONLY)
            try:
                os.fsync(dir_fd)
            finally:
                os.close(dir_fd)
    except Exception:
        # Clean up the temp file on failure
        try:
            os.unlink(temp_path)
        except OSError:
            pass
        raise

    return new_config
```
