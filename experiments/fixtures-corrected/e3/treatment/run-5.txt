import json
import os
import tempfile


def update_json_config(file_path: str, updates: dict) -> None:
    """Atomically update a JSON config file with the given updates.

    Args:
        file_path: Path to the JSON config file.
        updates: Dictionary of values to merge into the config.

    Raises:
        ValueError: If updates is not a valid JSON-serializable object.
        OSError: If the file cannot be written.
    """
    # Validate input at the trust boundary
    try:
        json.dumps(updates)  # Ensure it's JSON-serializable
    except (TypeError, ValueError) as e:
        raise ValueError(f"Invalid updates: {e}") from e

    # Read existing config, or start fresh if file doesn't exist
    try:
        with open(file_path, "r") as f:
            config = json.load(f)
    except FileNotFoundError:
        config = {}
    except (json.JSONDecodeError, OSError) as e:
        # If existing file is corrupt, fail rather than overwrite silently
        raise ValueError(f"Existing config is not valid JSON: {e}") from e

    # Merge updates (shallow merge for simplicity)
    if not isinstance(config, dict) or not isinstance(updates, dict):
        raise ValueError("Config and updates must be JSON objects")
    config.update(updates)

    # Atomic write: temp file in same directory, flush, rename
    directory = os.path.dirname(os.path.abspath(file_path))
    fd, temp_path = tempfile.mkstemp(dir=directory, prefix=".config-", suffix=".tmp")
    try:
        with os.fdopen(fd, "w") as f:
            json.dump(config, f, indent=2, sort_keys=True)
            f.write("\n")
            f.flush()
            os.fsync(f.fileno())  # Flush data to disk

        # Atomic rename — same filesystem, so it's atomic on POSIX
        os.replace(temp_path, file_path)
    except Exception:
        # Clean up temp file on failure
        try:
            os.unlink(temp_path)
        except OSError:
            pass
        raise
