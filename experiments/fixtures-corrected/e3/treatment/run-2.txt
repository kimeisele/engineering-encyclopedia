```python
import json
import os
import tempfile
from typing import Any, Dict

def update_json_config(filepath: str, updater: Dict[str, Any]) -> None:
    """
    Update a JSON config file by atomically replacing it with new content.

    The function reads the existing config (or starts with an empty dict),
    applies the updater dict on top of it (shallow merge), and writes the
    result to a temp file in the same directory, flushes it to disk, then
    renames it over the original. This ensures readers see either the old
    or the new content, never a truncated/mixed state.

    Args:
        filepath: Path to the JSON config file.
        updater: Dict of key/value pairs to update in the config.

    Raises:
        ValueError: If updater is not a dict.
        OSError: If the file cannot be written (e.g. permissions).
        json.JSONDecodeError: If the existing file contains invalid JSON.
    """
    if not isinstance(updater, dict):
        raise ValueError("updater must be a dict")

    # Validate the updater is JSON-serializable before touching the file
    try:
        json.dumps(updater)
    except (TypeError, ValueError) as e:
        raise ValueError(f"updater is not JSON-serializable: {e}")

    # Read existing config (or start fresh if file doesn't exist)
    if os.path.exists(filepath):
        with open(filepath, 'r', encoding='utf-8') as f:
            config = json.load(f)
    else:
        config = {}

    # Merge updater into config (shallow merge)
    config.update(updater)

    # Write atomically: temp file in same dir, flush, rename
    dirpath = os.path.dirname(os.path.abspath(filepath))
    fd, tmp_path = tempfile.mkstemp(dir=dirpath, prefix=".config-", suffix=".tmp")
    try:
        with os.fdopen(fd, 'w', encoding='utf-8') as f:
            json.dump(config, f, indent=2)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp_path, filepath)
    except Exception:
        # Clean up temp file if anything goes wrong before rename
        try:
            os.unlink(tmp_path)
        except OSError:
            pass
        raise
```
