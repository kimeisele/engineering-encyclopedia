```python
import json
import os
import tempfile
from typing import Any


def update_json_config(filepath: str, new_config: Any) -> None:
    """
    Atomically update a JSON config file on disk.

    The function validates the new config, writes it to a temporary file in the
    same directory, flushes it to disk, and then atomically renames it over the
    target file. This ensures readers see either the old or the new content,
    never a truncated or partial state.

    Args:
        filepath: Path to the JSON config file to update.
        new_config: The new config data (must be JSON-serializable).
    """
    # Input validation at the trust boundary
    if not isinstance(filepath, str) or not filepath:
        raise ValueError("filepath must be a non-empty string")
    
    if not isinstance(new_config, dict):
        raise ValueError("new_config must be a dict")
    
    if not _is_json_serializable(new_config):
        raise ValueError("new_config must be JSON-serializable")

    directory = os.path.dirname(filepath) or "."
    basename = os.path.basename(filepath)
    
    # Ensure target is a .json file (basic check, not a security boundary)
    if not basename.lower().endswith(".json"):
        raise ValueError("filepath must end with .json")

    temp_path = None
    try:
        # Create temp file in same directory for atomic rename
        fd, temp_path = tempfile.mkstemp(
            dir=directory,
            prefix=f".{basename}.",
            suffix=".tmp",
            text=True
        )
        
        # Write JSON content to temp file
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(new_config, f, indent=2, sort_keys=True, ensure_ascii=False)
            f.write("\n")
            # Flush data to OS buffer
            f.flush()
            # fsync to ensure data is durably written to disk
            os.fsync(f.fileno())
        
        # Atomically rename temp file over target
        os.replace(temp_path, filepath)
        temp_path = None  # Prevent cleanup on success
        
    except Exception as e:
        raise RuntimeError(f"Failed to update config file: {e}") from e
    finally:
        # Clean up temp file if it still exists (write failed or crash)
        if temp_path and os.path.exists(temp_path):
            os.unlink(temp_path)


def _is_json_serializable(obj: Any) -> bool:
    """Check if an object can be serialized to JSON."""
    try:
        json.dumps(obj)
        return True
    except (TypeError, ValueError):
        return False
```
