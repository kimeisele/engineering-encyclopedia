```python
import json
import os
import tempfile
from typing import Any, Union, Type, Optional


def update_json_config(
    path: str,
    updater: Union[dict, list, None] = None,
    *,
    schema: Optional[Type] = None,
    **kwargs: Any,
) -> None:
    """
    Atomically update a JSON config file on disk.

    Args:
        path: Path to the JSON config file.
        updater: New config data (dict or list) to write.
                  If None and kwargs provided, kwargs are merged with existing.
        schema: Optional type to validate the final data against (e.g., dict, list).
        **kwargs: If updater is None, these are merged into the existing config.

    Raises:
        FileNotFoundError: If the config file does not exist and no updater provided.
        ValueError: If the final data fails schema validation or JSON serialization.
    """
    # Determine the target data
    if updater is not None:
        new_data = updater
    else:
        # Read existing file, preserve if it exists, else start empty
        if not os.path.exists(path):
            new_data = {}
        else:
            with open(path, "r") as f:
                try:
                    new_data = json.load(f)
                except (json.JSONDecodeError, ValueError):
                    raise ValueError(f"Existing file {path} is not valid JSON")
        new_data.update(kwargs)

    # Validate against schema if given
    if schema is not None and not isinstance(new_data, schema):
        raise ValueError(f"Data must be of type {schema}, got {type(new_data)}")

    # Serialize to JSON (will raise TypeError on non-serializable objects)
    try:
        content = json.dumps(new_data, indent=2, sort_keys=True) + "\n"
    except (TypeError, ValueError) as e:
        raise ValueError(f"Serialization failed: {e}")

    # Atomic write: temp file in same dir, flush, fsync, rename
    dir_path = os.path.dirname(os.path.abspath(path))
    fd, tmp_path = tempfile.mkstemp(dir=dir_path, prefix=".config-", suffix=".tmp")
    try:
        with os.fdopen(fd, "w") as f:
            f.write(content)
            f.flush()
            os.fsync(f.fileno())  # durability for file content

        # Ensure directory metadata is flushed (durability for the rename)
        dir_fd = os.open(dir_path, os.O_DIRECTORY)
        try:
            os.fsync(dir_fd)
        finally:
            os.close(dir_fd)

        os.replace(tmp_path, path)  # atomic on POSIX, works on Windows (move)
        tmp_path = None  # mark success so cleanup skips
    finally:
        if tmp_path is not None:
            try:
                os.unlink(tmp_path)
            except OSError:
                pass
```
