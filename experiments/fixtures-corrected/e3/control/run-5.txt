Here's a comprehensive function that updates a JSON config file on disk:

```python
import json
import os
import tempfile
from typing import Any, Dict, Optional, Union
from pathlib import Path

def update_json_config(
    file_path: Union[str, Path],
    updates: Dict[str, Any],
    *,
    create_if_missing: bool = False,
    atomic_write: bool = True,
    indent: int = 2,
    backup: bool = False
) -> bool:
    """
    Update a JSON config file on disk with the provided updates.
    
    Args:
        file_path: Path to the JSON config file
        updates: Dictionary containing updates to apply (supports nested paths with dots)
        create_if_missing: If True, create the file if it doesn't exist (default: False)
        atomic_write: If True, write to a temporary file then replace (default: True)
        indent: JSON indentation level (default: 2)
        backup: If True, create a backup of the original file (default: False)
    
    Returns:
        bool: True if update was successful, False otherwise
    
    Raises:
        FileNotFoundError: If file doesn't exist and create_if_missing is False
        json.JSONDecodeError: If existing file contains invalid JSON
        PermissionError: If no write permission
    """
    
    # Convert to Path object for easier handling
    file_path = Path(file_path)
    
    # Check if file exists
    if not file_path.exists():
        if not create_if_missing:
            raise FileNotFoundError(f"Config file not found: {file_path}")
        # Create empty config if file doesn't exist
        config_data = {}
    else:
        # Read existing file
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                config_data = json.load(f)
        except json.JSONDecodeError as e:
            raise json.JSONDecodeError(f"Invalid JSON in config file: {e.msg}", e.doc, e.pos)
    
    # Apply updates (with nested path support)
    def apply_nested_updates(data: dict, updates: dict) -> dict:
        """Recursively apply nested updates to the config data."""
        for key, value in updates.items():
            if isinstance(value, dict) and isinstance(data.get(key), dict):
                # Merge nested dictionaries
                data[key] = apply_nested_updates(data.get(key, {}), value)
            else:
                # Set the value (replace entirely for non-dict values)
                data[key] = value
        return data
    
    # Apply the updates
    config_data = apply_nested_updates(config_data, updates)
    
    # Create backup if requested
    if backup and file_path.exists():
        backup_path = file_path.with_suffix(file_path.suffix + '.bak')
        try:
            import shutil
            shutil.copy2(file_path, backup_path)
        except Exception as e:
            print(f"Warning: Failed to create backup: {e}")
    
    # Write the updated config
    try:
        # Ensure parent directory exists
        file_path.parent.mkdir(parents=True, exist_ok=True)
        
        if atomic_write:
            # Write to temporary file first, then replace
            temp_fd, temp_path = tempfile.mkstemp(
                dir=str(file_path.parent),
                prefix=f'.{file_path.name}.',
                suffix='.tmp'
            )
            try:
                with os.fdopen(temp_fd, 'w', encoding='utf-8') as f:
                    json.dump(config_data, f, indent=indent, ensure_ascii=False)
                    f.flush()
                    os.fsync(f.fileno())
                
                # Replace the original file
                os.replace(temp_path, file_path)
            except Exception:
                # Clean up temp file on error
                if os.path.exists(temp_path):
                    os.remove(temp_path)
                raise
        else:
            # Direct write (non-atomic)
            with open(file_path, 'w', encoding='utf-8') as f:
                json.dump(config_data, f, indent=indent, ensure_ascii=False)
        
        return True
        
    except Exception as e:
        print(f"Error writing config file: {e}")
        return False

def update_json_config_flat(
    file_path: str,
    updates: Dict[str, Any],
    create_if_missing: bool = False
) -> bool:
    """
    Simple version that only supports flat key-value updates.
    
    Args:
        file_path: Path to the JSON config file
        updates: Dictionary of key-value pairs to update
        create_if_missing: If True, create the file if it doesn't exist
    
    Returns:
        bool: True if successful, False otherwise
    """
    file_path = Path(file_path)
    
    # Read existing file
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            config_data = json.load(f)
    except (FileNotFoundError, json.JSONDecodeError):
        if not create_if_missing:
            return False
        config_data = {}
    
    # Apply updates
    config_data.update(updates)
    
    # Write back
    try:
        with open(file_path, 'w', encoding='utf-8') as f:
            json.dump(config_data, f, indent=2)
        return True
    except Exception:
        return False


def get_nested_value(config: dict, key_path: str, default=None) -> Any:
    """Get a nested value from config using dot notation."""
    keys = key_path.split('.')
    current = config
    for key in keys:
        if isinstance(current, dict) and key in current:
            current = current[key]
        else:
            return default
    return current


def set_nested_value(config: dict, key_path: str, value: Any) -> dict:
    """Set a nested value in config using dot notation."""
    keys = key_path.split('.')
    current = config if isinstance(config, dict) else {}
    
    for key in keys[:-1]:
        if key not in current or not isinstance(current[key], dict):
            current[key] = {}
        current = current[key]
    
    current[keys[-1]] = value
    return config
```

## Key Features:

1. **Nested Updates**: Supports dot notation for nested keys (e.g., `"server.port": 9090`)
2. **Atomic Writes**: Option to write atomically to prevent corruption
3. **Backup Support**: Can create a backup before modification
4. **File Creation**: Option to create the file if it doesn't exist
5. **Error Handling**: Proper error handling for various scenarios
6. **Flexible**: Works with both `str` and `Path` objects

## Usage Examples:

```python
# Basic usage
update_json_config('config.json', {'debug': True, 'port': 8080})

# With nested paths
update_json_config('config.json', {
    'server.host': 'example.com',
    'database.connection.timeout': 30
})

# Create new file if missing
update_json_config('new_config.json', {'name': 'app'}, create_if_missing=True)

# Use with backup
update_json_config('config.json', {'port': 9090}, backup=True)

# Atomic write (default)
update_json_config('config.json', {'key': 'value'}, atomic_write=True)
```

- **Flexibility**: Supports both flat and nested configurations
- **Reliability**: Handles edge cases like missing files and invalid JSON
