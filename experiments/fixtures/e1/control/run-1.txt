        """Get next job from queue"""

        """Mark task as done"""
        self.queue.task_done()
        
    def delete_job(self, job_id: str):
        """Remove job from records"""
        self.job_records.pop(job_id, None)

    """
    Main task to process a payment
    
    Expected job_data format:
    {
        'job_id': str,
        'customer_id': str,
        'amount': float,
        'currency': str,
        'description': str,
        'order_id': str,
        'payment_method': str  # 'card' or 'paypal'
    }
    """
    job_id = job_data.get('job_id')
    customer_id = job_data.get('customer_id')
    amount = job_data.get('amount')
    currency = job_data.get('currency', 'USD')

    try:
        # Step 1: Fetch customer information
        customer = fetch_customer(customer_id)
        if not customer:
            raise Exception(f"Customer {customer_id} not found")

        # Step 4: Process payment with payment gateway
        payment_result = charge_customer(
            customer_id=customer_id,
            amount=amount,
            currency=currency,
            payment_method=customer.get('payment_method_token'),
            description=job_data.get('description', ''),

                customer_id=customer_id,
                job_id=job_id,
                amount=amount,
                transaction_id=payment_result['transaction_id']
            )
            
            return {
                'status': 'success',
                'job_id': job_id,
                'transaction_id': payment_result['transaction_id'],
                'message': 'Payment processed successfully'
            }

            # Payment failed

            logger.error(f"Payment failed for job {job_id}: {error_msg}")
            
            # Retry with exponential backoff
            if self.request.retries < self.max_retries:
                retry_delay = min(30, 2 ** self.request.retries * 5)

            customer_id=customer_id,

            error_message=str(exc),

                'error': str(exc),

            }

@celery_app.task(name='process_payment.webhook')
def handle_payment_webhook(payload: Dict[str, Any]):
    """Handle payment gateway webhooks"""
    event_type = payload.get('type')
    
    if event_type == 'payment_intent.succeeded':
        # Payment succeeded
        payment_id = payload.get('data', {}).get('object', {}).get('id')
        logger.info(f"Payment succeeded: {payment_id}")
        
    elif event_type == 'payment_intent.payment_failed':
        # Payment failed
        payment_id = payload.get('data', {}).get('object', {}).get('id')
        logger.error(f"Payment failed via webhook: {payment_id}")
        
    elif event_type == 'charge.refunded':
        # Refund processed
        charge_id = payload.get('data', {}).get('object', {}).get('id')
        logger.info(f"Refund processed: {charge_id}")
    
    return {'status': 'received'}
