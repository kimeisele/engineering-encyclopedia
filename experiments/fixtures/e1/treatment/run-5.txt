```python
"""
Charging worker: consumes jobs from a queue, charges a customer exactly once.

Design notes:
- Each job carries an idempotency key provided by the producer.
- Completion is recorded in a journal *before* the charge is attempted, and the
  journal entry is fsynced. This makes the "already charged" check durable:
  if we crash after charging but before writing the in-memory state, the
  journal already says "started", and the recovery path resolves the ambiguity.
- The deduplication write (started marker) and the charge attempt are guarded
  by an atomic file claim so two workers cannot both see the job as unclaimed.
- A temp file in the same directory, fsync + rename = atomic claim.
"""

import json
import os
import tempfile
import time
import random
from pathlib import Path


class ChargeWorker:
    def __init__(self, queue, claim_dir, max_attempts=5):
        self.queue = queue
        self.claim_dir = Path(claim_dir)
        self.claim_dir.mkdir(parents=True, exist_ok=True)
        self.max_attempts = max_attempts

    def run(self):
        while True:
            job = self.queue.dequeue()  # may block
            if job is None:
                continue
            self.process_job(job)

    def process_job(self, job):
        idem_key = job["idempotency_key"]

        # -- claim: atomically create a started-marker (dedup record) ----------
        # The claim file is the durable "completion record" for our purposes.
        # Writing it with tempfile + fsync + rename ensures no two workers can
        # both observe the job as not-yet-done.
        claim_path = self.claim_dir / f"{idem_key}.claim"
        if not self._atomic_claim(claim_path):
            # Another worker claimed it; we are not the one to charge.
            return

        # -- charge with bounded retries ---------------------------------------
        for attempt in range(1, self.max_attempts + 1):
            try:
                self._charge(job)
                self._mark_completed(claim_path)
                return
            except Exception as exc:
                if attempt == self.max_attempts:
                    self._mark_failed(claim_path, str(exc))
                    return
                # exponential backoff with jitter (avoid retry storms)
                delay = (2 ** (attempt - 1)) + random.uniform(0, 1)
                time.sleep(delay)

    def _atomic_claim(self, claim_path):
        """Atomically create claim file. Returns True if we won the claim."""
        # tempfile in the SAME directory as target (python.files.atomic-replacement)
        fd, tmp_name = tempfile.mkstemp(dir=self.claim_dir, prefix=".claim-tmp-")
        try:
            with os.fdopen(fd, "w") as f:
                json.dump({"status": "in-flight"}, f)
                f.flush()
                os.fsync(f.fileno())  # durable before rename
            # O_EXCL rename: fails if target already exists → atomic claim.
            os.rename(tmp_name, claim_path)
            return True
        except FileExistsError:
            os.unlink(tmp_name)  # cleanup leftover temp
            return False
        except OSError:
            os.unlink(tmp_name)
            return False

    def _charge(self, job):
        # This would call a payment gateway; here we simulate.
        # The idempotency_key is sent to the gateway so the gateway itself
        # can deduplicate (belt-and-suspenders), but our journal is the
        # source of truth for "done" that survives restart.
        print(f"Charging customer {job['customer_id']} amount {job['amount']}")

    def _mark_completed(self, claim_path):
        with open(claim_path, "r+") as f:
            data = json.load(f)
            data["status"] = "completed"
            f.seek(0)
            json.dump(data, f)
            f.truncate()
            f.flush()
            os.fsync(f.fileno())

    def _mark_failed(self, claim_path, error):
        with open(claim_path, "r+") as f:
            data = json.load(f)
            data["status"] = "failed"
            data["error"] = error
            f.seek(0)
            json.dump(data, f)
            f.truncate()
            f.flush()
            os.fsync(f.fileno())
```
