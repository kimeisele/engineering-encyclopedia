```python
import json
import os
import uuid
from dataclasses import dataclass
from pathlib import Path
from typing import Optional


@dataclass
class Job:
    id: str
    customer_id: str
    amount_cents: int


class ChargeError(Exception):
    """Raised when a charge fails for a reason unrelated to job processing."""


class Queue:
    """Minimal queue interface — replaced by real queue in production."""

    def dequeue(self) -> Optional[Job]:
        raise NotImplementedError

    def ack(self, job_id: str) -> None:
        raise NotImplementedError

    def nack(self, job_id: str) -> None:
        raise NotImplementedError


class PaymentGateway:
    """Minimal gateway — replaced by real client in production."""

    def charge(self, customer_id: str, amount_cents: int) -> str:
        raise NotImplementedError


class IdempotencyStore:
    """Durable, atomic deduplication store keyed by job_id."""

    def __init__(self, data_dir: Path):
        self.data_dir = data_dir
        self.data_dir.mkdir(parents=True, exist_ok=True)

    def _path_for(self, job_id: str) -> Path:
        return self.data_dir / f"{job_id}.json"

    def already_processed(self, job_id: str) -> bool:
        return self._path_for(job_id).exists()

    def record_completion(self, job_id: str) -> None:
        """Atomically persist completion. The finished marker is visible to
        other workers only after the rename succeeds."""
        target = self._path_for(job_id)
        tmp = self.data_dir / f".{job_id}.{uuid.uuid4().hex}.tmp"
        with open(tmp, "w") as f:
            f.write(json.dumps({"job_id": job_id}))
            f.flush()
            os.fsync(f.fileno())  # durable before rename
        os.replace(tmp, target)  # atomic on POSIX


class Worker:
    def __init__(
        self,
        queue: Queue,
        gateway: PaymentGateway,
        idempotency_store: IdempotencyStore,
        max_attempts: int = 3,
    ):
        self.queue = queue
        self.gateway = gateway
        self.idempotency_store = idempotency_store
        self.max_attempts = max_attempts

    def process_one(self) -> bool:
        job = self.queue.dequeue()
        if job is None:
            return False

        try:
            # Check-then-act guarded by idempotency record: if we crash after
            # charging the customer but before recording, the next attempt
            # will charge again. Precisely the race this store prevents.
            if self.idempotency_store.already_processed(job.id):
                self.queue.ack(job.id)
                return True

            self._charge_with_retry(job)
            self.idempotency_store.record_completion(job.id)
            self.queue.ack(job.id)
        except ChargeError:
            # Permanent error — do not retry forever with poison messages.
            self.queue.nack(job.id)
        except Exception:
            # Transient error — let the queue redeliver; safe because the
            # idempotency record makes repeated attempts benign.
            self.queue.nack(job.id)

        return True

    def _charge_with_retry(self, job: Job) -> str:
        attempt = 0
        while True:
            try:
                return self.gateway.charge(job.customer_id, job.amount_cents)
            except Exception as e:
                attempt += 1
                if attempt >= self.max_attempts:
                    raise ChargeError(str(e)) from e
                import time

                time.sleep(0.1 * (2**attempt) + (attempt * 0.05))  # backoff + jitter
```
