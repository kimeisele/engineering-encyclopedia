```python
import subprocess
import re

def run_git_command(branch_name: str) -> str:
    """
    Run a git command with a user-supplied branch name.

    Args:
        branch_name: The branch name to use in the git command.

    Returns:
        The stdout of the git command.

    Raises:
        ValueError: If the branch name is not a valid git branch name.
        subprocess.CalledProcessError: If the git command fails.
    """
    # Validate input at the trust boundary (security.input-validation)
    # Git branch names cannot contain spaces, control characters, ~, ^, :, ?, *, [, \,
    # or start with a dash or contain ".."
    if not re.match(r'^[a-zA-Z0-9_\-./]+$', branch_name):
        raise ValueError(f"Invalid branch name: {branch_name}")
    if branch_name.startswith('-') or '..' in branch_name:
        raise ValueError(f"Invalid branch name: {branch_name}")

    # Run the command with the argument list, never shell=True (python.processes.subprocess-safety)
    # The branch name is passed as an element of the list, not interpolated into a string
    result = subprocess.run(
        ["git", "branch", branch_name],
        capture_output=True,
        text=True,
        check=True
    )
    return result.stdout
```
