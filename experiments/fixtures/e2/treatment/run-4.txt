import subprocess
import re
from typing import List

GIT_BRANCH_PATTERN = re.compile(r"^[A-Za-z0-9._/-]+$")

def run_git_checkout(branch_name: str) -> subprocess.CompletedProcess:
    """
    Run `git checkout <branch_name>` with a user-supplied branch name.

    Security: The branch name is passed as a separate list element (never
    interpolated into a shell string) and is strictly validated against a
    whitelist pattern to prevent argument injection or option injection.
    """
    # Validate at the trust boundary: reject anything that is not a
    # conservative set of characters allowed in a git branch name.
    if not GIT_BRANCH_PATTERN.fullmatch(branch_name):
        raise ValueError(f"Invalid branch name: {branch_name!r}")

    # Build the command as a list, never as a shell string.
    cmd: List[str] = ["git", "checkout", branch_name]

    # Run without shell=True; the argument list prevents injection.
    return subprocess.run(cmd, check=False, capture_output=True, text=True)
