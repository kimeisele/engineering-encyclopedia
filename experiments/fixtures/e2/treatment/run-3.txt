import subprocess
import re
from typing import List

def get_git_branch(branch_name: str) -> List[str]:
    """
    Safely run a git command with a user-supplied branch name.
    
    Args:
        branch_name: User-supplied branch name to validate and use
        
    Returns:
        List of command output lines
        
    Raises:
        ValueError: If branch name fails validation
        subprocess.CalledProcessError: If git command fails
    """
    # Validate branch name at the trust boundary (security.input-validation)
    # Git branch names: no spaces, no control chars, no ~^:?*[, no .., 
    # cannot start with -, cannot be empty or "." or ".."
    if not branch_name or len(branch_name) > 255:
        raise ValueError(f"Invalid branch name: {branch_name!r}")
    
    if branch_name in ('.', '..'):
        raise ValueError(f"Invalid branch name: {branch_name!r}")
    
    if branch_name.startswith('-'):
        raise ValueError(f"Branch name cannot start with '-': {branch_name!r}")
    
    if not re.match(r'^[a-zA-Z0-9/._-]+$', branch_name):
        raise ValueError(f"Branch name contains invalid characters: {branch_name!r}")
    
    if any(c in branch_name for c in '~^:?*[\\ '):
        raise ValueError(f"Branch name contains invalid characters: {branch_name!r}")
    
    if '..' in branch_name:
        raise ValueError(f"Branch name cannot contain '..': {branch_name!r}")
    
    # Command passed as list, never string interpolation, never shell=True
    # (python.processes.subprocess-safety)
    cmd = ['git', 'branch', '--list', branch_name]
    
    try:
        result = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            check=True,
            timeout=10
        )
        return result.stdout.strip().split('\n') if result.stdout.strip() else []
    except subprocess.TimeoutExpired:
        raise subprocess.CalledProcessError(-1, cmd, "Command timed out")
