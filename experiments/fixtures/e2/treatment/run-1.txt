import subprocess
from pathlib import Path


def run_git_branch_command(branch_name: str, repo_path: str) -> str:
    """
    Runs `git branch <branch_name>` in the given repository.

    The branch name is passed as a separate argument to the git executable,
    never interpolated into a shell string, so no shell injection is possible.
    """

    # Reject branch names that would be ambiguous or dangerous as a path.
    # This is defense-in-depth: the argument list already prevents shell
    # injection, but this blocks confusion attacks like "-u" or names with
    # control characters.
    if not branch_name or branch_name.startswith("-"):
        raise ValueError("Branch name must be non-empty and must not start with '-'")

    if any(ch in branch_name for ch in "\x00\r\n\t /\\"):
        raise ValueError("Branch name contains invalid characters")

    repo = Path(repo_path).resolve()
    if not repo.is_dir():
        raise ValueError(f"Not a directory: {repo_path}")

    # Run git as a list of arguments. No shell=True, no string interpolation.
    result = subprocess.run(
        ["git", "branch", branch_name],
        cwd=repo,
        capture_output=True,
        text=True,
        timeout=30,
    )

    if result.returncode != 0:
        raise RuntimeError(f"git branch failed: {result.stderr.strip()}")

    return result.stdout.strip()
