import subprocess
import re
import sys

def run_git_command(branch_name: str) -> str:
    """
    Run a git command (git rev-parse) with a user-supplied branch name.

    Args:
        branch_name: The branch name supplied by the user.

    Returns:
        The full commit hash of the branch.

    Raises:
        ValueError: If the branch name is invalid.
        subprocess.CalledProcessError: If the git command fails.
    """
    # Validate the branch name at the trust boundary (security.input-validation)
    # Git branch names cannot contain: space, ~, ^, :, ?, *, [, \, or start/end with '/'
    # or contain '..', or end with '.', or contain '@{'
    GIT_INVALID_CHARS = r'[\s~^:?*\[\\]|(\.\.)|(@\{)|(/$)|(^/)'
    if branch_name is None or not isinstance(branch_name, str) or len(branch_name) == 0:
        raise ValueError("Branch name must be a non-empty string")
    
    if re.search(GIT_INVALID_CHARS, branch_name) or branch_name.startswith('-'):
        raise ValueError(f"Invalid git branch name: {branch_name!r}")
    
    # Additional checks for reserved patterns
    if branch_name == '@' or branch_name.startswith('@{'):
        raise ValueError(f"Invalid git branch name: {branch_name!r}")
    
    if branch_name == '.' or branch_name == '..' or '.lock' in branch_name:
        raise ValueError(f"Invalid git branch name: {branch_name!r}")
    
    # Run the git command
    # python.processes.subprocess-safety: Use list arguments, never shell=True
    try:
        result = subprocess.run(
            ["git", "rev-parse", "--verify", f"refs/heads/{branch_name}"],
            capture_output=True,
            text=True,
            check=False  # we handle errors manually to add context
        )
    except subprocess.SubprocessError as e:
        # Wrap the error with context
        raise subprocess.CalledProcessError(
            returncode=getattr(e, 'returncode', 1),
            cmd=['git', 'rev-parse', '--verify', f'refs/heads/{branch_name}'],
            stderr=f"Failed to execute git command: {str(e)}"
        )
    
    if result.returncode != 0:
        stderr_msg = result.stderr.strip() if result.stderr else "unknown error"
        raise subprocess.CalledProcessError(
            returncode=result.returncode,
            cmd=['git', 'rev-parse', '--verify', f'refs/heads/{branch_name}'],
            stderr=stderr_msg
        )
    
    return result.stdout.strip()
