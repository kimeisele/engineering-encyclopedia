```python
import json
import os
import tempfile
from typing import Any, Callable, Dict

def update_json_config(
    filepath: str, 
    mutator: Callable[[Dict[str, Any]], Dict[str, Any]],
    validator: Callable[[Dict[str, Any]], bool] = lambda data: True
) -> None:
    """
    Atomically update a JSON config file.
    
    The function applies `mutator` to the current config content, validates
    the result, writes it to a temp file in the same directory, fsyncs,
    and renames over the target. This ensures readers see either old or new
    content, never a partial write.
    """
    # Read current content
    with open(filepath, 'r') as f:
        current = json.load(f)
    
    # Apply mutation and validate result
    new_data = mutator(current)
    if not validator(new_data):
        raise ValueError("Mutation produced invalid config")
    
    # Write to temp file in same directory for atomic rename
    dir_path = os.path.dirname(os.path.abspath(filepath))
    fd, temp_path = tempfile.mkstemp(dir=dir_path, prefix='.config.tmp.', suffix='.json')
    try:
        with os.fdopen(fd, 'w') as f:
            json.dump(new_data, f, indent=2)
            f.flush()
            os.fsync(f.fileno())  # Ensure data hits disk before rename
        
        # Atomic rename - same filesystem guarantees atomicity on POSIX
        os.replace(temp_path, filepath)
    except:
        # Clean up temp file on any failure
        try:
            os.unlink(temp_path)
        except OSError:
            pass
        raise
```
