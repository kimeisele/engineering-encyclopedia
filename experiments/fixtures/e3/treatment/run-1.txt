```python
import json
import os
import tempfile
from typing import Any, Dict


def update_json_config(file_path: str, updates: Dict[str, Any]) -> None:
    """
    Atomically update a JSON config file on disk.

    The function reads the existing file (if present), applies the updates,
    validates the result, and writes it back atomically by writing a temp
    file in the same directory, flushing it, and renaming it over the target.

    Args:
        file_path: Path to the JSON config file.
        updates: Dictionary of key/value pairs to merge into the config.

    Raises:
        ValueError: If the resulting config is not a valid JSON object.
        IOError: If the file cannot be read or written.
    """
    config = {}
    if os.path.exists(file_path):
        with open(file_path, "r", encoding="utf-8") as f:
            config = json.load(f)

    config.update(updates)

    # Validate the config is a JSON-serializable dict (security.input-validation)
    if not isinstance(config, dict):
        raise ValueError("Config must be a JSON object")

    # Serialize before writing to catch serialization errors early
    serialized = json.dumps(config, indent=2, ensure_ascii=False) + "\n"

    # Atomic write: temp file in same directory + flush + rename
    dir_path = os.path.dirname(os.path.abspath(file_path))
    fd, temp_path = tempfile.mkstemp(dir=dir_path, prefix=".config-", suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            f.write(serialized)
            f.flush()
            os.fsync(f.fileno())

        os.replace(temp_path, file_path)
    except Exception:
        # Clean up temp file if the write fails (python.files.atomic-replacement)
        if os.path.exists(temp_path):
            os.unlink(temp_path)
        raise
```
