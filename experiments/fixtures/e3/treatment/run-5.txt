```python
import json
import os
import tempfile
from typing import Any


def update_json_config(filepath: str, new_config: dict[str, Any]) -> None:
    """Atomically update a JSON config file.

    Args:
        filepath: Path to the JSON config file.
        new_config: The new configuration to write.

    Raises:
        ValueError: If the new_config is not a JSON-serializable dict.
        OSError: If the file cannot be written or renamed.
    """
    # Validate input structure before touching the filesystem.
    if not isinstance(new_config, dict):
        raise ValueError("Configuration must be a dictionary.")
    try:
        serialized = json.dumps(new_config, indent=2, sort_keys=True)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"Invalid JSON configuration: {exc}")

    directory = os.path.dirname(os.path.abspath(filepath))
    os.makedirs(directory, exist_ok=True)

    # Create the temp file in the *same* directory as the target so the
    # final rename is atomic (same filesystem).
    fd, tmp_path = tempfile.mkstemp(dir=directory, prefix=".config-", suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            f.write(serialized)
            f.flush()
            os.fsync(f.fileno())

        # Replace the target. Atomic on POSIX.
        os.replace(tmp_path, filepath)

        # Flush the directory so the rename itself is durable.
        dir_fd = os.open(directory, os.O_RDONLY)
        try:
            os.fsync(dir_fd)
        finally:
            os.close(dir_fd)

    except BaseException:
        # Clean up the temp file on any failure (including Ctrl-C).
        try:
            os.unlink(tmp_path)
        except OSError:
            pass
        raise
```
