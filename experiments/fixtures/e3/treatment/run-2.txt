```python
import json
import os
import tempfile


def update_json_config(filepath: str, new_content: dict) -> None:
    """
    Atomically update a JSON config file on disk.

    Writes the new content to a temporary file in the same directory as the
    target, flushes it, and renames it over the target. The rename is atomic
    on POSIX filesystems.

    Args:
        filepath: Absolute or relative path to the JSON config file.
        new_content: Dictionary to be written as the new JSON config.

    Raises:
        TypeError: If new_content is not a dict.
        OSError: If the directory doesn't exist or cannot be written to.
        json.JSONEncodeError: If new_content cannot be serialized to JSON.
    """
    # Security: input validation at the trust boundary. We explicitly accept
    # only a dict, rejecting all other shapes (e.g., lists, strings, ints).
    if not isinstance(new_content, dict):
        raise TypeError("new_content must be a dict")

    target_dir = os.path.dirname(os.path.abspath(filepath))

    # Create the temp file in the same directory as the target. This ensures
    # os.replace() is a rename within one filesystem, which is atomic.
    fd, temp_path = tempfile.mkstemp(
        dir=target_dir, prefix=".config_tmp_", suffix=".json"
    )

    try:
        # Serialize to bytes before writing to avoid encoding issues.
        # ensure_ascii=True (default) keeps output ASCII-safe for portability.
        data = json.dumps(new_content, indent=2).encode("utf-8")

        # Write and flush. os.fsync() pushes the file data to disk so that a
        # power loss after the rename still leaves a valid file.
        with os.fdopen(fd, "wb") as f:
            f.write(data)
            f.flush()
            os.fsync(f.fileno())

        # Atomic replacement: either the old file or the new file exists,
        # never a partially written version.
        os.replace(temp_path, filepath)
    except BaseException:
        # Clean up the temp file on any failure (write error, serialization
        # error, etc.). Without this, we'd leave orphaned temp files behind.
        try:
            os.unlink(temp_path)
        except FileNotFoundError:
            pass
        raise
```
