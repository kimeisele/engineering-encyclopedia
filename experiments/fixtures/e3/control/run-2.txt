Here's a comprehensive function that updates a JSON config file on disk with various options for flexibility:

```python
import json
import os
import tempfile
from typing import Any, Dict, Optional, Union
from pathlib import Path
import copy


def update_json_config(
    file_path: Union[str, Path],
    updates: Dict[str, Any],
    *,
    create_if_missing: bool = False,
    default_content: Optional[Dict[str, Any]] = None,
    backup: bool = False,
    backup_suffix: str = ".bak",
    atomic_write: bool = True,
    indent: Optional[int] = 2,
    sort_keys: bool = False,
) -> bool:
    """
    Update a JSON config file with the provided updates.

    Args:
        file_path: Path to the JSON file
        updates: Dictionary of updates to apply (nested dictionaries will be merged)
        create_if_missing: If True, create the file if it doesn't exist
        default_content: Default content to use if file doesn't exist (only used with create_if_missing)
        backup: If True, create a backup file before modifying
        backup_suffix: Suffix for backup file (e.g., ".bak")
        atomic_write: If True, write to temp file then rename (safer)
        indent: JSON indentation for output file
        sort_keys: If True, sort keys in JSON output

    Returns:
        True if successful, False otherwise

    Raises:
        FileNotFoundError: If file doesn't exist and create_if_missing is False
        json.JSONDecodeError: If file contains invalid JSON
        PermissionError: If no write permission
    """
    
    file_path = Path(file_path)
    
    # Check if file exists
    if not file_path.exists():
        if not create_if_missing:
            raise FileNotFoundError(f"Config file not found: {file_path}")
        
        # Create file with default content or empty dict
        config_data = default_content or {}
    else:
        # Read existing config
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                config_data = json.load(f)
        except json.JSONDecodeError as e:
            raise json.JSONDecodeError(f"Invalid JSON in {file_path}: {e}", e.doc, e.pos)
    
    # Ensure we're working with a dict
    if not isinstance(config_data, dict):
        raise ValueError(f"Config file must contain a JSON object, got {type(config_data).__name__}")
    
    # Apply updates (deep merge)
    updated_data = deep_merge(config_data, updates)
    
    # Create backup if requested
    if backup and file_path.exists():
        backup_path = file_path.with_suffix(file_path.suffix + backup_suffix)
        try:
            import shutil
            shutil.copy2(file_path, backup_path)
        except Exception as e:
            print(f"Warning: Could not create backup: {e}")
    
    # Write the updated config
    try:
        if atomic_write:
            return _atomic_write_json(file_path, updated_data, indent=indent, sort_keys=sort_keys)
        else:
            with open(file_path, 'w', encoding='utf-8') as f:
                json.dump(updated_data, f, indent=indent, sort_keys=sort_keys)
            return True
    except Exception as e:
        print(f"Error writing config file: {e}")
        return False


def deep_merge(base: Dict[str, Any], updates: Dict[str, Any]) -> Dict[str, Any]:
    """
    Deep merge two dictionaries, merging nested dictionaries recursively.
    Lists and other types are replaced.
    """
    result = copy.deepcopy(base)
    
    for key, value in updates.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = copy.deepcopy(value)
    
    return result


def _atomic_write_json(
    file_path: Path,
    data: Dict[str, Any],
    indent: Optional[int] = 2,
    sort_keys: bool = False
) -> bool:
    """
    Write JSON data atomically using a temporary file.
    """
    temp_path = None
    
    try:
        # Create temporary file in same directory
        fd, temp_path = tempfile.mkstemp(
            dir=file_path.parent,
            prefix=f".{file_path.name}.",
            suffix=".tmp"
        )
        
        with os.fdopen(fd, 'w', encoding='utf-8') as f:
            json.dump(data, f, indent=indent, sort_keys=sort_keys)
        
        # Replace original file
        os.replace(temp_path, file_path)
        return True
        
    except Exception:
        if temp_path and os.path.exists(temp_path):
            os.remove(temp_path)
        raise
    finally:
        # Clean up temp file if it still exists
        if temp_path and os.path.exists(temp_path):
            try:
                os.remove(temp_path)
            except:
                pass

if __name__ == "__main__":

    config_file = "config.json"

    # Read and display result

    # Example 2: Create new file

        create_if_missing=True,
        default_content={"version": "1.0.0"}
    )
    print(f"\nCreate successful: {success}")
    
    # Example 3: With backup and atomic write

        backup=True,
        atomic_write=True
    )
    print(f"\nUpdate with backup successful: {success}")
    
    # Clean up
    for f in ["config.json", "new_config.json", "config.json.bak"]:
        if os.path.exists(f):
            os.remove(f)
```

Here's also a more concise version if you need something simpler:

```python
import json
from pathlib import Path

def update_json_simple(file_path, updates):
    """Simple version for basic updates"""
    path = Path(file_path)
    
    # Read existing or create empty
    data = {}
    if path.exists():
        with open(path, 'r') as f:
            data = json.load(f)
    
    # Update (shallow merge - for simple configs)
    data.update(updates)
    
    # Write back
    with open(path, 'w') as f:
        json.dump(data, f, indent=2)
    
    return True

# Example usage
update_json_simple("config.json", {"new_setting": True, "app_port": 8080})
```

**Key features of the comprehensive version:**

1. **Deep merge** - Nested dictionaries are merged recursively
2. **Atomic writes** - Prevents corruption from interrupted writes
3. **Backup support** - Creates a backup before modification
4. **Error handling** - Proper exceptions for missing files or invalid JSON
5. **Flexible creation** - Can create new files if needed
6. **Configurable formatting** - Control indentation and key sorting

**Usage examples:**
