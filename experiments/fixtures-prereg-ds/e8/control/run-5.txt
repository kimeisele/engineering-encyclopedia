Here's a comprehensive solution that ensures cached values are never stale by implementing automatic invalidation when the source data changes:

```python
import time
import threading
import hashlib
from typing import Any, Callable, Optional, Dict, Tuple
from collections import OrderedDict

class SafeCache:
    """
    A thread-safe cache with automatic invalidation when source data changes.
    Uses a combination of:
    - TTL (Time To Live) for time-based expiration
    - Data version tracking for change detection
    - LRU eviction for memory management
    """
    
    def __init__(self, max_size: int = 100, default_ttl: float = 300, 
                 check_interval: float = 60):
        """
        Initialize the cache.
        
        Args:
            max_size: Maximum number of items to store (LRU eviction)
            default_ttl: Default time-to-live in seconds
            check_interval: How often to check for data changes
        """
        self.max_size = max_size
        self.default_ttl = default_ttl
        self.check_interval = check_interval
        
        # Main cache storage
        self._cache = OrderedDict()  # key -> (value, timestamp, version, last_checked)
        self._lock = threading.RLock()
        
        # Track source data versions
        self._data_sources = {}  # source_key -> (current_version, checker_func)
        
        # Background thread for periodic checks
        self._monitor_thread = None
        self._stop_monitor = False
        
    def start_monitor(self):
        """Start background monitoring for data changes."""
        if self._monitor_thread is None or not self._monitor_thread.is_alive():
            self._stop_monitor = False
            self._monitor_thread = threading.Thread(target=self._monitor_loop, daemon=True)
            self._monitor_thread.start()
    
    def stop_monitor(self):
        """Stop background monitoring."""
        self._stop_monitor = True
        if self._monitor_thread:
            self._monitor_thread.join(timeout=5)
    
    def _monitor_loop(self):
        """Background loop to check for data changes."""
        while not self._stop_monitor:
            time.sleep(self.check_interval)
            try:
                self._check_all_sources()
            except Exception as e:
                print(f"Error checking data sources: {e}")
    
    def register_source(self, source_key: str, version_checker: Callable[[], Any]):
        """
        Register a data source with a function that returns its current version.
        
        Args:
            source_key: Unique identifier for the source
            version_checker: Function that returns the current version/state of the source
        """
        with self._lock:
            self._data_sources[source_key] = (version_checker(), version_checker)
    
    def _get_source_version(self, source_key: str) -> Optional[Any]:
        """Get current version from a data source."""
        if source_key in self._data_sources:
            checker = self._data_sources[source_key][1]
            try:
                return checker()
            except Exception as e:
                print(f"Error getting version for {source_key}: {e}")
                return None
        return None
    
    def _check_source_changed(self, source_key: str, cached_version: Any) -> bool:
        """Check if a source has changed since it was cached."""
        current_version = self._get_source_version(source_key)
        if current_version is None:
            return False  # Can't check, assume unchanged
        return current_version != cached_version
    
    def _check_all_sources(self):
        """Check all cached items against their data sources."""
        with self._lock:
            keys_to_remove = []
            
            for key, (value, timestamp, version, last_checked) in self._cache.items():
                source_key = key.split(':')[0]  # Assuming key format "source_key:item_key"
                
                if source_key in self._data_sources:
                    if self._check_source_changed(source_key, version):
                        keys_to_remove.append(key)
            
            # Remove stale items
            for key in keys_to_remove:
                del self._cache[key]
    
    def get(self, key: str, compute_func: Optional[Callable[[], Any]] = None, 
            ttl: Optional[float] = None) -> Any:
        """
        Get a cached value. If not in cache or stale, compute and cache it.
        
        Args:
            key: Cache key
            compute_func: Function to compute the value if not in cache
            ttl: Optional TTL override (default uses default_ttl)
        
        Returns:
            The cached or computed value
        """
        with self._lock:
            current_time = time.time()
            
            # Check if key exists in cache
            if key in self._cache:
                value, timestamp, version, last_checked = self._cache[key]
                
                # Check TTL
                if current_time - timestamp <= (ttl or self.default_ttl):
                    # Check if source data has changed
                    source_key = key.split(':')[0]
                    if source_key in self._data_sources:
                        if self._check_source_changed(source_key, version):
                            # Source changed, invalidate cache
                            del self._cache[key]
                            print(f"Cache invalidated for {key}: source data changed")
                        else:
                            # Cache is valid, return it
                            self._cache.move_to_end(key)  # LRU update
                            return value
                    else:
                        # No source registered, assume not changed
                        self._cache.move_to_end(key)
                        return value
                else:
                    # TTL expired
                    del self._cache[key]
            
            # Cache miss or stale, compute new value
            if compute_func:
                value = compute_func()
                
                # Get current source version
                source_key = key.split(':')[0]
                version = self._get_source_version(source_key) if source_key in self._data_sources else None
                
                # Store in cache
                self._cache[key] = (value, current_time, version, current_time)
                self._cache.move_to_end(key)
                
                # Evict least recently used if cache is too large
                self._evict_if_needed()
                
                return value
            
            return None
    
    def set(self, key: str, value: Any, ttl: Optional[float] = None):
        """Manually set a value in the cache."""
        with self._lock:
            current_time = time.time()
            source_key = key.split(':')[0]
            version = self._get_source_version(source_key) if source_key in self._data_sources else None
            
            self._cache[key] = (value, current_time, version, current_time)
            self._cache.move_to_end(key)
            
            self._evict_if_needed()
    
    def invalidate(self, key: str):
        """Remove a specific key from cache."""
        with self._lock:
            if key in self._cache:
                del self._cache[key]
    
    def clear(self):
        """Clear entire cache."""
        with self._lock:
            self._cache.clear()
    
    def _evict_if_needed(self):
        """Evict least recently used entries if cache exceeds max size."""
        while len(self._cache) > self.max_size:
            self._cache.popitem(last=False)  # Remove LRU
    
    def get_stats(self) -> Dict[str, int]:
        """Get cache statistics."""
        with self._lock:
            return {
                'size': len(self._cache),
                'max_size': self.max_size,
                'sources': len(self._data_sources)
            }


# Example usage with a simulated data source
class DataSource:
    def __init__(self):
        self.data = {"users": 100, "items": 50}
        self.version = 1
    
    def update_data(self, key, value):
        """Simulate data update"""
        self.data[key] = value
        self.version += 1  # Increment version on any change
    
    def get_version(self):
        """Return current version as change indicator"""
        return self.version


def example_usage():
    # Create data source
    source = DataSource()
    
    # Create cache and register source
    cache = SafeCache(max_size=10, default_ttl=30)
    cache.register_source("users", source.get_version)
    cache.start_monitor()  # Start background monitoring
    
    # Function to compute expensive data
    def compute_users():
        print("Computing users value...")
        return {"total": source.data["users"], "timestamp": time.time()}
    
    # Test cache
    print("=== First access (cache miss) ===")
    result1 = cache.get("users:stats", compute_users)
    print(f"Result 1: {result1}")
    
    print("\n=== Second access (cache hit) ===")
    result2 = cache.get("users:stats", compute_users)
    print(f"Result 2: {result2}")
    print(f"Same object: {result1 is result2}")  # Should be True
    
    print("\n=== Update source data ===")
    source.update_data("users", 150)
    
    print("\n=== Access after update ===")
    result3 = cache.get("users:stats", compute_users)
    print(f"Result 3: {result3}")
    print(f"New value: {result3['total']}")  # Should be 150
    
    # Clean up
    cache.stop_monitor()
    cache.clear()


if __name__ == "__main__":
    example_usage()
```

This solution provides:

## Key Features:

1. **Automatic Invalidation**: Detects when source data changes and automatically invalidates cached values
2. **Version Tracking**: Uses version numbers/checkers to detect source changes
3. **TTL Support**: Time-based expiry as additional safety
4. **Thread-Safe**: Uses reentrant locks for concurrent access
5. **Background Monitoring**: Optionally checks for changes periodically
6. **LRU Eviction**: Prevents memory issues with large caches

## Usage Patterns:

```python
# Register data sources with change detection
cache.register_source("users", data_source.get_version)

# Get cached values with auto-invalidation
value = cache.get("users:profile", compute_expensive_function)

# Manual invalidation
cache.invalidate("users:profile")

# Clear all cache
cache.clear()
```

## Key Design Decisions:

1. **Version-based invalidation**: Instead of comparing full data (which could be expensive), we use version numbers or change detectors
2. **Key structure**: Assumes keys follow "source_key:item_key" pattern for source association
3. **Dual safety**: Both TTL and version checking provide defense against stale data
4. **Non-blocking reads**: Background monitoring doesn't lock during cache reads
5. **Failure handling**: If version checker fails, cache isn't invalidated (fail-safe)

This approach ensures that:
- Cache values are never stale when source data changes
- The system remains performant (no full data comparison)
- Memory is managed properly with LRU
- Thread safety is maintained for concurrent access
