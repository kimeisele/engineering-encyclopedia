from typing import Any, Callable, Dict, Optional
import time
import threading
from functools import wraps
import hashlib
import pickle

class CacheWithVersioning:
    """
    A cache system that automatically detects when source data changes
    and invalidates stale entries.
    """
    
    def __init__(self, data_loader: Callable, ttl: Optional[float] = None):
        """
        Initialize cache with a data loader function
        
        Args:
            data_loader: Function that fetches the source data
            ttl: Time-to-live in seconds (if None, cache never expires by time)
        """
        self.data_loader = data_loader
        self.ttl = ttl
        self._cache: Dict[str, dict] = {}
        self._lock = threading.Lock()
        self._version_checkers: Dict[str, Callable] = {}
        
    def _compute_data_version(self, data: Any) -> str:
        """
        Compute a version/hash of the data to detect changes
        """
        return hashlib.md5(pickle.dumps(data)).hexdigest()
    
    def get(self, key: str, *args, **kwargs) -> Any:
        """
        Get cached value for key, refreshing if stale
        
        Args:
            key: The cache key
            *args, **kwargs: Arguments to pass to the data loader
        
        Returns:
            The current value for the key
        """
        with self._lock:
            # Check if cache entry exists
            if key in self._cache:
                cached_entry = self._cache[key]
                
                # Check TTL expiry
                if self.ttl and (time.time() - cached_entry['timestamp'] > self.ttl):
                    print(f"Cache TTL expired for key: {key}")
                    return self._refresh(key, *args, **kwargs)
                
                # Check if data has changed by fetching fresh source
                fresh_data = self.data_loader(key, *args, **kwargs)
                current_version = self._compute_data_version(fresh_data)
                
                if current_version != cached_entry['version']:
                    print(f"Data changed for key: {key}, refreshing cache")
                    # Update cache with fresh data
                    self._cache[key] = {
                        'value': fresh_data,
                        'version': current_version,
                        'timestamp': time.time()
                    }
                    return fresh_data
                else:
                    # Data hasn't changed, return cached value
                    return cached_entry['value']
            else:
                # No cache entry, load and cache
                return self._refresh(key, *args, **kwargs)
    
    def _refresh(self, key: str, *args, **kwargs) -> Any:
        """
        Refresh cache entry for key
        """
        data = self.data_loader(key, *args, **kwargs)
        version = self._compute_data_version(data)
        
        self._cache[key] = {
            'value': data,
            'version': version,
            'timestamp': time.time()
        }
        
        return data
    
    def invalidate(self, key: str):
        """
        Manually invalidate a cache entry
        """
        with self._lock:
            if key in self._cache:
                del self._cache[key]
                print(f"Cached entry invalidated for key: {key}")

def cached_data_loader(timeout: float = 5.0):
    """
    Decorator version that creates a cached data loader
    
    Args:
        timeout: Maximum time to wait for data loading in seconds
    
    Returns:
        A decorator that adds caching with automatic invalidation
    """
    def decorator(func: Callable) -> Callable:
        cache_dict: Dict[str, dict] = {}
        
        @wraps(func)
        def wrapper(*args, **kwargs):
            # Create a cache key from function name and arguments
            cache_key = f"{func.__name__}:{args}:{kwargs}"
            
            # Check if we have cached data
            if cache_key in cache_dict:
                cached_entry = cache_dict[cache_key]
                
                # Always re-fetch to check if source changed
                fresh_data = func(*args, **kwargs)
                
                # Compare versions
                current_version = hashlib.md5(
                    pickle.dumps(fresh_data)
                ).hexdigest()
                
                if current_version != cached_entry['version']:
                    # Data changed, update cache
                    print(f"Data changed for {cache_key}")
                    cache_dict[cache_key] = {
                        'value': fresh_data,
                        'version': current_version,
                        'timestamp': time.time()
                    }
                    return fresh_data
                else:
                    # Return cached version (data unchanged)
                    return cached_entry['value']
            else:
                # Load and cache initial data
                data = func(*args, **kwargs)
                version = hashlib.md5(pickle.dumps(data)).hexdigest()
                cache_dict[cache_key] = {
                    'value': data,
                    'version': version,
                    'timestamp': time.time()
                }
                return data
        
        # Add cache management methods
        wrapper.clear_cache = lambda: cache_dict.clear()
        wrapper.get_cache_size = lambda: len(cache_dict)
        
        return wrapper
    
    return decorator

class VersionedCache:
    """
    Cache that uses version checks to detect stale data
    """
    
    def __init__(self, version_provider: Callable, data_loader: Callable):
        """
        Args:
            version_provider: Function that returns a version/hash of the source
            data_loader: Function that loads data for a key
        """
        self.version_provider = version_provider
        self.data_loader = data_loader
        self._cache = {}
        self._lock = threading.Lock()
    
    def get(self, key: str) -> Any:
        """
        Get data for key, checking version first
        """
        with self._lock:
            # Get current version of source data
            current_version = self.version_provider(key)
            
            # Check if cache exists and version matches
            if key in self._cache:
                cached_version = self._cache[key]['version']
                
                if cached_version == current_version:
                    # Version matches, return cached value
                    return self._cache[key]['value']
                else:
                    print(f"Version changed for {key}: {cached_version} -> {current_version}")
                    # Need to refresh
                    data = self.data_loader(key)
                    self._cache[key] = {
                        'value': data,
                        'version': current_version
                    }
                    return data
            else:
                # First request for this key
                data = self.data_loader(key)
                self._cache[key] = {
                    'value': data,
                    'version': current_version
                }
                return data

This implementation provides several key features:

1. **Automatic Invalidation**: The cache always checks if the source data has changed by comparing versions
2. **TTL Support**: Optional time-based expiry
3. **Thread Safety**: Uses locks to prevent race conditions
4. **Version Comparison**: Uses hash values to detect changes
5. **Both Class and Decorator**: Provides flexible usage patterns

The key insight is that the cache **always** checks the source data version before serving cached values. If the version has changed, it automatically refreshes the cache. This ensures you never serve stale data.
