```python
from typing import Callable, Generic, TypeVar
from dataclasses import dataclass
import threading

K = TypeVar("K")
V = TypeVar("V")


@dataclass(frozen=True)
class CachedValue(Generic[V]):
    value: V
    source_version: object


class VersionedCache(Generic[K, V]):
    """
    A cache that never serves stale values.

    Correctness strategy:
    - The cache key is (user_key, source_version).
    - The source_version is obtained atomically from the source before
      reading the value, so the cache entry is always tied to that version.
    - On each read, we first ask the source for its current version.
      If the cached entry's version matches, it is fresh. If not, we
      re-fetch and cache under the new version.
    - Source version check and value fetch are done while holding a per-key
      lock, so concurrent readers cannot observe a torn read/write.
    """

    def __init__(
        self,
        get_version: Callable[[K], object],
        get_value: Callable[[K], V],
    ):
        self._get_version = get_version
        self._get_value = get_value
        self._store: dict[tuple[K, object], CachedValue[V]] = {}
        self._locks: dict[K, threading.Lock] = {}
        self._locks_guard = threading.Lock()

    def _lock_for(self, key: K) -> threading.Lock:
        with self._locks_guard:
            if key not in self._locks:
                self._locks[key] = threading.Lock()
            return self._locks[key]

    def get(self, key: K) -> V:
        """Return the current value for key, never a stale one."""
        lock = self._lock_for(key)
        with lock:
            current_version = self._get_version(key)
            cache_key = (key, current_version)

            cached = self._store.get(cache_key)
            if cached is not None:
                return cached.value

            # Cache miss: fetch value under the observed version.
            value = self._get_value(key)

            # We must re-check the version because the source may have
            # changed during our fetch. If it changed, we cannot cache
            # this value — return it directly, but don't store stale.
            if self._get_version(key) == current_version:
                self._store[cache_key] = CachedValue(value, current_version)

            return value


# NOTE: The caller is responsible for ensuring that source changes are
# visible through get_version (e.g., the version is derived from the
# underlying data row's version/hash). This design does not use TTL;
# TTL alone is insufficient (see performance.cache-invalidation).


# --- Example usage ---
if __name__ == "__main__":
    import time

    class DataSource:
        def __init__(self):
            self.data = {"config": "v1"}
            self.version_counter = 0

        def get_version(self, key):
            # Version increments on every change — observable atomicity.
            return self.version_counter

        def get_value(self, key):
            # Simulate a read that sees the current data.
            return self.data[key]

        def update(self, key, new_value):
            self.data[key] = new_value
            self.version_counter += 1

    ds = DataSource()
    cache = VersionedCache(ds.get_version, ds.get_value)

    print(cache.get("config"))    # v1
    ds.update("config", "v2")     # source changes
    print(cache.get("config"))    # v2 (not stale)
```
