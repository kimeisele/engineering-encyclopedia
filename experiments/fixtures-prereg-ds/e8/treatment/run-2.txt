```python
from typing import Any, Callable, Hashable
import threading

class VersionedCache:
    """
    A cache that never serves stale data.

    The cache key is a tuple of (user_key, source_version).  When the
    source changes, the caller must provide a new source_version, which
    changes the effective cache key.  This guarantees that a reader
    cannot observe an old value after the source has changed.

    Invalidation is explicit: the caller must call `invalidate` or
    provide a new version to `get`.  There is no TTL-based guessing.
    """

    def __init__(self) -> None:
        self._store: dict[tuple[Hashable, Hashable], Any] = {}
        self._lock = threading.Lock()

    def get(self, key: Hashable, source_version: Hashable,
            load_fn: Callable[[], Any]) -> Any:
        """
        Return the cached value for `key` at `source_version`.

        If no entry exists for the (key, source_version) pair, call
        load_fn() to obtain the value, store it, and return it.
        """
        cache_key = (key, source_version)
        with self._lock:
            if cache_key in self._store:
                return self._store[cache_key]
            value = load_fn()
            self._store[cache_key] = value
            return value

    def invalidate(self, key: Hashable) -> None:
        """Remove all cached entries for `key` (any version)."""
        with self._lock:
            keys_to_delete = [k for k in self._store if k[0] == key]
            for k in keys_to_delete:
                del self._store[k]
```
