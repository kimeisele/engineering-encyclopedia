from typing import Any, Callable, Hashable
import threading
from collections import OrderedDict


class VersionedCache:
    """
    A cache that stores both the value and the source version it was fetched
    from. The source version is captured atomically with the value, so a
    reader can never observe a value from an older source version after the
    source has changed.
    
    Thread-safety: a single lock guards the internal dict, and the
    read-check-act sequence is atomic. This prevents two readers from both
    seeing a missing entry and both fetching (thundering herd) and prevents
    a reader from getting a value whose version does not match the current
    source version.
    """

    def __init__(self, loader: Callable[[Hashable], tuple[Any, Any]]):
        """
        :param loader: A function that takes a key and returns
            (value, source_version).  The source_version must be a value
            that changes whenever the underlying source data changes.
            For example, a monotonically increasing revision number, or a
            checksum of the source data.
        """
        self._loader = loader
        self._cache: OrderedDict[Hashable, tuple[Any, Any]] = OrderedDict()
        self._lock = threading.Lock()

    def get(self, key: Hashable) -> Any:
        """
        Returns the cached value for `key`, or fetches it if not present.
        Never serves a stale value: if the source data has changed since the
        value was cached, this fetches a fresh value.
        """
        with self._lock:
            # Atomically fetch the current source version and the value.
            # This is the key invariant: the version we compare against the
            # cache is the same version that produced the value we store.
            value, source_version = self._loader(key)

            cached = self._cache.get(key)
            if cached is not None:
                cached_value, cached_version = cached
                if cached_version == source_version:
                    # Move to end to indicate recent use (LRU-style, though
                    # eviction is not required for correctness).
                    self._cache.move_to_end(key)
                    return cached_value

            # Cache miss or stale entry: store the freshly fetched value.
            self._cache[key] = (value, source_version)
            self._cache.move_to_end(key)
            return value


# Example usage:
#
# source = {"revision": 0, "data": {"user1": "alice"}}
#
# def load_user(user_id):
#     # Capture the version atomically with the data read. In a real system
#     # this might be a DB row with a version column, or a file with a
#     # checksum. The key point: the version must change whenever the data
#     # changes, and the read must see both consistently.
#     return source["data"][user_id], source["revision"]
#
# cache = VersionedCache(load_user)
# print(cache.get("user1"))  # "alice"
# source["data"]["user1"] = "bob"
# source["revision"] += 1
# print(cache.get("user1"))  # "bob"  <- never stale
