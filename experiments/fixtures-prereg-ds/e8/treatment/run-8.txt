```python
from typing import Callable, Generic, TypeVar
import threading
import time

K = TypeVar("K")
V = TypeVar("V")


def cached_value(
    key: K,
    get_source: Callable[[K], V],
    source_version: Callable[[K], int],
    max_staleness_seconds: float = 0.0,
    cache_size: int = 128,
) -> Callable[[], V]:
    """
    Returns a function that returns the cached value for the given key.
    The cache is always consistent with the source: before every read, the
    value is validated against the source version. If the source has changed,
    the value is refreshed.

    This makes staleness impossible: the cache key includes the source version,
    and the version check is done atomically with the read.

    :param key: The key to look up.
    :param get_source: Function that retrieves the source value for a key.
    :param source_version: Function that returns a version identifier for the
                           source. This must change whenever the source data
                           changes (e.g., a row version, a file mtime, a
                           sequence number).
    :param max_staleness_seconds: Maximum time to serve a cached value without
                                  checking the source. 0.0 means always check
                                  before serving (strict consistency).
    :param cache_size: Maximum number of keys to hold in the cache.
    """
    cache: dict[tuple[K, int], tuple[V, float]] = {}
    lock = threading.Lock()

    def get_value() -> V:
        while True:
            version = source_version(key)

            # Fast path: we have a cached value for this exact version and it's
            # still within the staleness window.
            with lock:
                entry = cache.get((key, version))
                if entry is not None:
                    value, cached_at = entry
                    elapsed = time.monotonic() - cached_at
                    if elapsed <= max_staleness_seconds or max_staleness_seconds == 0.0:
                        return value

            # Slow path: fetch fresh value from source and cache it.
            fresh_value = get_source(key)
            # Re-check the version after fetching — the source could have changed
            # while we were fetching.
            new_version = source_version(key)
            if new_version != version:
                # Source changed during fetch; we might have a stale value.
                # Loop again to get the newest version. This avoids serving
                # a value that doesn't match the version we'll cache.
                continue

            with lock:
                # Invalidate entry if cache is too big (simple approach).
                if len(cache) >= cache_size and (key, new_version) not in cache:
                    # evict arbitrary entry (not perfect, but fine for demo)
                    cache.pop(next(iter(cache)))

                cache[(key, new_version)] = (fresh_value, time.monotonic())
                return fresh_value

    return get_value
```
