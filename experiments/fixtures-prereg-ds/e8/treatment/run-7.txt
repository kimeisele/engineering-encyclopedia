```python
from typing import Callable, Generic, Hashable, TypeVar
import threading
import time

KT = TypeVar('KT', bound=Hashable)
VT = TypeVar('VT')


class VersionedCache(Generic[KT, VT]):
    """
    A cache that never serves stale data by tying cache entries to a version
    of the source data. When the source data changes, the version changes,
    and the cache is automatically invalidated.

    The key insight is that the cache key is composed of (user_key, source_version).
    If the source's version changes, the lookup will miss, and the value
    will be refetched from the source.
    """

    def __init__(self, source_version_func: Callable[[], int], fetch_func: Callable[[KT], VT],
                 max_age_seconds: float = 0.0):
        """
        :param source_version_func: Returns the current version of the source data.
                                     Must be monotonic and change whenever source data changes.
        :param fetch_func: Fetches a fresh value from the source for a given key.
        :param max_age_seconds: If > 0, entries older than this are skipped as stale
                                (secondary guard, but the primary mechanism is the version).
        """
        self._version_func = source_version_func
        self._fetch = fetch_func
        self._max_age = max_age_seconds
        self._cache: dict[tuple[KT, int], tuple[VT, float]] = {}
        self._lock = threading.Lock()

    def get(self, key: KT) -> VT:
        """
        Returns the cached value for the key, or fetches a fresh one.
        The value is guaranteed not to be stale relative to the source version
        that was current when this method was called.
        """
        current_version = self._version_func()

        # Fast path: check if we have a valid entry for this version
        cache_key = (key, current_version)
        with self._lock:
            entry = self._cache.get(cache_key)
            if entry is not None:
                value, fetched_at = entry
                if self._max_age <= 0 or (time.monotonic() - fetched_at) < self._max_age:
                    return value

        # Slow path: need to fetch from source (this also updates the cache)
        # This is done outside the lock to avoid holding the lock during slow I/O.
        # However, we re-check under the lock afterwards to avoid a thundering herd.
        value = self._fetch(key)

        # Store under the current version only. Old versions will be evicted naturally
        # when they are no longer referenced, or we can evict them here.
        with self._lock:
            self._cache[cache_key] = (value, time.monotonic())
            # Evict old versions of this key if they exist (prevent unbounded growth)
            self._cache = {k: v for k, v in self._cache.items() if k[0] != key or k[1] == current_version}
        return value


# Example usage
if __name__ == "__main__":
    # Simulate a source database with a version counter
    source_data = {"a": 1}
    source_version = [0]

    def get_source_version() -> int:
        return source_version[0]

    def fetch_from_source(key: str) -> int:
        # In real code, this would be a database call
        return source_data[key]

    cache = VersionedCache(get_source_version, fetch_from_source)

    # Initial fetch
    print(cache.get("a"))  # 1

    # Change the source data and bump the version
    source_data["a"] = 2
    source_version[0] += 1

    # Cache must not serve stale data - it should return 2
    print(cache.get("a"))  # 2
```
