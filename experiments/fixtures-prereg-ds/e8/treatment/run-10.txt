```python
import threading
import time
from typing import Any, Callable, Dict, Generic, Hashable, Optional, TypeVar

K = TypeVar("K", bound=Hashable)
V = TypeVar("V")


class CachedValue(Generic[K, V]):
    """
    A cache that invalidates entries based on source version changes,
    not just TTL. Never serves stale values after the source changes.
    """

    def __init__(self, source_getter: Callable[[K], V], version_getter: Callable[[K], Any]):
        self._source_getter = source_getter
        self._version_getter = version_getter
        self._cache: Dict[K, tuple[Any, V]] = {}
        self._lock = threading.RLock()  # RLock allows re-entrant use within callbacks

    def get(self, key: K) -> V:
        """Return the cached value for key, re-fetching if the source version changed."""
        with self._lock:
            current_version = self._version_getter(key)
            cached = self._cache.get(key)

            if cached is not None:
                cached_version, cached_value = cached
                if cached_version == current_version:
                    return cached_value

            # Source has changed (or first access) — fetch fresh value
            value = self._source_getter(key)
            self._cache[key] = (current_version, value)
            return value

    def invalidate(self, key: K) -> None:
        """Explicitly invalidate an entry (call this after the source is updated)."""
        with self._lock:
            self._cache.pop(key, None)


# --- Example usage ---
if __name__ == "__main__":
    # Example source: a dict that can be updated
    data = {"user1": "Alice"}
    data_version = {"user1": 1}

    def get_source(key):
        return data[key]

    def get_version(key):
        return data_version[key]

    cache = CachedValue(get_source, get_version)

    print(cache.get("user1"))  # "Alice" (fetched)

    # Simulate source data change
    data["user1"] = "Bob"
    data_version["user1"] = 2

    # Cache must not serve stale value
    print(cache.get("user1"))  # "Bob" (re-fetched because version changed)
```
