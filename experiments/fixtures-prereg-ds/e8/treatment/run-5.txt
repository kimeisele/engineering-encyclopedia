from threading import Lock
import time

class CachedValue:
    """
    A cache that never serves stale data by tying the cache entry to the source
    version. The key is a composite of (user_key, source_version), so when the
    source data changes, the old entry is simply no longer looked up.
    """

    def __init__(self, source_loader):
        """
        source_loader: callable taking (key) -> (value, version)
        where version is a monotonic identifier that changes each time the
        source data for that key changes.
        """
        self._source_loader = source_loader
        self._cache = {}  # (user_key, version) -> value
        self._lock = Lock()  # protects _cache and the check-then-act sequence

    def get(self, user_key, max_age_seconds=None):
        """
        Returns the current value for user_key.
        Never returns a value older than the source's current version.

        If max_age_seconds is given, entries older than that are re-validated
        even if the source version hasn't changed — this is a secondary
        freshness control, not the correctness mechanism.

        Handles thundering herd: only one thread performs the source load;
        others wait on the lock and then get the fresh value.
        """
        # Load current source version outside the lock — this is the
        # "check" step. We must see the latest version.
        value, version = self._source_loader(user_key)

        composite_key = (user_key, version)

        # Fast path: the exact (key, version) we want is already cached.
        with self._lock:
            if composite_key in self._cache:
                cached = self._cache[composite_key]
                if max_age_seconds is None or (
                    time.time() - cached[1] < max_age_seconds
                ):
                    return cached[0]

        # Check-then-act must be atomic: between our version check and the
        # cache lookup, another thread could have loaded a newer version.
        # The lock ensures we don't race into an interleaving where we
        # serve stale data.
        with self._lock:
            # Re-check the source version inside the lock — the source may
            # have changed since we read it above. This closes the race.
            value_inside, version_inside = self._source_loader(user_key)
            composite_key_inside = (user_key, version_inside)

            if composite_key_inside in self._cache:
                cached = self._cache[composite_key_inside]
                if max_age_seconds is None or (
                    time.time() - cached[1] < max_age_seconds
                ):
                    return cached[0]

            # Not cached (or too old) — load and store. We hold the lock,
            # so no thundering herd: others will wait and find our entry.
            value, version = value_inside, version_inside
            self._cache[composite_key_inside] = (value, time.time())
            # Optional: prune old versions for this user_key to bound memory
            self._prune(user_key)
            return value

    def _prune(self, user_key):
        """Remove all entries for user_key except the current version."""
        current_version = None
        for (uk, ver) in list(self._cache.keys()):
            if uk == user_key and current_version is None:
                current_version = ver
            elif uk == user_key and ver != current_version:
                del self._cache[(uk, ver)]

    def invalidate(self, user_key):
        """
        Explicit invalidation: drop all cached entries for a key.
        Useful when the source has an out-of-band update that doesn't
        change version (or to free memory).
        """
        with self._lock:
            keys_to_delete = [k for k in self._cache if k[0] == user_key]
            for k in keys_to_delete:
                del self._cache[k]


# Example usage
if __name__ == "__main__":
    # Simulated source: value changes every few seconds with version increment
    class FakeSource:
        def __init__(self):
            self._data = {"a": ("v1", 1)}

        def load(self, key):
            # In real code this hits the database/filesystem
            return self._data[key]

        def update(self, key, new_value):
            _, old_ver = self._data[key]
            self._data[key] = (new_value, old_ver + 1)

    src = FakeSource()
    cache = CachedValue(src.load)

    print(cache.get("a"))      # v1
    src.update("a", "v2")
    print(cache.get("a"))      # v2 — never v1, because version changed

    # Simulate concurrent readers
    import threading
    results = []
    def worker():
        results.append(cache.get("a"))
    threads = [threading.Thread(target=worker) for _ in range(5)]
    for t in threads: t.start()
    for t in threads: t.join()
    print(results)  # all v2
