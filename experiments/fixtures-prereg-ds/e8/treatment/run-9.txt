from typing import Any, Callable, Generic, TypeVar
import threading
import time

T = TypeVar('T')

class VersionedCache(Generic[T]):
    """
    A cache that never serves stale data by tying each cached value to a
    source version. On each read, it checks whether the source has changed
    and invalidates accordingly.
    """
    def __init__(self, source: Callable[[], tuple[Any, T]], ttl: float = 60.0):
        self._source = source
        self._ttl = ttl
        self._lock = threading.Lock()
        self._cache: dict[Any, tuple[Any, T, float]] = {}  # key -> (version, value, expiry)
        self._last_version: dict[Any, Any] = {}  # key -> last seen source version

    def get(self, key: str) -> T:
        """
        Returns the cached value for key, or fetches from source if missing.
        Never returns stale data: each call re-validates the source version.
        """
        with self._lock:
            now = time.time()
            cached = self._cache.get(key)
            
            # If we have a cache entry, check if the source has changed
            if cached is not None:
                version, value, expiry = cached
                
                # Short-circuit: if TTL has not expired and version is still current,
                # we still need to check source to avoid staleness
                # Actually fetch current version to ensure correctness
                current_version, _ = self._get_source_version(key)
                
                if current_version == version and now < expiry:
                    return value
                elif current_version != version:
                    # Source changed, invalidate
                    del self._cache[key]
                    del self._last_version[key]
            
            # Cache miss or stale - fetch from source
            source_version, value = self._fetch_from_source(key)
            self._cache[key] = (source_version, value, now + self._ttl)
            self._last_version[key] = source_version
            return value

    def _get_source_version(self, key: str) -> tuple[Any, T]:
        """Gets the source version and value - in real usage this might be a checksum, mtime, or version API"""
        version, value = self._source()
        return version, value

    def _fetch_from_source(self, key: str) -> tuple[Any, T]:
        """Fetches from source with validation"""
        version, value = self._source()
        # Validate the value shape (example - could be schema validation)
        if value is None:
            raise ValueError(f"Source returned invalid value for key '{key}'")
        return version, value

    def invalidate(self, key: str) -> None:
        """Explicit invalidation - useful when source notifies changes"""
        with self._lock:
            self._cache.pop(key, None)
            self._last_version.pop(key, None)

    def clear(self) -> None:
        """Clear all cache entries"""
        with self._lock:
            self._cache.clear()
            self._last_version.clear()


# Example usage with a source that tracks file modifications
def file_source(filepath: str):
    import os, hashlib
    def _get():
        stat = os.stat(filepath)
        version = (stat.st_mtime, stat.st_size)
        with open(filepath, 'rb') as f:
            content = f.read()
        # Simple validation - file must not be empty
        if not content:
            raise ValueError("File is empty")
        return version, content
    return _get


if __name__ == "__main__":
    # Demo
    cache = VersionedCache(file_source("example.txt"))
    print(cache.get("file"))  # First call fetches
    print(cache.get("file"))  # Second call - checks version, returns cached if unchanged
    # After modifying example.txt...
    print(cache.get("file"))  # Now returns new content
