```python
from typing import Any, Callable, Hashable
import threading
import time


class VersionedCache:
    """
    A cache that never serves stale data by tying the cache key to the source version.
    
    The cache key is (user_key, source_version), so when the source data changes,
    the old entry is simply no longer found. This avoids explicit invalidation races.
    """

    def __init__(self, load: Callable[[Hashable], Any], get_version: Callable[[], Any]):
        """
        Args:
            load: function that loads the value for a key from the source
            get_version: function that returns a version identifier for the source
                        (e.g. a hash, mtime, or revision number) - must change
                        whenever the source data changes
        """
        self._load = load
        self._get_version = get_version
        self._cache: dict[tuple[Hashable, Any], Any] = {}
        self._lock = threading.Lock()

    def get(self, key: Hashable, ttl: float | None = None) -> Any:
        """
        Get the cached value for key, loading it if needed.
        Never returns stale data because the version is part of the cache key.
        
        Args:
            key: the user-facing key
            ttl: optional time-to-live in seconds; if None, cache until version changes
        """
        version = self._get_version()
        cache_key = (key, version)

        # Fast path: check cache without lock
        if ttl is None:
            cached = self._cache.get(cache_key)
            if cached is not None:
                return cached
        else:
            cached = self._cache.get(cache_key)
            if cached is not None:
                value, timestamp = cached
                if time.monotonic() - timestamp < ttl:
                    return value

        # Slow path: load with lock to avoid thundering herd
        with self._lock:
            # Re-check after acquiring lock (another thread may have loaded it)
            cached = self._cache.get(cache_key)
            if cached is not None:
                if ttl is None:
                    return cached
                value, timestamp = cached
                if time.monotonic() - timestamp < ttl:
                    return value
            
            # Load fresh value
            value = self._load(key)
            if ttl is None:
                self._cache[cache_key] = value
            else:
                self._cache[cache_key] = (value, time.monotonic())
            return value


# Example usage:
# cache = VersionedCache(
#     load=lambda key: read_from_source(key),
#     get_version=lambda: get_source_revision()
