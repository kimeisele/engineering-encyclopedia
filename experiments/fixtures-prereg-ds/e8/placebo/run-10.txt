from typing import Any, Callable
import threading
import time

class SourceAwareCache:
    """
    A cache that prevents stale reads by validating the source data's
    version/timestamp on every access. If the source has changed since
    the value was cached, the cache is refreshed transparently.
    """
    def __init__(self, source_loader: Callable[[str], Any],
                 version_provider: Callable[[str], Any],
                 ttl: float | None = None):
        """
        Args:
            source_loader: function that loads the actual value for a key
            version_provider: function that returns a version/timestamp for
                              the source of a key (e.g. file mtime, DB row version)
            ttl: optional time-to-live in seconds; if set, values older than
                 this are considered stale even if version hasn't changed
        """
        self._source_loader = source_loader
        self._version_provider = version_provider
        self._ttl = ttl
        self._store: dict[str, tuple[Any, Any, float]] = {}  # key -> (value, version, cached_at)
        self._lock = threading.RLock()

    def get(self, key: str) -> Any:
        with self._lock:
            current_version = self._version_provider(key)

            entry = self._store.get(key)
            if entry is not None:
                value, cached_version, cached_at = entry
                version_match = (cached_version == current_version)
                ttl_valid = (self._ttl is None) or ((time.time() - cached_at) < self._ttl)

                if version_match and ttl_valid:
                    return value  # cached value is still fresh

            # Cache miss or stale → reload from source
            fresh_value = self._source_loader(key)
            self._store[key] = (fresh_value, current_version, time.time())
            return fresh_value

    def invalidate(self, key: str) -> None:
        """Force removal of a key from cache (e.g. after known source update)."""
        with self._lock:
            self._store.pop(key, None)

    def clear(self) -> None:
        with self._lock:
            self._store.clear()


# === Example usage ===
if __name__ == "__main__":
    # Simulated source: we can manually "change" the data
    source_data = {"user:1": "Alice", "user:2": "Bob"}
    file_mtimes = {"user:1": 100, "user:2": 100}

    def load(key):
        return source_data[key]

    def version(key):
        return file_mtimes[key]

    cache = SourceAwareCache(load, version)

    print(cache.get("user:1"))  # "Alice" (loaded from source)
    print(cache.get("user:1"))  # "Alice" (served from cache, version unchanged)

    # Simulate source change for user:1
    source_data["user:1"] = "Alice2"
    file_mtimes["user:1"] += 1

    print(cache.get("user:1"))  # "Alice2" (stale value NOT served)
