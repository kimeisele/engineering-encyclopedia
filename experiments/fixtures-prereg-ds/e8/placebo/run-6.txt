from typing import Callable, Generic, TypeVar
from datetime import datetime, timedelta

T = TypeVar('T')

class TTLCache(Generic[T]):
    """
    A cache that stores values with a TTL and never serves stale data.
    If the source changes, the TTL expires and the value is refreshed.
    """
    
    def __init__(self, load_func: Callable[[str], T], ttl_seconds: float = 60.0):
        """
        Args:
            load_func: Function that loads a fresh value for a key.
            ttl_seconds: Time-to-live for each cached value in seconds.
        """
        self._load = load_func
        self._ttl = timedelta(seconds=ttl_seconds)
        self._cache: dict[str, tuple[T, datetime]] = {}
    
    def get(self, key: str) -> T:
        """
        Return the cached value for key, or load it fresh.
        Never returns a value older than the TTL.
        """
        cached = self._cache.get(key)
        now = datetime.now()
        
        if cached is None:
            value = self._load(key)
            self._cache[key] = (value, now)
            return value
        
        value, timestamp = cached
        if now - timestamp >= self._ttl:
            value = self._load(key)
            self._cache[key] = (value, now)
            return value
        
        return value
    
    def invalidate(self, key: str) -> None:
        """Force the next get() to reload the value for key."""
        self._cache.pop(key, None)
