from typing import Any, Callable, Dict
import threading

class CacheWithInvalidation:
    """
    A cache that stores a value per key and invokes a reload callback
    when the key is accessed, ensuring the value is never stale.
    """

    def __init__(self, loader: Callable[[str], Any]):
        self._loader = loader
        self._cache: Dict[str, Any] = {}
        self._lock = threading.Lock()

    def get(self, key: str) -> Any:
        """
        Returns the cached value for the key, but always reloads from the
        source (loader) to avoid serving stale data.
        """
        with self._lock:
            # Always fetch fresh data from the source; no stale value is served.
            fresh_value = self._loader(key)
            self._cache[key] = fresh_value
            return fresh_value

    def get_cached_only(self, key: str) -> Any:
        """
        Returns only the cached value without hitting the source.
        Raises KeyError if the key is not cached.
        """
        with self._lock:
            return self._cache[key]
