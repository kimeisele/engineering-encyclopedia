from typing import Any, Callable
from collections import OrderedDict
import threading

class KeyedCache:
    def __init__(self, source_loader: Callable[[str], Any], max_size: int = 100):
        """
        source_loader: function that loads fresh data for a given key from the source
        max_size: maximum number of cached entries (LRU eviction)
        """
        self._loader = source_loader
        self._cache: OrderedDict[str, Any] = OrderedDict()
        self._lock = threading.RLock()
        self._max_size = max_size

    def get(self, key: str) -> Any:
        """
        Returns the cached value for the key, refreshing from source if:
        1. Key is not in cache, OR
        2. The source data has changed since last load.

        Change detection: re-load from source and compare with cached value.
        This is expensive but guarantees freshness. Use a faster checker if available.
        """
        with self._lock:
            # Load fresh value from source
            fresh_value = self._loader(key)

            # Check if cached value exists and matches fresh value
            if key in self._cache:
                cached_value = self._cache[key]
                if cached_value == fresh_value:
                    # Source hasn't changed, serve cached and refresh LRU position
                    self._cache.move_to_end(key)
                    return cached_value

            # Source changed or not cached — store fresh value
            self._cache[key] = fresh_value
            self._cache.move_to_end(key)

            # LRU eviction
            if len(self._cache) > self._max_size:
                self._cache.popitem(last=False)

            return fresh_value

if __name__ == "__main__":
    # Simulate a source that changes (e.g., database, external API)
    source_state = {"a": 1, "b": 2}

    def load_from_source(key: str):
        # In real scenario, this would query the actual data source each time
        return source_state[key]

    cache = KeyedCache(load_from_source)

    print(cache.get("a"))  # 1, fresh loaded
    print(cache.get("a"))  # 1, cached (source same)
    
    # Source changes
    source_state["a"] = 100
    print(cache.get("a"))  # 100, detects change, returns fresh value (not stale)
