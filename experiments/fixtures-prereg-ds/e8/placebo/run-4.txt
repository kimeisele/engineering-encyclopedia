from typing import Callable, Hashable, Any
import threading
import time

class SourceAwareCache:
    """
    A cache that invalidates entries when the source data changes.

    The key insight is to pair each cached value with a snapshot of the
    source's state (e.g. a version number, a hash, or a modification
    timestamp). On every read, we re-check the source state; if it differs
    from the snapshot, we treat the entry as stale and refresh it.
    """

    def __init__(self, source_getter: Callable[[], dict[Hashable, Any]],
                 state_getter: Callable[[], Hashable]):
        """
        :param source_getter: returns the full source data as a dict,
                              or a function that fetches just the value for key.
        :param state_getter: returns a comparable snapshot of the source's
                             state (e.g. version, mtime, hash). If the state
                             changes, all cached entries for that source are stale.
        """
        self._source_getter = source_getter
        self._state_getter = state_getter
        self._cache: dict[Hashable, tuple[Hashable, Any]] = {}
        self._lock = threading.Lock()

    def get(self, key: Hashable) -> Any:
        """Return cached value for key, refreshing if source state changed."""
        current_state = self._state_getter()
        with self._lock:
            cached = self._cache.get(key)
            if cached is not None:
                cached_state, value = cached
                if cached_state == current_state:
                    return value
            # stale or missing: refresh from source
            data = self._source_getter()
            value = data[key]          # KeyError propagates to caller
            self._cache[key] = (current_state, value)
            return value
