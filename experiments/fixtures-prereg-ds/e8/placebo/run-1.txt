from typing import Any, Callable, Dict
import threading
import time


class CacheWithInvalidation:
    """
    A cache that avoids stale reads by using a validation function or
    a maximum time-to-live per key.
    """

    def __init__(
        self,
        load_func: Callable[[str], Any],
        is_fresh: Callable[[str, Any], bool] | None = None,
        ttl: float | None = None,
    ):
        """
        Args:
            load_func: atomic function to load the source value for a key.
            is_fresh: optional predicate (key, cached_value) -> bool.
                      If provided, this takes precedence over ttl.
            ttl: optional time-to-live in seconds (checked on each get).
        """
        self._load = load_func
        self._is_fresh = is_fresh
        self._ttl = ttl
        self._cache: Dict[str, Any] = {}
        self._timestamps: Dict[str, float] = {}
        self._lock = threading.RLock()

    def get(self, key: str) -> Any:
        """
        Return the cached value for key only if it's still fresh.
        Otherwise reload the source and update the cache.
        """
        with self._lock:
            if key not in self._cache:
                value = self._load(key)
                self._cache[key] = value
                self._timestamps[key] = time.monotonic()
                return value

            cached = self._cache[key]

            if self._is_fresh is not None:
                if self._is_fresh(key, cached):
                    return cached
            elif self._ttl is not None:
                age = time.monotonic() - self._timestamps[key]
                if age < self._ttl:
                    return cached
            else:
                # No freshness check – unsafe, but caller chose this.
                return cached

            # stale – reload
            new_value = self._load(key)
            self._cache[key] = new_value
            self._timestamps[key] = time.monotonic()
            return new_value

    def invalidate(self, key: str) -> None:
        """Force next get to reload the source."""
        with self._lock:
            self._cache.pop(key, None)
            self._timestamps.pop(key, None)
