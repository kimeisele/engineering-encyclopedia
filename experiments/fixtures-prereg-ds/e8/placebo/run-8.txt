from typing import Any, Callable, Dict, Hashable, Optional
import threading
import time


class CacheWithVersioning:
    """
    A cache that stores each key with a version number. The source data
    version is provided by a user-supplied `version_provider` callable.
    If the version changes, the cached value is considered stale and must
    be recomputed.
    """

    def __init__(
        self,
        load_func: Callable[[Hashable], Any],
        version_provider: Callable[[], Any],
        ttl: Optional[float] = None,
    ):
        """
        :param load_func: Function that loads the value for a given key.
        :param version_provider: Function returning a monotonic-ish version
                                 identifier for the source data.
        :param ttl: Optional time-to-live in seconds. If set, values older
                    than this are considered stale even if version unchanged.
        """
        self._load_func = load_func
        self._version_provider = version_provider
        self._ttl = ttl
        self._cache: Dict[Hashable, tuple[Any, Any, float]] = {}  # key -> (value, version, timestamp)
        self._lock = threading.Lock()

    def get(self, key: Hashable) -> Any:
        """
        Return the cached value for `key`, refreshing if stale.
        """
        current_version = self._version_provider()
        now = time.monotonic()

        with self._lock:
            entry = self._cache.get(key)

            if entry is not None:
                value, cached_version, cached_time = entry
                # Check version change first, then TTL
                if cached_version == current_version:
                    if self._ttl is None or (now - cached_time) < self._ttl:
                        return value
                    # TTL expired, need to refresh
                # else: version changed, stale

            # Cache miss or stale — load fresh value
            fresh_value = self._load_func(key)
            self._cache[key] = (fresh_value, current_version, now)
            return fresh_value

    def invalidate(self, key: Hashable) -> None:
        """Force removal of a key from cache (optional API)."""
        with self._lock:
            self._cache.pop(key, None)


# ========== Example usage ==========
if __name__ == "__main__":
    # Simulate a changing source
    source_data = {"user:1": "Alice"}
    source_version = [1]

    def get_source_version():
        return source_version[0]

    def load_user(user_id):
        # In real usage, fetch from DB/file/etc.
        print(f"Loading {user_id} from source...")
        return source_data[user_id]

    cache = CacheWithVersioning(load_user, get_source_version)

    print(cache.get("user:1"))  # → Loading... Alice
    print(cache.get("user:1"))  # → Alice (cached)

    # Source changes: update data and version
    source_data["user:1"] = "Alicia"
    source_version[0] += 1

    print(cache.get("user:1"))  # → Loading... Alicia (stale not served)
