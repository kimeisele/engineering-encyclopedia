def make_cached_with_invalidation(source_func: Callable[[str], Any], version_func: Callable[[], Any] = None):
    """
    Caches values per key. re-fetches if the source data changes.
    version_func should return a hash/timestamp that changes when source data does.
    """
    _cache = {}
    _last_version = None
    
    def get(key: str):
        nonlocal _last_version
        current_version = version_func() if version_func else None
        
        if current_version != _last_version:
            _cache.clear()
            _last_version = current_version
            
        if key not in _cache:
            _cache[key] = source_func(key)
        return _cache[key]
    
    return get
