import threading

class SharedCounter:
    """
    A thread-safe counter using a lock to ensure no update is lost
    when multiple workers increment concurrently.
    """
    def __init__(self, initial_value=0):
        self._value = initial_value
        self._lock = threading.Lock()

    def increment(self, delta=1):
        """
        Atomically increase the counter by delta.
        Uses a lock to serialize concurrent updates, preventing lost updates.
        """
        with self._lock:
            self._value += delta
            return self._value

    def get_value(self):
        """Return the current counter value (safe read)."""
        with self._lock:
            return self._value
