```python
import threading

class SharedCounter:
    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()

    def increment(self, amount=1):
        """Safely increment the counter."""
        with self._lock:
            self._value += amount

    def get(self):
        """Return the current counter value."""
        with self._lock:
            return self._value


# Usage example
counter = SharedCounter()

def worker(n):
    for _ in range(n):
        counter.increment()

# Simulating concurrent updates
t1 = threading.Thread(target=worker, args=(1000,))
t2 = threading.Thread(target=worker, args=(1000,))
t1.start()
t2.start()
t1.join()
t2.join()

assert counter.get() == 2000, "Counter updates were lost!"
print(f"Final counter value: {counter.get()}")
```
