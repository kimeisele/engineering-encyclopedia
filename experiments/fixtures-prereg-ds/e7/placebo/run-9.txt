```python
import threading

class SharedCounter:
    """A thread-safe counter that never loses updates."""
    
    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()
    
    def increment(self, by=1):
        """Update the counter by a non-negative amount atomically."""
        if by < 0:
            raise ValueError("Increment must be non-negative")
        with self._lock:
            self._value += by
    
    def get(self):
        """Read the current value atomically."""
        with self._lock:
            return self._value

# Example usage with two workers
def worker(counter, iterations=1000):
    for _ in range(iterations):
        counter.increment()

if __name__ == "__main__":
    counter = SharedCounter()
    t1 = threading.Thread(target=worker, args=(counter,))
    t2 = threading.Thread(target=worker, args=(counter,))
    t1.start()
    t2.start()
    t1.join()
    t2.join()
    
    print(f"Final counter value: {counter.get()}")
    # Should print 2000 (no lost updates)
```
