import threading

class SharedCounter:
    """Thread-safe counter using a lock (mutex)."""
    
    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()
    
    def increment(self, amount=1):
        """Atomically increment the counter by `amount`. Never loses updates."""
        with self._lock:
            # Lock ensures only one thread can modify at a time
            self._value += amount
        return self._value
    
    def get_value(self):
        """Return the current value (read is also protected for consistency)."""
        with self._lock:
            return self._value


# Alternative: optimistic conflict detection using compare-and-swap (CAS)
def update_counter_atomic(counter, delta):
    """
    Lock-free CAS approach — detects conflicts and retries.
    Counter must expose get_value() and a CAS primitive.
    """
    while True:
        old = counter.value
        new = old + delta
        # If CAS fails (another thread changed value), retry
        if counter.compare_and_swap(old, new):
            return new
