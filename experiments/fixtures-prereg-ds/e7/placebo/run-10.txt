import threading

class SharedCounter:
    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()

    def increment(self, amount=1):
        """Atomically increment the counter by amount."""
        with self._lock:
            self._value += amount
            return self._value

    def get(self):
        with self._lock:
            return self._value
