```python
import threading

class SharedCounter:
    """A thread-safe counter using a lock to prevent lost updates."""

    def __init__(self, initial_value=0):
        self._value = initial_value
        self._lock = threading.Lock()

    def increment(self, amount=1):
        """Atomically increment the counter by the given amount."""
        with self._lock:
            self._value += amount
            return self._value

    def get_value(self):
        """Return the current value (not atomic for read, but acceptable)."""
        with self._lock:
            return self._value
```
