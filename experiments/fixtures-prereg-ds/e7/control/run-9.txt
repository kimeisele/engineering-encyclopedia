## Solution 1: Using Locking (Thread-safe with mutex)

```python
import threading
import time
import random

class LockedCounter:
    def __init__(self):
        self.value = 0
        self.lock = threading.Lock()
    
    def increment(self, amount=1):
        """Thread-safe increment using a lock"""
        with self.lock:
            self.value += amount
            return self.value
    
    def get_value(self):
        with self.lock:
            return self.value

# Example usage with worker threads
def worker_locked(counter, updates):
    for _ in range(updates):
        time.sleep(random.uniform(0.001, 0.01))  # Simulate work
        counter.increment(1)

# Test with locking
counter = LockedCounter()
print("Testing with Lock approach:")
print(f"Initial value: {counter.get_value()}")

threads = []
for i in range(2):
    t = threading.Thread(target=worker_locked, args=(counter, 100))
    threads.append(t)
    t.start()

for t in threads:
    t.join()

print(f"Final value with lock: {counter.get_value()}")
```

## Solution 2: Using Optimistic Conflict Detection (Compare-and-Swap)

```python
import threading
import time
import random

class OptimisticCounter:
    def __init__(self):
        self.value = 0
        self.lock = threading.Lock()  # Lock only for CAS operation, not for reading
    
    def increment_optimistic(self, amount=1):
        """Increment using compare-and-swap with retries"""
        while True:
            current = self.read_value()
            new_value = current + amount
            
            # CAS operation - atomic in Python due to GIL, but we'll simulate
            if self.compare_and_set(current, new_value):
                return new_value
            
            # If CAS failed, another thread modified it; retry
            time.sleep(random.uniform(0.0001, 0.001))
    
    def read_value(self):
        # In real-world, this might be an atomic read
        with self.lock:
            return self.value
    
    def compare_and_set(self, expected, new_value):
        # Simulated CAS - in real Python we'd use hardware CAS or GIL guarantees
        with self.lock:
            if self.value == expected:
                self.value = new_value
                return True
            return False
    
    def get_value(self):
        return self.read_value()

# Example usage
def worker_optimistic(counter, updates):
    for _ in range(updates):
        time.sleep(random.uniform(0.001, 0.01))  # Simulate work
        counter.increment_optimistic(1)

# Test with optimistic approach
counter2 = OptimisticCounter()
print("\nTesting with Optimistic approach:")
print(f"Initial value: {counter2.get_value()}")

threads = []
for i in range(2):
    t = threading.Thread(target=worker_optimistic, args=(counter2, 100))
    threads.append(t)
    t.start()

for t in threads:
    t.join()

print(f"Final value with optimistic: {counter2.get_value()}")
```

## Solution 3: Using Python's built-in synchronization (Recommended)

```python
import threading
import time
import random

class PythonCounter:
    def __init__(self):
        self.value = 0
        self.lock = threading.Lock()
    
    def increment(self, amount=1):
        # Python 3.7+ has a simpler approach with `threading.RLock` or just use lock
        with self.lock:
            self.value += amount
            return self.value

# Most Pythonic solution using Lock as context manager
class PythonicCounter:
    def __init__(self):
        self._value = 0
        self._lock = threading.Lock()
    
    @property
    def value(self):
        with self._lock:
            return self._value
    
    def increment(self, amount=1):
        with self._lock:
            self._value += amount
            return self._value

# Test with most Pythonic solution
def worker_pythonic(counter, updates):
    for _ in range(updates):
        time.sleep(random.uniform(0.001, 0.01))
        counter.increment(1)

pythonic_counter = PythonicCounter()
print("\nTesting with Pythonic approach:")
print(f"Initial value: {pythonic_counter.value}")

threads = []
for i in range(2):
    t = threading.Thread(target=worker_pythonic, args=(pythonic_counter, 1000))
    threads.append(t)
    t.start()

for t in threads:
    t.join()

print(f"Final value with Pythonic: {pythonic_counter.value}")
```

## Key Differences:

**Locking (Synchronous)**: Simple, guarantees no lost updates, but can cause contention and waiting.

**Optimistic (Compare-and-Swap)**: Better for read-heavy workloads, can retry on conflict, better scalability but more complex.

**Best Practice**: Use `threading.Lock()` with context manager for most Python applications - it's simple, safe, and efficient.

**Why no updates get lost:**
- **Lock**: Each thread acquires the lock before modifying, ensuring mutual exclusion
- **Optimistic**: CAS detects conflicts and retries, ensuring the final value includes all increments

**Output would show**: Both counter values would equal `200` (2 workers × 100 updates each) showing no updates were lost.
