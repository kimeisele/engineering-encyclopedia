## 1. Using a Lock (Threading Lock)

```python
import threading
import time
import random

class Counter:
    def __init__(self):
        self.value = 0
        self.lock = threading.Lock()
    
    def update(self, amount):
        """Thread-safe update using a lock"""
        with self.lock:
            self.value += amount
            return self.value
    
    def get_value(self):
        with self.lock:
            return self.value

# Usage
def worker(counter, iterations=100):
    for _ in range(iterations):
        counter.update(1)
        time.sleep(random.uniform(0, 0.001))

# Test
counter = Counter()
threads = []
for _ in range(5):
    t = threading.Thread(target=worker, args=(counter,))
    threads.append(t)
    t.start()

for t in threads:
    t.join()

print(f"Final counter value: {counter.get_value()} (expected: 500)")
```

## 2. Using Atomic Operations (with multiprocessing)

```python
import multiprocessing
import os

class AtomicCounter:
    def __init__(self):
        self.value = multiprocessing.Value('i', 0)
        self.lock = multiprocessing.Lock()
    
    def update(self, amount):
        """Atomic update using compare-and-swap pattern"""
        with self.lock:
            self.value.value += amount
            return self.value.value
    
    def get_value(self):
        with self.lock:
            return self.value.value

# Worker function for multiprocessing
def worker(counter, iterations=100):
    for _ in range(iterations):
        counter.update(1)

if __name__ == '__main__':
    counter = AtomicCounter()
    processes = []
    for _ in range(5):
        p = multiprocessing.Process(target=worker, args=(counter,))
        processes.append(p)
        p.start()
    
    for p in processes:
        p.join()
    
    print(f"Final counter value: {counter.get_value()} (expected: 500)")
```

## 3. Using Optimistic Concurrency Control (Conflict Detection)

```python
import threading
import time

class OptimisticCounter:
    """Uses optimistic locking with version numbers"""
    
    def __init__(self):
        self.value = 0
        self.version = 0
        self.lock = threading.Lock()
    
    def update(self, amount):
        """Update with optimistic concurrency control"""
        # Read phase
        with self.lock:
            current_value = self.value
            current_version = self.version
        
        # Simulate some work/processing
        new_value = current_value + amount
        
        # Write phase with conflict detection
        with self.lock:
            # Check if version hasn't changed (no conflict)
            if self.version == current_version:
                self.value = new_value
                self.version += 1
                return True
            else:
                # Conflict detected - retry or raise
                print(f"Conflict detected! Retrying...")
                return False
    
    def update_with_retry(self, amount, max_retries=10):
        """Update with automatic retry on conflict"""
        for attempt in range(max_retries):
            if self.update(amount):
                return True
            time.sleep(0.01)  # Small delay before retry
        raise Exception("Max retries exceeded on update")
    
    def get_value(self):
        with self.lock:
            return self.value

# Usage with retry mechanism
def worker_optimistic(counter, iterations=100):
    for _ in range(iterations):
        counter.update_with_retry(1)

# Test
counter = OptimisticCounter()
threads = []
for _ in range(5):
    t = threading.Thread(target=worker_optimistic, args=(counter,))
    threads.append(t)
    t.start()

for t in threads:
    t.join()

print(f"Final counter value: {counter.get_value()} (expected: 500)")
```

## 4. Using a Read-Write Lock (for mixed workloads)

```python
import threading

class ReadWriteCounter:
    """Allows multiple readers but only one writer"""
    
    def __init__(self):
        self.value = 0
        self.read_condition = threading.Condition()
        self.write_condition = threading.Condition()
        self.active_readers = 0
        self.active_writers = 0
    
    def update(self, amount):
        """Thread-safe update using write lock"""
        with self.write_condition:
            self.active_writers += 1
            try:
                self.value += amount
            finally:
                self.active_writers -= 1
    
    def read(self):
        """Thread-safe read using read lock (allows concurrent reads)"""
        with self.read_condition:
            self.active_readers += 1
            try:
                return self.value
            finally:
                self.active_readers -= 1

# Test
counter = ReadWriteCounter()

def writer_worker(counter, iterations=100):
    for _ in range(iterations):
        counter.update(1)

def reader_worker(counter, iterations=100):
    for _ in range(iterations):
        val = counter.read()

# Create threads
threads = []
for _ in range(3):
    threads.append(threading.Thread(target=writer_worker, args=(counter,)))
    threads.append(threading.Thread(target=reader_worker, args=(counter,)))

for t in threads:
    t.start()
for t in threads:
    t.join()

print(f"Final counter value: {counter.read()} (expected: 300)")
```

## 5. Using Queue-based approach (message passing)

```python
import threading
import queue

class QueueCounter:
    """Counter using message passing via queue"""
    
    def __init__(self):
        self.queue = queue.Queue()
        self.value = 0
        self.worker_thread = threading.Thread(target=self._process_updates, daemon=True)
        self.worker_thread.start()
    
    def _process_updates(self):
        """Single thread processes all updates - no conflicts possible"""
        while True:
            try:
                update = self.queue.get(timeout=0.1)
                if update == 'STOP':
                    break
                self.value += update
            except queue.Empty:
                continue
    
    def update(self, amount):
        """Add update to queue (non-blocking)"""
        self.queue.put(amount)
    
    def get_value(self):
        return self.value
    
    def close(self):
        """Gracefully shut down the worker"""
        self.queue.put('STOP')

# Usage
def worker(counter, iterations=100):
    for _ in range(iterations):
        counter.update(1)

counter = QueueCounter()
threads = []
for _ in range(5):
    t = threading.Thread(target=worker, args=(counter,))
    threads.append(t)
    t.start()

for t in threads:
    t.join()

# Wait a bit for all queued updates to process
import time
time.sleep(0.1)

print(f"Final counter value: {counter.get_value()} (expected: 500)")
counter.close()
```

## Key Strategies Explained:

1. **Lock-Based**: Simple, guarantees correctness, but can cause bottlenecks.

2. **Atomic Operations**: Fastest, but limited to simple operations.

3. **Optimistic Concurrency**: Good for low contention, enables retry on conflict.

4. **Read-Write Locks**: Optimal when reads are frequent, writes are rare.

5. **Queue-Based**: Decouples updates from actual operations, no conflicts possible.

## Choosing the Right Strategy:

- **Low concurrency**: Use simple locks
- **High read volume**: Read-write locks
- **High write volume**: Queue-based approach
- **Distributed systems**: Optimistic concurrency with version numbers
- **Simple atomic operations**: Use CPU atomic operations (if available)

The key requirement is ensuring that **no update is ever lost**, which all these strategies guarantee by preventing concurrent writes to the same memory location without proper synchronization.
