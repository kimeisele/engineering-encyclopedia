## 1. Mutex Locking (Thread-safe)
```python
import threading

class SharedCounter:
    def __init__(self):
        self._counter = 0
        self._lock = threading.Lock()
    
    def increment(self, value=1):
        """Thread-safe increment using mutex lock"""
        with self._lock:
            self._counter += value
            return self._counter
    
    def get_value(self):
        with self._lock:
            return self._counter
```

## 2. Atomic Operation with Lock (Python's GIL provides this)

import threading

## 3. Optimistic Locking with Version Numbers
```python
import threading
import time

class VersionedCounter:
    def __init__(self):
        self._counter = 0
        self._version = 0
        self._lock = threading.Lock()
    
    def increment_optimistic(self, value=1, max_retries=10):
        """Optimistic locking with version checking"""
        for attempt in range(max_retries):
            # Snapshot current state
            with self._lock:
                old_version = self._version
                old_value = self._counter
            
            # Simulate work (non-critical section)
            new_value = old_value + value
            
            # Try to commit with version check
            with self._lock:
                if self._version == old_version:  # No conflict
                    self._counter = new_value
                    self._version += 1
                    return new_value
                # Conflict detected, retry
                time.sleep(0.001)  # small backoff
        
        raise Exception("Failed to update after max retries")
```

## 4. Read-Write Lock
```python
import threading

class ReadWriteCounter:
    def __init__(self):
        self._counter = 0
        self._read_lock = threading.Lock()
        self._write_lock = threading.Lock()
        self._readers = 0
    
    def increment(self, value=1):
        """Write operation - exclusive access"""
        with self._write_lock:
            self._counter += value
            return self._counter
    
    def get_value(self):
        """Read operation - shared access"""
        with self._read_lock:
            self._readers += 1
            if self._readers == 1:
                self._write_lock.acquire()
        
        try:
            return self._counter
        finally:
            with self._read_lock:
                self._readers -= 1
                if self._readers == 0:
                    self._write_lock.release()
```

## 6. Lock-free approach using CAS (Compare-and-Swap) pattern
```python
import threading
import time

class LockFreeCounter:
    """Simulates CAS (Compare-And-Swap) pattern"""
    def __init__(self):
        self._counter = 0
        self._lock = threading.Lock()  # Still need basic atomicity for CAS
    
    def increment_cas(self, value=1):
        """CAS-style update with retry"""
        while True:
            old_value = self._read_value()
            new_value = old_value + value
            
            if self._compare_and_swap(old_value, new_value):
                return new_value
            # CAS failed, another thread updated - retry
            time.sleep(0.0001)  # small backoff
    
    def _read_value(self):
        with self._lock:
            return self._counter
    
    def _compare_and_swap(self, expected, new):
        with self._lock:
            if self._counter == expected:
                self._counter = new
                return True
            return False
```

## 7. Complete Example with Worker Threads
```python
import threading
import random
import time

class ThreadSafeCounter:
    """Complete implementation with proper locking"""
    
    def __init__(self):
        self._value = 0
        self._lock = threading.RLock()  # Reentrant lock for nested calls
    
    def increment(self, amount=1):
        """Increment counter by given amount"""
        with self._lock:
            # Critical section - protected
            self._value += amount
            return self._value
    
    def decrement(self, amount=1):
        """Decrement counter"""
        with self._lock:
            self._value -= amount
            return self._value
    
    def get_value(self):
        """Get current value"""
        with self._lock:
            return self._value
    
    def reset(self):
        """Reset counter to zero"""
        with self._lock:
            self._value = 0

def worker(counter, updates, delay=0.01):
    """Worker function that performs multiple updates"""
    for _ in range(updates):
        counter.increment()
        time.sleep(delay)

def run_concurrent_test():
    """Test with multiple threads"""
    counter = ThreadSafeCounter()
    
    # Create threads
    threads = []
    num_threads = 5
    updates_per_thread = 100
    
    for i in range(num_threads):
        t = threading.Thread(target=worker, args=(counter, updates_per_thread))
        threads.append(t)
        t.start()
    
    # Wait for all threads
    for t in threads:
        t.join()
    
    expected = num_threads * updates_per_thread
    actual = counter.get_value()
    
    print(f"Expected: {expected}, Actual: {actual}")
    assert expected == actual, f"Race condition detected! Expected {expected}, got {actual}"
    print("✓ No updates were lost!")

if __name__ == "__main__":
    run_concurrent_test()
```

## Key Recommendations:

1. **For simple increments**: Python's GIL makes `+=` atomic, but use explicit locks for clarity
2. **For high concurrency**: Use optimistic locking (version numbers)
3. **For read-heavy workloads**: Use Read-Write locks
4. **For production**: Consider using `multiprocessing` with proper locks if using multiple processes

Choose based on your specific requirements:
- **Performance**: Atomic operations with GIL are fastest
- **Correctness**: Always prefer explicit locks over GIL reliance
- **Complexity**: Simple mutex locks are easiest to understand and debug
