import threading
import time

class SharedCounter:
    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()

    def increment(self):
        """Increment the counter safely. No update is ever lost."""
        with self._lock:
            old = self._value
            # Simulate some work to increase the chance of interleaving
            time.sleep(0.001)
            self._value = old + 1

    def get(self):
        with self._lock:
            return self._value


# Example usage
def worker(counter, iterations):
    for _ in range(iterations):
        counter.increment()

if __name__ == "__main__":
    counter = SharedCounter()
    threads = []
    for _ in range(2):
        t = threading.Thread(target=worker, args=(counter, 100))
        threads.append(t)
        t.start()
    for t in threads:
        t.join()
    print(f"Final count: {counter.get()}")  # Should be 200
