import threading
import time


class SharedCounter:
    """A counter that safely handles concurrent updates using a lock."""

    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()

    def increment(self, steps=1):
        """
        Increment the counter atomically.

        Uses a lock to ensure that concurrent increments never lose updates.
        The lock is acquired before reading, and released after writing,
        making the read-modify-write sequence atomic.
        """
        with self._lock:
            current = self._value
            # Simulate a small amount of work to demonstrate the race condition
            time.sleep(0.001)
            self._value = current + steps
        return self._value

    @property
    def value(self):
        with self._lock:
            return self._value


# Alternative: optimistic locking with version check (for database-like systems)
class OptimisticCounter:
    """
    Alternative implementation using optimistic locking.

    This simulates a conflict-detection strategy where each update checks
    the version before applying the change, and retries if conflict occurs.
    """
    def __init__(self, initial=0):
        self._value = initial
        self._version = 0
        self._lock = threading.Lock()

    def increment(self, steps=1, max_retries=10):
        with self._lock:
            for attempt in range(max_retries):
                expected_version = self._version
                current = self._value
                # Simulate work that might allow another thread to interleave
                time.sleep(0.001)
                if expected_version != self._version:
                    # Conflict: another thread updated between our read and write
                    continue
                self._value = current + steps
                self._version += 1
                return self._value
            raise RuntimeError(f"Failed to update after {max_retries} retries")

    @property
    def value(self):
        with self._lock:
            return self._value


# Test function to demonstrate the race condition is handled
def test_concurrent_updates():
    counter = SharedCounter()
    threads = []
    for _ in range(100):
        t = threading.Thread(target=counter.increment, args=(1,))
        threads.append(t)

    for t in threads:
        t.start()
    for t in threads:
        t.join()

    print(f"SharedCounter final value: {counter.value} (expected 100)")

    optimistic = OptimisticCounter()
    threads = []
    for _ in range(100):
        t = threading.Thread(target=optimistic.increment, args=(1,))
        threads.append(t)

    for t in threads:
        t.start()
    for t in threads:
        t.join()

    print(f"OptimisticCounter final value: {optimistic.value} (expected 100)")


if __name__ == "__main__":
    test_concurrent_updates()
