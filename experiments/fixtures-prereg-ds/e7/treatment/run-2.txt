```python
import threading
import time

class SharedCounter:
    """
    A thread-safe counter using optimistic concurrency control with
    conflict detection via compare-and-swap (CAS).
    """
    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()

    def increment(self, n=1):
        """
        Atomically increment the counter by n.
        Uses a lock-free approach with CAS, retrying on conflict.
        """
        while True:
            # Read current value (snapshot)
            with self._lock:
                current = self._value
            
            # Compute new value
            new_value = current + n
            
            # Attempt atomic compare-and-swap
            with self._lock:
                if self._value == current:
                    self._value = new_value
                    return new_value
                # If conflict detected, loop and retry
                # (another thread updated between our read and CAS attempt)
            
            # Small backoff to reduce contention
            time.sleep(0)

    def get(self):
        with self._lock:
            return self._value

# Example usage with two worker threads
def worker(counter, iterations=1000):
    for _ in range(iterations):
        counter.increment()

if __name__ == "__main__":
    counter = SharedCounter()
    threads = [
        threading.Thread(target=worker, args=(counter, 1000)),
        threading.Thread(target=worker, args=(counter, 1000))
    ]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    assert counter.get() == 2000, f"Lost updates! Final: {counter.get()}"
    print(f"Final counter value: {counter.get()} (expected 2000)")
```
