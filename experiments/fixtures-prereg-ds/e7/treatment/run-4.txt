import threading

class SharedCounter:
    """Thread-safe counter using optimistic locking (compare-and-swap)."""
    
    def __init__(self, initial=0):
        self._value = initial
        self._lock = threading.Lock()
    
    def increment(self):
        """Atomically increment the counter using a lock-based critical section."""
        # Use pessimistic locking: hold the lock only for the critical section
        with self._lock:
            self._value += 1
        return self._value
    
    def get(self):
        """Return the current value."""
        with self._lock:
            return self._value


# Alternative: optimistic locking without a lock, using compare-and-swap
class OptimisticCounter:
    """Counter using CAS loop; requires retry on conflict."""
    
    def __init__(self, initial=0):
        self._value = initial
        self._version = 0
    
    def increment(self):
        """Increment using CAS; retry on version conflict."""
        while True:
            old_value = self._value
            old_version = self._version
            # Simulated CAS (in real code, use cmpxchg or database transaction)
            if self._version == old_version:
                self._value = old_value + 1
                self._version += 1
                return self._value
            # else: retry (conflict detected before write)
    
    def get(self):
        return self._value


# For the specific "two workers updating same shared counter" case:
def update_shared_counter(counter, iterations=1000):
    """Worker that increments the counter multiple times."""
    for _ in range(iterations):
        counter.increment()


# Demo with two threads
if __name__ == "__main__":
    counter = SharedCounter()
    t1 = threading.Thread(target=update_shared_counter, args=(counter, 10000))
    t2 = threading.Thread(target=update_shared_counter, args=(counter, 10000))
