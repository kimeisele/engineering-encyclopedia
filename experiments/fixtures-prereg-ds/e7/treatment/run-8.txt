import threading

class SharedCounter:
    """Thread-safe counter using optimistic locking with a version field."""
    
    def __init__(self, initial=0):
        self._value = initial
        self._version = 0
        self._lock = threading.Lock()
    
    def update(self, delta):
        """
        Atomically add delta to the counter.
        Uses a version-check-retry loop (optimistic concurrency control).
        """
        while True:
            # Snapshot current state under lock
            with self._lock:
                current_value = self._value
                current_version = self._version
            
            # Compute new value based on snapshot
            new_value = current_value + delta
            
            # Try to commit - only succeeds if version hasn't changed
            with self._lock:
                if self._version == current_version:
                    self._value = new_value
                    self._version += 1
                    return new_value
                # else: version changed, retry with fresh snapshot

    def get(self):
        with self._lock:
            return self._value
