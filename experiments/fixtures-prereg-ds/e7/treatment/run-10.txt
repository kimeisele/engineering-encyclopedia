```python
from threading import Lock

class SharedCounter:
    def __init__(self, initial=0):
        self._value = initial
        self._lock = Lock()

    def increment(self, amount=1):
        with self._lock:
            self._value += amount
        return self._value

    @property
    def value(self):
        with self._lock:
            return self._value
```
