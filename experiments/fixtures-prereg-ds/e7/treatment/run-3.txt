import threading
import time
import itertools

class SharedCounter:
    """
    A thread-safe counter using optimistic concurrency (version check).
    
    This implementation ensures no update is ever lost, even if two workers
    try to increment at the exact same time.
    """
    
    def __init__(self, initial_value=0):
        self._lock = threading.Lock()  # protects the internal state
        self._value = initial_value
        self._version = 0  # monotonic version counter for conflict detection
    
    def increment(self, delta=1):
        """
        Atomically increment the counter.
        
        Uses a pessimistic lock to ensure the read-modify-write cycle is
        atomic. This is the simplest way to guarantee no lost updates.
        
        Args:
            delta: amount to add to the counter (default 1)
        
        Returns:
            New value of the counter after increment
        """
        with self._lock:
            self._value += delta
            self._version += 1
            return self._value
    
    def get_value(self):
        """Return current counter value (safely)."""
        with self._lock:
            return self._value
    
    def get_version(self):
        """Return current version (for optimistic checks)."""
        with self._lock:
            return self._version


class OptimisticCounter:
    """
    Alternative implementation using optimistic concurrency with retry.
    
    This approach doesn't use a lock for the update itself, but instead
    detects conflicts via version numbers and retries when needed.
    """
    
    def __init__(self, initial_value=0, max_retries=10):
        self._lock = threading.Lock()  # only protects version check
        self._value = initial_value
        self._version = 0
        self._max_retries = max_retries
    
    def increment(self, delta=1):
        """
        Increment counter with optimistic conflict detection.
        
        Reads version, computes new value, then attempts to CAS (compare-and-swap).
        If version changed, retry until success or max retries exceeded.
        
        Args:
            delta: amount to add
        
        Returns:
            New value after successful increment
        
        Raises:
            RuntimeError: if max retries exceeded
        """
        for attempt in range(self._max_retries + 1):
            # optimistic read
            with self._lock:
                expected_version = self._version
                current_value = self._value
            
            # compute what we'd like to write
            new_value = current_value + delta
            
            # try to commit - this is the critical section
            with self._lock:
                # check if version still matches (no one else changed it)
                if self._version == expected_version:
                    self._value = new_value
                    self._version += 1
                    return self._value
                # version changed - someone else updated, retry
                # (we loop and re-read)
        
        raise RuntimeError(f"Failed to increment after {self._max_retries} retries")
    
    def get_value(self):
        with self._lock:
            return self._value


# For simplicity and safety, use the pessimistic lock version as the primary
def update_shared_counter(counter, worker_id, num_updates):
    """
    Worker function that updates the counter multiple times.
    
    Args:
        counter: SharedCounter instance
        worker_id: identifier for this worker
        num_updates: how many times to increment
    """
    for i in range(num_updates):
        new_val = counter.increment()
        # small delay to simulate real work and increase chance of interleaving
        time.sleep(0.0001)


def main():
    """Example usage with two concurrent workers."""
    counter = SharedCounter()
    
    # Create two workers that each try to increment 1000 times
    threads = []
    import threading
    
    for worker_id in range(2):
        t = threading.Thread(target=update_shared_counter, 
                           args=(counter, worker_id, 1000))
        threads.append(t)
    
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    
    final_value = counter.get_value()
    print(f"Final counter value: {final_value}")
    print(f"Expected value: 2000")
    assert final_value == 2000, f"Lost update! Got {final_value}, expected 2000"


if __name__ == "__main__":
    main()
