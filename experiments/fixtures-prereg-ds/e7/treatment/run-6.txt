import threading
import time

class SharedCounter:
    """Thread-safe counter using a reentrant lock (pessimistic locking)."""
    
    def __init__(self, initial_value=0):
        self._value = initial_value
        self._lock = threading.RLock()
    
    def increment(self, amount=1):
        """
        Atomically increment the counter by the given amount.
        
        Uses an RLock to ensure only one worker updates at a time.
        The lock is held only during the critical section (read-modify-write),
        not across any I/O or network calls.
        """
        with self._lock:
            # Critical section: check-and-act is atomic under the lock
            new_value = self._value + amount
            self._value = new_value
            return new_value
    
    def get(self):
        """Return the current value without modifying it."""
        with self._lock:
            return self._value

# Alternative: optimistic locking with retry (for when contention is low)
class OptimisticCounter:
    """
    Counter using optimistic concurrency with version checking.
    
    Each update reads the current value and version, then attempts a
    compare-and-swap. If the version changed, retry with a bounded number
    of attempts.
    """
    
    def __init__(self, initial_value=0):
        self._value = initial_value
        self._version = 0
        self._lock = threading.Lock()
    
    def increment(self, amount=1, max_retries=10):
        """
        Increment the counter, retrying if a conflict is detected.
        
        Uses a lock only to protect the read-modify-write on the internal
        state, but the real conflict detection is the version check.
        In practice, for true optimistic control without a lock, you'd use
        Compare-And-Swap (CAS) atomic operations (e.g., with a DB or
        multiprocessing.shared_memory).
        """
        for attempt in range(max_retries):
            with self._lock:
                current_value = self._value
                current_version = self._version
            
            # Simulate the compare-and-swap outside the lock (not truly possible
            # in pure Python threads, but in a database or with atomics you'd
            # do it like this). For demonstration, we rely on the GIL + lock.
            with self._lock:
                if self._version != current_version:
                    continue  # conflict, retry
                self._value = current_value + amount
                self._version += 1
                return self._value
        
        raise RuntimeError(f"Failed to increment after {max_retries} retries due to conflicts")
    
    def get(self):
        with self._lock:
            return self._value

# Example usage
if __name__ == "__main__":
    counter = SharedCounter()
    
    def worker():
        for _ in range(1000):
            counter.increment()
    
    threads = [threading.Thread(target=worker) for _ in range(5)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    
    print(f"Final counter value: {counter.get()} (expected 5000)")
