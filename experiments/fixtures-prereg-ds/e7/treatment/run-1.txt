import threading
import time

class SharedCounter:
    """
    A thread-safe counter using a mutex (pessimistic locking).
    The lock is held only for the critical section (read-increment-write),
    ensuring no update is ever lost even with concurrent workers.
    """
    def __init__(self, initial_value=0):
        self._value = initial_value
        self._lock = threading.Lock()

    def increment(self):
        """
        Increment the counter by 1 atomically.
        The lock ensures read-modify-write is atomic across threads.
        """
        with self._lock:  # lock held only for the critical section
            current = self._value
            time.sleep(0.001)  # simulate slight work to expose race if unlocked
            self._value = current + 1
        return self._value

    def get_value(self):
        """Return current counter value."""
        with self._lock:
            return self._value


# Alternative optimistic implementation (commented for clarity)
class OptimisticCounter:
    """
    Optimistic concurrency control: retry on conflict detection.
    Uses a version field to detect if state changed before committing.
    """
    def __init__(self, initial_value=0):
        self._value = initial_value
        self._version = 0
        self._lock = threading.Lock()

    def increment(self):
        # Retry loop with bounded attempts
        for _ in range(100):
            with self._lock:
                version = self._version
                value = self._value
            
            # Simulate work outside lock
            time.sleep(0.001)
            new_value = value + 1
            
            # Attempt commit with version check
            with self._lock:
                if self._version == version:
                    self._value = new_value
                    self._version += 1
                    return new_value
            # Conflict detected, retry
            
        raise RuntimeError("Failed to update after 100 retries")


# Example usage with two concurrent workers
if __name__ == "__main__":
    counter = SharedCounter(0)
    
    def worker(updates):
        for _ in range(updates):
            counter.increment()
    
    t1 = threading.Thread(target=worker, args=(1000,))
    t2 = threading.Thread(target=worker, args=(1000,))
    t1.start()
    t2.start()
    t1.join()
    t2.join()
    
    print(f"Final value: {counter.get_value()} (expected 2000)")
