import time
from threading import Lock

class FreshnessCache:
    def __init__(self, source_func, ttl=60):
        """
        Initialize cache with a source function and TTL (time-to-live).

        Args:
            source_func: Function that returns fresh data when called
            ttl: Maximum age of cached value in seconds (default: 60)
        """
        self.source_func = source_func
        self.ttl = ttl
        self._cache = {}
        self._lock = Lock()

    def get(self, key):
        """
        Get value for key, ensuring it's not stale.

        Args:
            key: Cache key to retrieve

        Returns:
            Cached value if fresh, otherwise fresh value from source
        """
        with self._lock:
            cached = self._cache.get(key)
            if cached and (time.time() - cached['timestamp']) < self.ttl:
                return cached['value']

            fresh_value = self.source_func(key)
            self._cache[key] = {
                'value': fresh_value,
                'timestamp': time.time()
            }
            return fresh_value

    def invalidate(self, key=None):
        """
        Invalidate cache entry(s).

        Args:
            key: Specific key to invalidate, or None to clear all
        """
        with self._lock:
            if key is None:
                self._cache.clear()
            else:
                self._cache.pop(key, None)
