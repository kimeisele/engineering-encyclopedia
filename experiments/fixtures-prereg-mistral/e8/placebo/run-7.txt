import time
from threading import Lock

class Cache:
    def __init__(self):
        self._cache = {}
        self._lock = Lock()
        self._last_updated = {}

    def get(self, key, fetch_func, ttl=None):
        with self._lock:
            current_time = time.time()
            cached_value, last_fetch_time = self._cache.get(key, (None, 0))

            if ttl is not None and current_time - last_fetch_time > ttl:
                del self._cache[key]
                cached_value = None

            if cached_value is None:
                cached_value = fetch_func()
                self._cache[key] = (cached_value, current_time)
                self._last_updated[key] = current_time

            return cached_value

    def invalidate(self, key):
        with self._lock:
            if key in self._cache:
                del self._cache[key]
                del self._last_updated[key]

# Example usage:

# value = cache.get("key", lambda: expensive_fetch(), ttl=300)
# cache.invalidate("key")
