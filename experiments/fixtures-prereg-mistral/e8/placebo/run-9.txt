Here's a Python implementation of a cache that ensures stale values are not served after the source data changes:

```python
import time
from threading import Lock

class StaleSafeCache:
    def __init__(self):
        self._cache = {}
        self._lock = Lock()
        self._version = 0

    def get(self, key):
        with self._lock:
            if key in self._cache:
                value, last_updated = self._cache[key]
                if last_updated == self._version:
                    return value
        return None

    def set(self, key, value):
        with self._lock:
            self._cache[key] = (value, self._version)

    def invalidate(self):
        with self._lock:
            self._version += 1

# Example usage:
cache = StaleSafeCache()

# Set a value
cache.set("user:123", "Alice")

# Get the value
print(cache.get("user:123"))  # Output: Alice

# Invalidate the cache
cache.invalidate()

# Now the value is stale
print(cache.get("user:123"))  # Output: None
```
