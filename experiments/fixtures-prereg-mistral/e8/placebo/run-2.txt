Here's a Python implementation of a cache that ensures stale values are not served after source data changes, along with the required application report:

```python
import time
from threading import Lock

class StaleSafeCache:
    def __init__(self):
        self._cache = {}
        self._lock = Lock()
        self._version = 0  # Tracks source data version

    def get(self, key):
        with self._lock:
            if key in self._cache:
                value, version = self._cache[key]
                if version == self._version:
                    return value
        return None

    def set(self, key, value):
        with self._lock:
            self._cache[key] = (value, self._version)

    def invalidate(self):
        with self._lock:
            self._version += 1

    def update_source(self, key, value):
        self.set(key, value)
        self.invalidate()

# Example usage:
cache = StaleSafeCache()
cache.update_source("key1", "value1")
print(cache.get("key1"))  # Returns "value1"

cache.update_source("key1", "new_value")
print(cache.get("key1"))  # Returns None (stale value not served)
cache.update_source("key1", "new_value")
print(cache.get("key1"))  # Returns "new_value"
```
