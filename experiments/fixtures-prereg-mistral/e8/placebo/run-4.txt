import time
from threading import Lock

class FreshnessCache:
    def __init__(self):
        self._cache = {}
        self._lock = Lock()
        self._version = 0  # Global version counter

    def get(self, key, source_func, freshness_threshold=0):
        """
        Returns cached value if fresh enough, otherwise fetches new value.

        Args:
            key: Cache key
            source_func: Function to fetch fresh data when needed
            freshness_threshold: Max age of cached value in seconds

        Returns:
            Cached or fresh value
        """
        with self._lock:
            # Check if we have a cached value
            if key in self._cache:
                cached_value, timestamp, version = self._cache[key]
                current_time = time.time()

                # If cache is fresh and source hasn't changed
                if (current_time - timestamp <= freshness_threshold and
                    version == self._version):
                    return cached_value

            # Get fresh data
            value = source_func()
            self._version += 1  # Increment version to invalidate old caches
            self._cache[key] = (value, time.time(), self._version)
            return value

    def invalidate(self):
        """Force cache invalidation by incrementing version"""
        with self._lock:
            self._version += 1

# Example usage:
if __name__ == "__main__":
    cache = FreshnessCache()
    expensive_data = [1, 2, 3]

    def get_data():
        return expensive_data.copy()

    # First call - cache miss
    print(cache.get("data", get_data))  # [1, 2, 3]

    # Second call - cache hit (if within threshold)
    print(cache.get("data", get_data))  # [1, 2, 3]

    # Modify source data
    expensive_data.append(4)

    # Cache still returns old value until invalidated or threshold exceeded
    print(cache.get("data", get_data))  # [1, 2, 3] (stale)
    cache.invalidate()  # Force invalidation
    print(cache.get("data", get_data))  # [1, 2, 3, 4] (fresh)
