import time
from threading import Lock

class FreshCache:
    def __init__(self):
        self._cache = {}
        self._lock = Lock()
        self._version = 0

    def get(self, key):
        with self._lock:
            if key in self._cache:
                value, version = self._cache[key]
                if version == self._version:
                    return value
        return None

    def set(self, key, value):
        with self._lock:
            self._cache[key] = (value, self._version)

    def invalidate(self):
        with self._lock:
            self._version += 1

# Example usage:
cache = FreshCache()

# When source data changes:
cache.invalidate()
