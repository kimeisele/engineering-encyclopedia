import time
from threading import Lock

class FreshCache:
    def __init__(self, source_func, ttl=None):
        self.source_func = source_func
        self.ttl = ttl
        self.cache = {}
        self.lock = Lock()
        self.last_updated = {}

    def get(self, key):
        current_time = time.time()

        with self.lock:
            # Check if we have a cached value and it's still fresh
            if key in self.cache:
                cached_time = self.last_updated.get(key, 0)
                if self.ttl is None or (current_time - cached_time) < self.ttl:
                    return self.cache[key]

            # Either no cached value or it's stale - fetch fresh data
            value = self.source_func(key)
            self.cache[key] = value
            self.last_updated[key] = current_time
            return value

    def invalidate(self, key):
        with self.lock:
            if key in self.cache:
                del self.cache[key]
                if key in self.last_updated:
                    del self.last_updated[key]
