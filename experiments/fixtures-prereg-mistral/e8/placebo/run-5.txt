import time
from threading import Lock

class FreshCache:
    def __init__(self):
        self._cache = {}
        self._lock = Lock()
        self._version = 0

    def get(self, key, source_func, ttl=None):
        """Get a value from cache or compute it fresh if needed.

        Args:
            key: The cache key
            source_func: Function to compute the value if not cached or stale
            ttl: Optional time-to-live in seconds (None means no expiration)

        Returns:
            The cached or fresh value
        """
        with self._lock:
            # Check if we have a cached value
            if key in self._cache:
                entry = self._cache[key]
                # Check if entry is fresh
                if ttl is None or (time.time() - entry['timestamp']) <= ttl:
                    return entry['value']

            # Either no cache entry or stale - compute fresh value
            value = source_func()
            self._cache[key] = {
                'value': value,
                'timestamp': time.time(),
                'version': self._version
            }
            self._version += 1
            return value

    def invalidate(self, key=None):
        """Invalidate cache entries.

        Args:
            key: Specific key to invalidate, or None to clear all
        """
        with self._lock:
            if key is None:
                self._cache.clear()
            else:
                self._cache.pop(key, None)
            self._version += 1
