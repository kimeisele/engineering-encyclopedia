import threading
from typing import Any, Callable, Optional

class Cache:
    def __init__(self):
        self._cache = {}
        self._lock = threading.Lock()
        self._version = 0

    def get_cached(self, key: str, source: Callable[[], Any]) -> Any:
        """Return cached value for key if available and valid, otherwise fetch fresh."""
        with self._lock:
            if key in self._cache:
                cached_value, version = self._cache[key]
                if version == self._version:
                    return cached_value

            # Cache miss or stale cache - fetch fresh value
            value = source()
            self._cache[key] = (value, self._version)
            return value

    def invalidate(self) -> None:
        """Invalidate all cached entries by incrementing version."""
        with self._lock:
            self._version += 1
            self._cache.clear()
