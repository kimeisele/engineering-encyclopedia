import time
from threading import Lock
from typing import Any, Callable, Optional

class StaleSafeCache:
    def __init__(self):
        self._cache: dict[Any, tuple[Any, int]] = {}  # key -> (value, version)
        self._lock = Lock()
        self._current_version = 0

    def get(self, key: Any, source_func: Callable[[], Any]) -> Any:
        """Returns cached value if valid, otherwise fetches fresh value.

        The cache key implicitly includes the source version through the
        version tracking mechanism. When source changes, version increments
        and stale cache entries are automatically invalidated.
        """
        with self._lock:
            # Check cache first
            if key in self._cache:
                cached_value, cached_version = self._cache[key]
                if cached_version == self._current_version:
                    return cached_value

            # Cache miss or stale - fetch fresh value
            new_value = source_func()
            self._cache[key] = (new_value, self._current_version)
            return new_value

    def invalidate(self) -> None:
        """Marks all cache entries as stale by incrementing version.

        This must be called after source data changes to prevent serving
        stale values. The version increment ensures no stale reads occur.
        """
        with self._lock:
            self._current_version += 1
            self._cache.clear()  # Optional: clear immediately or let get() handle it
