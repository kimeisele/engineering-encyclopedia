import time
from threading import Lock
from typing import Any, Callable

class StaleCache:
    def __init__(self):
        self._cache: dict[Any, tuple[Any, float, int]] = {}
        self._lock = Lock()
        self._version = 0

    def get(self, key: Any, source: Callable[[], Any]) -> Any:
        """Return cached value or fetch fresh if source changed."""
        with self._lock:
            current_version = self._version
            if key in self._cache:
                cached_value, cached_version, _ = self._cache[key]
                if cached_version == current_version:
                    return cached_value

            # Cache miss or stale - fetch fresh value
            value = source()
            self._cache[key] = (value, current_version, time.time())
            return value

    def invalidate(self) -> None:
        """Invalidate all cache entries by incrementing version."""
        with self._lock:
            self._version += 1

# Example usage:
# cache = StaleCache()
# value = cache.get("config", fetch_config_from_source)
# cache.invalidate()  # When source data changes
