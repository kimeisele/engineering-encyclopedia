import threading
from typing import Any, Callable, Optional

class Cache:
    def __init__(self):
        self._cache: dict[Any, Any] = {}
        self._lock = threading.Lock()
        self._version = 0

    def get(self, key: Any, source: Callable[[], Any]) -> Any:
        with self._lock:
            if key not in self._cache:
                self._cache[key] = source()
                return self._cache[key]

            current_version = self._version
            value = self._cache[key]

        # Recompute if source changed (version incremented)
        if current_version != self._version:
            with self._lock:
                # Double-check under lock to avoid race
                if key not in self._cache or self._version != current_version:
                    self._cache[key] = source()
                    return self._cache[key]
                return self._cache[key]

        return value

    def invalidate(self) -> None:
        with self._lock:
            self._version += 1
            self._cache.clear()

def cached_value(cache: Cache, key: Any, source: Callable[[], Any]) -> Any:
    return cache.get(key, source)
