import threading
from typing import Any, Callable, Optional

class Cache:
    def __init__(self):
        self._cache: dict[str, tuple[Any, int]] = {}
        self._lock = threading.Lock()
        self._version = 0

    def get(self, key: str, source: Callable[[], Any]) -> Any:
        # Check cache first
        with self._lock:
            cached_value, cached_version = self._cache.get(key, (None, -1))
            if cached_value is not None:
                return cached_value

        # Cache miss - compute value
        value = source()

        # Update cache with new version
        with self._lock:
            self._version += 1
            self._cache[key] = (value, self._version)

        return value

    def invalidate(self, key: str) -> None:
        with self._lock:
            if key in self._cache:
                del self._cache[key]

# Global cache instance
cache = Cache()
