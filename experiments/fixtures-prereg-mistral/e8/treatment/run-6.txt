import time
from threading import Lock
from typing import Any, Callable, Optional

class Cache:
    def __init__(self):
        self._cache: dict[Any, Any] = {}
        self._lock = Lock()
        self._version = 0

    def get(self, key: Any, source: Callable[[], Any], version: int) -> Any:
        with self._lock:
            cached_value, cached_version = self._cache.get(key, (None, None))
            if cached_value is not None and cached_version == version:
                return cached_value
            new_value = source()
            self._cache[key] = (new_value, version)
            return new_value

def cached_value(cache: Cache, key: Any, source: Callable[[], Any]) -> Any:
    version = cache._version
    value = cache.get(key, source, version)
    return value

def invalidate_cache(cache: Cache) -> None:
    with cache._lock:
        cache._version += 1
        cache._cache.clear()
