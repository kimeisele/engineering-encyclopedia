```python
import time
import threading
from typing import Any, Callable, Optional

class Cache:
    def __init__(self):
        self._cache = {}
        self._lock = threading.Lock()
        self._version = 0

    def get(self, key: str, source_func: Callable[[], Any], version: Optional[int] = None) -> Any:
        """Get a cached value for a key, ensuring it's not stale after source changes.

        Args:
            key: The cache key
            source_func: Function to fetch fresh data when needed
            version: Optional version to check against (for external version tracking)

        Returns:
            The cached or fresh value
        """
        with self._lock:
            # Check cache first
            if key in self._cache:
                cached_value, cached_version = self._cache[key]
                if version is None or cached_version >= version:
                    return cached_value

            # Fetch fresh data
            fresh_value = source_func()
            self._version += 1
            self._cache[key] = (fresh_value, self._version)
            return fresh_value

    def invalidate(self, key: str = None):
        """Invalidate cache entries. If key is None, clears entire cache."""
        with self._lock:
            if key is None:
                self._cache.clear()
            else:
                self._cache.pop(key, None)
```
