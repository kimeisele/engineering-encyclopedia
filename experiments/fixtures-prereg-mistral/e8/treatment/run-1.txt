import threading
from typing import Any, Callable, Optional, TypeVar

T = TypeVar('T')

class CacheEntry:
    def __init__(self, value: T, version: int):
        self.value = value
        self.version = version
        self.lock = threading.Lock()

class VersionedCache:
    def __init__(self):
        self._cache: dict[str, CacheEntry] = {}
        self._current_version = 0
        self._version_lock = threading.Lock()

    def get(self, key: str, source: Callable[[], T]) -> T:
        # Fast path: try to get cached value with current version
        with self._version_lock:
            current_version = self._current_version

        entry = self._cache.get(key)
        if entry is not None and entry.version == current_version:
            return entry.value

        # Slow path: need to update cache
        with self._version_lock:
            # Double-check after acquiring lock
            entry = self._cache.get(key)
            if entry is not None and entry.version == self._current_version:
                return entry.value

            # Update version and fetch new value
            self._current_version += 1
            new_version = self._current_version
            value = source()

            # Store with atomic write
            self._cache[key] = CacheEntry(value, new_version)
            return value

    def invalidate(self):
        """Increment version to invalidate all cached entries"""
        with self._version_lock:
            self._current_version += 1

# Global cache instance
cache = VersionedCache()

def get_cached_value(key: str, source: Callable[[], T]) -> T:
    """
    Returns a cached value for a key, ensuring no stale value is served
    after source data changes.

    The cache uses a versioning system where each cache entry is tagged with
    the version of the source at the time of caching. When the source changes,
    the version is incremented, making all existing cache entries stale.

    Args:
        key: The cache key
        source: Function that returns the current value from source

    Returns:
        The current value (either cached or freshly fetched)
    """
    return cache.get(key, source)

def invalidate_cache():
    """Invalidate all cached values by incrementing the version"""
    cache.invalidate()
