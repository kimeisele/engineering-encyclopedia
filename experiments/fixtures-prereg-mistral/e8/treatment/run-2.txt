```python
import threading
from typing import Callable, Any

class Cache:
    def __init__(self):
        self._cache = {}
        self._locks = {}
        self._lock = threading.Lock()

    def get_cached(self, key: str, fetch_func: Callable[[], Any]) -> Any:
        # Check cache first
        with self._lock:
            if key in self._cache:
                return self._cache[key]

        # If not in cache, fetch and store
        result = fetch_func()

        with self._lock:
            # Double-check in case another thread cached it while we were fetching
            if key not in self._cache:
                self._cache[key] = result
            return self._cache[key]

    def invalidate(self, key: str) -> None:
        with self._lock:
            if key in self._cache:
                del self._cache[key]
```
