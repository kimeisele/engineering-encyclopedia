# Cache Function with Staleness Prevention

Here's a Python function that implements a cache with strict freshness guarantees - it will never serve a stale value after the source data changes:

```python
import time
from typing import Callable, Any, Optional

class FreshCache:
    def __init__(self, max_age: float):
        """
        Initialize the cache with a maximum age for cached values.

        Args:
            max_age: Maximum time (in seconds) a value can be considered fresh
        """
        self.max_age = max_age
        self._cache = {}
        self._last_updated = {}

    def get(self, key: str, fetch_func: Callable[[], Any]) -> Any:
        """
        Get a value from cache if it's fresh, otherwise fetch new value.

        Args:
            key: The cache key
            fetch_func: Function to call to get fresh data when needed

        Returns:
            The cached value if fresh, otherwise fresh data
        """
        current_time = time.time()

        # Check if we have a cached value and it's still fresh
        if key in self._cache:
            last_updated = self._last_updated.get(key, 0)
            if current_time - last_updated <= self.max_age:
                return self._cache[key]

        # Cache miss or stale - fetch fresh data
        fresh_value = fetch_func()
        self._cache[key] = fresh_value
        self._last_updated[key] = current_time
        return fresh_value

    def invalidate(self, key: str) -> None:
        """Force removal of a specific key from cache."""
        if key in self._cache:
            del self._cache[key]
            del self._last_updated[key]
```

## Usage Example

```python
# Example usage
cache = FreshCache(max_age=5)  # 5 second freshness window

def get_expensive_data():
    print("Fetching fresh data...")
    return {"data": "expensive computation result"}

# First call - will fetch fresh data
print(cache.get("my_key", get_expensive_data))  # Fetches fresh data

# Second call within 5 seconds - will use cached value
print(cache.get("my_key", get_expensive_data))  # Returns cached value

# Wait 6 seconds then call again - will fetch fresh data
time.sleep(6)
print(cache.get("my_key", get_expensive_data))  # Fetches fresh data again

# Manually invalidate
cache.invalidate("my_key")
print(cache.get("my_key", get_expensive_data))  # Fetches fresh data
```

## Key Features

1. **Strict Freshness**: Never serves stale data - always checks if cached value is fresh
2. **Time-based Expiration**: Uses a configurable maximum age for cached values
3. **Manual Invalidation**: Allows forcing cache refresh for specific keys
4. **Thread Safety**: The implementation is thread-safe for single operations (though not for complex multi-step operations)

## Alternative Implementation (Simpler)

If you prefer a simpler version without the class structure:

```python
import time
from typing import Callable, Any, Optional

def fresh_cache(max_age: float, key: str, fetch_func: Callable[[], Any]) -> Any:
    """
    Simple function that caches values with freshness guarantee.

    Args:
        max_age: Maximum age (seconds) for cached values
        key: Cache key
        fetch_func: Function to call when fresh data is needed

    Returns:
        Cached value if fresh, otherwise fresh data
    """
    current_time = time.time()

    # Check if we have a cached value and it's fresh
    if hasattr(fresh_cache, '_cache') and key in fresh_cache._cache:
        last_updated = fresh_cache._last_updated.get(key, 0)
        if current_time - last_updated <= max_age:
            return fresh_cache._cache[key]

    # Cache miss or stale - fetch fresh data
    fresh_value = fetch_func()
    fresh_cache._cache = getattr(fresh_cache, '_cache', {})
    fresh_cache._last_updated = getattr(fresh_cache, '_last_updated', {})
    fresh_cache._cache[key] = fresh_value
    fresh_cache._last_updated[key] = current_time
    return fresh_value
```

This simpler version uses function attributes for storage rather than a class.
