```python
import threading
from typing import Optional

class AtomicCounter:
    def __init__(self, initial_value: int = 0):
        self._value = initial_value
        self._lock = threading.Lock()

    def increment(self, amount: int = 1) -> None:
        with self._lock:
            self._value += amount

    @property
    def value(self) -> int:
        with self._lock:
            return self._value

def update_counter(counter: AtomicCounter, amount: int) -> None:
    counter.increment(amount)
```
