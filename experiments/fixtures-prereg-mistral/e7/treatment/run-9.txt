from typing import Optional

class SharedCounter:
    def __init__(self, initial_value: int = 0):
        self._value = initial_value
        self._lock = threading.Lock()

    def increment(self, amount: int = 1) -> None:
        """Thread-safe increment of the shared counter using pessimistic locking."""
        with self._lock:
            self._value += amount

    def get_value(self) -> int:
        """Thread-safe read of the current counter value."""
        with self._lock:
            return self._value

# Alternative optimistic locking version
class OptimisticSharedCounter:
    def __init__(self, initial_value: int = 0):
        self._value = initial_value

    def increment(self, amount: int = 1) -> bool:
        """
        Attempt to increment the counter using optimistic locking.
        Returns True if successful, False if conflict detected.
        """
        while True:
            old_value = self._value
            new_value = old_value + amount
            # Simulate atomic compare-and-swap operation
            if self._value == old_value:
                self._value = new_value
                return True
            # Conflict detected, retry

    def get_value(self) -> int:
