import threading
from dataclasses import dataclass

@dataclass
class Counter:
    value: int
    version: int = 0

class CounterService:
    def __init__(self):
        self._counter = Counter(0)
        self._lock = threading.Lock()

    def increment(self) -> bool:
        """Atomically increment the counter if no concurrent updates occurred.
        Returns True if the increment succeeded, False if a conflict was detected."""
        with self._lock:
            current = self._counter
            new_value = current.value + 1
            new_version = current.version + 1

        # Try to commit the update
        with self._lock:
            if self._counter.version != current.version:
                return False  # Conflict detected
            self._counter = Counter(new_value, new_version)
