Here's the implementation using an atomic counter with optimistic locking (version field) to handle concurrent updates:

```python
import threading
from dataclasses import dataclass

@dataclass
class SharedCounter:
    value: int
    version: int

class CounterUpdater:
    def __init__(self):
        self._counter = SharedCounter(0, 0)
        self._lock = threading.Lock()

    def increment(self) -> None:
        with self._lock:
            self._counter.value += 1
            self._counter.version += 1

    def get_value(self) -> int:
        with self._lock:
            return self._counter.value

# Alternative implementation using atomic operations (no lock needed)
import ctypes

class AtomicCounter:
    def __init__(self):
        self._counter = ctypes.c_long(0)

    def increment(self) -> None:
        ctypes.atomic_increment(self._counter)

    def get_value(self) -> int:
        return self._counter.value
```
