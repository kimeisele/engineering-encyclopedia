```python
import threading

class SharedCounter:
    def __init__(self, initial_value=0):
        self.value = initial_value
        self.lock = threading.Lock()

    def increment(self, amount=1):
        with self.lock:
            self.value += amount
            return self.value

# Example usage:
counter = SharedCounter()
# Worker 1
counter.increment(5)
# Worker 2
counter.increment(3)
# Final value is 8 (no updates lost)
```
