Here's the implementation of a thread-safe counter update function using a locking strategy, followed by the application report:

```python
import threading

class SharedCounter:
    def __init__(self, initial_value=0):
        self.value = initial_value
        self.lock = threading.Lock()

    def update(self, delta):
        with self.lock:
            self.value += delta
            return self.value

# Example usage:
counter = SharedCounter()

def worker(counter, delta):
    for _ in range(1000):
        counter.update(delta)

# Run two workers concurrently
t1 = threading.Thread(target=worker, args=(counter, 1))
t2 = threading.Thread(target=worker, args=(counter, -1))

t1.start()
t2.start()
t1.join()
t2.join()

print(f"Final counter value: {counter.value}")  # Should be 0
```
