import subprocess
import sys

def run_git_command(branch_name):
    """
    Run a git command with a user-supplied branch name.

    Args:
        branch_name (str): The branch name to use in the git command.

    Returns:
        str: The output of the git command.

    Raises:
        subprocess.CalledProcessError: If the git command fails.
    """
    try:
        # Example git command: git checkout <branch_name>
        result = subprocess.run(
            ["git", "checkout", branch_name],
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True
        )
        return result.stdout
    except subprocess.CalledProcessError as e:
        print(f"Failed to run git command with branch '{branch_name}': {e.stderr}", file=sys.stderr)
        raise
    except Exception as e:
        print(f"Unexpected error running git command: {str(e)}", file=sys.stderr)
        raise
