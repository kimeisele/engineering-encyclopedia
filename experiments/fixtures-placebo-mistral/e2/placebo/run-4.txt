import subprocess

def run_git_command(branch_name):
    """
    Runs a git command with the user-supplied branch name.

    Args:
        branch_name (str): The name of the branch to use in the git command.

    Returns:
        str: The output of the git command.

    Raises:
        subprocess.CalledProcessError: If the git command fails.
    """
    try:
        # Example git command: git checkout <branch_name>
        result = subprocess.run(
            ["git", "checkout", branch_name],
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True
        )
        return result.stdout
    except subprocess.CalledProcessError as e:
        # Include operation, identifier (branch_name), and failing step (git command)
        raise RuntimeError(
            f"Git command failed for branch '{branch_name}'. "
            f"Command: git checkout {branch_name}. "
            f"Error: {e.stderr}"
        ) from e
